   480→   480→function fmtTime(ts: number): string {

   504→   504→/** 长按手势（指针事件实现；移动超过阈值视为滚动，取消触发） */
   505→   505→function useLongPress(onLongPress: () => void, ms = 480) {
   506→   506→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
   507→   507→  const fired = useRef(false);
   508→   508→  const startPoint = useRef<{ x: number; y: number } | null>(null);
   509→   509→  const clear = () => {
   510→   510→    if (timer.current) {
   511→   511→      clearTimeout(timer.current);
   512→   512→      timer.current = null;
   513→   513→    }
   514→   514→  };
   515→   515→  return {
   516→   516→    onPointerDown: (e: ReactPointerEvent<HTMLButtonElement>) => {
   517→   517→      fired.current = false;
   518→   518→      startPoint.current = { x: e.clientX, y: e.clientY };
   519→   519→      clear();
   520→   520→      timer.current = setTimeout(() => {
   521→   521→        fired.current = true;
   522→   522→        onLongPress();
   523→   523→      }, ms);
   524→   524→    },
   525→   525→    onPointerMove: (e: ReactPointerEvent<HTMLButtonElement>) => {
   526→   526→      const s = startPoint.current;
   527→   527→      if (s && Math.hypot(e.clientX - s.x, e.clientY - s.y) > 8) clear();
   528→   528→    },
   529→   529→    onPointerUp: clear,
   530→   530→    onPointerLeave: clear,
   531→   531→    onPointerCancel: clear,
   532→   532→    didFire: () => fired.current,
   533→   533→  };
   534→   534→}
