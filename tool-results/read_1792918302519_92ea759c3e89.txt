    87→/** HSL → #rrggbb（彩虹色相条选色用） */
    88→function hslToHex(h: number, s: number, l: number): string {
    89→  const f = (n: number) => {
    90→    const k = (n + h / 30) % 12;
    91→    const c = l - s * Math.min(l, 1 - l) * Math.max(-1, Math.min(k - 3, 9 - k, 1));
    92→    return Math.round(255 * c)
    93→      .toString(16)
    94→      .padStart(2, '0');
    95→  };
    96→  return `#${f(0)}${f(8)}${f(4)}`;
    97→}

   103→/** #rrggbb → 色相 0..360（无彩色返回 0） */
   104→function hexToHue(hex: string): number {
   105→  const n = parseInt(hex.slice(1), 16);
   106→  const r = ((n >> 16) & 255) / 255;
   107→  const g = ((n >> 8) & 255) / 255;
   108→  const b = (n & 255) / 255;
   109→  const mx = Math.max(r, g, b);
   110→  const mn = Math.min(r, g, b);
   111→  if (mx === mn) return 0;
   112→  const d = mx - mn;
   113→  let h: number;
   114→  if (mx === r) h = ((g - b) / d + (g < b ? 6 : 0)) * 60;
   115→  else if (mx === g) h = ((b - r) / d + 2) * 60;
   116→  else h = ((r - g) / d + 4) * 60;
   117→  return h;
   118→}

   534→export interface WardrobeProps {
   535→  open: boolean;
   536→  wear: GirlWear;
   537→  wear2: BoyWear;
   538→  onChange: (w: GirlWear) => void;
   539→  onChange2: (w: BoyWear) => void;

   542→  petWear?: PetWear;

   544→  petCoat?: PetCoat;
   545→  /** 宠物换装回调（patch 只带变更的字段） */
   546→  onPetChange?: (patch: { coat?: PetCoat; wear?: PetWear }) => void;

   551→export default function Wardrobe({
   552→  open,
   553→  wear,
   554→  wear2,
   555→  onChange,
   556→  onChange2,
   557→  onClose,
   558→  petWear,
   559→  petCoat,
   560→  onPetChange,
   561→  initialChar,
   562→}: WardrobeProps) {
   563→  const [char, setChar] = useState<'girl' | 'boy' | 'pet'>(initialChar ?? 'girl');
   564→  const [tab, setTab] = useState<WardTab>('hair');
   565→  const inDressMode = wear.dress >= 0;
   566→  const tabs = char === 'girl' ? TABS_GIRL : TABS_BOY;
   567→
   568→  /** 切换角色：保留当前分类（女生专属的连衣裙在男生侧回落到上衣；宠物栏不占用 tab） */
   569→  const switchChar = (c: 'girl' | 'boy' | 'pet') => {
   570→    if (c === char) return;
   571→    sfx.tap();
   572→    setChar(c);
   573→    if (c === 'boy' && tab === 'dress') setTab('top');
   574→  };
   575→
   576→  const items = useMemo(() => {
   577→    if (char === 'boy') {
   578→      if (tab === 'hair') return BOY_HAIR_ORDER.map((h) => ({ key: h, name: BOY_HAIR_NAMES[h] ?? h }));
   579→      if (tab === 'top') return BOY_TOPS.map((t, i) => ({ key: String(i), name: t.name }));
   580→      if (tab === 'bottom') return BOY_BOTTOMS.map((b, i) => ({ key: String(i), name: b.name }));
   581→      return BOY_SHOES.map((s, i) => ({ key: String(i), name: s.name }));
   582→    }
   583→    if (tab === 'hair') return HAIR_ORDER.map((h) => ({ key: h, name: HAIR_NAMES[h] ?? h }));
   584→    if (tab === 'dress') return DRESSES.map((d, i) => ({ key: String(i), name: d.name }));
   585→    if (tab === 'top') return TOPS.map((t, i) => ({ key: String(i), name: t.name }));
   586→    if (tab === 'bottom') return BOTTOMS.map((b, i) => ({ key: String(i), name: b.name }));
   587→    return SHOES.map((s, i) => ({ key: String(i), name: s.name }));
   588→  }, [char, tab]);
   589→
   590→  const isSel = (key: string) => {
   591→    if (char === 'boy') {
   592→      if (tab === 'hair') return wear2.hair === key;
   593→      if (tab === 'top') return wear2.top === Number(key);
   594→      if (tab === 'bottom') return wear2.bottom === Number(key);
   595→      return wear2.shoes === Number(key);
   596→    }
   597→    if (tab === 'hair') return wear.hair === key;
   598→    if (tab === 'dress') return inDressMode && wear.dress === Number(key);
   599→    if (tab === 'top') return !inDressMode && wear.top === Number(key);
   600→    if (tab === 'bottom') return !inDressMode && wear.bottom === Number(key);
   601→    return wear.shoes === Number(key);
   602→  };
   603→
   604→  const pick = (key: string) => {
   605→    sfx.tap();
   606→    if (char === 'boy') {
   607→      if (tab === 'hair') onChange2({ ...wear2, hair: key as BoyHairStyle });
   608→      else if (tab === 'top') onChange2({ ...wear2, top: Number(key) });
   609→      else if (tab === 'bottom') onChange2({ ...wear2, bottom: Number(key) });
   610→      else onChange2({ ...wear2, shoes: Number(key) });
   611→      return;
   612→    }
   613→    if (tab === 'hair') onChange({ ...wear, hair: key as GirlHairStyle });
   614→    else if (tab === 'dress') onChange({ ...wear, dress: Number(key) });
   615→    else if (tab === 'top') onChange({ ...wear, dress: -1, top: Number(key) });
   616→    else if (tab === 'bottom') onChange({ ...wear, dress: -1, bottom: Number(key) });
   617→    else onChange({ ...wear, shoes: Number(key) });
   618→  };
   619→
   620→  // —— 调色盘（当前角色 + 当前分类的槽位；null = 还原默认款色） ——
   621→  const slot = TINT_SLOT[tab];
   622→  const curTint = (char === 'girl' ? wear.tints?.[slot] : wear2.tints?.[slot]) ?? null;
   623→  const paletteColors = tab === 'hair' ? HAIR_COLORS : CLOTH_COLORS;

   627→    if (char === 'girl') {

   630→      else delete t[slot];
