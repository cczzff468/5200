     1→/**
     2→ * 群聊数据层（宿主：微信 / QQ 双 App）。
     3→ *
     4→ * 设计要点：
     5→ * - 群 = ChatGroup：独立会话实体（非联系人），成员是 AI 角色（char/npc）联系人 ID，机主固定参与；
     6→ * - 会话身份 sessionKey = `<app>:group:<id>`（复用流总线/未读/标志/时间感知等按字符串键隔离的设施）；
     7→ * - 存储：微信群 `wx-chat-groups`（历史数据兼容）、QQ 群 `qq-chat-groups` 两池分开，群 id 全局唯一，
     8→ *   getGroup 双池查找；消息落 `<app>-group-msgs:<id>`（微信群沿用历史 wx-group-msgs 前缀）；
     9→ * - 记忆互通开关 memoryInterop 按群独立（设置页唯一入口，不再有按成员覆盖）：开启时群记忆参与成员角色的
    10→ *   私聊召回、角色私聊记忆参与群聊召回；effectiveInterop 统一解析，聊天页把结果以 interopOn 回调
    11→ *   传给记忆召回；角色之间的隔离永远由记忆存储键（mem-frag:<contactId>）保证，与开关无关；

    13→ * - 群红包（普通/拼手气/专属）与指定成员转账：消息落本群消息库，状态随群持久化（按群 ID 隔离，
    14→ *   与单聊互不相通）；成员按人设领取/收款/退回，纯本地模拟不涉及真实资金；
    15→ * - 解散群聊级联清理：群记录/群消息（含红包/转账卡片状态）/会话附属数据/群聊天背景 + 群来源记忆
    16→ *   （onGroupDissolved 钩子由 memory 层注册，删去各成员记忆库里 sourceGroupId=本群的碎片/总结，
    17→ *   依赖方向保持 memory → groups 单向，不反向 import）。
    18→ */

   194→function readJSON<T>(key: string): T | null {
   195→  try {
   196→    return kvGet<T>(key);
   197→  } catch {
   198→    return null;
   199→  }
   200→}

   202→function writeJSON(key: string, value: unknown): void {
   203→  try {
   204→    kvSet(key, value);
   205→  } catch {
   206→    // 存储失败不中断（群是增强能力）
   207→  }

   215→function normalizeGroup(g: unknown): ChatGroup | null {
   216→  if (!g || typeof g !== 'object') return null;
   217→  const r = g as Partial<ChatGroup>;
   218→  if (typeof r.id !== 'string' || !Array.isArray(r.memberIds)) return null;
   219→  return {
   220→    id: r.id,
   221→    app: r.app === 'qq' ? 'qq' : 'wx',
   222→    name: typeof r.name === 'string' ? r.name : '未命名群聊',
   223→    avatar: typeof r.avatar === 'string' ? r.avatar : null,
   224→    ownerId: typeof r.ownerId === 'string' ? r.ownerId : '',
   225→    memberIds: r.memberIds.filter((x): x is string => typeof x === 'string'),
   226→    // 管理员/禁言为新增字段：旧数据缺省补空值（normalize 保证读写形状一致）
   227→    adminIds: Array.isArray(r.adminIds) ? r.adminIds.filter((x): x is string => typeof x === 'string') : [],
   228→    mutes:
   229→      r.mutes && typeof r.mutes === 'object' && !Array.isArray(r.mutes)
   230→        ? Object.fromEntries(
   231→            Object.entries(r.mutes as Record<string, unknown>).filter(
   232→              (entry): entry is [string, number | null] => typeof entry[0] === 'string' && (entry[1] === null || typeof entry[1] === 'number'),
   233→            ),
   234→          )
   235→        : {},
   236→    announcement: typeof r.announcement === 'string' ? r.announcement : '',
   237→    memoryInterop: r.memoryInterop === true,
   238→    createdAt: typeof r.createdAt === 'number' ? r.createdAt : 0,
   239→  };
   240→}

   242→function readPool(app: GroupApp): ChatGroup[] {
   243→  const raw = readJSON<ChatGroup[]>(POOLS[app]);
   244→  if (!Array.isArray(raw)) return [];
   245→  return raw.map(normalizeGroup).filter((g): g is ChatGroup => g != null);
   246→}
   247→
   248→function writePool(app: GroupApp, list: ChatGroup[]): void {
   249→  writeJSON(POOLS[app], list);
   250→}

   262→export function createGroup(input: {
   263→  name: string;
   264→  memberIds: string[];
   265→  ownerId: string;
   266→  /** 宿主 App（缺省 wx，兼容旧调用） */
   267→  app?: GroupApp;
   268→  avatar?: string | null;
   269→  announcement?: string;
   270→}): ChatGroup {
   271→  const app: GroupApp = input.app ?? 'wx';
   272→  const group: ChatGroup = {
   273→    id: genId(),
   274→    app,
   275→    name: input.name.trim() || '未命名群聊',
   276→    avatar: input.avatar ?? null,
   277→    ownerId: input.ownerId,
   278→    memberIds: Array.from(new Set(input.memberIds)).slice(0, GROUP_MEMBER_CAP),
   279→    adminIds: [],
   280→    mutes: {},
   281→    announcement: input.announcement?.trim() ?? '',
   282→    memoryInterop: false,
   283→    createdAt: Date.now(),
   284→  };
   285→  writePool(app, [...readPool(app), group]);
   286→  // 建群系统消息（三.9）：居中灰字「群聊创建」，随群消息持久化
   287→  pushGroupEvent(group.id, '群聊创建', { type: 'create' });
   288→  return group;
   289→}

   291→/**
   292→ * 更新群（名字/头像/公告/互通开关/成员等）；返回更新后的群（群不存在返回 null）。
   293→ * 群资料变更（改名/换头像/改公告）自动落一条群事件系统消息（三.7/8：所有操作要生成对应系统消息），
   294→ * 机主无关调用方是谁（信息页/聊天页/宿主壳层统一在此处发事件，避免遗漏）。
   295→ */
   296→export function updateGroup(
   297→  groupId: string,
   298→  patch: Partial<Pick<ChatGroup, 'name' | 'avatar' | 'announcement' | 'memoryInterop' | 'memberIds'>>
   299→): ChatGroup | null {
   300→  const app = getGroup(groupId)?.app;
   301→  if (!app) return null;
   302→  const list = readPool(app);
   303→  const idx = list.findIndex((g) => g.id === groupId);
   304→  if (idx === -1) return null;
   305→  const prev = list[idx];
   306→  const next: ChatGroup = { ...prev, ...patch };
   307→  if (patch.name !== undefined) next.name = patch.name.trim() || next.name;
   308→  list[idx] = next;
   309→  writePool(app, list);
   310→  // 群资料事件（只在值真正变化时发，成员增删走专用函数不经过这里）
   311→  if (patch.name !== undefined && next.name !== prev.name) {
   312→    pushGroupEvent(groupId, `群名被修改为「${next.name}」`, { type: 'rename', extra: next.name });
   313→  }
   314→  if (patch.avatar !== undefined && (patch.avatar ?? null) !== (prev.avatar ?? null)) {
   315→    pushGroupEvent(groupId, '群头像已更新', { type: 'avatar' });
   316→  }
   317→  if (patch.announcement !== undefined && (patch.announcement ?? '') !== (prev.announcement ?? '')) {
   318→    pushGroupEvent(groupId, '群公告已更新', { type: 'announcement' });
   319→  }
   320→  return next;
   321→}

   323→/** 邀请成员：已在群里原样返回；人数达上限（GROUP_MEMBER_CAP）返回 null（UI 提示「已达上限」）。
   324→ *  成功时落「XX加入了群聊」系统消息（三.1；opts.name = 新成员显示名，供事件文本使用）。 */
   325→export function addGroupMember(groupId: string, contactId: string, opts?: { name?: string }): ChatGroup | null {
   326→  const g = getGroup(groupId);
   327→  if (!g) return null;
   328→  if (g.memberIds.includes(contactId)) return g;
   329→  if (g.memberIds.length >= GROUP_MEMBER_CAP) return null;
   330→  const next = updateGroup(groupId, { memberIds: [...g.memberIds, contactId] });
   331→  if (next && opts?.name) {
   332→    pushGroupEvent(groupId, `${opts.name}加入了群聊`, { type: 'join', targetId: contactId });
   333→  }
   334→  return next;
   335→}

   337→/** 原始移除（不发事件；供联系人删除级联等内部路径使用；UI 踢人请用 kickGroupMember） */
   338→export function removeGroupMember(groupId: string, contactId: string): ChatGroup | null {
   339→  const g = getGroup(groupId);
   340→  if (!g) return null;
   341→  return updateGroup(groupId, { memberIds: g.memberIds.filter((id) => id !== contactId) });
   342→}

   347→export function groupRoleOf(g: ChatGroup, contactId: string): GroupMemberRole {

   353→/** 是否处于禁言中（永久禁言或未到解禁时间；过期即视为自动解禁，无需写回） */
   354→export function isGroupMuted(g: ChatGroup, contactId: string, now: number = Date.now()): boolean {
   355→  const until = g.mutes[contactId];
   356→  if (until === null) return true;
   357→  if (typeof until === 'number' && until > now) return true;
   358→  return false;
   359→}

   361→/** 禁言剩余描述（「剩 X 分钟 / X 小时 / X 天」或「永久」；未禁言返回 null） */
   362→export function groupMuteLeftText(g: ChatGroup, contactId: string, now: number = Date.now()): string | null {
   363→  const until = g.mutes[contactId];
   364→  if (until === null) return '永久';
   365→  if (typeof until !== 'number' || until <= now) return null;
   366→  const left = Math.max(1, Math.ceil((until - now) / 60_000));
   367→  if (left < 60) return `剩 ${left} 分钟`;
   368→  const hours = Math.ceil(left / 60);
   369→  if (hours < 24) return `剩 ${hours} 小时`;
   370→  return `剩 ${Math.ceil(hours / 24)} 天`;
   371→}

   373→/** 时长毫秒 → 人类可读（事件文本用；固定档位优先用 GROUP_MUTE_PRESETS 的 label） */
   374→function muteDurationText(ms: number | null): string {
   375→  if (ms === null) return '永久';
   376→  const preset = GROUP_MUTE_PRESETS.find((p) => p.ms === ms);
   377→  if (preset) return preset.label;
   378→  if (ms % 60_000 === 0) {
   379→    const mins = ms / 60_000;
   380→    if (mins % 60 === 0) return `${mins / 60} 小时`;
   381→    return `${mins} 分钟`;
   382→  }
   383→  return `${Math.round(ms / 60_000)} 分钟`;
   384→}

   433→  });
   434→  return list[idx];

   530→/** 事件相对时间标签（刚刚 / X 分钟前 / X 小时前 / 昨天 / M月D日 HH:MM） */
   531→export function groupEventAgo(ts: number, now: number = Date.now()): string {
   532→  const diff = now - ts;
   533→  if (diff < 60_000) return '刚刚';
   534→  if (diff < 60 * 60_000) return `${Math.floor(diff / 60_000)} 分钟前`;
   535→  if (diff < 24 * 60 * 60_000) return `${Math.floor(diff / (60 * 60_000))} 小时前`;
   536→  const d = new Date(ts);
   537→  const hh = String(d.getHours()).padStart(2, '0');
   538→  const mm = String(d.getMinutes()).padStart(2, '0');
   539→  const yest = new Date(now - 24 * 60 * 60_000);
   540→  if (d.getFullYear() === yest.getFullYear() && d.getMonth() === yest.getMonth() && d.getDate() === yest.getDate()) {
   541→    return `昨天 ${hh}:${mm}`;
   542→  }
   543→  return `${d.getMonth() + 1}月${d.getDate()}日 ${hh}:${mm}`;
   544→}

   554→/** 从 localStorage JSON map 里删掉一个键（unreads/flags/hidden/time-aware 同构清理用） */

   572→ *  钩子异常互不影响：单个失败不阻塞其它清理，也不阻塞解散本身。 */
   573→export function onGroupDissolved(fn: GroupDissolveHook): void {

   577→/**
   578→ * 解散群聊：删除群 + 群消息（红包/转账卡片状态随消息一并清除）+ 会话级附属数据
   579→ * （未读/标志/隐藏/时间感知/每成员的群记忆提取轮次计数/群聊天背景）+ 群来源记忆
   580→ * （通过 onGroupDissolved 钩子级联，见 memory 层 memPurgeGroupSource）。
   581→ * 返回被解散的群（供 UI 提示），群不存在返回 null。
   582→ */
   583→export function dissolveGroup(groupId: string): ChatGroup | null {
   584→  const g = getGroup(groupId);
   585→  if (!g) return null;
   586→  writePool(g.app, readPool(g.app).filter((x) => x.id !== groupId));
   587→  try {
   588→    kvDel(groupMsgsKey(g.app, groupId));
   589→    // 未读/标志/隐藏为 localStorage JSON map（键 = 会话 id；群会话 id = `group:<gid>`）
   590→    const maps = LS_MAPS[g.app];
   591→    removeLocalMapKey(maps.unreads, `group:${groupId}`);
   592→    removeLocalMapKey(maps.flags, `group:${groupId}`);
   593→    removeLocalMapKey(maps.hidden, `group:${groupId}`);
   594→    // 时间感知开关（localStorage map 里的键）
   595→    removeLocalMapKey('chat-time-aware', `${g.app}:group:${groupId}`);
   596→    // 每个成员的群记忆提取轮次计数（mem-round:<contactId>:<app>:group:<gid>）
   597→    for (const cid of g.memberIds) {
   598→      kvDel(`mem-round:${cid}:${g.app}:group:${groupId}`);
   599→    }
   600→    // 群聊天背景图片本体（IndexedDB settings store，键同 contacts-store 的 chat-bg 前缀；群背景与单聊相互独立）
   601→    void localDB.delete('settings', `chat-bg:${g.app}:group:${groupId}`).catch(() => undefined);
   602→  } catch {
   603→    // 清理失败不阻塞解散
   604→  }
   605→  // 群来源记忆级联清理（钩子逐个调用；单个钩子异常不影响其余清理，更不阻塞解散）
   606→  for (const fn of dissolveHooks) {
   607→    try {
   608→      fn(g);
   609→    } catch {
   610→      // 忽略
   611→    }
   612→  }
   613→  return g;
   614→}

   617→ *  opts.name 提供时给留下的成员落「XX退出了群聊」事件（三.2；AI 由此知道人为什么不见了）。 */

   631→    }
   632→  }
   633→  return affected;
   634→}

   638→function normalizeMsg(m: unknown): WxGroupMsg | null {
   639→  if (!m || typeof m !== 'object') return null;
   640→  const r = m as Partial<WxGroupMsg>;
   641→  if (typeof r.id !== 'string' || typeof r.content !== 'string' || typeof r.time !== 'number') return null;
   642→  return {
   643→    id: r.id,
   644→    role: r.role === 'me' ? 'me' : 'peer',
   645→    senderId: typeof r.senderId === 'string' ? r.senderId : 'unknown',
   646→    senderName: typeof r.senderName === 'string' ? r.senderName : '',
   647→    content: r.content,
   648→    time: r.time,
   649→    kind:
   650→      r.kind === 'notice' || r.kind === 'image' || r.kind === 'location' || r.kind === 'sticker' || r.kind === 'redpacket' || r.kind === 'transfer' || r.kind === 'forward'
   651→        ? r.kind
   652→        : 'text',
   653→    rp:
   654→      r.kind === 'redpacket' && r.rp && typeof r.rp.amount === 'number'
   655→        ? {
   656→            amount: r.rp.amount,
   657→            count: typeof r.rp.count === 'number' && r.rp.count >= 1 ? Math.floor(r.rp.count) : 1,
   658→            mode: r.rp.mode === 'lucky' || r.rp.mode === 'exclusive' ? r.rp.mode : 'normal',
   659→            blessing: typeof r.rp.blessing === 'string' ? r.rp.blessing : '恭喜发财，大吉大利',
   660→            targetId: typeof r.rp.targetId === 'string' ? r.rp.targetId : undefined,
   661→            targetName: typeof r.rp.targetName === 'string' ? r.rp.targetName : undefined,
   662→            claims: Array.isArray(r.rp.claims)
   663→              ? r.rp.claims
   664→                  .filter((c): c is GroupRpData['claims'][number] => Boolean(c) && typeof (c as { name?: unknown }).name === 'string')
   665→                  .map((c) => ({
   666→                    contactId: typeof c.contactId === 'string' ? c.contactId : '',
   667→                    name: c.name,
   668→                    avatar: typeof c.avatar === 'string' ? c.avatar : null,
   669→                    amount: typeof c.amount === 'number' ? c.amount : 0,
   670→                    ts: typeof c.ts === 'number' ? c.ts : 0,
   671→                  }))
   672→              : [],
   673→            sentAt: typeof r.rp.sentAt === 'number' ? r.rp.sentAt : r.time,
   674→            expired: r.rp.expired === true || undefined,
   675→            expiredAt: typeof r.rp.expiredAt === 'number' ? r.rp.expiredAt : undefined,
   676→            cid: typeof r.rp.cid === 'string' ? r.rp.cid : undefined,
   677→          }
   678→        : undefined,
   679→    tr:
   680→      r.kind === 'transfer' && r.tr && typeof r.tr.amount === 'number'
   681→        ? {
   682→            amount: r.tr.amount,
   683→            note: typeof r.tr.note === 'string' ? r.tr.note : '',
   684→            toId: typeof r.tr.toId === 'string' ? r.tr.toId : '',
   685→            toName: typeof r.tr.toName === 'string' ? r.tr.toName : '',
   686→            received: r.tr.received === true || undefined,
   687→            receivedAt: typeof r.tr.receivedAt === 'number' ? r.tr.receivedAt : undefined,
   688→            status: r.tr.status === 'returned' || r.tr.status === 'rejected' ? r.tr.status : undefined,
   689→            cid: typeof r.tr.cid === 'string' ? r.tr.cid : undefined,
   690→          }
   691→        : undefined,
   692→    noticeText: typeof r.noticeText === 'string' ? r.noticeText : undefined,
   693→    evt:
   694→      r.evt && typeof r.evt.type === 'string' && EVENT_TYPES.has(r.evt.type)
   695→        ? {
   696→            type: r.evt.type as GroupEventDetail['type'],
   697→            actorId: typeof r.evt.actorId === 'string' ? r.evt.actorId : undefined,
   698→            targetId: typeof r.evt.targetId === 'string' ? r.evt.targetId : undefined,
   699→            extra: typeof r.evt.extra === 'string' ? r.evt.extra : undefined,
   700→          }
   701→        : undefined,
   702→    notice:
   703→      r.notice && typeof r.notice.pre === 'string' && typeof r.notice.accent === 'string'
   704→        ? { icon: r.notice.icon === 'tr' || r.notice.icon === 'fam' ? r.notice.icon : 'rp', pre: r.notice.pre, accent: r.notice.accent }
   705→        : undefined,
   706→    img: r.img && typeof r.img.src === 'string' ? { src: r.img.src } : undefined,
   707→    loc:
   708→      r.loc && typeof r.loc.name === 'string' && typeof r.loc.address === 'string'
   709→        ? { name: r.loc.name, address: r.loc.address }
   710→        : undefined,
   711→    stk:
   712→      r.stk && typeof r.stk.url === 'string'
   713→        ? { url: r.stk.url, meaning: typeof r.stk.meaning === 'string' ? r.stk.meaning : '', sid: typeof r.stk.sid === 'string' ? r.stk.sid : undefined }
   714→        : undefined,
   715→    quote:
   716→      r.quote && typeof r.quote.name === 'string' && typeof r.quote.content === 'string'
   717→        ? {
   718→            name: r.quote.name,
   719→            content: r.quote.content,
   720→            // 引用源消息 ID（删除/撤回后把引用内容改写为「原消息已删除」用；旧数据无此字段照常渲染）
   721→            id: typeof r.quote.id === 'string' ? r.quote.id : undefined,
   722→          }
   723→        : undefined,
   724→    recalled: r.recalled === true,
   725→    fwd:
   726→      r.kind === 'forward' && r.fwd && typeof r.fwd.from === 'string'
   727→        ? {
   728→            from: r.fwd.from,
   729→            merged: r.fwd.merged === true || undefined,
   730→            title: typeof r.fwd.title === 'string' ? r.fwd.title : undefined,
   731→            records: Array.isArray(r.fwd.records)
   732→              ? r.fwd.records
   733→                  .filter((x): x is GroupFwdRecord => Boolean(x) && typeof (x as GroupFwdRecord).name === 'string' && typeof (x as GroupFwdRecord).text === 'string')
   734→                  .map((x) => ({
   735→                    name: x.name,
   736→                    role: x.role === 'me' ? ('me' as const) : ('peer' as const),
   737→                    text: x.text,
   738→                    quote: typeof x.quote === 'string' ? x.quote : undefined,
   739→                    time: typeof x.time === 'number' ? x.time : 0,
   740→                    avatar: typeof x.avatar === 'string' ? x.avatar : x.avatar === null ? null : undefined,
   741→                    kind: x.kind === 'sticker' || x.kind === 'image' ? x.kind : undefined,
   742→                    imgSrc: typeof x.imgSrc === 'string' ? x.imgSrc : undefined,
   743→                    stkMeaning: typeof x.stkMeaning === 'string' ? x.stkMeaning : undefined,
   744→                  }))
   745→              : undefined,
   746→          }
   747→        : undefined,
   748→  };
   749→}

   760→  const app = getGroup(groupId)?.app ?? 'wx';

   762→}

   764→/** 该群最后一条非通知消息（会话列表预览用） */
   765→export function groupPreview(groupId: string): { text: string; time: number } {
   766→  const msgs = loadGroupMsgs(groupId);
   767→  const last = [...msgs].reverse().find((m) => m.kind !== 'notice');
   768→  if (!last) return { text: '', time: 0 };
   769→  if (last.recalled) return { text: `${last.role === 'me' ? '你' : last.senderName || '有人'}撤回了一条消息`, time: last.time };
   770→  if (last.kind === 'image') return { text: '[图片]', time: last.time };
   771→  if (last.kind === 'sticker') return { text: `[表情]${last.stk?.meaning ? ` ${last.stk.meaning}` : ''}`, time: last.time };
   772→  if (last.kind === 'location') return { text: `[位置] ${last.loc?.name ?? ''}`.trim(), time: last.time };
   773→  if (last.kind === 'redpacket') return { text: '[红包]', time: last.time };
   774→  if (last.kind === 'transfer') return { text: '[转账]', time: last.time };
   775→  if (last.kind === 'forward') return { text: last.fwd?.merged ? '[聊天记录]' : last.content, time: last.time };
   776→  const prefix = last.role === 'me' ? '我' : last.senderName;
   777→  return { text: `${prefix}：${last.content}`, time: last.time };
   778→}
