   152→function GuitarPanel({
   153→  timbre,
   154→  heldChord,
   155→  heldChordRef,
   156→}: {
   157→  timbre: GuitarTimbre;
   158→  heldChord: GuitarChord | null;
   159→  heldChordRef: RefObject<GuitarChord | null>;
   160→}) {
   161→  const [vibing, setVibing] = useState<number[]>([]); // 正在抖动的弦
   162→  const vibeTimers = useRef<Map<number, ReturnType<typeof setTimeout>>>(new Map());
   163→  // 按住滑动扫弦：记录上一根被拨的弦，滑入新弦时再拨（每根弦每次按住只拨一次入列去重）
   164→  const lastStrum = useRef<{ pid: number; idx: number }[]>([]);
   165→
   166→  // 弦抖动动效（回放/外部音符事件点亮时只动不发声）
   167→  const flashString = useCallback((idx: number) => {
   168→    setVibing((v) => (v.includes(idx) ? v : [...v, idx]));
   169→    const old = vibeTimers.current.get(idx);
   170→    if (old) clearTimeout(old);
   171→    vibeTimers.current.set(
   172→      idx,
   173→      setTimeout(() => {
   174→        setVibing((v) => v.filter((i) => i !== idx));
   175→        vibeTimers.current.delete(idx);
   176→      }, 520),
   177→    );
   178→  }, []);
   179→
   180→  const twang = useCallback(
   181→    (idx: number) => {
   182→      const chord = heldChordRef.current;
   183→      const semi = chord ? chord.offsets[idx] : null;
   184→      if (chord && semi === null) return; // 按和弦时该弦被左手闷住，不出声
   185→      const freq = chordStringFreq(GUITAR_STRINGS[idx].freq, semi);
   186→      pluckString(freq, 1, timbre);
   187→      emitNote({
   188→        kind: 'guitar',
   189→        freq,
   190→        name: chord ? `${chord.name}·${GUITAR_STRINGS[idx].name}` : GUITAR_STRINGS[idx].name,
   191→      });
   192→      flashString(idx);
   193→    },
   194→    [flashString, heldChordRef, timbre],
   195→  );
   196→
   197→  // 音符事件（外部弹奏等）点亮就近开放弦
   198→  useEffect(() => {
   199→    return onNote((e) => {
   200→      if (e.kind !== 'guitar') return;
   201→      let best = 0;
   202→      let bestD = Infinity;
   203→      GUITAR_STRINGS.forEach((s, i) => {
   204→        const d = Math.abs(Math.log(s.freq / e.freq));
   205→        if (d < bestD) {
   206→          bestD = d;
   207→          best = i;
   208→        }
   209→      });
   210→      flashString(best);
   211→    });
   212→  }, [flashString]);
   213→
   214→  const idxFromEvent = useCallback((e: React.PointerEvent<HTMLDivElement>) => {
   215→    const rect = e.currentTarget.getBoundingClientRect();
   216→    const x = e.clientX - rect.left;
   217→    const w = rect.width / GUITAR_STRINGS.length;
   218→    return Math.max(0, Math.min(GUITAR_STRINGS.length - 1, Math.floor(x / w)));
   219→  }, []);
   220→
   221→  return (
   222→    <div className="relative min-h-0 flex-1 overflow-hidden rounded-[26px] shadow-2xl">
   223→      {/* 木纹指板 */}
   224→      <div
   225→        className="absolute inset-0"

   229→        }}
   230→      />
   231→      {/* 品丝 + 品记 */}
   232→      <div className="absolute inset-0">
   233→        {[12, 28, 44, 60, 76, 90].map((top) => (
   234→          <div
   235→            key={top}
   236→            className="absolute inset-x-0 h-[5px] rounded-full bg-gradient-to-b from-[#cfd4d8] via-[#9aa3ab] to-[#7c848c] shadow"
   237→            style={{ top: `${top}%` }}
   238→          />
   239→        ))}
   240→        {[20, 36, 52].map((top) => (
   241→          <div
   242→            key={top}
   243→            className="absolute left-1/2 h-3.5 w-3.5 -translate-x-1/2 rounded-full bg-[#efe6d4]/90 shadow-inner"
   244→            style={{ top: `${top}%` }}
   245→          />
   246→        ))}
   247→      </div>
   248→      {/* 六根弦（可拨/可扫） */}
   249→      <div
   250→        className="absolute inset-0 flex touch-none"
   251→        style={{ WebkitTapHighlightColor: 'transparent' }}
   252→        onPointerDown={(e) => {
   253→          e.currentTarget.setPointerCapture?.(e.pointerId);
   254→          const idx = idxFromEvent(e);
   255→          twang(idx);
   256→          lastStrum.current = [{ pid: e.pointerId, idx }];
   257→        }}
   258→        onPointerMove={(e) => {
   259→          if (e.buttons === 0 && e.pointerType === 'mouse') return;
   260→          if (!lastStrum.current.some((s) => s.pid === e.pointerId)) return;
   261→          const idx = idxFromEvent(e);
   262→          const mine = lastStrum.current.find((s) => s.pid === e.pointerId);
   263→          if (mine && mine.idx !== idx) {
   264→            twang(idx);
   265→            mine.idx = idx;
   266→          }
   267→        }}
   268→        onPointerUp={(e) => {
   269→          lastStrum.current = lastStrum.current.filter((s) => s.pid !== e.pointerId);
   270→        }}
   271→        onPointerCancel={(e) => {
   272→          lastStrum.current = lastStrum.current.filter((s) => s.pid !== e.pointerId);
   273→        }}
   274→      >
   275→        {GUITAR_STRINGS.map((s, i) => (
   276→          <div key={s.name + i} className="relative flex-1">
   277→            {/* 弦体：低音弦缠丝金色 / 高音弦银亮，拨动时抖动 */}

   288→            {/* 弦枕环 */}

   290→            {/* 弦名 */}

   300→          已按 {heldChord.name} 和弦

   306→        }}

   308→    </div>

   368→// ---------------- 钢琴舞台（深色舞台风：工具条 + 竖纹幕布 + 音符气泡 + 真实感键床） ----------------

   370→/** 键名 → 键中心横向位置（键床宽度百分比，音符气泡对齐用） */
   371→function keyCenterPct(name: string): number {
   372→  const wi = PIANO_WHITE.findIndex((k) => k.name === name);
   373→  if (wi >= 0) return ((wi + 0.5) / PIANO_WHITE.length) * 100;
   374→  const bk = PIANO_BLACK.find((b) => b.name === name);
   375→  if (bk) return ((bk.after + 1) / PIANO_WHITE.length) * 100;
   376→  return 50;
   377→}

   379→interface NoteBubble {
   380→  id: number;

   385→function PianoStage({
   386→  timbre,
   387→  onTimbre,
   388→  onClose,
   389→}: {
   390→  timbre: PianoTimbre;
   391→  onTimbre: (v: PianoTimbre) => void;
   392→  onClose: () => void;
   393→}) {
   394→  const [pressed, setPressed] = useState<string[]>([]);
   395→  const timers = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());
   396→  // 滑音：按住拖动时经过新键也发声
   397→  const lastKey = useRef<{ pid: number; key: string }[]>([]);
   398→  // 外部音符（小星星演示等）点亮的键
   399→  const [lit, setLit] = useState<string[]>([]);
   400→  const litTimers = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());
   401→  // 舞台上浮的音符气泡
   402→  const [bubbles, setBubbles] = useState<NoteBubble[]>([]);
   403→  const bubbleSeq = useRef(0);
   404→  const bubbleTimers = useRef<Map<number, ReturnType<typeof setTimeout>>>(new Map());
   405→
   406→  // 演示/弹奏期间读最新音色（effect 中同步，避免渲染期写 ref）
   407→  const timbreRef = useRef(timbre);
   408→  useEffect(() => {
   409→    timbreRef.current = timbre;
   410→  }, [timbre]);
   411→
   412→  const spawnBubble = useCallback((label: string) => {
   413→    const id = ++bubbleSeq.current;
   414→    setBubbles((b) => [...b.slice(-10), { id, label, x: keyCenterPct(label) }]);
   415→    const t = setTimeout(() => {
   416→      setBubbles((b) => b.filter((x) => x.id !== id));
   417→      bubbleTimers.current.delete(id);
   418→    }, 1080);
   419→    bubbleTimers.current.set(id, t);
   420→  }, []);
   421→
   422→  const hit = useCallback(
   423→    (name: string, freq: number) => {
   424→      pianoNote(freq, 1, timbreRef.current);
   425→      emitNote({ kind: 'piano', freq, name });
   426→      setPressed((p) => (p.includes(name) ? p : [...p, name]));
   427→      const old = timers.current.get(name);
   428→      if (old) clearTimeout(old);
   429→      timers.current.set(
   430→        name,
   431→        setTimeout(() => {
   432→          setPressed((p) => p.filter((n) => n !== name));
   433→          timers.current.delete(name);
   434→        }, 260),
   435→      );
   436→    },
   437→    [],
   438→  );
   439→
   440→  // 音符事件点亮对应琴键 + 上浮气泡（手弹经 emitNote 回路、演示曲同样点亮）
   441→  useEffect(() => {
   442→    return onNote((e) => {
   443→      if (e.kind !== 'piano') return;
   444→      setLit((l) => (l.includes(e.name) ? l : [...l, e.name]));
   445→      const old = litTimers.current.get(e.name);
   446→      if (old) clearTimeout(old);
   447→      litTimers.current.set(
   448→        e.name,
   449→        setTimeout(() => {
   450→          setLit((l) => l.filter((n) => n !== e.name));
   451→          litTimers.current.delete(e.name);
   452→        }, 300),
   453→      );
   454→      spawnBubble(e.name);
   455→    });
   456→  }, [spawnBubble]);
   457→
   458→  // —— 《小星星》演示（红点开关） ——
   459→  const [demoOn, setDemoOn] = useState(false);
   460→  const demoTimers = useRef<ReturnType<typeof setTimeout>[]>([]);
   461→
   462→  const stopDemo = useCallback(() => {
   463→    demoTimers.current.forEach(clearTimeout);
   464→    demoTimers.current = [];
   465→    setDemoOn(false);
   466→  }, []);
   467→
   468→  const startDemo = useCallback(() => {
   469→    stopDemo();
   470→    setDemoOn(true);
   471→    let t = 0;
   472→    const ts: ReturnType<typeof setTimeout>[] = [];
   473→    for (const n of TWINKLE) {
   474→      const freq = NOTE_FREQ.get(n.name);
   475→      if (freq == null) continue;
   476→      const delay = t;
   477→      ts.push(
   478→        setTimeout(() => {
   479→          pianoNote(freq, 1, timbreRef.current);
   480→          emitNote({ kind: 'piano', freq, name: n.name });
   481→        }, delay),
   482→      );
   483→      t += twinkleTempoMs * n.beats;
   484→    }
   485→    ts.push(
   486→      setTimeout(() => {
   487→        demoTimers.current = [];
   488→        setDemoOn(false);
   489→      }, t + 400),
   490→    );
   491→    demoTimers.current = ts;
   492→  }, [stopDemo]);
   493→
   494→  // —— 节拍器（100BPM 短促打拍，WebAudio 合成，无音频文件） ——
   495→  const [metroOn, setMetroOn] = useState(false);
   496→  const metroInt = useRef<ReturnType<typeof setInterval> | null>(null);
   497→  useEffect(() => {
   498→    if (!metroOn) return;
   499→    const tick = () => {
   500→      const ctx = ensureAudio();
   501→      if (!ctx) return;
   502→      const o = ctx.createOscillator();
   503→      const gn = ctx.createGain();
   504→      o.type = 'square';
   505→      o.frequency.value = 1500;
   506→      gn.gain.setValueAtTime(0.14, ctx.currentTime);
   507→      gn.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.045);
   508→      o.connect(gn);
   509→      gn.connect(ctx.destination);
   510→      o.start();
   511→      o.stop(ctx.currentTime + 0.05);
   512→    };
   513→    tick();
   514→    metroInt.current = setInterval(tick, 600); // ♩=100
   515→    return () => {
   516→      if (metroInt.current) clearInterval(metroInt.current);
   517→      metroInt.current = null;
   518→    };
   519→  }, [metroOn]);
   520→
   521→  // 卸载时清掉全部定时器
   522→  useEffect(
   523→    () => () => {
   524→      litTimers.current.forEach((t) => clearTimeout(t));
   525→      litTimers.current.clear();
   526→      timers.current.forEach((t) => clearTimeout(t));
   527→      timers.current.clear();
   528→      bubbleTimers.current.forEach((t) => clearTimeout(t));
   529→      bubbleTimers.current.clear();
   530→      demoTimers.current.forEach(clearTimeout);
   531→      demoTimers.current = [];
   532→    },
   533→    [],
   534→  );
   535→
   536→  /** 白键/黑键共用的按压手势（点按 + 按住滑键滑音 + 多指和弦） */
   537→  const pressHandlers = useCallback(
   538→    (name: string, freq: number) => ({
   539→      onPointerDown: (e: React.PointerEvent<HTMLDivElement>) => {
   540→        e.currentTarget.setPointerCapture?.(e.pointerId);
   541→        hit(name, freq);
   542→        lastKey.current = [
   543→          ...lastKey.current.filter((s) => s.pid !== e.pointerId),
   544→          { pid: e.pointerId, key: name },
   545→        ];
   546→      },
   547→      onPointerMove: (e: React.PointerEvent<HTMLDivElement>) => {
   548→        if (e.buttons === 0 && e.pointerType === 'mouse') return;
   549→        const mine = lastKey.current.find((s) => s.pid === e.pointerId);
   550→        if (mine && mine.key !== name) {
   551→          hit(name, freq);
   552→          mine.key = name;
   553→        }
   554→      },
   555→      onPointerUp: (e: React.PointerEvent<HTMLDivElement>) => {
   556→        lastKey.current = lastKey.current.filter((s) => s.pid !== e.pointerId);
   557→      },
   558→      onPointerCancel: (e: React.PointerEvent<HTMLDivElement>) => {
   559→        lastKey.current = lastKey.current.filter((s) => s.pid !== e.pointerId);
   560→      },
   561→    }),
   562→    [hit],
   563→  );
   564→
   565→  const unit = 100 / PIANO_WHITE.length; // 单个白键宽度百分比
   566→
   567→  return (
   568→    <div className="relative flex min-h-0 flex-1 flex-col overflow-hidden rounded-[22px] bg-[#0a0a0c] shadow-2xl ring-1 ring-black/60">
   569→      {/* 顶部工具条：红点演示｜蓝色琴标·标题｜FX 音色｜节拍器｜结束 */}
   570→      <div className="flex h-11 shrink-0 items-center gap-2.5 border-b border-white/10 bg-gradient-to-b from-[#232327] to-[#131316] px-3">
   571→        <button
   572→          type="button"
   573→          aria-label={demoOn ? '停止小星星演示' : '播放小星星演示'}
   574→          onClick={() => (demoOn ? stopDemo() : startDemo())}
   575→          className="flex h-7 w-7 shrink-0 items-center justify-center transition active:scale-90"
   576→          style={{ WebkitTapHighlightColor: 'transparent' }}
   577→        >
   578→          <span
   579→            className={`block h-4 w-4 rounded-full bg-[#e14b41] shadow-[0_0_10px_rgba(225,75,65,0.55),inset_0_-2px_3px_rgba(0,0,0,0.35)] ${
   580→              demoOn ? 'animate-pulse ring-2 ring-white/70' : ''
   581→            }`}
   582→          />
   583→        </button>
   584→        <div className="flex min-w-0 items-center gap-1.5">
   585→          <PianoIcon className="h-[18px] w-[18px] shrink-0 text-[#4f8ef7]" />
   586→          <span className="truncate text-[12px] font-semibold text-[#9a9aa2]">弹钢琴 · C3–C6</span>
   587→        </div>
   588→        <div className="ml-auto flex shrink-0 items-center gap-2">
   589→          <TimbrePicker
   590→            variant="chip"
   591→            options={PIANO_TIMBRES}
   592→            value={timbre}
   593→            onChange={(v) => onTimbre(v as PianoTimbre)}
   594→          />
   595→          <button
   596→            type="button"
   597→            aria-label={metroOn ? '关闭节拍器' : '开启节拍器'}
   598→            aria-pressed={metroOn}
   599→            onClick={() => setMetroOn((m) => !m)}
   600→            className={`flex h-8 w-8 items-center justify-center rounded-full border-[1.5px] transition active:scale-90 ${
   601→              metroOn
   602→                ? 'border-[#f4b3c2] bg-[#f4b3c2]/20 text-[#f4b3c2] shadow-[0_0_10px_rgba(244,179,194,0.4)]'
   603→                : 'border-white/75 bg-white/5 text-white'
   604→            }`}
   605→            style={{ WebkitTapHighlightColor: 'transparent' }}
   606→          >
   607→            <Gauge className="h-4 w-4" />
   608→          </button>
   609→          <button
   610→            type="button"
   611→            onClick={onClose}
   612→            aria-label="结束演奏"
   613→            className="flex h-8 w-8 items-center justify-center rounded-full border-[1.5px] border-white/75 bg-white/5 text-white transition active:scale-90"
   614→            style={{ WebkitTapHighlightColor: 'transparent' }}
   615→          >
   616→            <X className="h-4 w-4" />
   617→          </button>
   618→        </div>
   619→      </div>
   620→      {/* 键床（v6.7：删中部幕布舞台，工具条与琴键直接相贴）：象牙白键 + 亮面黑键（2+3 真实排布）+ 键上音符气泡 */}
   621→      <div className="relative min-h-0 flex-1 border-t border-white/10 bg-[#050506]">
   622→        {demoOn && (
   623→          <div className="pointer-events-none absolute left-1/2 top-2 z-20 -translate-x-1/2 whitespace-nowrap rounded-full bg-white/10 px-3 py-1 text-[10.5px] font-semibold text-white/85 backdrop-blur-sm">
   624→            ♪ 小星星 演示中 · 点红点停止
   625→          </div>
   626→        )}
   627→        {bubbles.map((b) => (
   628→          <span
   629→            key={b.id}
   630→            className="note-bubble pointer-events-none absolute top-[30%] z-10 rounded-full border border-white/15 bg-white/10 px-2 py-0.5 text-[10px] font-bold text-white/90"
   631→            style={{ left: `${b.x}%` }}
   632→          >
   633→            {b.label}
   634→          </span>
   635→        ))}
   636→        <div className="relative mx-[3px] h-full">
   637→          <div className="absolute inset-0 flex">
   638→            {PIANO_WHITE.map((k) => {
   639→              const on = pressed.includes(k.name) || lit.includes(k.name);
   640→              return (
   641→                <div
   642→                  key={k.name}
   643→                  className="relative flex-1 touch-none border-r border-[#c9c6ba]/80 last:border-r-0"
   644→                  style={{
   645→                    WebkitTapHighlightColor: 'transparent',
   646→                    borderRadius: '0 0 4px 4px',
   647→                    background: on
   648→                      ? 'linear-gradient(180deg,#ffe0ea 0%,#f9cfdd 100%)'
   649→                      : 'linear-gradient(180deg,#ffffff 0%,#f4f3ee 62%,#e7e5dc 100%)',
   650→                    boxShadow: on
   651→                      ? 'inset 0 2px 4px rgba(180,90,110,0.28), inset 0 -3px 4px -2px rgba(120,60,80,0.3)'
   652→                      : 'inset -1px 0 1.5px -1px rgba(60,55,40,0.35), inset 1px 0 1.5px -1px rgba(255,255,255,0.95), inset 0 -5px 6px -4px rgba(70,64,48,0.5)',
   653→                    transform: on ? 'scaleY(0.975)' : undefined,
   654→                    transformOrigin: 'top',

   656→                  {...pressHandlers(k.name, k.freq)}

   687→                  transformOrigin: 'top',
   688→                }}
   689→                {...pressHandlers(b.name, b.freq)}
   690→              />
   691→            );
   692→          })}
   693→        </div>
   694→      </div>
   695→      <style
   696→        dangerouslySetInnerHTML={{

   699→        }}
   700→      />
   701→    </div>
   702→  );
