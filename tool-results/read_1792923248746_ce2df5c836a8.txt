   123→/** 家具在任意水平角下的占地 AABB（fp 未旋转脚印 × 旋转） */
   124→export function footprintOf(entry: CatalogEntry, ry: number): [number, number] {
   125→  const [w, d] = entry.fp;
   126→  const c = Math.abs(Math.cos(ry));
   127→  const s = Math.abs(Math.sin(ry));
   128→  return [w * c + d * s, w * s + d * c];
   129→}

   207→/** 找离 (x,z) 最近的挂墙槽位并投影到墙上（返回槽位 + 贴墙坐标） */
   208→export function snapToWallSlot(
   209→  x: number,
   210→  z: number,
   211→  room?: RoomId,
   212→): { slot: WallSlot; x: number; z: number } {
   213→  let best: { slot: WallSlot; x: number; z: number } | null = null;
   214→  let bd = Infinity;
   215→  for (const s of WALL_SLOTS) {
   216→    if (room && s.room !== room) continue;
   217→    const along = s.axis === 'z' ? x : z;
   218→    const a = Math.max(s.from, Math.min(s.to, along));
   219→    const px = s.axis === 'z' ? a : s.at;
   220→    const pz = s.axis === 'z' ? s.at : a;
   221→    const d = Math.hypot(px - x, pz - z);
   222→    if (d < bd) {
   223→      bd = d;
   224→      best = { slot: s, x: px, z: pz };
   225→    }
   226→  }
   227→  const s = best?.slot ?? WALL_SLOTS[0];
   228→  const a = Math.max(s.from, Math.min(s.to, s.axis === 'z' ? x : z));
   229→  return { slot: s, x: s.axis === 'z' ? a : s.at, z: s.axis === 'z' ? s.at : a };
   230→}

   299→/** 同墙邻居（用于滑行避让：同墙的窗与门互不重叠；用当前坐标，缺省用默认位） */
   300→function wallNeighbors(
   301→  selfId: string,
   302→  axis: 'x' | 'z',
   303→  at: number,
   304→  winPos?: number[],
   305→  doorPos?: { bed: number; kt: number; bath: number; bal?: number },
   306→): { along: number; half: number }[] {
   307→  const out: { along: number; half: number }[] = [];
   308→  for (const [i, w] of WINS.entries()) {
   309→    if (`win-${i}` === selfId || w.axis !== axis || Math.abs(w.at - at) > 0.01) continue;
   310→    out.push({ along: winPos?.[i] ?? w.def, half: WIN_HALF });
   311→  }
   312→  for (const d of DOORS) {
   313→    if (`door-${d.id}` === selfId || d.axis !== axis || Math.abs(d.at - at) > 0.01) continue;
   314→    out.push({ along: doorPos?.[d.id] ?? d.def, half: DOOR_HALF });
   315→  }
   316→  return out;
   317→}

   319→/** 把沿墙坐标钳回滑行范围并避让同墙邻居（门/窗滑行共用） */
   320→export function clampWallAlong(selfId: string, def: { axis: 'x' | 'z'; at: number; min: number; max: number }, along: number, neighbors: { along: number; half: number }[]): number {
   321→  let a = Math.max(def.min, Math.min(def.max, along));
   322→  for (const o of neighbors) {
   323→    const need = o.half + (selfId.startsWith('door-') ? DOOR_HALF : WIN_HALF) + 0.12;
   324→    if (Math.abs(a - o.along) < need) {
   325→      a = o.along > a ? o.along - need : o.along + need;
   326→    }
   327→  }
   328→  return Math.max(def.min, Math.min(def.max, a));
   329→}

   331→/** 编辑模式拖拽/微调一扇窗：返回沿墙钳制后的完整位置 */
   332→export function resolveWinPos(i: number, along: number, winPos?: number[], doorPos?: { bed: number; kt: number; bath: number; bal?: number }): { x: number; z: number; along: number } {
   333→  const w = WINS[i];
   334→  const cur = winPos?.[i] ?? w.def;
   335→  const others = wallNeighbors(`win-${i}`, w.axis, w.at, winPos, doorPos);
   336→  const a = clampWallAlong(`win-${i}`, w, Number.isFinite(along) ? along : cur, others);
   337→  const x = w.axis === 'z' ? a : w.at + w.side * 0.08;
   338→  const z = w.axis === 'z' ? w.at + w.side * 0.08 : a;
   339→  return { x, z, along: a };
   340→}

   342→/** 编辑模式拖拽/微调一扇门：返回沿墙钳制后的完整位置（门组在墙中心线上） */
   343→export function resolveDoorPos(id: 'bed' | 'kt' | 'bath' | 'bal', along: number, winPos?: number[], doorPos?: { bed: number; kt: number; bath: number; bal?: number }): { x: number; z: number; along: number } {
   344→  const d = DOORS.find((x) => x.id === id)!;
   345→  const cur = doorPos?.[id] ?? d.def;
   346→  const others = wallNeighbors(`door-${id}`, d.axis, d.at, winPos, doorPos);
   347→  const a = clampWallAlong(`door-${id}`, d, Number.isFinite(along) ? along : cur, others);
   348→  const x = d.axis === 'z' ? a : d.at;
   349→  const z = d.axis === 'z' ? d.at : a;
   350→  return { x, z, along: a };
   351→}
