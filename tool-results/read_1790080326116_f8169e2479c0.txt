    60→/** 聊天页背景层样式：color → 纯色；image → 图片（cover 居中）；default → undefined（不渲染层） */
    61→export function chatBgLayerStyle(bg: ChatSettingsBg, imageUrl: string | null): CSSProperties | undefined {
    62→  if (bg.mode === 'color' && bg.color) {
    63→    return { backgroundColor: bg.color };
    64→  }
    65→  if (bg.mode === 'image' && imageUrl) {
    66→    return {
    67→      backgroundImage: `url(${imageUrl})`,
    68→      backgroundSize: 'cover',
    69→      backgroundPosition: 'center',
    70→      backgroundRepeat: 'no-repeat',
    71→    };
    72→  }
    73→  return undefined;
    74→}

   702→function fmtSearchTime(ts: number): string {
   703→  const d = new Date(ts);
   704→  const now = new Date();
   705→  const hm = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
   706→  if (d.getFullYear() !== now.getFullYear()) return `${d.getFullYear()}年${d.getMonth() + 1}月${d.getDate()}日 ${hm}`;
   707→  if (d.toDateString() === now.toDateString()) return hm;
   708→  return `${d.getMonth() + 1}月${d.getDate()}日 ${hm}`;
   709→}
