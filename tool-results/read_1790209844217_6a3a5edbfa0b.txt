   185→/** 模拟网格 auto-flow（sparse 不回填）：依次摆放 tiles，记录每个 tile 的 flow 起点
   186→ *  与摆完后的游标。用于空白位放置时把指针位置（行,列）换算成插入下标与需要垫的
   187→ *  空占位格数——「指针压哪格，App 就落到哪格」。 */
   188→function flowPositions(tiles: Tile[]): {
   189→  /** 每个 tile 摆放时的起点格（行,列） */
   190→  starts: { r: number; c: number }[];
   191→  /** 摆完第 i 个 tile 后的游标（after[i] = 第 i+1 个 tile 的理论起点） */
   192→  after: { r: number; c: number }[];
   193→  /** 全部摆完后的游标（= 第一个空闲格） */
   194→  end: { r: number; c: number };
   195→} {
   196→  const occupied = new Set<string>();
   197→  let cr = 0;
   198→  let cc = 0;
   199→  const starts: { r: number; c: number }[] = [];
   200→  const after: { r: number; c: number }[] = [];
   201→  const spanOf = (t: Tile): { c: number; r: number } => (t.kind === 'widget' ? WIDGET_SPAN_SIZE[t.widget] : { c: 1, r: 1 });
   202→  const fits = (r: number, c: number, cs: number, rs: number): boolean => {
   203→    if (c + cs > 4) return false;
   204→    for (let i = 0; i < rs; i++) for (let j = 0; j < cs; j++) if (occupied.has(`${r + i}:${c + j}`)) return false;
   205→    return true;
   206→  };
   207→  const advance = (n: number) => {
   208→    cc += n;
   209→    while (cc >= 4) {
   210→      cc -= 4;
   211→      cr++;
   212→    }
   213→  };
   214→  for (const t of tiles) {
   215→    if (t.kind === 'empty') {
   216→      starts.push({ r: cr, c: cc });
   217→      occupied.add(`${cr}:${cc}`);
   218→      advance(1);
   219→      after.push({ r: cr, c: cc });
   220→      continue;
   221→    }
   222→    const s = spanOf(t);
   223→    while (!fits(cr, cc, s.c, s.r)) {
   224→      advance(1);
   225→    }
   226→    starts.push({ r: cr, c: cc });
   227→    for (let i = 0; i < s.r; i++) for (let j = 0; j < s.c; j++) occupied.add(`${cr + i}:${cc + j}`);
   228→    advance(s.c);
   229→    after.push({ r: cr, c: cc });
   230→  }
   231→  return { starts, after, end: { r: cr, c: cc } };
   232→}

   285→ *  世界书自 v9 起纳入页尾——否则 v9 重置路径提前 return 不会走 App 补位逻辑，世界书会从主屏消失） */
   286→const PAGE3_APP_IDS: AppId[] = ['music', 'wechat', 'appstore', 'qq', 'memory', 'worldbook'];
   287→/** 其余小组件（默认收起不显示：写入 hidden，需要时从「+」画廊/主题小组件页找回；
   288→ *  网易云小组件自 v6 起默认上屏第 3 页，不再收起；
   289→ *  v9 起 widgets-7 的四个新小组件 iCity/日历/表盘时钟/拍立得也默认收起——用户要求新添加的小组件不显示） */
   290→const DEFAULT_HIDDEN_WIDGETS: WidgetKind[] = ['diary', 'listen', 'dialog', 'vinyl', 'icity', 'calendar', 'tickclock', 'polaroid'];

   292→function defaultLayout(): HomeLayout {
   293→  return {
   294→    v: LAYOUT_VERSION,
   295→    pages: [
   296→      [
   297→        { kind: 'widget', widget: 'clock' },
   298→        { kind: 'widget', widget: 'weather' },
   299→        ...PAGE1_APP_IDS.map((id) => ({ kind: 'app' as const, id })),
   300→      ],
   301→      [
   302→        { kind: 'widget', widget: 'profile' },
   303→        { kind: 'widget', widget: 'bubble' },
   304→        ...PAGE2_APP_IDS.map((id) => ({ kind: 'app' as const, id })),
   305→      ],
   306→      [{ kind: 'widget', widget: 'netease' }, ...PAGE3_APP_IDS.map((id) => ({ kind: 'app' as const, id }))],
   307→    ],
   308→    dock: DOCK_APPS.map((a) => a.id),
   309→    hidden: DEFAULT_HIDDEN_WIDGETS.map(widgetKey),
   310→  };
   311→}

   313→/** 某页容量：页内有小组件则 14 格，否则 20 格（插入小组件时按插入后计算） */
   314→function pageCap(page: Tile[], insertingWidget = false): number {
   315→  return page.some((t) => t.kind === 'widget') || insertingWidget ? PAGE_WIDGET_CAP : PAGE_CAP;
   316→}

   318→/**
   319→ * 校验/修复持久化布局：
   320→ * - 兼容旧单页 {grid}：迁移为第 1 页；
   321→ * - v6 起布局重排：旧版本直接重置为新默认（保留 hidden 删除记录）；
   322→ * - 小组件全布局唯一、App 去重、Dock ≤4、缺失 App 按容量补到末尾有空间的页（hidden 的除外）。
   323→ */
   324→function sanitizeLayout(raw: unknown): HomeLayout {
   325→  const allIds = APPS.map((a) => a.id);
   326→  const def = defaultLayout();
   327→  if (!raw || typeof raw !== 'object') return def;
   328→  const r = raw as { v?: unknown; pages?: unknown; grid?: unknown; dock?: unknown; hidden?: unknown };
   329→
   330→  const hidden = new Set<string>();
   331→  if (Array.isArray(r.hidden)) {
   332→    for (const h of r.hidden) {
   333→      if (typeof h !== 'string') continue;
   334→      // 旧格式 'widget'（全布局唯一时代）→ 天气小组件；小组件键需为已知种类
   335→      if (h === 'widget' || h === widgetKey('weather')) hidden.add(widgetKey('weather'));
   336→      else if (isWidgetKey(h) && (WIDGET_KINDS as string[]).includes(h.slice('widget:'.length))) hidden.add(h);
   337→      else if (!isWidgetKey(h) && (allIds as string[]).includes(h)) hidden.add(h);
   338→    }
   339→  }
   340→
   341→  // 布局版本升级（v6/v7 重排等）：旧版本布局直接重置为新默认（App 全部按用户指定的新排布落位）。
   342→  // hidden 合并「默认收起的第三/四页其余小组件」与存量删除记录，并从默认页里滤掉 hidden 项
   343→  // （否则被删过的 App/小组件会随默认页短暂重现，直到下一次 sanitize 才被剔除）；
   344→  // 网易云旧收起记录放行——v5 及更早它是「默认收起」而非「用户删除」，v6 要上屏第 3 页
   345→  if ((typeof r.v === 'number' ? r.v : 1) < LAYOUT_VERSION) {
   346→    const merged = new Set([...def.hidden, ...hidden]);
   347→    merged.delete(widgetKey('netease'));
   348→    const pages = def.pages.map((p) =>
   349→      p.filter((t) => (t.kind === 'widget' ? !merged.has(widgetKey(t.widget)) : t.kind === 'app' ? !merged.has(t.id) : true))
   350→    );
   351→    return { v: LAYOUT_VERSION, pages, dock: def.dock, hidden: Array.from(merged) };
   352→  }
   353→
   354→  const dock: AppId[] = [];
   355→  if (Array.isArray(r.dock)) {
   356→    for (const id of r.dock) {
   357→      if (typeof id === 'string' && (allIds as string[]).includes(id) && !dock.includes(id as AppId) && !hidden.has(id)) {
   358→        if (dock.length < DOCK_CAPACITY) dock.push(id as AppId);
   359→      }
   360→    }
   361→  }
   362→
   363→  // 解析页列表：新格式 {pages}；旧格式 {grid} → 第 1 页（语音备忘录挪出）+ 第 2 页（语音备忘录）
   364→  let rawPages: unknown[][] = [];
   365→  if (Array.isArray(r.pages)) {
   366→    rawPages = (r.pages as unknown[]).map((p) => (Array.isArray(p) ? p : []));
   367→  } else if (Array.isArray(r.grid)) {
   368→    const page0 = (r.grid as unknown[]).filter((t) => {
   369→      const o = t as { kind?: unknown; id?: unknown } | null;
   370→      return !(o && typeof o === 'object' && o.kind === 'app' && o.id === 'recorder');
   371→    });
   372→    rawPages = [page0];
   373→    if (!hidden.has('recorder') && !dock.includes('recorder')) {
   374→      rawPages.push([{ kind: 'app', id: 'recorder' as AppId }]);
   375→    }
   376→  }
   377→
   378→  const pages: Tile[][] = [];
   379→  const widgetSeen: Record<WidgetKind, boolean> = {
   380→    weather: hidden.has(widgetKey('weather')),
   381→    clock: hidden.has(widgetKey('clock')),
   382→    profile: hidden.has(widgetKey('profile')),
   383→    bubble: hidden.has(widgetKey('bubble')),
   384→    diary: hidden.has(widgetKey('diary')),
   385→    listen: hidden.has(widgetKey('listen')),
   386→    netease: hidden.has(widgetKey('netease')),
   387→    dialog: hidden.has(widgetKey('dialog')),
   388→    vinyl: hidden.has(widgetKey('vinyl')),
   389→    calendar: hidden.has(widgetKey('calendar')),
   390→    icity: hidden.has(widgetKey('icity')),
   391→    tickclock: hidden.has(widgetKey('tickclock')),
   392→    polaroid: hidden.has(widgetKey('polaroid')),
   393→  };
   394→  const placed = new Set<AppId>(dock);
   395→  for (const list of rawPages) {
   396→    const page: Tile[] = [];
   397→    for (const t of list) {
   398→      if (!t || typeof t !== 'object') continue;
   399→      const o = t as { kind?: unknown; id?: unknown; widget?: unknown };
   400→      if (o.kind === 'empty') {
   401→        // 空占位格（拖到行尾空位/空列时补齐列位）：原样保留，否则重排会让 App 位置漂移
   402→        page.push({ kind: 'empty' });
   403→        continue;
   404→      }
   405→      if (o.kind === 'widget') {
   406→        // 已知种类（未知/无 widget 字段 → 天气，兼容最老格式）；每种全布局唯一，重复丢弃
   407→        const w = typeof o.widget === 'string' && (WIDGET_KINDS as string[]).includes(o.widget) ? (o.widget as WidgetKind) : 'weather';
   408→        if (widgetSeen[w]) continue;
   409→        widgetSeen[w] = true;
   410→        page.push({ kind: 'widget', widget: w });
   411→      } else if (o.kind === 'app' && typeof o.id === 'string' && (allIds as string[]).includes(o.id)) {
   412→        const id = o.id as AppId;
   413→        if (hidden.has(id) || placed.has(id)) continue;
   414→        placed.add(id);
   415→        page.push({ kind: 'app', id });
   416→      }
   417→    }
   418→    pages.push(page);
   419→  }
   420→  if (pages.length === 0) pages.push([]);
   421→  // 缺失的小组件按默认位置补回（hidden 的不补——默认收起的第 3/4 页小组件与被删除的
   422→  // 小组件都在 hidden 里，不会自动复活；用户主动加回后此处不再介入）：时钟→第 1 页头、
   423→  // 天气→时钟旁、信息卡片→第 2 页头、气泡→信息卡片后、日记/一起听/网易云→第 3 页、
   424→  // 对话气泡/黑胶→第 4 页（存量布局不足时新建页）
   425→  if (!widgetSeen.clock) pages[0].unshift({ kind: 'widget', widget: 'clock' });
   426→  if (!widgetSeen.weather) {
   427→    const at = pages[0].length > 0 && pages[0][0].kind === 'widget' && pages[0][0].widget === 'clock' ? 1 : 0;
   428→    pages[0].splice(at, 0, { kind: 'widget', widget: 'weather' });
   429→  }
   430→  if (!widgetSeen.profile) {
   431→    while (pages.length < 2) pages.push([]);
   432→    pages[1].unshift({ kind: 'widget', widget: 'profile' });
   433→  }
   434→  if (!widgetSeen.bubble) {
   435→    while (pages.length < 2) pages.push([]);
   436→    const p2 = pages[1];
   437→    const pi = p2.findIndex((t) => t.kind === 'widget' && t.widget === 'profile');
   438→    p2.splice(pi >= 0 ? pi + 1 : p2.length, 0, { kind: 'widget', widget: 'bubble' });
   439→  }
   440→  if (!widgetSeen.diary) {
   441→    while (pages.length < 3) pages.push([]);
   442→    pages[2].unshift({ kind: 'widget', widget: 'diary' });
   443→  }
   444→  if (!widgetSeen.listen) {
   445→    while (pages.length < 3) pages.push([]);
   446→    const p3 = pages[2];
   447→    const di = p3.findIndex((t) => t.kind === 'widget' && t.widget === 'diary');
   448→    p3.splice(di >= 0 ? di + 1 : p3.length, 0, { kind: 'widget', widget: 'listen' });
   449→  }
   450→  if (!widgetSeen.netease) {
   451→    while (pages.length < 3) pages.push([]);
   452→    const p3 = pages[2];
   453→    const li = p3.findIndex((t) => t.kind === 'widget' && t.widget === 'listen');
   454→    p3.splice(li >= 0 ? li + 1 : p3.length, 0, { kind: 'widget', widget: 'netease' });
   455→  }
   456→  // 第四页（用户要求）：对话气泡 → 第 4 页头（不足四页新建）、黑胶 → 对话气泡后
   457→  if (!widgetSeen.dialog) {
   458→    while (pages.length < 4) pages.push([]);
   459→    pages[3].unshift({ kind: 'widget', widget: 'dialog' });
   460→  }
   461→  if (!widgetSeen.vinyl) {
   462→    while (pages.length < 4) pages.push([]);
   463→    const p4 = pages[3];
   464→    const di = p4.findIndex((t) => t.kind === 'widget' && t.widget === 'dialog');
   465→    p4.splice(di >= 0 ? di + 1 : p4.length, 0, { kind: 'widget', widget: 'vinyl' });
   466→  }
   467→  // widgets-7 四个新小组件缺失时的默认落位：iCity→第 2 页气泡后、日历→第 3 页头、
   468→  // 表盘时钟→第 4 页头、拍立得→表盘时钟后（hidden 里的不补——v9 起默认收起，存量布局已在
   469→  // hidden 里，不会自动复活；用户从画廊主动加回后此处不再介入）
   470→  if (!widgetSeen.icity) {
   471→    while (pages.length < 2) pages.push([]);
   472→    const p2 = pages[1];
   473→    const bi = p2.findIndex((t) => t.kind === 'widget' && t.widget === 'bubble');
   474→    p2.splice(bi >= 0 ? bi + 1 : p2.length, 0, { kind: 'widget', widget: 'icity' });
   475→  }
   476→  if (!widgetSeen.calendar) {
   477→    while (pages.length < 3) pages.push([]);
   478→    pages[2].unshift({ kind: 'widget', widget: 'calendar' });
   479→  }
   480→  if (!widgetSeen.tickclock) {
   481→    while (pages.length < 4) pages.push([]);
   482→    pages[3].unshift({ kind: 'widget', widget: 'tickclock' });
   483→  }
   484→  if (!widgetSeen.polaroid) {
   485→    while (pages.length < 4) pages.push([]);
   486→    const p4 = pages[3];
   487→    const ti = p4.findIndex((t) => t.kind === 'widget' && t.widget === 'tickclock');
   488→    p4.splice(ti >= 0 ? ti + 1 : p4.length, 0, { kind: 'widget', widget: 'polaroid' });
   489→  }
   490→
   491→  // 缺失的 App：从末尾往前找还有容量的页补入，都满了就开新页
   492→  for (const id of allIds) {
   493→    if (hidden.has(id) || placed.has(id)) continue;
   494→    let target = -1;
   495→    for (let i = pages.length - 1; i >= 0; i--) {
   496→      if (pages[i].length < pageCap(pages[i])) {
   497→        target = i;
   498→        break;
   499→      }
   500→    }
   501→    if (target === -1) {
   502→      pages.push([]);
   503→      target = pages.length - 1;
   504→    }
   505→    pages[target].push({ kind: 'app', id });
   506→    placed.add(id);
   507→  }
   508→  return { v: LAYOUT_VERSION, pages, dock, hidden: Array.from(hidden) };
   509→}

   523→/**
   524→ * 把拖拽项移动到目标区/目标页/目标索引（跨区、跨页可搬；小组件不可进 Dock；
   525→ * Dock 超员时挤出最后一个回网格；目标页超容量则整体放弃）。
   526→ * 命中索引来自拖拽开始时捕获的静态几何，先移除被拖项再插入目标索引即可。
   527→ */
   528→function reorder(layout: HomeLayout, id: string, to: Hit): HomeLayout {
   529→  const pages = layout.pages.map((p) => p.slice());
   530→  const dock = layout.dock.slice();
   531→
   532→  const isWidget = isWidgetKey(id);
   533→  let fromZone: Zone | null = null;
   534→  let fromPage = -1;
   535→  let fromIndex = -1;
   536→  for (let p = 0; p < pages.length; p++) {
   537→    const i = pages[p].findIndex((t) => tileKey(t) === id);
   538→    if (i >= 0) {
   539→      fromZone = 'grid';
   540→      fromPage = p;
   541→      fromIndex = i;
   542→      break;
   543→    }
   544→  }
   545→  if (fromZone === null) {
   546→    const di = dock.indexOf(id as AppId);
   547→    if (di >= 0) {
   548→      fromZone = 'dock';
   549→      fromIndex = di;
   550→    } else {
   551→      return layout;
   552→    }
   553→  }
   554→
   555→  // 取出被拖项
   556→  if (fromZone === 'grid') pages[fromPage].splice(fromIndex, 1);
   557→  else dock.splice(fromIndex, 1);
   558→
   559→  if (to.zone === 'dock') {
   560→    if (isWidget) {
   561→      // 小组件不可进 Dock：放回原位
   562→      pages[fromPage].splice(Math.max(0, Math.min(pages[fromPage].length, fromIndex)), 0, {
   563→        kind: 'widget',
   564→        widget: widgetOfKey(id),
   565→      });
   566→      return { ...layout, pages, dock };
   567→    }
   568→    dock.splice(Math.max(0, Math.min(dock.length, to.index)), 0, id as AppId);
   569→    if (dock.length > DOCK_CAPACITY) {
   570→      const moved = dock.pop() as AppId;
   571→      const homePage = fromZone === 'grid' ? fromPage : pages.length - 1;
   572→      const insertAt = fromZone === 'grid' ? Math.max(0, Math.min(pages[homePage].length, fromIndex)) : pages[homePage].length;
   573→      pages[homePage].splice(insertAt, 0, { kind: 'app', id: moved });
   574→    }
   575→    return { ...layout, pages, dock };
   576→  }
   577→
   578→  // 落入网格某页：超容量整体放弃（放回原位）；同页内换位不改变页内数量，始终放行
   579→  const target = pages[Math.min(to.page, pages.length - 1)];
   580→  const samePageMove = fromZone === 'grid' && fromPage === to.page;
   581→  const padEmpty = to.zone === 'grid' ? Math.max(0, Math.min(8, to.padEmpty ?? 0)) : 0;
   582→  if (!samePageMove && target.length + padEmpty >= pageCap(target, isWidget)) {
   583→    if (fromZone === 'grid') {
   584→      pages[fromPage].splice(
   585→        Math.max(0, Math.min(pages[fromPage].length, fromIndex)),
   586→        0,
   587→        isWidget ? { kind: 'widget', widget: widgetOfKey(id) } : { kind: 'app', id: id as AppId }
   588→      );
   589→    } else {
   590→      dock.splice(fromIndex, 0, id as AppId);
   591→    }
   592→    return { ...layout, pages, dock };
   593→  }
   594→  const idx = Math.max(0, Math.min(target.length, to.index));
   595→  // 空占位格：把新项从自然落点推到指针所指的（行,列）（拖到行尾空位/任意空白格的精确定位）
   596→  const empties: Tile[] = Array.from({ length: padEmpty }, () => ({ kind: 'empty' as const }));
   597→  const tile: Tile = isWidget ? { kind: 'widget', widget: widgetOfKey(id) } : { kind: 'app', id: id as AppId };
   598→  target.splice(idx, 0, ...empties, tile);
   599→  return { ...layout, pages, dock };
   600→}

   602→function sameHit(a: Hit | null, b: Hit | null): boolean {
   603→  if (!a || !b) return false;
   604→  if (a.zone !== b.zone) return false;
   605→  if (a.zone === 'dock' && b.zone === 'dock') return a.index === b.index;
   606→  if (a.zone === 'grid' && b.zone === 'grid')
   607→    // padEmpty 也要比：拖到自己紧后方的空白位时 index 与原位相同但需要垫空占位格，
   608→    // 若只比 page/index 会被误判为「拖回原位」而放弃（预览/松手都弹回，用户实测）
   609→    return a.page === b.page && a.index === b.index && (a.padEmpty ?? 0) === (b.padEmpty ?? 0);
   610→  return false;
   611→}

  1216→  /** 命中检测（静态网格几何版）：返回指针所指的落位。
  1217→   *  用拖拽开始时捕获的格子矩形/行列线判定，与动画/渲染状态完全无关：
  1218→   *  ① Dock：指针在 Dock 条内（App）→ 按指针 x 与各图标中心线推导插入下标——
  1219→   *     压在图标上=以其中心为界插到前/后，压在空位=按中心线计数；整条 Dock 是
  1220→   *     连续插入轨道，不再要求「先与某个 Dock App 换位」；
  1221→   *  ② 矩形命中：指针落在某网格槽位矩形（外扩 HIT_PAD_PX）内 → 插入到该槽位
  1222→   *     （插入语义：被拖项落在此格，其余项顺移让位；被拖项自己当前格 → hit === cur，
  1223→   *     reorder 自然跳过）；
  1224→   *  ③ 空白位兜底：未命中任何槽位且指针在主屏内 → 按静态槽位几何拟合「行列线」
  1225→   *     （列 = App 槽宽 + 列间隙；行 = 槽位 top 按行距聚类），把指针换算成目标格
  1226→   *     （行,列），再在「当前布局 − 被拖项 − 本次预览垫格」上模拟 flow：插入点 =
  1227→   *     第一个 flow 起点 ≥ 目标格的 tile 之前，不足目标行/列则垫空占位格（受页容量
  1228→   *     约束）——指针压哪格 App 就落哪格，目标旁边有没有 App 都能直接放入
  1229→   *     （用户实测：目标旁有 App 只能换位、没 App 就放不进去）。
  1230→   *  跨页只走边缘滞留翻页：非当前页槽位一律不参与命中——邻页槽位平移后紧贴视口
  1231→   *  左右缘，贴边放置时会抓到邻页槽位，把 App 弹回另一页
  1232→   *  （用户实测：第二页 App 拖到第一页右缘，松手又回到第二页）。 */
  1233→  const hitTestAt = (x: number, y: number): Hit | null => {
  1234→    const W = dragPageW.current;
  1235→    const dragId0 = dragMeta.current?.id;
  1236→    // ① Dock 连续插入轨道：指针在 Dock 条内（略上探 24px 兼容从网格拖入的过渡区）且
  1237→    //    被拖项是 App → 插入下标 = 中心线在指针左侧的（非被拖项）图标数。
  1238→    //    下标基于「去掉被拖项后的序列」，停在自己当前位置时自然得到原位（防抖动）。
  1239→    const dockRect = dragDockRect.current;
  1240→    if (
  1241→      dockRect &&
  1242→      dragId0 !== undefined &&
  1243→      !isWidgetKey(dragId0) &&
  1244→      x >= dockRect.left - 8 &&
  1245→      x <= dockRect.right + 8 &&
  1246→      y >= dockRect.top - 24 &&
  1247→      y <= dockRect.bottom
  1248→    ) {
  1249→      const dockSlots = dragGeo.current
  1250→        .filter((g) => g.zone === 'dock' && g.id !== dragId0)
  1251→        .sort((a, b) => a.index - b.index);
  1252→      let idx = 0;
  1253→      for (const s of dockSlots) {
  1254→        if (s.left + s.w / 2 < x) idx++;
  1255→      }
  1256→      const cur = zoneIndexOf(dragId0);
  1257→      if (cur && cur.zone === 'dock' && cur.index === idx) return cur;
  1258→      return { zone: 'dock', index: idx };
  1259→    }
  1260→    // ② 矩形命中（网格槽位；Dock 图标矩形仅用于小组件——App 已走上面的轨道）
  1261→    let best: Hit | null = null;
  1262→    let bestDist = Number.POSITIVE_INFINITY;
  1263→    for (const g of dragGeo.current) {
  1264→      if (g.zone === 'grid' && g.page !== pageRef.current) continue;
  1265→      const gx = g.zone === 'grid' ? g.left + (g.page - pageRef.current) * W : g.left;
  1266→      const pad = g.zone === 'dock' ? DOCK_HIT_PAD_PX : HIT_PAD_PX;
  1267→      if (x < gx - pad || x > gx + g.w + pad || y < g.top - pad || y > g.top + g.h + pad) continue;
  1268→      const ddx = gx + g.w / 2 - x;
  1269→      const ddy = g.top + g.h / 2 - y;
  1270→      const d = ddx * ddx + ddy * ddy;
  1271→      if (d < bestDist) {
  1272→        bestDist = d;
  1273→        best = g.zone === 'grid' ? { zone: 'grid', page: g.page, index: g.index } : { zone: 'dock', index: g.index };
  1274→      }
  1275→    }
  1276→    if (best) return best;
  1277→    // ③ 空白位兜底（行+列双定位）
  1278→    const rect = dragRootRect.current;
  1279→    if (rect && x > rect.left && x < rect.right && y > rect.top && y < rect.bottom) {
  1280→      const p = Math.min(pageRef.current, layoutRef.current.pages.length - 1);
  1281→      const cur = dragId0 !== undefined ? zoneIndexOf(dragId0) : null;
  1282→      const isW = dragId0 !== undefined && isWidgetKey(dragId0);
  1283→      const slots = dragGeo.current.filter((g) => g.zone === 'grid' && g.page === p);
  1284→      if (slots.length === 0) {
  1285→        // 空页：直接放到页首
  1286→        if (cur && cur.zone === 'grid' && cur.page === p && cur.index === 0) return cur;
  1287→        return { zone: 'grid', page: p, index: 0 };
  1288→      }
  1289→      // 列线：优先 App 槽宽；无 App 槽的页用小组件槽宽反推单列宽
  1290→      const appSlot = slots.find((s) => s.kind === 'app');
  1291→      const refSlot = appSlot ?? slots[0];
  1292→      const refSpanC = refSlot.kind === 'widget' ? WIDGET_SPAN_SIZE[widgetOfKey(refSlot.id)].c : 1;
  1293→      const colW = (refSlot.w - (refSpanC - 1) * GRID_GAP_X) / refSpanC;
  1294→      const minLeft = Math.min(...slots.map((s) => s.left));
  1295→      const targetCol = Math.max(0, Math.min(3, Math.round((x - colW / 2 - minLeft) / (colW + GRID_GAP_X))));
  1296→      // 行线：槽位 top 聚类（相邻 >12px 视为新行）；行距 = App 槽高 + 行间隙
  1297→      //（无 App 槽时从通栏/方格小组件槽高反推；self-center 的方格小组件高度不含行间隙，跳过）
  1298→      const sortedTops = slots.map((s) => s.top).sort((a, b) => a - b);
  1299→      const rowTops: number[] = [];
  1300→      for (const t of sortedTops) {
  1301→        if (rowTops.length === 0 || t - rowTops[rowTops.length - 1] > 12) rowTops.push(t);
  1302→      }
  1303→      const pitchSrc = appSlot ?? slots.find((s) => !(s.kind === 'widget' && ['weather', 'bubble'].includes(widgetOfKey(s.id)))) ?? slots[0];
  1304→      const pitchSrcR = pitchSrc.kind === 'widget' ? WIDGET_SPAN_SIZE[widgetOfKey(pitchSrc.id)].r : 1;
  1305→      const pitch = pitchSrc.h / pitchSrcR + GRID_GAP_Y;
  1306→      const top0 = rowTops[0];
  1307→      // 行边界取行间隙中点：round(f - 0.41)（行高 ≈76、行距 ≈92 → 间隙中点在 0.913 行距处）
  1308→      const gridRow = Math.max(0, Math.round((y - top0) / pitch - 0.41));
  1309→      // 在「当前布局 − 被拖项 − 本次预览垫格」上模拟 flow（该序列在整次拖拽中不变，
  1310→      // 判定结果因此与预览状态无关、停同一指针位置恒得同一落点）
  1311→      const recEmpties = dragEmpties.current;
  1312→      const pageTiles = layoutRef.current.pages[p] ?? [];
  1313→      const sim = pageTiles.filter((t, j) => {
  1314→        if (dragId0 !== undefined && tileKey(t) === dragId0) return false;
  1315→        if (recEmpties && recEmpties.page === p && j >= recEmpties.at && j < recEmpties.at + recEmpties.count && t.kind === 'empty') return false;
  1316→        return true;
  1317→      });
  1318→      const { starts, after, end } = flowPositions(sim);
  1319→      // 目标行夹到 flow 末行：内容下方空白不新开行，落到 flow 末行并按列定位
  1320→      const targetRow = Math.min(gridRow, end.r);
  1321→      // 插入点：第一个 flow 起点 ≥ 目标格（行主序）的 tile 之前；没有则页尾
  1322→      let idx = sim.length;
  1323→      for (let i = 0; i < starts.length; i++) {
  1324→        const s = starts[i];
  1325→        if (s.r > targetRow || (s.r === targetRow && s.c >= targetCol)) {
  1326→          idx = i;
  1327→          break;
  1328→        }
  1329→      }
  1330→      // 垫格：从插入点游标推进到目标格（跨行时垫完整空行）
  1331→      const cb = idx === 0 ? { r: 0, c: 0 } : after[idx - 1];
  1332→      let padEmpty = 0;
  1333→      if (cb.r < targetRow) padEmpty = 4 - cb.c + (targetRow - cb.r - 1) * 4 + targetCol;
  1334→      else if (cb.r === targetRow && cb.c < targetCol) padEmpty = targetCol - cb.c;
  1335→      // 页容量约束：sim 项 + 垫格 + 被拖项 ≤ 页容量
  1336→      padEmpty = Math.max(0, Math.min(padEmpty, pageCap(pageTiles, isW) - sim.length - 1, 8));
  1337→      // 被拖项本来就停在该插入位置 → 返回其当前位置，避免无意义的换位抖动
  1338→      if (padEmpty === 0 && cur && cur.zone === 'grid' && cur.page === p && cur.index === idx) return cur;
  1339→      return { zone: 'grid', page: p, index: idx, padEmpty };
  1340→    }
  1341→    return null;
  1342→  };

  1344→  /** 拖拽预览专用 reorder：先摘除上一次预览垫入的空占位格再换位，并记录新垫入的空占位格
  1345→   *  （只记本次插入、且恰好紧邻被拖项之前的那几个，不碰既有布局里用户保留的空占位格） */
  1346→  const reorderForDrag = (id: string, hit: Hit) => {
  1347→    const rec = dragEmpties.current;
  1348→    dragEmpties.current = null;
  1349→    let work = layoutRef.current;
  1350→    if (rec) {
  1351→      const pages = work.pages.map((p, i) => {
  1352→        if (i !== rec.page) return p;
  1353→        const out = p.slice();
  1354→        let removed = 0;
  1355→        while (rec.at + removed < out.length && removed < rec.count && out[rec.at + removed].kind === 'empty') {
  1356→          out.splice(rec.at + removed, 1);
  1357→          removed++;
  1358→        }
  1359→        return out;
  1360→      });
  1361→      work = { ...work, pages };
  1362→    }
  1363→    const next = reorder(work, id, hit);
  1364→    layoutRef.current = next;
  1365→    if (hit.zone === 'grid') {
  1366→      const pad = Math.max(0, Math.min(3, hit.padEmpty ?? 0));
  1367→      if (pad > 0) {
  1368→        const pi = Math.min(hit.page, next.pages.length - 1);
  1369→        const page = next.pages[pi];
  1370→        const pos = page.findIndex((t) => tileKey(t) === id);
  1371→        if (pos >= pad && page.slice(pos - pad, pos).every((t) => t.kind === 'empty')) {
  1372→          dragEmpties.current = { page: pi, at: pos - pad, count: pad };
  1373→        }
  1374→      }
  1375→    }
  1376→    return next;
  1377→  };

  1379→  /** 拖拽中贴近屏幕左右边缘 → 停留 EDGE_FLIP_MS 后自动翻页（手指仍停在边缘则原地重新计时，
  1380→   *  可连续翻，跨页拖拽用）；已在最后一页仍贴住右缘 → 新建一页接住被拖 App
  1381→   *  （添加界面的方式，每次拖拽限建一页防止连建） */
  1382→  const updateEdgeFlip = (cx: number) => {

  1393→        window.clearTimeout(flipTimer.current);
  1394→        flipTimer.current = null;
  1395→      }

  1400→          edgeFlip.current = 0;

  1404→            if (!pageCreated.current) {
  1405→              pageCreated.current = true;
  1406→              addPage();
  1407→            }
  1408→          } else {
  1409→            flipTo(from + d);

  1411→          if (pageRef.current !== from) {

  1413→              navigator.vibrate?.(8);
  1414→            } catch {
  1415→              /* 震动不可用 */
  1416→            }

  1419→            // 不主动搬的话松手后 App 会留在原页）
  1420→            moveDraggedToPageEnd();
  1421→            // 手指仍停在边缘 → 原地重新计时，可连续翻页
  1422→            const p = edgePointer.current;
  1423→            if (p) updateEdgeFlip(p.x);
  1424→          }
  1425→        }, EDGE_FLIP_MS);
  1426→      }
  1427→    }
