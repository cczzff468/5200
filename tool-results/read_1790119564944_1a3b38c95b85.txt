    84→async function readStorageEstimate(): Promise<{ usage: number; quota: number }> {
    85→  try {
    86→    if (typeof navigator !== 'undefined' && navigator.storage?.estimate) {
    87→      const est = await navigator.storage.estimate();
    88→      return { usage: est.usage ?? 0, quota: est.quota ?? 0 };
    89→    }
    90→  } catch {
    91→    // 忽略，返回 0
    92→  }
    93→  return { usage: 0, quota: 0 };
    94→}

    96→function formatMB(bytes: number): string {
    97→  if (!Number.isFinite(bytes) || bytes <= 0) return '0 MB';
    98→  const mb = bytes / (1024 * 1024);
    99→  return mb >= 100 ? `${Math.round(mb)} MB` : `${mb.toFixed(1)} MB`;
   100→}

   405→function NotificationPage({ onBack }: { onBack: () => void }) {
   406→  const [enabled, setEnabled] = useState<boolean>(() =>
   407→    typeof window !== 'undefined' && typeof Notification !== 'undefined'
   408→      ? Notification.permission === 'granted'
   409→      : false
   410→  );
   411→  const [denied, setDenied] = useState<boolean>(() =>
   412→    typeof window !== 'undefined' && typeof Notification !== 'undefined'
   413→      ? Notification.permission === 'denied'
   414→      : false
   415→  );
   416→
   417→  const handleToggle = async (checked: boolean) => {
   418→    if (!checked) {
   419→      setEnabled(false);
   420→      return;
   421→    }
   422→    if (typeof window === 'undefined' || typeof Notification === 'undefined') {
   423→      setEnabled(false);
   424→      return;
   425→    }
   426→    if (Notification.permission === 'granted') {
   427→      setEnabled(true);
   428→      setDenied(false);
   429→      return;
   430→    }
   431→    if (Notification.permission === 'denied') {
   432→      setEnabled(false);
   433→      setDenied(true);
   434→      return;
   435→    }
   436→    // 'default'：向浏览器请求授权
   437→    const result = await Notification.requestPermission();
   438→    if (result === 'granted') {
   439→      setEnabled(true);
   440→      setDenied(false);
   441→    } else {
   442→      setEnabled(false);
   443→      setDenied(result === 'denied');
   444→    }
   445→  };
   446→
   447→  return (
   448→    <DetailShell title="通知" onBack={onBack}>
   449→      <GroupCard>
   450→        <div className="flex h-[46px] items-center justify-between px-4">
   451→          <span className="text-[16px]">允许通知</span>
   452→          <Switch
   453→            checked={enabled}
   454→            onCheckedChange={(v) => void handleToggle(v)}
   455→            aria-label="允许通知"
   456→          />
   457→        </div>
   458→      </GroupCard>
   459→      {denied && (
   460→        <p className="mt-3 px-1 text-[13px] leading-relaxed" style={{ color: IOS_RED }}>
   461→          通知已被浏览器拒绝，请在浏览器设置中恢复。
   462→        </p>
   463→      )}
   464→      <p className="mt-3 px-1 text-[12px] leading-relaxed text-muted-foreground">
   465→        闹钟和计时器到点时将通过系统通知提醒。
   466→      </p>
   467→    </DetailShell>
   468→  );
   469→}

   473→interface StorageCounts {
   474→  photos: number;
   475→  recordings: number;
   476→  music: number;
   477→  notes: number;
   478→  events: number;
   479→}
   480→
   481→function StoragePage({ onBack }: { onBack: () => void }) {
   482→  const [usage, setUsage] = useState<number | null>(null);
   483→  const [quota, setQuota] = useState<number | null>(null);
   484→  const [counts, setCounts] = useState<StorageCounts | null>(null);
   485→  const [cleaning, setCleaning] = useState(false);
   486→  const [cleanMsg, setCleanMsg] = useState('');
   487→
   488→  const loadStats = useCallback(async () => {
   489→    const est = await readStorageEstimate();
   490→    setUsage(est.usage);
   491→    setQuota(est.quota);
   492→    try {
   493→      const [photos, recordings, music, notes, events] = await Promise.all([
   494→        localDB.count('photos'),
   495→        localDB.count('recordings'),
   496→        localDB.count('music'),
   497→        localDB.count('notes'),
   498→        localDB.count('events'),
   499→      ]);
   500→      setCounts({ photos, recordings, music, notes, events });
   501→    } catch {
   502→      setCounts({ photos: 0, recordings: 0, music: 0, notes: 0, events: 0 });
   503→    }
   504→  }, []);
   505→
   506→  useEffect(() => {
   507→    void loadStats();
   508→  }, [loadStats]);
   509→
   510→  const handleClean = async () => {
   511→    if (cleaning) return;

  1874→/** 读取图片文件 → 居中裁剪缩放为 256px 头像 data URL（控住 IndexedDB 体积） */
  1875→function fileToAvatarDataUrl(file: File): Promise<string> {
  1876→  return new Promise((resolve, reject) => {
  1877→    const url = URL.createObjectURL(file);
  1878→    const img = new Image();
  1879→    img.onload = () => {
  1880→      try {
  1881→        const size = 256;
  1882→        const canvas = document.createElement('canvas');
  1883→        canvas.width = size;
  1884→        canvas.height = size;
  1885→        const ctx = canvas.getContext('2d');
  1886→        if (!ctx) throw new Error('canvas 不可用');
  1887→        const scale = Math.max(size / img.width, size / img.height);
  1888→        const w = img.width * scale;
  1889→        const h = img.height * scale;
  1890→        ctx.drawImage(img, (size - w) / 2, (size - h) / 2, w, h);
  1891→        URL.revokeObjectURL(url);
  1892→        resolve(canvas.toDataURL('image/jpeg', 0.85));
  1893→      } catch (err) {
  1894→        URL.revokeObjectURL(url);
  1895→        reject(err instanceof Error ? err : new Error('图片处理失败'));
  1896→      }
  1897→    };
  1898→    img.onerror = () => {
  1899→      URL.revokeObjectURL(url);
  1900→      reject(new Error('图片读取失败'));
  1901→    };
  1902→    img.src = url;
  1903→  });
  1904→}
