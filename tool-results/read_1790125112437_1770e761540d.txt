   180→   180→      ? (raw.threshold as MemSettings['threshold'])

   183→   183→      ? (raw.longThreshold as MemSettings['longThreshold'])

   268→   268→/**
   269→   269→ * 时间感知过期清扫：把 expiresAt 已到期的碎片标记 expiredAt（自动归档）。
   270→   270→ * 惰性执行：召回（memRecallBlock）、一轮对话结束（memAfterAiTurn）、记忆库刷新时调用；
   271→   271→ * 召回/总结的过滤都按 isMemExpired 实时判断，本清扫只负责落库标记与 UI 展示。
   272→   272→ * 返回本次标记的条数。

   274→   274→export function memSweepExpiry(contactId: string): number {
   275→   275→  const now = Date.now();
   276→   276→  const list = readFragments(contactId);
   277→   277→  let changed = 0;
   278→   278→  const next = list.map((f) => {
   279→   279→    if (f.expiresAt != null && now > f.expiresAt && !f.expiredAt && !f.supersededAt) {
   280→   280→      changed++;
   281→   281→      return { ...f, expiredAt: now };
   282→   282→    }
   283→   283→    return f;
   284→   284→  });
   285→   285→  if (changed > 0) writeJSON(fragKey(contactId), next);
   286→   286→  return changed;
   287→   287→}

   315→   315→
   316→   316→/**
   317→   317→ * 手动时间编辑入参（时间感知：用户在记忆库手动修改事件/过期时间）：
   318→   318→ * - eventTime/expiresAt 传 null 表示清除（事件时间清空 / 恢复永不过期），传时间戳表示设置；
   319→   319→ * - undefined 表示不改动该字段。
   320→   320→ */
   321→   321→export interface MemTimePatch {
   322→   322→  eventTime?: number | null;
   323→   323→  expiresAt?: number | null;
   324→   324→}
   325→   325→
   326→   326→/**
   327→   327→ * 三层通用：应用手动时间补丁。被手动编辑过的记忆会带上 timeEditedAt 标记，
   328→   328→ * 此后自动流程（提取合并/去重/总结）一律以用户设置为准，不得改写其 eventTime/expiresAt。
   329→   329→ */
   330→   330→function applyTimePatch<
   331→   331→  T extends { eventTime?: number; expiresAt?: number; expiredAt?: number; timeEditedAt?: number; editedAt?: number }
   332→   332→>(rec: T, patch: MemTimePatch, now: number): T {
   333→   333→  const next = { ...rec };
   334→   334→  if (patch.eventTime !== undefined) {
   335→   335→    if (patch.eventTime == null) delete next.eventTime;
   336→   336→    else if (Number.isFinite(patch.eventTime)) next.eventTime = patch.eventTime;
   337→   337→  }
   338→   338→  if (patch.expiresAt !== undefined) {
   339→   339→    if (patch.expiresAt == null) delete next.expiresAt;
   340→   340→    else if (Number.isFinite(patch.expiresAt)) next.expiresAt = patch.expiresAt;
   341→   341→  }
   342→   342→  next.timeEditedAt = now;

   512→   512→
   513→   513→/** 相关性打分：内容与当前上下文的重叠 + 新近加成（长期/核心/碎片各自组内排序用） */
   514→   514→function relevanceScore(content: string, createdAt: number, context: string): number {
   515→   515→  const ctx = contextGramsCache.get(context) ?? bigrams(context);
   516→   516→  contextGramsCache.set(context, ctx);
   517→   517→  if (contextGramsCache.size > 8) contextGramsCache.delete(contextGramsCache.keys().next().value as string);
   518→   518→  let score = overlap(bigrams(content), ctx) * 2;
   519→   519→  const age = Date.now() - createdAt;
   520→   520→  if (age < 86400_000) score += 3;
   521→   521→  else if (age < 7 * 86400_000) score += 2;
   522→   522→  else if (age < 30 * 86400_000) score += 1;
   523→   523→  return score;
   524→   524→}
   525→   525→
   526→   526→function overlap(a: Set<string>, b: Set<string>): number {
   527→   527→  let n = 0;
   528→   528→  a.forEach((g) => {
   529→   529→    if (b.has(g)) n++;
   530→   530→  });
   531→   531→  return n;
   532→   532→}
   533→   533→
   534→   534→/** 上下文文本的 gram 缓存（同一次召回内复用） */
   535→   535→const contextGramsCache = new Map<string, Set<string>>();
   536→   536→
   537→   537→/** 碎片召回得分：相关性 × 权重系数（高权重优先）× 淡化系数（淡化中 0.4；已归档直接不参与） */
   538→   538→function fragRecallScore(f: MemFragment, forget: MemForget, context: string, now: number): { s: number; st: ReturnType<typeof fadeState> } {
   539→   539→  const st = fadeState(f as FadeInput, forget, now);
   540→   540→  const s =
   541→   541→    relevanceScore(f.content, f.reinforcedAt ?? f.sourceTime, context) *
   542→   542→    weightFactor(f.weight) *
   543→   543→    (st === 'fading' ? 0.4 : 1);
   544→   544→  return { s, st };
   545→   545→}

   547→   547→/**
   548→   548→ * 召回该联系人（互通开关限定范围）的记忆，组装为注入 system 的文本块。
   549→   549→ * 注入顺序（层级从高到低）：长期记忆 → 核心记忆 → 近期碎片；跨层去重：与已选内容相似（≥0.6）
   550→   550→ * 的低层记忆不再重复注入（长期已收编的由长期代表、核心已收编碎片的由核心代表）。
   551→   551→ * 容量上限：长期 8 条、核心 12 条、碎片 5 条（各按 权重×相关性×淡化 降序截断）；
   552→   552→ * 已入长期（archivedAt）的核心、已入核心（consumedAt）与已归档的碎片不再召回。
   553→   553→ * 该联系人没有记忆 / 互通关闭且当前 App 无来源记忆 → 返回空串（正常聊天，不报错）。
   554→   554→ *
   555→   555→ * 群聊互通（opts）：
   556→   556→ * - mode 'private'（缺省）：跨 App 互通开关照旧；另读群聊来源记忆——碎片/总结按来源群
   557→   557→ *   memoryInterop 判断（互斥于跨 App share 开关，两层开关都开才可见）；
   558→   558→ * - mode 'group'（groupId 必传）：只读当前群来源记忆 + （该群互通开启时）自己的非群聊记忆；
   559→   559→ *   其他群的记忆永不参与（群间隔离），角色隔离由存储键天然保证。
   560→   560→ */
   561→   561→export interface MemRecallOpts {
   562→   562→  mode?: 'private' | 'group';
   563→   563→  /** mode='group' 时必传：当前群聊 ID */
   564→   564→  groupId?: string;
   565→   565→  /** 群互通开关解析（缺省读群数据层的 memoryInterop） */
   566→   566→  interopOn?: (groupId: string) => boolean;
   567→   567→  /** 群名解析（来源标注「群聊·群名」用；缺省读群数据层） */
   568→   568→  groupLabel?: (groupId: string) => string | null;
   569→   569→}

   571→   571→export function memRecallBlock(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts): string {
   572→   572→  // 顶层容错（与 collectWbBlocks 同款约定）：记忆注入任何意外异常都不阻断消息发送，
   573→   573→  // 返回空块 = 无记忆正常聊天（群聊/私聊同此规则）
   574→   574→  try {
   575→   575→    return memRecallBlockInner(contactId, app, contextText, opts);
   576→   576→  } catch {
   577→   577→    return '';
   578→   578→  }
   579→   579→}
   580→   580→
   581→   581→function memRecallBlockInner(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts): string {
   582→   582→  if (!contactId) return '';
   583→   583→  memSweepExpiry(contactId);
   584→   584→  const { share, forget } = getMemSettings(contactId);
   585→   585→  const now = Date.now();
   586→   586→  const isGroupMode = opts?.mode === 'group' && !!opts.groupId;
   587→   587→  const curGroup = opts?.groupId ?? '';
   588→   588→  const interopOn = opts?.interopOn ?? defaultGroupInteropOn;
   589→   589→  const groupLabelOf = opts?.groupLabel ?? defaultGroupLabel;
   590→   590→  /** 核心/长期层级的群来源可见性：纯私聊总结恒可见（旧数据兼容）；带群来源的按各群互通判断 */
   591→   591→  const summaryVisiblePrivate = (m: Pick<MemCore, 'groupIds' | 'privateSource'>): boolean => {
   592→   592→    const gids = m.groupIds ?? [];
   593→   593→    if (gids.length === 0) return true; // 纯私聊/朋友圈来源（含旧数据）
   594→   594→    if (m.privateSource !== false) return true; // 混合来源（含私聊部分）：私聊恒可见
   595→   595→    return gids.some((g) => interopOn(g)); // 纯群聊来源：任一来源群互通开启才可见
   596→   596→  };
   597→   597→  /** 群聊模式下的总结可见性：本群来源恒可见；纯私聊总结在互通开时可见；其他群的总结永不可见（群间隔离） */
   598→   598→  const summaryVisibleInGroup = (m: Pick<MemCore, 'groupIds' | 'privateSource'>): boolean => {
   599→   599→    const gids = m.groupIds ?? [];
   600→   600→    if (gids.includes(curGroup)) return true;
   601→   601→    if (gids.some((g) => g !== curGroup)) return false; // 沾了其他群来源：不进本群上下文
   602→   602→    return interopOn(curGroup) && m.privateSource !== false;
   603→   603→  };
   604→   604→  /** 碎片级可见性（来源精确到条） */
   605→   605→  const fragVisiblePrivate = (f: MemFragment): boolean => {
   606→   606→    if (f.source !== 'group') return true;
   607→   607→    return f.sourceGroupId ? interopOn(f.sourceGroupId) : false;
   608→   608→  };
   609→   609→  const fragVisibleInGroup = (f: MemFragment): boolean => {
   610→   610→    if (f.source === 'group') return f.sourceGroupId === curGroup; // 只读当前群（群间隔离）
   611→   611→    return interopOn(curGroup); // 自己的私聊/朋友圈记忆：互通开启才可见
   612→   612→  };
   613→   613→  // 长期记忆（顶层画像）：全量注入（安全上限 8 条防失控）；用户手动设置过期的长期不再注入
   614→   614→  const longs = listLongTerm(contactId)
   615→   615→    .filter((m) => !isMemExpired(m, now) && (share || m.apps.includes(app)))
   616→   616→    .filter((m) => (isGroupMode ? summaryVisibleInGroup(m) : summaryVisiblePrivate(m)))
   617→   617→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, contextText) }))
   618→   618→    .sort((a, b) => b.s - a.s)
   619→   619→    .slice(0, 8);
   620→   620→  // 核心记忆：全量注入（安全上限 12 条）；已被长期收编的不再注入（由长期记忆代表）；
   621→   621→  // 用户手动设置过期的核心不再注入（默认永不过期不受影响）
   622→   622→  const cores = listCores(contactId)
   623→   623→    .filter((m) => !m.archivedAt && !isMemExpired(m, now) && (share || m.apps.includes(app)))
   624→   624→    .filter((m) => (isGroupMode ? summaryVisibleInGroup(m) : summaryVisiblePrivate(m)))
   625→   625→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, contextText) }))
   626→   626→    .sort((a, b) => b.s - a.s)
   627→   627→    .slice(0, 12);
   628→   628→  // 近期记忆碎片：按相关性取前 5 条；已消费的由核心记忆代表，已更新/已过期的不召回
   629→   629→  const frags = listFragments(contactId)
   630→   630→    .filter((f) => !f.consumedAt && !f.supersededAt && !isMemExpired(f, now) && (share || f.app === app))
   631→   631→    .filter((f) => (isGroupMode ? fragVisibleInGroup(f) : fragVisiblePrivate(f)))
   632→   632→    .map((f) => ({ f, ...fragRecallScore(f, forget, contextText, now) }))
   633→   633→    .filter((x) => x.st !== 'faded')
   634→   634→    .sort((a, b) => b.s - a.s)
   635→   635→    .slice(0, 5);
   636→   636→  if (longs.length === 0 && cores.length === 0 && frags.length === 0) return '';
   637→   637→  // 跨层去重：按注入顺序遍历，与已选内容相似（≥0.6）的条目跳过（信息已由高层代表）
   638→   638→  const accepted: string[] = [];
   639→   639→  const isDup = (content: string) => accepted.some((a) => similarity(a, content) >= SIMILAR_MERGE_THRESHOLD);
   640→   640→  const keepLongs = longs.filter(({ m }) => {
   641→   641→    if (isDup(m.content)) return false;
   642→   642→    accepted.push(m.content);
   643→   643→    return true;
   644→   644→  });
   645→   645→  const keepCores = cores.filter(({ m }) => {
   646→   646→    if (isDup(m.content)) return false;
   647→   647→    accepted.push(m.content);
   648→   648→    return true;
   649→   649→  });
   650→   650→  const keepFrags = frags.filter(({ f }) => {
   651→   651→    if (isDup(f.content)) return false;
   652→   652→    accepted.push(f.content);
   653→   653→    return true;
   654→   654→  });
   655→   655→  if (keepLongs.length === 0 && keepCores.length === 0 && keepFrags.length === 0) return '';
   656→   656→  // 时间感知：层内按有效时间（事件时间优先，其次加强/来源/创建时间）从新到旧排序
   657→   657→  keepLongs.sort((a, b) => memEffectiveTime(b.m) - memEffectiveTime(a.m));
   658→   658→  keepCores.sort((a, b) => memEffectiveTime(b.m) - memEffectiveTime(a.m));
   659→   659→  keepFrags.sort((a, b) => memEffectiveTime(b.f) - memEffectiveTime(a.f));
   660→   660→  // 头部说明按互通状态分文案（跨 App 互通修复）：角色必须被告知「多端同一个人」以及
   661→   661→  // 本次能看到哪些来源的记忆——否则带「·微信」标注的碎片会被当成无关信息，甚至否认在别处的对话
   662→   662→  const headScope = isGroupMode
   663→   663→    ? '记忆库自动整理'
   664→   664→    : share

   699→   699→/** 召回预览（设置页）：与 memRecallBlock 同一套层级与排序，返回排序后的记忆对象 */
   700→   700→export function memRecallPreview(contactId: string): { longs: MemLongTerm[]; cores: MemCore[]; frags: MemFragment[] } {
   701→   701→  const { forget } = getMemSettings(contactId);
   702→   702→  const now = Date.now();

   704→   704→    .filter((m) => !isMemExpired(m, now))
   705→   705→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, '') }))
   706→   706→    .sort((a, b) => b.s - a.s)
   707→   707→    .slice(0, 3)
   708→   708→    .map((x) => x.m);
   709→   709→  const cores = listCores(contactId)
   710→   710→    .filter((m) => !m.archivedAt && !isMemExpired(m, now))

   712→   712→    .sort((a, b) => b.s - a.s)
   713→   713→    .slice(0, 3)
   714→   714→    .map((x) => x.m);
   715→   715→  const frags = listFragments(contactId)
   716→   716→    .filter((f) => !f.consumedAt && !f.supersededAt && !isMemExpired(f, now))

   718→   718→    .filter((x) => x.st !== 'faded')
   719→   719→    .sort((a, b) => b.s - a.s)
   720→   720→    .slice(0, 5)
   721→   721→    .map((x) => x.f);
   722→   722→  return { longs, cores, frags };

   822→   822→/**
   823→   823→ * 模型时间字符串 → 时间戳（无时区字符串按北京时间解析，与提取锚点/注入标签一致）：
   824→   824→ * - "YYYY-MM-DD" → 北京时间 0 点（仅日期事件，标签不带出无意义的 00:00）
   825→   825→ * - "YYYY-MM-DD HH:mm" → 北京时间该时刻
   826→   826→ * - 其余（含完整 ISO/带偏移）→ Date.parse 兜底
   827→   827→ * 范围距当前 ±MEM_TIME_RANGE_YEARS 年，超出/解析失败返回 null（绝不让幻觉时间入库）。
   828→   828→ */
   829→   829→function parseMemTime(v: unknown, now: number): number | null {
   830→   830→  if (typeof v !== 'string') return null;
   831→   831→  const s = v.trim();
   832→   832→  if (!s || s === 'null' || s === 'undefined') return null;
   833→   833→  let d: Date | null = null;
   834→   834→  const md = /^(\d{4})-(\d{1,2})-(\d{1,2})$/.exec(s);
   835→   835→  const mdt = /^(\d{4})-(\d{1,2})-(\d{1,2})[ T](\d{1,2}):(\d{2})(?::\d{2})?$/.exec(s);
   836→   836→  // 无时区字符串统一按北京时间（UTC+8）解析 —— 与提取锚点（Asia/Shanghai）、注入标签（bjParts）
   837→   837→  // 一致，与设备时区无关；带时区/ISO 带偏移的字符串走 Date.parse 原语义
   838→   838→  if (md) d = new Date(Date.UTC(Number(md[1]), Number(md[2]) - 1, Number(md[3])) - 8 * 3_600_000);
   839→   839→  else if (mdt)
   840→   840→    d = new Date(
   841→   841→      Date.UTC(Number(mdt[1]), Number(mdt[2]) - 1, Number(mdt[3]), Number(mdt[4]), Number(mdt[5])) - 8 * 3_600_000
   842→   842→    );
   843→   843→  else {
   844→   844→    const t = Date.parse(s);
   845→   845→    if (!Number.isNaN(t)) d = new Date(t);
   846→   846→  }
   847→   847→  if (!d) return null;
   848→   848→  const t = d.getTime();
   849→   849→  const span = MEM_TIME_RANGE_YEARS * 366 * 86_400_000;
   850→   850→  if (t < now - span || t > now + span) return null;
   851→   851→  return t;
   852→   852→}

   854→   854→/** 把 extract 接口返回的碎片归一化（字符串/对象混合兼容 + 非法值兜底自动分类 + 时间解析/防幻觉） */
   855→   855→function normalizeExtract(res: ExtractApiResult): ExtractItem[] {
   856→   856→  const raw = Array.isArray(res.fragments) ? res.fragments : [];
   857→   857→  const now = Date.now();
   858→   858→  const out: ExtractItem[] = [];
   859→   859→  for (const x of raw) {
   860→   860→    if (typeof x === 'string') {
   861→   861→      const t = x.trim();
   862→   862→      if (t) out.push({ text: t, weight: autoWeight(t) });
   863→   863→    } else if (x && typeof x === 'object' && typeof x.text === 'string' && x.text.trim()) {
   864→   864→      const o = x as { text: string; weight?: unknown; eventTime?: unknown; expiresAt?: unknown; supersedes?: unknown };
   865→   865→      const item: ExtractItem = { text: o.text.trim(), weight: normalizeWeight(o.weight, o.text) };
   866→   866→      const et = parseMemTime(o.eventTime, now);
   867→   867→      const ex = parseMemTime(o.expiresAt, now);
   868→   868→      // 双保险：过期时间不得早于事件时间（服务端已校验一次，这里兜底防历史残留/异常数据）
   869→   869→      if (et != null) item.eventTime = et;
   870→   870→      if (ex != null && (et == null || ex >= et)) item.expiresAt = ex;
   871→   871→      if (Array.isArray(o.supersedes)) {
   872→   872→        const ids = o.supersedes
   873→   873→          .filter((id): id is string => typeof id === 'string' && id.trim().length > 0)
   874→   874→          .map((id) => id.trim())
   875→   875→          .slice(0, 6);
   876→   876→        if (ids.length > 0) item.supersedes = ids;
   877→   877→      }
   878→   878→      out.push(item);
   879→   879→    }
   880→   880→  }
   881→   881→  return out.slice(0, 6);
   882→   882→}

   885→   885→  op: 'extract' | 'summarize',
   886→   886→  body: Record<string, unknown>,
   887→   887→  apiConfig: ApiConfig
   888→   888→): Promise<T> {

   890→   890→    method: 'POST',

   898→   898→
   899→   899→/** 归一化文本（去空白）——精确去重键 */
   900→   900→const normText = (s: string) => s.replace(/\s+/g, '');
   901→   901→
   902→   902→/**
   903→   903→ * 追加碎片（矛盾更新优先，其次三层去重）：
   904→   904→ * 0) 矛盾更新（模型标注 supersedes 且目标存在）：旧记忆标记 supersededAt/supersededBy（「已更新」，
   905→   905→ *    不再参与召回/总结），本条直接新增（事件/过期时间一并入库）；矛盾条目不走相似合并
   906→   906→ *    （「不吃辣」绝不能被合并进「爱吃辣」）；
   907→   907→ * 1) 精确重复 → 记一次「加强」（reinforcedAt/次数刷新），不新增；
   908→   908→ * 2) 相似（2-gram ≥0.6，同一事实的不同说法）→ 合并进已有条目：内容取更完整的、权重取更高、
   909→   909→ *    加强计时刷新、计数 +1，不新增占位（已更新/已过期/已消费的旧记忆不作为合并目标）；
   910→   910→ * 3) 全新 → 新建（权重：LLM 判定优先，缺省按关键词自动分类）。
   911→   911→ * 返回 { added: 新增条目; merged: 合并/加强次数; superseded: 矛盾更新条数 }
   912→   912→ */
   913→   913→function appendFragments(
   914→   914→  contactId: string,
   915→   915→  app: MemApp,
   916→   916→  items: ExtractItem[],
   917→   917→  sourceTime: number,
   918→   918→  sourceMsgId?: string,
   919→   919→  /** 额外字段（动态/群聊来源记忆用：source/sourcePostId/sourceKind/sourceCommentId/sourceGroupId/groupMembers 直通新碎片） */
   920→   920→  extra?: Partial<Pick<MemFragment, 'source' | 'sourcePostId' | 'sourceKind' | 'sourceCommentId' | 'sourceGroupId' | 'groupMembers'>>
   921→   921→): { added: MemFragment[]; merged: number; superseded: number } {
   922→   922→  const list = readFragments(contactId);
   923→   923→  const now = Date.now();
   924→   924→  const seen = new Set(list.filter((f) => !f.supersededAt && !isMemExpired(f, now)).map((f) => normText(f.content)));
   925→   925→  const added: MemFragment[] = [];
   926→   926→  let merged = 0;
   927→   927→  let superseded = 0;
   928→   928→  const newItem = (content: string, item: ExtractItem, id?: string): MemFragment => ({
   929→   929→    id: id ?? uid(),
   930→   930→    contactId,
   931→   931→    app,
   932→   932→    content,
   933→   933→    sourceTime,
   934→   934→    createdAt: now,
   935→   935→    eventTime: item.eventTime ?? undefined,
   936→   936→    expiresAt: item.expiresAt ?? undefined,
   937→   937→    weight: item.weight ?? autoWeight(content),
   938→   938→    reinforcedAt: now,
   939→   939→    reinforceCount: 0,
   940→   940→    sourceMsgId,
   941→   941→    ...(extra ?? {}),
   942→   942→  });
   943→   943→  for (const item of items) {
   944→   944→    const content = item.text.trim();
   945→   945→    if (!content) continue;
   946→   946→    const key = normText(content);
   947→   947→    // 0) 矛盾更新：标记被更正的旧记忆为「已更新」，本条直接新增
   948→   948→    const targets = (item.supersedes ?? [])
   949→   949→      .map((id) => list.find((f) => f.id === id))
   950→   950→      .filter((f): f is MemFragment => f != null && f.supersededAt == null);
   951→   951→    if (targets.length > 0) {
   952→   952→      const newId = uid();
   953→   953→      for (const t of targets) {
   954→   954→        t.supersededAt = now;
   955→   955→        t.supersededBy = newId;
   956→   956→      }
   957→   957→      superseded += targets.length;
   958→   958→      seen.add(key);
   959→   959→      added.push(newItem(content, item, newId));
   960→   960→      continue;
   961→   961→    }
   962→   962→    // 1) 精确重复 → 加强已有记忆（已更新/已过期的旧条目不是目标）
   963→   963→    if (seen.has(key)) {
   964→   964→      const hit = list.find((f) => normText(f.content) === key && !f.supersededAt && !isMemExpired(f, now));
   965→   965→      if (hit) {
   966→   966→        hit.reinforcedAt = now;
   967→   967→        hit.reinforceCount = (hit.reinforceCount ?? 0) + 1;
   968→   968→        hit.weight = higherWeight(hit.weight, item.weight);
   969→   969→        // 时间感知：未手动编辑过的记忆，重复提及且本次提取带新时间时刷新（事件时间保鲜）；
   970→   970→        // 用户手动设置过（timeEditedAt）的一律以用户为准，绝不覆盖
   971→   971→        if (!hit.timeEditedAt) {
   972→   972→          if (item.eventTime != null) hit.eventTime = item.eventTime;
   973→   973→          if (item.expiresAt != null) hit.expiresAt = item.expiresAt;
   974→   974→        }
   975→   975→        merged++;
   976→   976→        continue;
   977→   977→      }
   978→   978→      // 命中的只是已更新/已过期的旧条目 → 落到下面按新增处理
   979→   979→    }
   980→   980→    // 2) 相似说法 → 合并进已有记忆（优先未消费的；内容保留更完整的一条）
   981→   981→    const simHit = list
   982→   982→      .filter((f) => !f.supersededAt && !isMemExpired(f, now) && f.content.length >= 4 && content.length >= 4)
   983→   983→      .sort((a, b) => Number(Boolean(a.consumedAt)) - Number(Boolean(b.consumedAt)) || a.createdAt - b.createdAt)
   984→   984→      .find((f) => similarity(f.content, content) >= SIMILAR_MERGE_THRESHOLD);
   985→   985→    if (simHit) {
   986→   986→      if (content.length > simHit.content.length) simHit.content = content;
   987→   987→      simHit.reinforcedAt = now;
   988→   988→      simHit.reinforceCount = (simHit.reinforceCount ?? 0) + 1;
   989→   989→      simHit.weight = higherWeight(simHit.weight, item.weight);
   990→   990→      // 时间感知（与精确重复同规则）：未手动编辑过 → 新时间保鲜；手动设置过 → 用户优先不覆盖
   991→   991→      if (!simHit.timeEditedAt) {
   992→   992→        if (item.eventTime != null) simHit.eventTime = item.eventTime;
   993→   993→        if (item.expiresAt != null) simHit.expiresAt = item.expiresAt;
   994→   994→      }
   995→   995→      if (!simHit.sourceMsgId && sourceMsgId) simHit.sourceMsgId = sourceMsgId;
   996→   996→      // 溯源字段补齐（动态来源记忆）：点赞/评论同动态连续写入时内容相近会被合并成一条，
   997→   997→      // 合并目标（先写入的那条）若缺溯源字段，用本次传入的补齐——否则删评论时按
   998→   998→      // sourceCommentId 清理会漏掉这条合并碎片（六.3 级联清理依赖这些字段）
   999→   999→      if (extra) {
  1000→  1000→        if (!simHit.source && extra.source) simHit.source = extra.source;
  1001→  1001→        if (!simHit.sourcePostId && extra.sourcePostId) simHit.sourcePostId = extra.sourcePostId;
  1002→  1002→        if (!simHit.sourceKind && extra.sourceKind) simHit.sourceKind = extra.sourceKind;
  1003→  1003→        if (!simHit.sourceCommentId && extra.sourceCommentId) simHit.sourceCommentId = extra.sourceCommentId;
  1004→  1004→        if (!simHit.sourceGroupId && extra.sourceGroupId) simHit.sourceGroupId = extra.sourceGroupId;
  1005→  1005→        if (!simHit.groupMembers && extra.groupMembers) simHit.groupMembers = extra.groupMembers;
  1006→  1006→      }
  1007→  1007→      seen.add(normText(simHit.content));
  1008→  1008→      merged++;
  1009→  1009→      continue;
  1010→  1010→    }
  1011→  1011→    // 3) 全新记忆
  1012→  1012→    seen.add(key);
  1013→  1013→    added.push(newItem(content, item));
  1014→  1014→  }
  1015→  1015→  if (added.length > 0 || merged > 0 || superseded > 0) writeJSON(fragKey(contactId), [...list, ...added]);
  1016→  1016→  return { added, merged, superseded };
  1017→  1017→}

  1019→  1019→/** 把当前待总结（未被更新/未过期/未消费/未归档）的碎片总结为核心记忆，并标记来源碎片已消费（阈值判断由调用方负责） */
  1020→  1020→async function summarizePendingIntoCore(contactId: string, apiConfig: ApiConfig, names?: MemNames | null): Promise<MemCore> {
  1021→  1021→  const { forget } = getMemSettings(contactId);
  1022→  1022→  const now = Date.now();

  1024→  1024→    (f) =>

  1028→  1028→      fadeState(f as FadeInput, forget, now) !== 'faded'
  1029→  1029→  );

  1033→  1033→    {

  1036→  1036→      userName,
  1037→  1037→      peerName,
  1038→  1038→    },
  1039→  1039→    apiConfig
  1040→  1040→  );
  1041→  1041→  const summary = (res.summary || '').trim();
  1042→  1042→  if (!summary) throw new Error('总结结果为空');

  1044→  1044→    id: uid(),
  1045→  1045→    contactId,
  1046→  1046→    content: summary,
  1047→  1047→    fragmentCount: pending.length,
  1048→  1048→    sourceIds: pending.map((f) => f.id),

  1050→  1050→    createdAt: Date.now(),

  1067→  1067→  );
  1068→  1068→  return core;
  1069→  1069→}

  1071→  1071→/** 把当前未归档且未过期的核心记忆总结为一条长期记忆，并标记来源核心已归档（方案A：不重复总结） */

  1124→  1124→/** 群聊轮次的来源/作用域选项（memAfterAiTurn opts）：群记忆碎片带来源标记，轮次计数与私聊分开 */

  1130→  1130→}
  1131→  1131→
  1132→  1132→/**
  1133→  1133→ * 一轮 AI 对话结束后的记忆管线（各聊天 App 的 finalize 成功分支调用）：
  1134→  1134→ * 1) 轮次 +1；2) 达到提取间隔 → 从最近对话提取记忆碎片（后台异步，不阻塞聊天）；

  1136→  1136→ * 失败静默（console.warn），不打断聊天。
  1137→  1137→ * buildConvo 惰性调用：只有真的需要提取时才读取/整理对话文本。
  1138→  1138→ * getSourceMsgId 惰性调用：提取时取来源消息 ID（追溯用，可省略）。
  1139→  1139→ * names：用户真实名字 + 角色名字（视角统一注入提取/总结 prompt；缺省回退固定称呼）。
  1140→  1140→ * opts：群聊轮次传 roundScope + group（群记忆来源标记，轮次计数与私聊互不干扰）。
  1141→  1141→ */
  1142→  1142→export function memAfterAiTurn(
  1143→  1143→  contactId: string | null,
  1144→  1144→  app: MemApp,
  1145→  1145→  apiConfig: ApiConfig,
  1146→  1146→  buildConvo: () => MemConvoTurn[],
  1147→  1147→  getSourceMsgId?: () => string | undefined,
  1148→  1148→  names?: MemNames | null,
  1149→  1149→  opts?: MemTurnOpts
  1150→  1150→): void {
  1151→  1151→  if (!contactId) return;
  1152→  1152→  const scope = opts?.roundScope ? `:${opts.roundScope}` : '';
  1153→  1153→  try {
  1154→  1154→    // 时间感知：先把已到期的碎片标记归档（惰性清扫，召回/总结另有实时过滤兜底）
  1155→  1155→    memSweepExpiry(contactId);
  1156→  1156→    const key = `mem-round:${contactId}:${app}${scope}`;
  1157→  1157→    const count = (readJSON<number>(key) ?? 0) + 1;
  1158→  1158→    const { interval } = getMemSettings(contactId);
  1159→  1159→    if (count < interval) {
  1160→  1160→      writeJSON(key, count);
  1161→  1161→      return;
  1162→  1162→    }
  1163→  1163→    // 达到间隔：归零计数并异步提取（归零在先避免每轮重试轰炸；下个窗口自然重试）
  1164→  1164→    writeJSON(key, 0);
  1165→  1165→    const guard = `${contactId}:${app}${scope}`;
  1166→  1166→    if (inflight.has(guard)) return;
  1167→  1167→    inflight.add(guard);
  1168→  1168→    void (async () => {
  1169→  1169→      try {
  1170→  1170→        const convo = buildConvo();
  1171→  1171→        if (convo.length >= 4) {
  1172→  1172→          const res = await callMemoryApi<ExtractApiResult>(
  1173→  1173→            'extract',
  1174→  1174→            { conversation: convo, app, existing: existingForConflict(contactId), ...namesOf(names) },
  1175→  1175→            apiConfig
  1176→  1176→          );
  1177→  1177→          const items = normalizeExtract(res);
  1178→  1178→          if (items.length > 0) {
  1179→  1179→            // 群聊轮次：碎片带群来源标记（source/sourceGroupId/groupMembers），供互通召回过滤
  1180→  1180→            const extra = opts?.group
  1181→  1181→              ? { source: 'group' as const, sourceGroupId: opts.group.id, groupMembers: opts.group.members }
  1182→  1182→              : undefined;
  1183→  1183→            appendFragments(contactId, app, items, Date.now(), getSourceMsgId?.(), extra);
  1184→  1184→          }
  1185→  1185→        }
  1186→  1186→      } catch (err) {
  1187→  1187→        console.warn('[memory] 自动提取失败（下个窗口重试）', err);
  1188→  1188→      }
  1189→  1189→      // 核心总结与提取相互独立：即使本轮提取失败，只要已积累的未消费碎片达到阈值，
  1190→  1190→      // 仍应尝试凝结核心记忆（否则一次提取故障会连带把总结也卡到下个窗口）
  1191→  1191→      try {
  1192→  1192→        await maybeAutoSummarize(contactId, apiConfig, names);
  1193→  1193→      } catch (err) {
  1194→  1194→        console.warn('[memory] 自动总结核心记忆失败（下个窗口重试）', err);
  1195→  1195→      }
  1196→  1196→      // 长期总结与核心总结相互独立：核心达到长期阈值即触发（失败下个窗口重试）
  1197→  1197→      try {
  1198→  1198→        await maybeAutoLongSummarize(contactId, apiConfig, names);
  1199→  1199→      } catch (err) {
  1200→  1200→        console.warn('[memory] 自动总结长期记忆失败（下个窗口重试）', err);
  1201→  1201→      } finally {
  1202→  1202→        inflight.delete(guard);
  1203→  1203→      }
  1204→  1204→    })();
  1205→  1205→  } catch {
  1206→  1206→    // 记忆是增强能力：任何异常都不影响聊天
  1207→  1207→  }
  1208→  1208→}

  1297→  1297→    inflight.delete(guard);

  1318→  1318→
  1319→  1319→/**
  1320→  1320→ * 全库去重整理（设置页按钮）：两两扫描该联系人全部碎片，相似（≥0.6）的合并为一条。
  1321→  1321→ * 保留更早创建的为正本（若另一条已入核心则保留已入核心者），内容取更完整的、
  1322→  1322→ * 权重取更高、加强次数累加、加强时间取较新。返回合并掉的条数。
  1323→  1323→ */
  1324→  1324→export function memDedupeNow(contactId: string): number {
  1325→  1325→  let list = readFragments(contactId);
  1326→  1326→  const before = list.length;
  1327→  1327→  let changed = true;
  1328→  1328→  while (changed) {
  1329→  1329→    changed = false;
  1330→  1330→    outer: for (let i = 0; i < list.length; i++) {
  1331→  1331→      for (let j = i + 1; j < list.length; j++) {
  1332→  1332→        const a = list[i];
  1333→  1333→        const b = list[j];
  1334→  1334→        if (!a.content || !b.content || a.content.length < 4 || b.content.length < 4) continue;
  1335→  1335→        if (similarity(a.content, b.content) < SIMILAR_MERGE_THRESHOLD) continue;
  1336→  1336→        // 正本：已入核心者优先，其次更早创建者
  1337→  1337→        const keep = Number(Boolean(a.consumedAt)) - Number(Boolean(b.consumedAt)) !== 0
  1338→  1338→          ? (a.consumedAt ? a : b)
  1339→  1339→          : (a.createdAt <= b.createdAt ? a : b);
  1340→  1340→        const drop = keep === a ? b : a;
  1341→  1341→        if (drop.content.length > keep.content.length) keep.content = drop.content;
  1342→  1342→        keep.weight = higherWeight(keep.weight, drop.weight);
  1343→  1343→        // 时间感知（手动优先）：正本未被手动编辑时从副本补齐缺失的时间字段；
  1344→  1344→        // 副本被手动编辑过则把时间连同标记一并继承（合并体继续受用户设置保护）
  1345→  1345→        if (!keep.timeEditedAt) {
  1346→  1346→          if (keep.eventTime == null && drop.eventTime != null) keep.eventTime = drop.eventTime;
  1347→  1347→          if (keep.expiresAt == null && drop.expiresAt != null) keep.expiresAt = drop.expiresAt;
  1348→  1348→          if (drop.timeEditedAt) keep.timeEditedAt = drop.timeEditedAt;
  1349→  1349→        }
  1350→  1350→        keep.reinforceCount = (keep.reinforceCount ?? 0) + 1 + (drop.reinforceCount ?? 0);
  1351→  1351→        keep.reinforcedAt = Math.max(keep.reinforcedAt ?? 0, drop.reinforcedAt ?? 0);
  1352→  1352→        keep.sourceMsgId = keep.sourceMsgId ?? drop.sourceMsgId;
  1353→  1353→        keep.consumedAt = keep.consumedAt ?? drop.consumedAt;
  1354→  1354→        list = list.filter((f) => f.id !== drop.id);
  1355→  1355→        changed = true;
  1356→  1356→        break outer;
  1357→  1357→      }
  1358→  1358→    }
  1359→  1359→  }
  1360→  1360→  if (list.length !== before) writeJSON(fragKey(contactId), list);
  1361→  1361→  return before - list.length;
  1362→  1362→}

  1365→  1365→
  1366→  1366→/** 同义反复短语：「小晨叫小晨 / 小晨名叫小晨 / 小晨的名字是小晨」等（名字身份事实无信息量） */
  1367→  1367→function tautologyClause(name: string): RegExp {
  1368→  1368→  const n = name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  1369→  1369→  return new RegExp(`${n}(?:叫|名叫|名字叫|的名字是|名字是)${n}[，,、]?`, 'g');
  1370→  1370→}

  1372→  1372→/**
  1373→  1373→ * 修复旧记忆的视角混乱（设置页按钮）：把碎片/核心/长期里「对方」替换为角色名字、「用户」
  1374→  1374→ * 替换为用户真实名字（旧版提取器固定用这两个代称，语义明确可安全替换）。
  1375→  1375→ * 只替换提供了对应名字的代称（名字缺失时代称保留，避免产生无主语碎片）；
  1376→  1376→ * 「我/你/TA」等代称语义不明不自动替换（可手动编辑）。
  1377→  1377→ * 同义反复清理：「用户叫小晨」→「小晨叫小晨」这类名字身份短语直接剥掉（已由全库用名隐含），
  1378→  1378→ * 剥完为空的条目删除。返回各层修复条数（含删除）。
  1379→  1379→ */
  1380→  1380→export function memRepairPerspectiveNow(
  1381→  1381→  contactId: string,
  1382→  1382→  names?: MemNames | null
  1383→  1383→): { frags: number; cores: number; longs: number } {
  1384→  1384→  const { userName, peerName } = namesOf(names);
  1385→  1385→  const hasUser = Boolean((names?.user ?? '').trim());
  1386→  1386→  const hasPeer = Boolean((names?.peer ?? '').trim());
  1387→  1387→  if (!hasUser && !hasPeer) return { frags: 0, cores: 0, longs: 0 };
  1388→  1388→  const fix = (content: string): string => {
  1389→  1389→    let out = content;
  1390→  1390→    if (hasPeer) out = out.replace(/对方/g, peerName);
  1391→  1391→    if (hasUser) out = out.replace(/用户/g, userName);
  1392→  1392→    // 同义反复清理：「用户叫小晨」替换后变成「小晨叫小晨」，名字身份事实已由全库用名隐含，
  1393→  1393→    // 直接剥掉该短语（拆不出有效内容则整条删除）
  1394→  1394→    out = out
  1395→  1395→      .replace(tautologyClause(userName), '')
  1396→  1396→      .replace(tautologyClause(peerName), '')
  1397→  1397→      .replace(/^[，,。、；;\s]+/, '')
  1398→  1398→      .replace(/[，,\s]+$/, '');
  1399→  1399→    return out;
  1400→  1400→  };
  1401→  1401→  /** 三层通用：替换 + 剥掉清空后为空的条目（返回修复条数） */
  1402→  1402→  const fixLayer = <T extends { content: string }>(list: T[]): { next: T[]; fixed: number } => {
  1403→  1403→    let fixed = 0;
  1404→  1404→    const out: T[] = [];
  1405→  1405→    for (const item of list) {
  1406→  1406→      const repaired = fix(item.content);
  1407→  1407→      if (repaired !== item.content) {
  1408→  1408→        fixed++;
  1409→  1409→        if (repaired) out.push({ ...item, content: repaired, editedAt: Date.now() });
  1410→  1410→      } else {
  1411→  1411→        out.push(item);
  1412→  1412→      }
  1413→  1413→    }
  1414→  1414→    return { next: out, fixed };
  1415→  1415→  };
  1416→  1416→  // 碎片
  1417→  1417→  const fragRes = fixLayer(readFragments(contactId));
  1418→  1418→  if (fragRes.fixed > 0) writeJSON(fragKey(contactId), fragRes.next);
  1419→  1419→  // 核心
  1420→  1420→  const coreRes = fixLayer(readCores(contactId));
  1421→  1421→  if (coreRes.fixed > 0) writeJSON(coreKey(contactId), coreRes.next);
  1422→  1422→  // 长期
  1423→  1423→  const longRes = fixLayer(readLongTerm(contactId));
  1424→  1424→  if (longRes.fixed > 0) writeJSON(longKey(contactId), longRes.next);
  1425→  1425→  return { frags: fragRes.fixed, cores: coreRes.fixed, longs: longRes.fixed };
  1426→  1426→}
