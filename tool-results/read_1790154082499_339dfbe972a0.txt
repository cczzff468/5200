   113→function loadMsgs(sessionKey: string): ChatMsg[] | null {
   114→  try {
   115→    // 持久化在 IndexedDB kv store（启动时由 idb-kv 从 localStorage 迁移，内存同步读）
   116→    const parsed: unknown = kvGet<ChatMsg[]>(lsMsgsKey(sessionKey));
   117→    if (!Array.isArray(parsed) || parsed.length === 0) return null;
   118→    const ok = parsed.every(
   119→      (m) =>
   120→        m &&
   121→        typeof m === 'object' &&
   122→        typeof (m as ChatMsg).content === 'string' &&
   123→        ((m as ChatMsg).role === 'user' || (m as ChatMsg).role === 'assistant')
   124→    );
   125→    if (!ok) return null;
   126→    const msgs = (parsed as ChatMsg[]).slice(-100);
   127→    // 欢迎语文案迁移：种子消息按固定 id 识别，旧文案就地替换为最新文案（用户消息不动）
   128→    if (sessionKey === 'assistant' && msgs[0]?.id === SEED_MSGS[0].id && msgs[0].content !== SEED_MSGS[0].content) {
   129→      msgs[0] = { ...msgs[0], content: SEED_MSGS[0].content };
   130→    }
   131→    return msgs;
   132→  } catch {
   133→    return null;
   134→  }
   135→}

   157→function fmtTime(t: number): string {
   158→  const d = new Date(t);
   159→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
   160→}
   161→
   162→function fmtDay(t: number): string {
   163→  const d = new Date(t);
   164→  return `${d.getMonth() + 1}月${d.getDate()}日`;
   165→}
   166→
   167→function dayKey(t: number): string {
   168→  const d = new Date(t);
   169→  return `${d.getFullYear()}-${d.getMonth()}-${d.getDate()}`;
   170→}
   171→
   172→/** 会话列表时间：今天显示 HH:MM，跨天显示 M月D日 */
   173→function fmtListTime(t: number): string {
   174→  if (!t) return '';
   175→  return dayKey(t) === dayKey(Date.now()) ? fmtTime(t) : fmtDay(t);
   176→}
   177→
   178→/** 聊天日期分隔行文案（与微信聊天时间规则一致）：今天只显 HH:MM（不加日期）；
   179→ *  昨天「昨天 HH:MM」；一周内（含前天）「星期X HH:MM」；一周前「M月D日 HH:MM」 */
   180→function fmtChatStamp(t: number): string {
   181→  const d = new Date(t);
   182→  const now = new Date();
   183→  if (d.toDateString() === now.toDateString()) return fmtTime(t);
   184→  const yest = new Date(now);
   185→  yest.setDate(now.getDate() - 1);
   186→  if (d.toDateString() === yest.toDateString()) return `昨天 ${fmtTime(t)}`;
   187→  if (now.getTime() - t < 7 * 86400000) return `星期${'日一二三四五六'[d.getDay()]} ${fmtTime(t)}`;
   188→  return `${fmtDay(t)} ${fmtTime(t)}`;
   189→}

   199→function useLongPress(onFire: (pos: LongPressPos) => void, ms = 500) {
   200→  const timer = useRef(0);
   201→  const fired = useRef(false);
   202→  const pos = useRef<LongPressPos>({ x: 0, rowBottom: 0 });
   203→  const stop = () => window.clearTimeout(timer.current);
   204→  useEffect(() => stop, []);
   205→  return {
   206→    onPointerDown: (e: React.PointerEvent<HTMLButtonElement>) => {
   207→      fired.current = false;
   208→      // rect 需在事件同步阶段捕获（timeout 里 currentTarget 已为 null）
   209→      pos.current = { x: e.clientX, rowBottom: e.currentTarget.getBoundingClientRect().bottom };
   210→      stop();
   211→      timer.current = window.setTimeout(() => {
   212→        fired.current = true;
   213→        if (typeof navigator !== 'undefined' && navigator.vibrate) navigator.vibrate(10);
   214→        onFire(pos.current);
   215→      }, ms);
   216→    },
   217→    onPointerUp: stop,
   218→    onPointerLeave: stop,
   219→    onPointerCancel: stop,
   220→    onContextMenu: (e: MouseEvent) => e.preventDefault(),
   221→    /** 长按已触发时拦截随后的 click（不打开会话） */
   222→    onClickCapture: (e: MouseEvent) => {
   223→      if (fired.current) {
   224→        e.preventDefault();
   225→        e.stopPropagation();
   226→        fired.current = false;
   227→      }
   228→    },
   229→  };
   230→}
