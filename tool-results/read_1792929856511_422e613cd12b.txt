   177→/** 角度插值（归一 [-π,π]） */
   178→function lerpAngle(cur: number, target: number, k: number): number {
   179→  let d = target - cur;
   180→  while (d > Math.PI) d -= Math.PI * 2;
   181→  while (d < -Math.PI) d += Math.PI * 2;
   182→  return cur + d * k;
   183→}

   185→/**
   186→ * 家具动作锚点（sit/lie/watch/lamp/play/cook/eat 落点与朝向；music 与 lamp 同款站位）。
   187→ * 与 CoupleHome.anchorOf 同口径的精简版 —— 只服务陪伴小人。
   188→ * 返回 null = 该家具不支持此动作。
   189→ */
   190→export function companionAnchor(
   191→  kind: 'sit' | 'lie' | 'watch' | 'lamp' | 'play' | 'cook' | 'eat' | 'music' | 'read' | 'water',
   192→  item: FurnitureItem,
   193→): Anchor | null {
   194→  const entry = CATALOG[item.kind];
   195→  if (!entry) return null;
   196→  const ry0 = item.ry ?? 0;
   197→  if (kind === 'lamp' || kind === 'music' || kind === 'water') {
   198→    // 走到灯/音响/花盆旁半步距离，面向它
   199→    const dz = entry.fp[1] / 2 + 0.5;
   200→    return {
   201→      tx: item.x + dz * Math.sin(ry0),
   202→      tz: item.z + dz * Math.cos(ry0),
   203→      ry: ry0 + Math.PI,
   204→      y: 0,
   205→    };
   206→  }
   207→  const a =
   208→    kind === 'sit' || kind === 'read'
   209→      ? entry.acts.seat
   210→      : kind === 'lie'
   211→        ? entry.acts.sleep
   212→        : kind === 'watch'
   213→          ? entry.acts.watch
   214→          : kind === 'play'
   215→            ? entry.acts.play
   216→            : kind === 'cook'
   217→              ? entry.acts.cook
   218→              : entry.acts.eat;
   219→  if (!a) return null;
   220→  const an = a as { y: number; dx?: number; dz?: number; ryOff?: number };
   221→  // watch/play/cook/eat 都是「面向家具」的动作：朝向 = 家具朝向 + π（与宿主 FACE_ACTS 同口径）
   222→  const ry = ry0 + (an.ryOff ?? 0) + (kind === 'watch' || kind === 'play' || kind === 'cook' || kind === 'eat' ? Math.PI : 0);
   223→  const cos = Math.cos(ry0);
   224→  const sin = Math.sin(ry0);
   225→  const lx = an.dx ?? 0;
   226→  let lz = an.dz ?? 0;
   227→  if (kind === 'sit' && an.dz == null) {
   228→    // 无锚点 dz 的坐具兜底：坐在家具中心偏前缘
   229→    lz = entry.fp[1] / 2 - 0.12;
   230→  }
   231→  return {
   232→    tx: item.x + lx * cos + lz * sin,
   233→    tz: item.z - lx * sin + lz * cos,
   234→    ry,
   235→    y: a.y ?? 0,
   236→  };
   237→}

   239→/** 找最近的可坐家具 */
   240→function nearestSeat(items: FurnitureItem[], from: Pt): FurnitureItem | null {
   241→  let best: FurnitureItem | null = null;
   242→  let bd = Infinity;
   243→  for (const it of items) {
   244→    if ((it.y ?? 0) > 0.45) continue;
   245→    const e = CATALOG[it.kind];
   246→    if (!e?.acts?.seat) continue;
   247→    const d = Math.hypot(it.x - from.x, it.z - from.z);
   248→    if (d < bd) {
   249→      bd = d;
   250→      best = it;
   251→    }
   252→  }
   253→  return best;
   254→}

   356→/** 找最近一扇关着的门（door）；返回门 id（bed/kt/bath/bal） */
   357→function nearestClosedDoor(
   358→  doors: Partial<Record<DoorKey, DoorGate>>,
   359→  from: Pt,
   360→): 'bed' | 'kt' | 'bath' | 'bal' | null {
   361→  let best: 'bed' | 'kt' | 'bath' | 'bal' | null = null;
   362→  let bd = Infinity;
   363→  (['bed', 'kt', 'bath', 'bal'] as const).forEach((k) => {
   364→    const d = doors[k];
   365→    if (!d || d.open) return;
   366→    // 门中点近似：axis 'z' 墙门在 (along, at) —— 这里用along与from的粗略距离即可
   367→    const dd = Math.abs(d.along - (Math.abs(d.along - from.x) < Math.abs(d.along - from.z) ? from.x : from.z));
   368→    if (dd < bd) {
   369→      bd = dd;
   370→      best = k;
   371→    }
   372→  });
   373→  return best;
   374→}

   376→/** v8.3 找身边 0.85m 内的关着的门（卡住自救开门用）：返回门 id 或 null */
   377→function closedDoorNear(
   378→  doors: Partial<Record<DoorKey, DoorGate>>,
   379→  from: Pt,
   380→  radius = 0.85,
   381→): 'bed' | 'kt' | 'bath' | 'bal' | null {
   382→  let best: 'bed' | 'kt' | 'bath' | 'bal' | null = null;
   383→  let bd = Infinity;
   384→  (['bed', 'kt', 'bath', 'bal'] as const).forEach((k) => {
   385→    const gate = doors[k];
   386→    if (!gate || gate.open) return;
   387→    const pts = doorPoints(k, gate.along);
   388→    const d = Math.min(Math.hypot(pts.aSide.x - from.x, pts.aSide.z - from.z), Math.hypot(pts.bSide.x - from.x, pts.bSide.z - from.z));
   389→    if (d < radius && d < bd) {
   390→      bd = d;
   391→      best = k;
   392→    }
   393→  });
   394→  return best;
   395→}

   397→/** 朝某点走一步（含碰撞回推；返回是否已到 stop 距离内） */
   398→function stepToward(
   399→  c: CharRT,
   400→  tx: number,
   401→  tz: number,
   402→  stop: number,
   403→  dt: number,
   404→  speed: number,
   405→  collide: (c: CharRT) => void,
   406→): boolean {
   407→  const dx = tx - c.x;
   408→  const dz = tz - c.z;
   409→  const d = Math.hypot(dx, dz);
   410→  if (d <= stop) {
   411→    c.moving = false;
   412→    return true;
   413→  }
   414→  const sp = Math.min(speed * dt, d - stop);
   415→  c.x += (dx / d) * sp;
   416→  c.z += (dz / d) * sp;
   417→  c.moving = true;
   418→  c.heading = lerpAngle(c.heading, Math.atan2(dx, dz), Math.min(1, dt * 10));
   419→  collide(c);
   420→  return false;
   421→}

   423→/** 沿路点前进；返回是否已消费完路径 */
   424→function followPath(c: CharRT, path: Pt[], dt: number, collide: (c: CharRT) => void): boolean {
   425→  const wp = path[0];
   426→  if (!wp) {
   427→    c.moving = false;
   428→    return true;
   429→  }
   430→  const dx = wp.x - c.x;
   431→  const dz = wp.z - c.z;
   432→  const d = Math.hypot(dx, dz);
   433→  if (d < 0.2) {
   434→    path.shift();
   435→    return path.length === 0;
   436→  }
   437→  const sp = Math.min(WALK_SPEED * dt, d);
   438→  c.x += (dx / d) * sp;
   439→  c.z += (dz / d) * sp;
   440→  c.moving = true;
   441→  c.heading = lerpAngle(c.heading, Math.atan2(dx, dz), Math.min(1, dt * 10));
   442→  collide(c);
   443→  return false;
   444→}

   463→/** 用户姿势是否可被陪伴系统接管（正在坐/躺/泡澡时不打断用户自己的动作） */
   464→function userPoseReleasable(poseRef: PoseSlot): boolean {
   465→  return poseRef.pose === 'stand' || poseRef.pose === 'hug' || poseRef.pose === 'holdhand';
   466→}
   467→
   468→/** 粘性姿势时长（秒）：cmd.dur 显式指定（0 = 无限保持），否则按姿势默认随机 */
   469→function stickyDur(cmd: CompanionCmd, kind: CompanionCmd['kind']): number {
   470→  if (typeof cmd.dur === 'number') return cmd.dur <= 0 ? Number.POSITIVE_INFINITY : cmd.dur;
   471→  switch (kind) {
   472→    case 'cook':
   473→      return DUR_COOK;
   474→    case 'play':
   475→      return DUR_PLAY[0] + Math.random() * (DUR_PLAY[1] - DUR_PLAY[0]);
   476→    case 'music':
   477→      return DUR_MUSIC[0] + Math.random() * (DUR_MUSIC[1] - DUR_MUSIC[0]);
   478→    case 'eat':
   479→      return DUR_EAT[0] + Math.random() * (DUR_EAT[1] - DUR_EAT[0]);
   480→    // —— v10 新动作 ——
   481→    case 'dance':
   482→      return 14 + Math.random() * 8;
   483→    case 'stretch':
   484→      return 4.5 + Math.random() * 2;
   485→    case 'water':
   486→      return 7 + Math.random() * 4;
   487→    case 'read':
   488→      return 30 + Math.random() * 25;
   489→    default:
   490→      return SIT_MIN + Math.random() * (SIT_MAX - SIT_MIN);
   491→  }
   492→}
   493→
   494→/**
   495→ * 每帧推进陪伴状态机。
   496→ * 返回 true = 本帧由陪伴逻辑接管（GameLoop 跳过 roam）。
   497→ */
   498→export function companionStep(
   499→  ta: CharRT,
   500→  me: CharRT,
   501→  rt: CompanionRT,
   502→  dt: number,
   503→  items: FurnitureItem[],
   504→  grid: BlockGrid,
   505→  doors: Partial<Record<DoorKey, DoorGate>>,
   506→  partnerPose: PoseSlot,
   507→  poseRef: PoseSlot,
   508→  collide: (c: CharRT) => void,
   509→  joyMag: number,
   510→  onEvent: (e: CompanionEvent) => void,
   511→): boolean {
   512→  // —— 消费新指令（seq 变化即触发） ——
   513→  if (rt.cmd && rt.cmd.seq !== rt.lastSeq) {
   514→    rt.lastSeq = rt.cmd.seq;
   515→    const cmd = rt.cmd;
   516→    if (cmd.kind === 'free') {
   517→      toIdle(rt, partnerPose);
   518→      if (userPoseReleasable(poseRef)) poseRef.pose = 'stand';
   519→      return true;
   520→    }
   521→    // 起手：起身（从坐/躺/牵手状态被叫起）
   522→    partnerPose.pose = 'stand';
   523→    partnerPose.y = 0;
   524→    if (userPoseReleasable(poseRef) && cmd.kind !== 'hug') poseRef.pose = 'stand';
   525→    rt.phase = 'walk';
   526→    rt.actT = 0;
   527→    rt.stuckT = 0;
   528→    rt.repathed = false;
   529→    rt.repaths = 0;
   530→    rt.stuckTotal = 0;
   531→    rt.sitUntil = 0;
   532→    rt.anchor = null;
   533→    rt.itemId = null;
   534→    // —— v10 原地动作：跳舞/伸懒腰（不需要走位，直接进粘性姿势） ——
   535→    if (cmd.kind === 'dance' || cmd.kind === 'stretch') {
   536→      ta.moving = false;
   537→      rt.phase = 'sit';
   538→      rt.actT = 0;
   539→      rt.sitUntil = rt.actT + stickyDur(cmd, cmd.kind);
   540→      partnerPose.pose = cmd.kind as CharPose;
   541→      partnerPose.y = 0;
   542→      onEvent({ ev: 'arrive', kind: cmd.kind, memo: cmd.memo });
   543→      return true;
   544→    }
   545→    // —— v10 开关窗：走到目标房间（缺省当前房间）空地 → 回报宿主开/关窗 ——
   546→    if (cmd.kind === 'window') {
   547→      const room = cmd.room ?? roomOf(ta.x, ta.z);
   548→      const dest = randPointInRoomFree(grid, room);
   549→      const route = planRoute(roomOf(ta.x, ta.z), room, doors);
   550→      rt.path = route ? [...route, dest] : (findPath(grid, { x: ta.x, z: ta.z }, dest) ?? []);
   551→      rt.target = dest;
   552→      return true;
   553→    }
   554→    // —— 计算目标与路径 ——
   555→    if (cmd.kind === 'come' || cmd.kind === 'hug' || cmd.kind === 'holdhand') {
   556→      const dx = ta.x - me.x;
   557→      const dz = ta.z - me.z;
   558→      const d = Math.max(0.001, Math.hypot(dx, dz));
   559→      const target = { x: me.x + (dx / d) * 0.6, z: me.z + (dz / d) * 0.6 };
   560→      rt.target = target;
   561→      rt.path = findPath(grid, { x: ta.x, z: ta.z }, target) ?? [];
   562→    } else if (cmd.kind === 'goto') {
   563→      const room = cmd.room ?? 'lr';
   564→      const dest = randPointInRoomFree(grid, room);
   565→      const route = planRoute(roomOf(ta.x, ta.z), room, doors);
   566→      rt.path = route ? [...route, dest] : (findPath(grid, { x: ta.x, z: ta.z }, dest) ?? []);
   567→      rt.target = dest;
   568→    } else if (cmd.kind === 'door') {
   569→      // 走到门旁：itemId = 门 id；缺省 = 最近一扇关着的门
   570→      const key = (['bed', 'kt', 'bath', 'bal'] as const).includes(cmd.itemId as 'bed')
   571→        ? (cmd.itemId as 'bed' | 'kt' | 'bath' | 'bal')
   572→        : nearestClosedDoor(doors, { x: ta.x, z: ta.z });
   573→      const gate = key ? doors[key] : undefined;
   574→      if (!key || !gate || gate.open) {
   575→        onEvent({ ev: 'giveup', kind: 'door', memo: cmd.memo ?? 'all-open' });
   576→        toIdle(rt, partnerPose);
   577→        return true;
   578→      }
   579→      rt.itemId = key;
   580→      const pts = doorPoints(key, gate.along);
   581→      const here = roomOf(ta.x, ta.z);
   582→      const p = roomOf(pts.aSide.x, pts.aSide.z) === here ? pts.aSide : pts.bSide;
   583→      rt.target = p;
   584→      rt.path = findPath(grid, { x: ta.x, z: ta.z }, p) ?? [];
   585→    } else {
   586→      // sit/lie/watch/lamp/play/cook/music/eat：解析家具 + 锚点
   587→      let item = cmd.itemId ? items.find((f) => f.id === cmd.itemId) ?? null : null;
   588→      if ((cmd.kind === 'sit' || cmd.kind === 'read') && !item) item = nearestSeat(items, { x: ta.x, z: ta.z });
   589→      if (cmd.kind === 'lie' && !item) item = nearestBed(items, { x: ta.x, z: ta.z });
   590→      if (cmd.kind === 'watch') item = items.find((f) => f.kind === 'tv') ?? item;
   591→      if (cmd.kind === 'play' && !item) item = nearestInstrument(items, { x: ta.x, z: ta.z });
   592→      if (cmd.kind === 'music' && !item) item = nearestMusicBox(items, { x: ta.x, z: ta.z });
   593→      if (cmd.kind === 'cook' && !item) item = nearestStove(items, { x: ta.x, z: ta.z });
   594→      if (cmd.kind === 'eat' && !item) item = nearestDiningTable(items, { x: ta.x, z: ta.z });
   595→      if (cmd.kind === 'water' && !item) item = nearestPlant(items, { x: ta.x, z: ta.z });
   596→      if (cmd.kind === 'lamp' && !item) {
   597→        item =
   598→          items.find((f) => CATALOG[f.kind]?.light && (f.y ?? 0) < 0.5) ??
   599→          items.find((f) => CATALOG[f.kind]?.light) ??
   600→          null;
   601→      }
   602→      if (!item) {
   603→        onEvent({ ev: 'giveup', kind: cmd.kind, memo: cmd.memo });
   604→        toIdle(rt, partnerPose);
   605→        return true;
   606→      }
   607→      const anchor = companionAnchor(
   608→        cmd.kind === 'lie'
   609→          ? 'lie'
   610→          : cmd.kind === 'watch'
   611→            ? 'watch'
   612→            : cmd.kind === 'lamp'
   613→              ? 'lamp'
   614→              : cmd.kind === 'music'
   615→                ? 'music'
   616→                : cmd.kind === 'play'
   617→                  ? 'play'
   618→                  : cmd.kind === 'cook'
   619→                    ? 'cook'
   620→                    : cmd.kind === 'eat'
   621→                      ? 'eat'
   622→                      : cmd.kind === 'read'
   623→                        ? 'read'
   624→                        : cmd.kind === 'water'
   625→                          ? 'water'
   626→                          : 'sit',
   627→        item,
   628→      );
   629→      if (!anchor) {
   630→        onEvent({ ev: 'giveup', kind: cmd.kind, memo: cmd.memo });
   631→        toIdle(rt, partnerPose);
   632→        return true;
   633→      }
   634→      rt.anchor = anchor;
   635→      rt.itemId = item.id;
   636→      rt.target = { x: anchor.tx, z: anchor.tz };
   637→      rt.path = findPath(grid, { x: ta.x, z: ta.z }, { x: anchor.tx, z: anchor.tz }) ?? [];
   638→    }
   639→    return true;
   640→  }
   641→
   642→  if (!rt.cmd || rt.phase === 'idle') return false;
   643→  const cmd = rt.cmd;
   644→
   645→  switch (rt.phase) {
   646→    case 'walk': {
   647→      const t0 = rt.target;
   648→      if (!t0) {
   649→        toIdle(rt, partnerPose);
   650→        return true;
   651→      }
   652→      const done = followPath(ta, rt.path, dt, collide);
   653→      if (!done) stepToward(ta, t0.x, t0.z, ARRIVE, dt, WALK_SPEED, collide);
   654→      // —— 卡死检测：0.9s 位移 < 0.05 → 自救阶梯（开门 → 重算×2 → 传送兜底，绝不放弃任务） ——
   655→      const moved = Math.hypot(ta.x - rt.stuckX, ta.z - rt.stuckZ);
   656→      if (moved > 0.05) {
   657→        rt.stuckT = 0;
   658→        rt.stuckTotal = 0;
   659→        rt.stuckX = ta.x;
   660→        rt.stuckZ = ta.z;
   661→      } else {
   662→        rt.stuckT += dt;
   663→        rt.stuckTotal += dt;
   664→        if (rt.stuckT > 0.9) {
   665→          // ① 身边有关着的门 → 自己开门（回报宿主开哪扇），门开后重算路径接着走
   666→          const gate = closedDoorNear(doors, { x: ta.x, z: ta.z });
   667→          if (gate) {
   668→            onEvent({ ev: 'arrive', kind: 'door', itemId: gate, memo: 'auto-open' });
   669→            rt.repathed = false; // 门开了允许再重算一次路径
   670→            rt.stuckT = 0;
   671→            rt.stuckX = ta.x;
   672→            rt.stuckZ = ta.z;
   673→          } else if (rt.repaths < 2) {
   674→            // ② 重算路径（最多 2 次：等门开/等家具状态刷新后常常就通了）
   675→            rt.repaths += 1;
   676→            rt.stuckT = 0;
   677→            rt.stuckX = ta.x;
   678→            rt.stuckZ = ta.z;
   679→            rt.path = findPath(grid, { x: ta.x, z: ta.z }, t0) ?? [];
   680→          } else if (rt.stuckTotal > 3) {
   681→            // ③ 传送兜底：落到目标点最近可达格 —— 视觉上瞬移一小步，但任务一定完成
   682→            const spot = nearestFree(grid.blocked, t0.x, t0.z);
   683→            ta.x = spot.x;
   684→            ta.z = spot.z;
   685→            ta.moving = false;
   686→            rt.path = [];
   687→            rt.stuckT = 0;
   688→            rt.stuckTotal = 0;
   689→            rt.repaths = 0;
   690→            rt.repathed = false;
   691→            rt.stuckX = ta.x;
   692→            rt.stuckZ = ta.z;
   693→          }
   694→        }
   695→      }
   696→      const remain = Math.hypot(t0.x - ta.x, t0.z - ta.z);
   697→      if (remain < ARRIVE + 0.06) {
   698→        // —— 到位：按指令分流 ——
   699→        ta.moving = false;
   700→        rt.stuckT = 0;
   701→        rt.stuckTotal = 0;
   702→        rt.repaths = 0;
   703→        rt.repathed = false;
   704→        if (
   705→          cmd.kind === 'sit' ||
   706→          cmd.kind === 'lie' ||
   707→          cmd.kind === 'watch' ||
   708→          cmd.kind === 'play' ||
   709→          cmd.kind === 'cook' ||
   710→          cmd.kind === 'eat' ||
   711→          cmd.kind === 'read' ||
   712→          cmd.kind === 'water'
   713→        ) {
   714→          const a = rt.anchor;
   715→          if (a) {
   716→            ta.x = a.tx;
   717→            ta.z = a.tz;
   718→            ta.heading = a.ry;
   719→          }
   720→          // play 按乐器分流姿势：钢琴坐姿 / 吉他站姿
   721→          const it = items.find((f) => f.id === rt.itemId);
   722→          partnerPose.pose =
   723→            cmd.kind === 'lie'
   724→              ? 'lie'
   725→              : cmd.kind === 'play'
   726→                ? it?.kind === 'guitar'
   727→                  ? 'guitar'
   728→                  : 'piano'
   729→                : cmd.kind === 'cook'
   730→                  ? 'cook'
   731→                  : cmd.kind === 'eat'
   732→                    ? 'eat'
   733→                    : cmd.kind === 'read'
   734→                      ? 'read'
   735→                      : cmd.kind === 'water'
   736→                        ? 'water'
   737→                        : 'sit';
   738→          partnerPose.y = a?.y ?? 0;
   739→          rt.phase = 'sit';
   740→          rt.sitUntil = rt.actT + stickyDur(cmd, cmd.kind);
   741→          rt.actT = 0;
   742→          onEvent({ ev: 'arrive', kind: cmd.kind, itemId: rt.itemId ?? undefined, memo: cmd.memo, song: cmd.song });
   743→        } else if (cmd.kind === 'lamp' || cmd.kind === 'music') {
   744→          if (rt.anchor) ta.heading = rt.anchor.ry;
   745→          onEvent({ ev: 'arrive', kind: cmd.kind, itemId: rt.itemId ?? undefined, memo: cmd.memo, song: cmd.song });
   746→          toIdle(rt, partnerPose);
   747→        } else if (cmd.kind === 'door') {
   748→          if (rt.target) ta.heading = Math.atan2(rt.target.x - ta.x, rt.target.z - ta.z + 0.6);
   749→          onEvent({ ev: 'arrive', kind: 'door', itemId: rt.itemId ?? undefined, memo: cmd.memo });
   750→          toIdle(rt, partnerPose);
   751→        } else if (cmd.kind === 'come') {
   752→          ta.heading = Math.atan2(me.x - ta.x, me.z - ta.z);
   753→          onEvent({ ev: 'arrive', kind: 'come', memo: cmd.memo });
   754→          toIdle(rt, partnerPose);
   755→        } else if (cmd.kind === 'hug') {
   756→          rt.phase = 'hug';
   757→          rt.actT = 0;
   758→          partnerPose.pose = 'hug';
   759→          if (userPoseReleasable(poseRef)) poseRef.pose = 'hug';
   760→          onEvent({ ev: 'hugStart', kind: 'hug', memo: cmd.memo });
   761→        } else if (cmd.kind === 'holdhand') {
   762→          rt.phase = 'hold';
   763→          rt.actT = 0;
   764→          rt.holdAt = 0;
   765→          rt.holdRoom = null;
   766→          partnerPose.pose = 'holdhand';
   767→          onEvent({ ev: 'holdStart', kind: 'holdhand', memo: cmd.memo });
   768→        } else if (cmd.kind === 'window') {
   769→          onEvent({ ev: 'arrive', kind: 'window', room: cmd.room, memo: cmd.memo ?? (cmd.open === true ? 'open' : 'close') });
   770→          toIdle(rt, partnerPose);
   771→        } else if (cmd.kind === 'goto') {
   772→          // 跨房间到位：回报（多步队列接续 / 自主行为记忆用）
   773→          onEvent({ ev: 'arrive', kind: 'goto', room: cmd.room, memo: cmd.memo });
   774→          toIdle(rt, partnerPose);
   775→        } else {
   776→          toIdle(rt, partnerPose);
   777→        }
   778→      }
   779→      return true;
   780→    }
   781→    case 'sit': {
   782→      // —— 粘性姿势：到点自行起身（Infinity = 夜睡/长驻，等被叫起）；新指令在上方消费块里已处理 ——
   783→      rt.actT += dt;
   784→      if (rt.sitUntil !== Number.POSITIVE_INFINITY && rt.actT >= rt.sitUntil) {
   785→        onEvent({ ev: 'release', kind: cmd.kind, itemId: rt.itemId ?? undefined, memo: 'finish' });
   786→        toIdle(rt, partnerPose);
   787→      }
   788→      return true;
   789→    }
   790→    case 'hug': {
   791→      // —— 拥抱保持：贴在用户身前；用户走动 → 自动松开 ——
   792→      rt.actT += dt;
   793→      const dx = me.x - ta.x;
   794→      const dz = me.z - ta.z;
   795→      const d = Math.hypot(dx, dz);
   796→      if (d > HUG_DIST + 0.08) stepToward(ta, me.x, me.z, HUG_DIST, dt, WALK_SPEED * 1.2, collide);
   797→      else {
   798→        ta.moving = false;
   799→        // v9 面对面抱抱：贴身后用户也转身面向 TA（双向对视，不再抱后背）
   800→        me.heading = lerpAngle(me.heading, Math.atan2(ta.x - me.x, ta.z - me.z), Math.min(1, dt * 5));
   801→      }
   802→      ta.heading = Math.atan2(dx, dz);
   803→      if (joyMag > 0.3 || rt.actT > HUG_MAX || d > 1.3) {
   804→        partnerPose.pose = 'stand';
   805→        if (poseRef.pose === 'hug') poseRef.pose = 'stand';
   806→        onEvent({ ev: 'release', kind: 'hug', memo: cmd.memo });
   807→        toIdle(rt, partnerPose);
   808→      }
   809→      return true;
   810→    }
   811→    case 'hold': {
   812→      // —— 牵手跟随：保持 0.5m 距离 + 防穿模；进新房间回报 room 事件 ——
   813→      rt.actT += dt;
   814→      const dx = me.x - ta.x;
   815→      const dz = me.z - ta.z;
   816→      const d = Math.hypot(dx, dz);
   817→      if (d > HOLD_DIST + 0.12) {
   818→        if (d > 1.15) {
   819→          // 较远（可能隔家具/门）：A* 绕障（节流 0.6s 重算）+ 直奔兜底
   820→          if (!rt.path.length || rt.actT - rt.holdAt > 0.6) {
   821→            rt.holdAt = rt.actT;
   822→            rt.path = findPath(grid, { x: ta.x, z: ta.z }, { x: me.x, z: me.z }) ?? [];
   823→          }
   824→          followPath(ta, rt.path, dt, collide);
   825→          if (Math.hypot(me.x - ta.x, me.z - ta.z) > HOLD_DIST + 0.1) {
   826→            stepToward(ta, me.x, me.z, HOLD_DIST, dt, WALK_SPEED * 1.5, collide);
   827→          }
   828→        } else {
   829→          stepToward(ta, me.x, me.z, HOLD_DIST, dt, WALK_SPEED * 1.5, collide);
   830→        }
   831→      } else if (d < HOLD_DIST - 0.14) {
   832→        // 太近：退半步防穿模
   833→        const inv = 1 / Math.max(0.001, d);
   834→        ta.x -= dx * inv * dt * 0.9;
   835→        ta.z -= dz * inv * dt * 0.9;
   836→        ta.moving = false;
   837→        collide(ta);
   838→      } else {
   839→        ta.moving = false;
   840→      }
   841→      ta.heading = lerpAngle(ta.heading, Math.atan2(me.x - ta.x, me.z - ta.z), Math.min(1, dt * (ta.moving ? 6 : 8)));
   842→      const room = roomOf(ta.x, ta.z);
   843→      if (rt.holdRoom === null) rt.holdRoom = room;
   844→      else if (room !== rt.holdRoom) {
   845→        rt.holdRoom = room;
   846→        onEvent({ ev: 'room', kind: 'holdhand', room, memo: cmd.memo });
   847→      }
   848→      return true;
   849→    }
   850→    default:
   851→      return false;
   852→  }
   853→}

   855→/** 陪伴小人是否空闲（宿主自主行动调度用：无指令且不在坐姿/拥抱/牵手） */
   856→export function companionIdle(rt: CompanionRT): boolean {
   857→  return rt.phase === 'idle' && !rt.cmd;
   858→}
   859→
