   148→interface SettingsState {
   149→  theme: ThemeMode;
   150→  wallpaperPreset: string;
   151→  /** 自定义壁纸的 ObjectURL（Blob 存 IndexedDB） */
   152→  customWallpaperUrl: string | null;
   153→  /** 锁屏壁纸预设（锁屏壁纸与主屏幕完全独立，各有各的预设/自定义） */
   154→  lockWallpaperPreset: string;
   155→  /** 锁屏自定义壁纸的 ObjectURL（Blob 存 IndexedDB） */
   156→  lockCustomWallpaperUrl: string | null;
   157→  apiConfig: ApiConfig;
   158→  /** 用户自己保存的 API 预设 */
   159→  apiPresets: ApiPreset[];
   160→  /** 识图模型配置（与 apiConfig 互相独立、互不覆盖；密文持久化） */
   161→  visionConfig: VisionConfig;
   162→  /** 用户自己保存的识图模型预设 */
   163→  visionPresets: VisionPreset[];
   164→  /** 锁屏密码配置（持久化） */
   165→  lockConfig: LockConfig;
   166→  /** 个人信息（头像/名字/标签，持久化） */
   167→  profile: Profile;
   168→  /** 自定义 App 图标（AppId → ObjectURL，Blob 存 IndexedDB settings.customIcons） */
   169→  customIcons: Record<string, string>;
   170→  loaded: boolean;
   171→
   172→  load: () => Promise<void>;
   173→  /** 锁屏总开关：关闭时同时停用密码并立即回到主屏幕 */
   174→  setLockScreen: (on: boolean) => void;
   175→  setTheme: (t: ThemeMode) => void;
   176→  setWallpaperPreset: (id: string) => void;
   177→  setCustomWallpaper: (blob: Blob | null) => void;
   178→  /** 设置锁屏壁纸预设（独立于主屏幕壁纸） */
   179→  setLockWallpaperPreset: (id: string) => void;
   180→  /** 设置锁屏自定义壁纸（null=移除自定义，回退到锁屏已选预设） */

   189→  setProfile: (patch: Partial<Profile>) => void;
   190→  /** 设置/移除某 App 的自定义图标（blob=null 恢复默认），同步持久化到 IndexedDB */

   192→  /** 清空全部自定义图标（全部恢复默认） */
   193→  resetAllCustomIcons: () => void;

   196→export const useSettings = create<SettingsState>((set, get) => ({
   197→  // 主题默认浅色（用户要求）；load 时无记录/一次性迁移后也回退到浅色
   198→  theme: 'light',
   199→  wallpaperPreset: 'graphite',
   200→  customWallpaperUrl: null,
   201→  lockWallpaperPreset: 'graphite',
   202→  lockCustomWallpaperUrl: null,
   203→  apiConfig: { ...DEFAULT_API_CONFIG },
   204→  apiPresets: [],
   205→  visionConfig: { ...DEFAULT_VISION_CONFIG },
   206→  visionPresets: [],
   207→  lockConfig: { lockScreen: true, enabled: false, code: '', len: 4 },
   208→  profile: { ...DEFAULT_PROFILE },
   209→  customIcons: {},
   210→  loaded: false,
   211→
   212→  load: async () => {
   213→    if (get().loaded) return;
   214→    try {
   215→      const [themeRec, wallpaperRec, lockWallpaperRec, apiRec, presetsRec, visionRec, visionPresetsRec, lockRec, profileRec, iconsRec] = await Promise.all([
   216→        localDB.get('settings', 'theme'),
   217→        localDB.get('settings', 'wallpaper'),
   218→        localDB.get('settings', 'lockWallpaper'),
   219→        localDB.get('settings', 'apiConfig'),
   220→        localDB.get('settings', 'apiPresets'),
   221→        localDB.get('settings', 'visionConfig'),
   222→        localDB.get('settings', 'visionPresets'),
   223→        localDB.get('settings', 'lock'),
   224→        localDB.get('settings', 'profile'),
   225→        localDB.get('settings', 'customIcons'),
   226→      ]);
   227→
   228→      // 自定义 App 图标：{ AppId: Blob } → 为每个 Blob 建 ObjectURL
   229→      let customIcons: Record<string, string> = {};
   230→      if (iconsRec && typeof iconsRec.value === 'object' && iconsRec.value !== null) {
   231→        for (const [k, v] of Object.entries(iconsRec.value as Record<string, unknown>)) {
   232→          if (v instanceof Blob) customIcons[k] = URL.createObjectURL(v);
   233→        }
   234→      }
   235→
   236→      let profile: Profile = { ...DEFAULT_PROFILE };
   237→      if (profileRec && typeof profileRec.value === 'object' && profileRec.value !== null) {
   238→        const v = profileRec.value as Partial<Profile>;
   239→        profile = {
   240→          name: typeof v.name === 'string' ? v.name : '',
   241→          tag: typeof v.tag === 'string' ? v.tag : DEFAULT_PROFILE.tag,
   242→          avatar:
   243→            typeof v.avatar === 'string' && v.avatar.startsWith('data:image/') && v.avatar.length < 600_000
   244→              ? v.avatar
   245→              : null,
   246→        };
   247→      }
   248→
   249→      let lockConfig: LockConfig = { lockScreen: true, enabled: false, code: '', len: 4 };
   250→      if (lockRec && typeof lockRec.value === 'object' && lockRec.value !== null) {
   251→        const v = lockRec.value as Partial<LockConfig>;
   252→        lockConfig = {
   253→          lockScreen: v.lockScreen !== false, // 旧记录无此字段 → 默认开启
   254→          enabled: v.enabled === true,
   255→          code: typeof v.code === 'string' ? v.code : '',
   256→          len: v.len === 6 ? 6 : 4,
   257→        };
   258→      }
   259→
   260→      // 主题默认浅色（用户要求）：无记录回退浅色；历史存储里的「深色」一次性重置为浅色
   261→      //（否则用户之前测试时持久化的 dark 会让默认值改动不可见；重置后用户再手动切深色仍可保留）
   262→      let theme: ThemeMode = 'light';
   263→      if (themeRec && (themeRec.value === 'light' || themeRec.value === 'dark' || themeRec.value === 'auto')) {
   264→        theme = themeRec.value;
   265→      }
   266→      const themeResetRec = await localDB.get('settings', 'themeDefaultLight');
   267→      const themeReset = !themeResetRec;
   268→      if (themeReset && theme === 'dark') {
   269→        theme = 'light';
   270→        void localDB.put('settings', { key: 'theme', value: 'light' });
   271→      }
   272→      if (themeReset) void localDB.put('settings', { key: 'themeDefaultLight', value: true });
   273→
   274→      let wallpaperPreset = 'graphite';
   275→      let customWallpaperUrl: string | null = null;
   276→      if (wallpaperRec && typeof wallpaperRec.value === 'object' && wallpaperRec.value !== null) {
   277→        const w = wallpaperRec.value as { preset?: unknown; customBlob?: unknown };
   278→        if (w.customBlob instanceof Blob) {
   279→          customWallpaperUrl = URL.createObjectURL(w.customBlob);
   280→        } else if (typeof w.preset === 'string' && WALLPAPER_PRESETS.some((p) => p.id === w.preset)) {
   281→          wallpaperPreset = w.preset;
   282→        }
   283→      }
   284→
   285→      // 锁屏壁纸（完全独立，默认 graphite）：{ preset, customBlob } 同时记录，移除自定义后可回退到已选预设
   286→      let lockWallpaperPreset = 'graphite';
   287→      let lockCustomWallpaperUrl: string | null = null;
   288→      if (lockWallpaperRec && typeof lockWallpaperRec.value === 'object' && lockWallpaperRec.value !== null) {
   289→        const lw = lockWallpaperRec.value as LockWallpaperRecord;
   290→        if (typeof lw.preset === 'string' && WALLPAPER_PRESETS.some((p) => p.id === lw.preset)) {
   291→          lockWallpaperPreset = lw.preset;
   292→        }
   293→        if (lw.customBlob instanceof Blob) {
   294→          lockCustomWallpaperUrl = URL.createObjectURL(lw.customBlob);
   295→        }
   296→      }
   297→
   298→      let apiConfig: ApiConfig = { ...DEFAULT_API_CONFIG };
   299→      if (apiRec && typeof apiRec.value === 'object' && apiRec.value !== null) {
   300→        // 兼容：旧版明文记录 / 新版密文信封都从这里解出
   301→        const v = ((await decryptValue<Partial<ApiConfig>>(apiRec.value)) ?? apiRec.value) as Partial<ApiConfig>;
   302→        apiConfig = {
   303→          baseUrl: typeof v.baseUrl === 'string' && v.baseUrl ? v.baseUrl : DEFAULT_API_CONFIG.baseUrl,
   304→          apiKey: typeof v.apiKey === 'string' ? v.apiKey : '',
   305→          model: typeof v.model === 'string' && v.model ? v.model : DEFAULT_API_CONFIG.model,
   306→          temperature: typeof v.temperature === 'number' ? v.temperature : DEFAULT_API_CONFIG.temperature,
   307→          maxTokens: typeof v.maxTokens === 'number' ? v.maxTokens : DEFAULT_API_CONFIG.maxTokens,
   308→        };
   309→        // 旧版明文静默升级：立即以密文回写（有 apiKey 才有必要；无 key 的占位配置不动）
   310→        const wasPlain = !('__enc' in (apiRec.value as object));
   311→        if (wasPlain && typeof apiConfig.apiKey === 'string' && apiConfig.apiKey.length > 0) {
   312→          void encryptValue(apiConfig)
   313→            .then((enc) => localDB.put('settings', { key: 'apiConfig', value: enc }))
   314→            .catch(() => undefined);
   315→        }
   316→      }
   317→
   318→      let apiPresets: ApiPreset[] = [];
   319→      if (presetsRec && presetsRec.value) {
   320→        // 兼容：旧版明文数组 / 新版密文信封
   321→        const pv = (await decryptValue<ApiPreset[]>(presetsRec.value)) ?? presetsRec.value;
   322→        if (Array.isArray(pv)) {
   323→          apiPresets = pv.filter(
   324→            (p): p is ApiPreset =>
   325→              typeof p === 'object' &&
   326→              p !== null &&
   327→              typeof (p as ApiPreset).id === 'string' &&
   328→              typeof (p as ApiPreset).name === 'string' &&
   329→              typeof (p as ApiPreset).config === 'object'
   330→          );
   331→          // 旧版明文含 apiKey 时静默升级为密文
   332→          const hasKey = apiPresets.some((p) => typeof p.config?.apiKey === 'string' && p.config.apiKey.length > 0);
   333→          if (!('__enc' in (presetsRec.value as object)) && hasKey) {
   334→            void encryptValue(apiPresets)
   335→              .then((enc) => localDB.put('settings', { key: 'apiPresets', value: enc }))
   336→              .catch(() => undefined);
   337→          }
   338→        }
   339→      }
   340→
   341→      // 识图模型配置（含 apiKey，密文信封，与 apiConfig 同策略；未配置 = baseUrl 空）
   342→      let visionConfig: VisionConfig = { ...DEFAULT_VISION_CONFIG };
   343→      if (visionRec && typeof visionRec.value === 'object' && visionRec.value !== null) {
   344→        const v = ((await decryptValue<Partial<VisionConfig>>(visionRec.value)) ?? visionRec.value) as Partial<VisionConfig>;
   345→        visionConfig = {
   346→          baseUrl: typeof v.baseUrl === 'string' ? v.baseUrl : '',
   347→          apiKey: typeof v.apiKey === 'string' ? v.apiKey : '',
   348→          model: typeof v.model === 'string' ? v.model : '',
   349→        };
   350→        const visionWasPlain = !('__enc' in (visionRec.value as object));
   351→        if (visionWasPlain && visionConfig.apiKey.length > 0) {
   352→          void encryptValue(visionConfig)
   353→            .then((enc) => localDB.put('settings', { key: 'visionConfig', value: enc }))
   354→            .catch(() => undefined);
   355→        }
   356→      }
   357→
   358→      let visionPresets: VisionPreset[] = [];
   359→      if (visionPresetsRec && visionPresetsRec.value) {
   360→        const vv = (await decryptValue<VisionPreset[]>(visionPresetsRec.value)) ?? visionPresetsRec.value;
   361→        if (Array.isArray(vv)) {
   362→          visionPresets = vv.filter(
   363→            (p): p is VisionPreset =>
   364→              typeof p === 'object' &&
   365→              p !== null &&
   366→              typeof (p as VisionPreset).id === 'string' &&
   367→              typeof (p as VisionPreset).name === 'string' &&
   368→              typeof (p as VisionPreset).config === 'object'
   369→          );
   370→          const visionHasKey = visionPresets.some((p) => typeof p.config?.apiKey === 'string' && p.config.apiKey.length > 0);
   371→          if (!('__enc' in (visionPresetsRec.value as object)) && visionHasKey) {
   372→            void encryptValue(visionPresets)
   373→              .then((enc) => localDB.put('settings', { key: 'visionPresets', value: enc }))
   374→              .catch(() => undefined);
   375→          }
   376→        }
   377→      }
   378→
   379→      // 开机门控收尾前预载主屏/锁屏壁纸（最多等 1.5s 兜底）：
   380→      // 锁屏/主屏挂载首帧即完整壁纸 + 正确前景色，修复刚打开网页时锁屏壁纸「闪一下」
   381→      const bootUrls: (string | null | undefined)[] = [customWallpaperUrl, lockCustomWallpaperUrl];
   382→      for (const pid of [wallpaperPreset, lockWallpaperPreset]) {
   383→        const bg = resolveWallpaperStyle(pid, null).backgroundImage;
   384→        if (bg) bootUrls.push(bgImageUrlOf(bg));
   385→      }
   386→      await preloadWallpapersForBoot(bootUrls);
   387→
   388→      set({
   389→        theme,
   390→        wallpaperPreset,
   391→        customWallpaperUrl,
   392→        lockWallpaperPreset,
   393→        lockCustomWallpaperUrl,
   394→        apiConfig,
   395→        apiPresets,
   396→        visionConfig,
   397→        visionPresets,
   398→        lockConfig,
   399→        profile,
   400→        customIcons,
   401→        loaded: true,
   402→      });
   403→      // 锁屏被用户关闭：本次开机直接进主屏幕（跳过锁屏）
   404→      if (!lockConfig.lockScreen) useUI.setState({ locked: false });
   405→    } catch {
   406→      set({ loaded: true });
   407→    }
   408→  },
   409→
   410→  setLockScreen: (on) => {
   411→    const cur = get().lockConfig;
   412→    const cfg: LockConfig = on
   413→      ? { ...cur, lockScreen: true }
   414→      : { lockScreen: false, enabled: false, code: '', len: cur.len }; // 关锁屏即停用密码
   415→    set({ lockConfig: cfg });
   416→    void localDB.put('settings', { key: 'lock', value: { ...cfg } });
   417→    if (!on) useUI.setState({ locked: false, screenOff: false, lockCameraOpen: false, torchOpen: false });
   418→  },
   419→
   420→  setTheme: (t) => {
   421→    set({ theme: t });
   422→    void localDB.put('settings', { key: 'theme', value: t });
   423→  },
   424→
   425→  setWallpaperPreset: (id) => {
   426→    const old = get().customWallpaperUrl;
   427→    if (old) URL.revokeObjectURL(old);
   428→    set({ wallpaperPreset: id, customWallpaperUrl: null });
   429→    void localDB.put('settings', { key: 'wallpaper', value: { preset: id } });
   430→  },
   431→
   432→  setCustomWallpaper: (blob) => {
   433→    const old = get().customWallpaperUrl;
   434→    if (old) URL.revokeObjectURL(old);
   435→    if (!blob) {
   436→      set({ customWallpaperUrl: null });
   437→      void localDB.put('settings', { key: 'wallpaper', value: { preset: get().wallpaperPreset } });
   438→      return;
   439→    }
   440→    const url = URL.createObjectURL(blob);
   441→    set({ customWallpaperUrl: url });
   442→    void localDB.put('settings', { key: 'wallpaper', value: { customBlob: blob } });
   443→  },
   444→
   445→  setLockWallpaperPreset: (id) => {
   446→    const old = get().lockCustomWallpaperUrl;
   447→    if (old) URL.revokeObjectURL(old);
   448→    set({ lockWallpaperPreset: id, lockCustomWallpaperUrl: null });
   449→    void localDB.put('settings', { key: 'lockWallpaper', value: { preset: id } });
   450→  },
   451→
   452→  setLockCustomWallpaper: (blob) => {
   453→    const old = get().lockCustomWallpaperUrl;
   454→    if (old) URL.revokeObjectURL(old);
   455→    const preset = get().lockWallpaperPreset;
   456→    if (!blob) {

   549→/** 由预设 id + 自定义 URL 解析壁纸背景样式（手机壳/锁屏/前景明暗共用，故导出）。

   552→ *  否则锁屏/主屏头几帧透明，会透出下层界面（刷新时主界面一闪而过的根因） */
   553→export function resolveWallpaperStyle(presetId: string, customUrl: string | null): CSSProperties {
   554→  if (customUrl) {
   555→    return {
   556→      backgroundColor: '#1c1c1e',
   557→      backgroundImage: `url(${customUrl})`,
   558→      backgroundSize: 'cover',
   559→      backgroundPosition: 'center',
   560→    };
   561→  }
   562→  const preset = WALLPAPER_PRESETS.find((w) => w.id === presetId) ?? WALLPAPER_PRESETS[0];
   563→  // 预设 css 可能是 background 简写（如 "url(...) center / cover no-repeat"），

   614→// ---------------- 壁纸亮度实测（锁屏文字跟随背景色） ----------------
   615→
   616→/** 区域平均亮度高于该值视为偏浅（该区域文字用黑色） */
   617→const LIGHT_LUMINANCE_THRESHOLD = 0.5;
   618→
   619→/** 图片壁纸亮度实测结果的模块级缓存（url → 三区域结果）：
   620→ *  同一张壁纸被多处独立 hook消费（图标底座 ×N / 锁屏 / 状态栏 / 页点），
   621→ *  共享缓存避免每个实例重复解码测量；后挂载的实例（如打开 Spotlight/多任务）
   622→ *  首帧即命中，不会闪「测量中回退静态标记」的错误颜色。上限 32 条防泄漏 */
   623→const wallpaperLightCache = new Map<string, { top: boolean; bottom: boolean; all: boolean }>();
   624→const WALLPAPER_LIGHT_CACHE_MAX = 32;
   625→
   626→/** 用已解码的图片实测三区域亮度并写入共享缓存（与 useMeasuredWallpaperLight 同一算法）。
   627→ *  开机预热调用：锁屏/主屏挂载首帧即拿到正确前景色，不闪「测量中回退静态标记」的跳变 */
   628→function measureWallpaperLightInto(url: string, img: HTMLImageElement): void {
   629→  if (wallpaperLightCache.has(url)) return;
   630→  try {
   631→    const W = 32;
   632→    const H = 64;
   633→    const canvas = document.createElement('canvas');
   634→    canvas.width = W;
   635→    canvas.height = H;
   636→    const ctx = canvas.getContext('2d', { willReadFrequently: true });
   637→    if (!ctx) return;
   638→    ctx.drawImage(img, 0, 0, W, H);
   639→    const { data } = ctx.getImageData(0, 0, W, H);
   640→    const rowLum: number[] = [];
   641→    for (let y = 0; y < H; y++) {
   642→      let sum = 0;
   643→      for (let x = 0; x < W; x++) {
   644→        const i = (y * W + x) * 4;
   645→        sum += luminanceOf([data[i], data[i + 1], data[i + 2]]);
   646→      }
   647→      rowLum.push(sum / W);
   648→    }
   649→    const avgRows = (a: number, b: number) => {
   650→      let sum = 0;
   651→      for (let y = a; y < b; y++) sum += rowLum[y];
   652→      return sum / (b - a);
   653→    };
   654→    if (wallpaperLightCache.size >= WALLPAPER_LIGHT_CACHE_MAX) {
   655→      wallpaperLightCache.delete(wallpaperLightCache.keys().next().value as string);
   656→    }
   657→    wallpaperLightCache.set(url, {
   658→      top: avgRows(0, Math.floor(H * 0.35)) > LIGHT_LUMINANCE_THRESHOLD,
   659→      bottom: avgRows(Math.floor(H * 0.86), H) > LIGHT_LUMINANCE_THRESHOLD,
   660→      all: avgRows(0, H) > LIGHT_LUMINANCE_THRESHOLD,
   661→    });
   662→  } catch {
   663→    // 画布被污染等异常：静默放弃，hook 内回退预设静态标记
   664→  }
   665→}
   666→
   667→/**
   668→ * 开机预载壁纸（修复刚打开网页时锁屏「闪一下」）：
   669→ * 黑屏开机门控期间把主屏/锁屏壁纸图提前拉取解码并预热亮度实测缓存，
   670→ * 锁屏/主屏挂载首帧即完整壁纸 + 正确前景色，不再「先铺底色 → 图片弹入」。
   671→ * data:/blob: 之外全部适用（data URL 无网络请求无需预载）；
   672→ * 整体带超时兜底，最坏多等 timeoutMs 即放行，绝不卡死开机。
   673→ */
   674→async function preloadWallpapersForBoot(urls: (string | null | undefined)[], timeoutMs = 1500): Promise<void> {
   675→  const list = [...new Set(urls.filter((u): u is string => !!u && !u.startsWith('data:')))];
   676→  if (list.length === 0) return;
   677→  const work = list.map(
   678→    (url) =>
   679→      new Promise<void>((res) => {
   680→        const img = new Image();
   681→        img.onload = () => {
   682→          measureWallpaperLightInto(url, img);
   683→          res();
   684→        };
   685→        img.onerror = () => res();
   686→        img.src = url;
   687→      })
   688→  );
   689→  await Promise.race([Promise.all(work), new Promise<void>((r) => setTimeout(r, timeoutMs))]);
   690→}

   692→export interface WallpaperLightness {
   693→  /** 顶部区域（状态栏/日期/大时钟/小组件，0–35%）是否偏浅：true=用黑字 */
   694→  top: boolean | null;
   695→  /** 底部区域（上滑提示/快捷按钮/Home 横杠，86–100%）是否偏浅：true=用黑字 */
   696→  bottom: boolean | null;
   697→  /** 整体平均亮度是否偏浅（主屏图标标签/页点分布在全屏，用整体判定）：true=用黑字 */
   698→  all: boolean | null;
   699→}

   701→/** 从 backgroundImage 里提取第一张 url(...) 的图片地址 */
   702→function bgImageUrlOf(bgImage: string): string | null {
   703→  const m = bgImage.match(/url\((['"]?)([^'")]+)\1\)/);
   704→  return m ? m[2] : null;
   705→}
   706→
   707→/** 解析 CSS 颜色为 [r,g,b]（支持 #rgb/#rrggbb/#rrggbbaa 与 rgb()/rgba()） */
   708→function parseCssColor(input: string): [number, number, number] | null {
   709→  const s = input.trim();
   710→  let m = s.match(/^#([0-9a-f]{3,8})$/i);
   711→  if (m) {
   712→    let hex = m[1];
   713→    if (hex.length === 3 || hex.length === 4) hex = [...hex].map((c) => c + c).join('');
   714→    const r = parseInt(hex.slice(0, 2), 16);
   715→    const g = parseInt(hex.slice(2, 4), 16);
   716→    const b = parseInt(hex.slice(4, 6), 16);
   717→    if (Number.isFinite(r) && Number.isFinite(g) && Number.isFinite(b)) return [r, g, b];
   718→    return null;
   719→  }
   720→  m = s.match(/^rgba?\(([^)]+)\)$/i);
   721→  if (m) {
   722→    const parts = m[1].split(/[\s,/]+/).filter(Boolean).map(Number);
   723→    if (parts.length >= 3 && parts.slice(0, 3).every((n) => Number.isFinite(n))) {
   724→      return [parts[0], parts[1], parts[2]];
   725→    }
   726→  }
   727→  return null;
   728→}
   729→
   730→function luminanceOf([r, g, b]: [number, number, number]): number {
   731→  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255;
   732→}
   733→
   734→/**
   735→ * 分区域实测壁纸亮度（锁屏文字随背景变色）：
   736→ * - 图片壁纸：解码后缩到 32×64 逐行取 Rec.709 亮度，顶部 0–35%（状态栏/时钟/小组件）与
   737→ *   底部 86–100%（上滑提示/快捷按钮/Home 横杠）分别求均值 —— 解决「上亮下暗」壁纸
   738→ *   （如雾山上白山+下深林）一刀切判色导致某端文字看不清的问题；
   739→ * - 纯色/渐变壁纸：解析样式里的色值，按出现顺序（渐变自上而下）前半=顶/后半=底；
   740→ * 返回 true=偏浅（黑字）/ false=偏深（白字）/ null=测量中或失败（调用方回退预设静态标记）。
   741→ */
   742→export function useMeasuredWallpaperLight(style: CSSProperties): WallpaperLightness {
   743→  const bgImage = typeof style.backgroundImage === 'string' ? style.backgroundImage : '';
   744→  const bgColor = typeof style.backgroundColor === 'string' ? style.backgroundColor : '';
   745→  const url = useMemo(() => bgImageUrlOf(bgImage), [bgImage]);
   746→
   747→  // 纯色/渐变壁纸：同步纯函数推导
   748→  const colorLight = useMemo<WallpaperLightness>(() => {
   749→    if (url) return { top: null, bottom: null, all: null }; // 图片壁纸走异步实测
   750→    const tokens = `${bgColor} ${bgImage}`.match(/#[0-9a-fA-F]{3,8}\b|rgba?\([^)]+\)/g) ?? [];
   751→    const lums = tokens
   752→      .map(parseCssColor)
   753→      .filter((c): c is [number, number, number] => c !== null)
   754→      .map(luminanceOf);
   755→    if (lums.length === 0) return { top: null, bottom: null, all: null };
   756→    const half = Math.max(1, Math.ceil(lums.length / 2));
   757→    const avg = (arr: number[]) => arr.reduce((acc, v) => acc + v, 0) / arr.length;
   758→    return {
   759→      top: avg(lums.slice(0, half)) > LIGHT_LUMINANCE_THRESHOLD,
   760→      bottom: avg(lums.slice(Math.max(0, lums.length - half))) > LIGHT_LUMINANCE_THRESHOLD,
   761→      all: avg(lums) > LIGHT_LUMINANCE_THRESHOLD,
   762→    };
   763→  }, [bgColor, bgImage, url]);
   764→
   765→  // 图片壁纸：解码后异步实测。结果绑定当时的 url，防止换壁纸瞬间新旧串值
   766→  const [imgState, setImgState] = useState<{ url: string; top: boolean; bottom: boolean; all: boolean } | null>(null);
   767→  useEffect(() => {
   768→    if (!url) return;
   769→    // 共享缓存已命中：渲染时已同步消费，无需重复解码测量
   770→    if (wallpaperLightCache.has(url)) return;
   771→    let cancelled = false;
   772→    const img = new Image();
   773→    img.onload = () => {
   774→      if (cancelled) return;
   775→      try {
   776→        const W = 32;
   777→        const H = 64;
   778→        const canvas = document.createElement('canvas');
   779→        canvas.width = W;
   780→        canvas.height = H;
   781→        const ctx = canvas.getContext('2d', { willReadFrequently: true });
   782→        if (!ctx) return; // 保持 null → 回退静态标记
   783→        ctx.drawImage(img, 0, 0, W, H);
   784→        const { data } = ctx.getImageData(0, 0, W, H);
   785→        const rowLum: number[] = [];
   786→        for (let y = 0; y < H; y++) {
   787→          let sum = 0;
   788→          for (let x = 0; x < W; x++) {
   789→            const i = (y * W + x) * 4;
   790→            sum += luminanceOf([data[i], data[i + 1], data[i + 2]]);
   791→          }
   792→          rowLum.push(sum / W);
   793→        }
   794→        const avgRows = (a: number, b: number) => {
   795→          let sum = 0;
   796→          for (let y = a; y < b; y++) sum += rowLum[y];
   797→          return sum / (b - a);
   798→        };
   799→        const result = {
   800→          top: avgRows(0, Math.floor(H * 0.35)) > LIGHT_LUMINANCE_THRESHOLD,
   801→          bottom: avgRows(Math.floor(H * 0.86), H) > LIGHT_LUMINANCE_THRESHOLD,
   802→          all: avgRows(0, H) > LIGHT_LUMINANCE_THRESHOLD,
   803→        };
   804→        // 写入共享缓存（简单上限淘汰，防长会话换壁纸累积泄漏）
   805→        if (wallpaperLightCache.size >= WALLPAPER_LIGHT_CACHE_MAX) {
   806→          wallpaperLightCache.delete(wallpaperLightCache.keys().next().value as string);
   807→        }
   808→        wallpaperLightCache.set(url, result);
   809→        setImgState({ url, ...result });
   810→      } catch {
   811→        // 画布被污染等异常：保持 null 回退静态标记
   812→      }
   813→    };
   814→    img.onerror = () => {
   815→      if (!cancelled) setImgState(null);
   816→    };
   817→    img.src = url;
   818→    return () => {
   819→      cancelled = true;
   820→    };
   821→  }, [url]);
   822→
   823→  if (url) {
   824→    // 共享缓存命中：同步返回（后挂载的图标/多任务/Spotlight 首帧即拿到结果，
   825→    // 不闪「测量中回退静态标记」的错误颜色，也避免 effect 内 setState）
   826→    const cached = wallpaperLightCache.get(url);
   827→    if (cached) return cached;
   828→    return imgState && imgState.url === url
   829→      ? { top: imgState.top, bottom: imgState.bottom, all: imgState.all }
   830→      : { top: null, bottom: null, all: null };
   831→  }
   832→  return colorLight;
   833→}

   882→const MAX_RECENT_APPS = 6;
   883→
   884→interface UIState {
   885→  activeApp: AppId | null;
   886→  phase: AppPhase;
   887→  /** 最近使用过的 App（多任务切换器卡片，最新在前；关闭 App 仍保留在后台） */
   888→  recentApps: AppId[];
   889→  /** 多任务切换器是否打开（可在主屏幕或任一 App 内打开） */
   890→  switcherOpen: boolean;
   891→  /** 锁屏是否激活（页面加载即锁定；锁定时覆盖全部交互） */
   892→  locked: boolean;
   893→  /** 熄屏（电源键：亮屏锁定 ↔ 熄屏，黑屏遮罩） */
   894→  screenOff: boolean;
   895→  /** 锁屏直达相机（不解锁，关闭相机回锁屏） */
   896→  lockCameraOpen: boolean;
   897→  /** 锁屏手电筒补光开启（全屏白，前景需转黑） */
   898→  torchOpen: boolean;
   899→  /** 闹钟响铃弹层全屏显示中（背景色，前景随主题） */
   900→  alarmRinging: boolean;
   901→  /** 电话 App 通话全屏层显示中（深色背景，状态栏/Home 横杠需白前景） */
   902→  callActive: boolean;
   903→  /** 主屏幕编辑（抖动）模式进行中：底部边缘上滑手势不触发多任务切换器 */
   904→  homeEdit: boolean;
   905→  openApp: (id: AppId) => void;
   906→  closeApp: () => void;
   907→  openSwitcher: () => void;
   908→  closeSwitcher: () => void;
   909→  /** 切换器关闭动画期间先把前台 App 退到后台（立即露出主屏幕） */
   910→  exitForegroundApp: () => void;
   911→  /** 在切换器里点卡片：切到该 App（无论当前是否已有 App 在前台） */
   912→  switchToApp: (id: AppId) => void;
   913→  /** 在切换器里上滑卡片：彻底关闭该 App（若是前台 App 则一并退出） */
   914→  killApp: (id: AppId) => void;
   915→  /** 解锁（锁屏上滑验证通过） */
   916→  unlock: () => void;
   917→  /** 立即锁定（关闭一切 App/切换器/锁屏相机，回到锁屏） */
   918→  lock: () => void;
   919→  /** 电源键：熄屏 ↔ 亮屏锁定 */
   920→  pressPower: () => void;
   921→  setLockCameraOpen: (v: boolean) => void;
   922→  setTorchOpen: (v: boolean) => void;
   923→  setAlarmRinging: (v: boolean) => void;
   924→  setCallActive: (v: boolean) => void;
   925→  /** 跨 App 跳转：电话 App 联系人详情点「信息」→ 打开信息 App 后自动进入该联系人会话（相当于添加好友），消费后置回 null */
   926→  pendingChatContact: string | null;
   927→  setPendingChatContact: (id: string | null) => void;
   928→  /** 跨 App 跳转：QQ 好友资料页「编辑资料」/ 微信好友详情页「朋友资料」→ 打开联系人 App 后直接进入该联系人的编辑页，消费后置回 null */
   929→  pendingContactEdit: string | null;
   930→  setPendingContactEdit: (id: string | null) => void;
   931→}
   932→
   933→export const useUI = create<UIState>((set, get) => ({
   934→  activeApp: null,
   935→  phase: 'closed',
   936→  recentApps: [],
   937→  switcherOpen: false,
   938→  locked: true,
   939→  screenOff: false,
   940→  lockCameraOpen: false,
   941→  torchOpen: false,
   942→  alarmRinging: false,
   943→  callActive: false,
   944→  homeEdit: false,
   945→  pendingChatContact: null,
   946→  setPendingChatContact: (v) => {
   947→    set({ pendingChatContact: v });
   948→  },
   949→  pendingContactEdit: null,
   950→  setPendingContactEdit: (v) => {
   951→    set({ pendingContactEdit: v });
   952→  },
   953→  openApp: (id) => {
   954→    if (get().locked) return; // 锁屏时禁止打开 App（锁屏直达相机走 lockCameraOpen）
   955→    if (get().activeApp || get().phase !== 'closed') return;
   956→    set({
   957→      activeApp: id,
   958→      phase: 'opening',
   959→      switcherOpen: false,
   960→      recentApps: [id, ...get().recentApps.filter((a) => a !== id)].slice(0, MAX_RECENT_APPS),
   961→    });
   962→    requestAnimationFrame(() => {
   963→      requestAnimationFrame(() => {
   964→        if (useUI.getState().phase === 'opening') set({ phase: 'open' });
   965→      });
   966→    });
   967→  },
   968→  closeApp: () => {
   969→    if (get().locked) return;
   970→    if (!get().activeApp || get().phase === 'closing') return;
   971→    set({ phase: 'closing' });
   972→    window.setTimeout(() => {
   973→      if (useUI.getState().phase === 'closing') set({ activeApp: null, phase: 'closed' });
   974→    }, 380);
   975→  },
   976→  openSwitcher: () => {
   977→    // 主屏幕或 App 内都可打开（打开/关闭动画进行中除外；锁定时禁止）
   978→    if (get().locked) return;
   979→    if (get().switcherOpen || get().phase === 'opening' || get().phase === 'closing') return;
   980→    set({ switcherOpen: true });
   981→  },
   982→  closeSwitcher: () => {
   983→    if (get().switcherOpen) set({ switcherOpen: false });
   984→  },
   985→  exitForegroundApp: () => {
   986→    if (get().activeApp) set({ activeApp: null, phase: 'closed' });
   987→  },
   988→  switchToApp: (id) => {
   989→    if (get().locked) return;
   990→    set({
   991→      activeApp: id,
   992→      phase: 'open',
   993→      switcherOpen: false,
   994→      recentApps: [id, ...get().recentApps.filter((a) => a !== id)].slice(0, MAX_RECENT_APPS),
   995→    });
   996→  },
   997→  killApp: (id) => {
   998→    const patch: Partial<UIState> = { recentApps: get().recentApps.filter((a) => a !== id) };
   999→    if (get().activeApp === id) {
  1000→      patch.activeApp = null;
  1001→      patch.phase = 'closed';
  1002→    }
  1003→    set(patch);
  1004→  },
  1005→  unlock: () => {
  1006→    set({ locked: false, screenOff: false, lockCameraOpen: false, torchOpen: false });
  1007→  },
  1008→  lock: () => {
  1009→    set({
  1010→      locked: true,
  1011→      screenOff: false,
  1012→      lockCameraOpen: false,
  1013→      torchOpen: false,
  1014→      activeApp: null,
  1015→      phase: 'closed',
  1016→      switcherOpen: false,
  1017→    });
  1018→  },
  1019→  pressPower: () => {
  1020→    const s = get();
  1021→    const lockScreen = useSettings.getState().lockConfig.lockScreen;
  1022→    if (s.screenOff) {
  1023→      // 熄屏 → 亮屏：有锁屏回锁定界面；无锁屏保持原界面
  1024→      set(lockScreen ? { screenOff: false, locked: true } : { screenOff: false });
  1025→    } else if (lockScreen) {
  1026→      // 熄屏并锁定（退出一切前台界面）
  1027→      set({
  1028→        screenOff: true,
  1029→        locked: true,
  1030→        lockCameraOpen: false,
  1031→        torchOpen: false,
  1032→        activeApp: null,
  1033→        phase: 'closed',
  1034→        switcherOpen: false,
  1035→      });
  1036→    } else {
  1037→      // 无锁屏：仅熄屏，保留当前界面
  1038→      set({ screenOff: true });
  1039→    }
  1040→  },
  1041→  setLockCameraOpen: (v) => {
  1042→    set({ lockCameraOpen: v });
  1043→  },
  1044→  setTorchOpen: (v) => {
  1045→    set({ torchOpen: v });
  1046→  },
  1047→  setAlarmRinging: (v) => {
  1048→    set({ alarmRinging: v });
  1049→  },
  1050→  setCallActive: (v) => {
  1051→    set({ callActive: v });
  1052→  },
  1053→}));
