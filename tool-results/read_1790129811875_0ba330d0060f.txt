     1→/**
     2→ * 群聊数据层（宿主：微信 / QQ 双 App）。
     3→ *
     4→ * 设计要点：
     5→ * - 群 = ChatGroup：独立会话实体（非联系人），成员是 AI 角色（char/npc）联系人 ID，机主固定参与；
     6→ * - 会话身份 sessionKey = `<app>:group:<id>`（复用流总线/未读/标志/时间感知等按字符串键隔离的设施）；
     7→ * - 存储：微信群 `wx-chat-groups`（历史数据兼容）、QQ 群 `qq-chat-groups` 两池分开，群 id 全局唯一，
     8→ *   getGroup 双池查找；消息落 `<app>-group-msgs:<id>`（微信群沿用历史 wx-group-msgs 前缀）；
     9→ * - 记忆互通开关 memoryInterop 按群独立（设置页唯一入口，不再有按成员覆盖）：开启时群记忆参与成员角色的
    10→ *   私聊召回、角色私聊记忆参与群聊召回；effectiveInterop 统一解析，聊天页把结果以 interopOn 回调
    11→ *   传给记忆召回；角色之间的隔离永远由记忆存储键（mem-frag:<contactId>）保证，与开关无关；
    12→ * - 谁来回复由每个角色按自己的人设与消息内容自判（无需回复只回 [SKIP]，不落盘），@ 成员始终优先；
    13→ * - 群红包（普通/拼手气/专属）与指定成员转账：消息落本群消息库，状态随群持久化（按群 ID 隔离，
    14→ *   与单聊互不相通）；成员按人设领取/收款/退回，纯本地模拟不涉及真实资金；
    15→ * - 解散群聊级联清理：群记录/群消息（含红包/转账卡片状态）/会话附属数据/群聊天背景 + 群来源记忆
    16→ *   （onGroupDissolved 钩子由 memory 层注册，删去各成员记忆库里 sourceGroupId=本群的碎片/总结，
    17→ *   依赖方向保持 memory → groups 单向，不反向 import）。
    18→ */

   207→function readJSON<T>(key: string): T | null {
   208→  try {
   209→    return kvGet<T>(key);
   210→  } catch {
   211→    return null;
   212→  }
   213→}

   215→function writeJSON(key: string, value: unknown): void {
   216→  try {
   217→    kvSet(key, value);
   218→  } catch {
   219→    // 存储失败不中断（群是增强能力）
   220→  }
   221→}

   228→function normalizeGroup(g: unknown): ChatGroup | null {
   229→  if (!g || typeof g !== 'object') return null;
   230→  const r = g as Partial<ChatGroup>;
   231→  if (typeof r.id !== 'string' || !Array.isArray(r.memberIds)) return null;
   232→  return {
   233→    id: r.id,
   234→    app: r.app === 'qq' ? 'qq' : 'wx',
   235→    name: typeof r.name === 'string' ? r.name : '未命名群聊',
   236→    // 群备注为新增字段：旧数据缺省补空串
   237→    remark: typeof r.remark === 'string' ? r.remark : '',
   238→    avatar: typeof r.avatar === 'string' ? r.avatar : null,
   239→    ownerId: typeof r.ownerId === 'string' ? r.ownerId : '',
   240→    memberIds: r.memberIds.filter((x): x is string => typeof x === 'string'),
   241→    // 管理员/禁言为新增字段：旧数据缺省补空值（normalize 保证读写形状一致）
   242→    adminIds: Array.isArray(r.adminIds) ? r.adminIds.filter((x): x is string => typeof x === 'string') : [],
   243→    mutes:
   244→      r.mutes && typeof r.mutes === 'object' && !Array.isArray(r.mutes)
   245→        ? Object.fromEntries(
   246→            Object.entries(r.mutes as Record<string, unknown>).filter(
   247→              (entry): entry is [string, number | null] => typeof entry[0] === 'string' && (entry[1] === null || typeof entry[1] === 'number'),
   248→            ),
   249→          )
   250→        : {},
   251→    announcement: typeof r.announcement === 'string' ? r.announcement : '',
   252→    annAt: typeof r.annAt === 'number' ? r.annAt : undefined,
   253→    no: typeof r.no === 'string' && /^\d{6,}$/.test(r.no) ? r.no : undefined,
   254→    memoryInterop: r.memoryInterop === true,
   255→    createdAt: typeof r.createdAt === 'number' ? r.createdAt : 0,
   256→  };
   257→}

   259→function readPool(app: GroupApp): ChatGroup[] {
   260→  const raw = readJSON<ChatGroup[]>(POOLS[app]);
   261→  if (!Array.isArray(raw)) return [];
   262→  return raw.map(normalizeGroup).filter((g): g is ChatGroup => g != null);
   263→}
   264→
   265→function writePool(app: GroupApp, list: ChatGroup[]): void {
   266→  writeJSON(POOLS[app], list);
   267→}

   270→export function listGroups(app?: GroupApp): ChatGroup[] {

   275→/** 生成唯一群号（9 位数字、首位非 0；跨双宿主查重，保证每个群的群号都不一样） */
   276→function genUniqueGroupNo(): string {
   277→  const used = new Set(listGroups().map((g) => g.no).filter((x): x is string => Boolean(x)));
   278→  for (let i = 0; i < 50; i++) {
   279→    const no = String(Math.floor(100000000 + Math.random() * 900000000));
   280→    if (!used.has(no)) return no;
   281→  }
   282→  return String(Date.now()).slice(-9); // 极端兜底：时间尾数（同毫秒建多群概率可忽略）
   283→}

   285→/** 群号（建群时已分配则直接返回；旧群缺省时惰性生成唯一号码并落盘，同群恒定） */
   286→export function ensureGroupNo(groupId: string): string {
   287→  const g = getGroup(groupId);
   288→  if (!g) return '';
   289→  if (g.no) return g.no;
   290→  const no = genUniqueGroupNo();
   291→  const list = readPool(g.app);
   292→  const idx = list.findIndex((x) => x.id === groupId);
   293→  if (idx !== -1) {
   294→    list[idx] = { ...list[idx], no };
   295→    writePool(g.app, list);
   296→  }
   297→  return no;
   298→}

   300→export function getGroup(groupId: string): ChatGroup | null {
   301→  return listGroups().find((g) => g.id === groupId) ?? null;
   302→}

   304→export function createGroup(input: {
   305→  name: string;
   306→  memberIds: string[];
   307→  ownerId: string;
   308→  /** 宿主 App（缺省 wx，兼容旧调用） */
   309→  app?: GroupApp;
   310→  avatar?: string | null;
   311→  announcement?: string;
   312→}): ChatGroup {
   313→  const app: GroupApp = input.app ?? 'wx';
   314→  const group: ChatGroup = {
   315→    id: genId(),
   316→    app,
   317→    name: input.name.trim() || '未命名群聊',
   318→    avatar: input.avatar ?? null,
   319→    ownerId: input.ownerId,
   320→    memberIds: Array.from(new Set(input.memberIds)).slice(0, GROUP_MEMBER_CAP),
   321→    adminIds: [],
   322→    mutes: {},
   323→    announcement: input.announcement?.trim() ?? '',
   324→    no: genUniqueGroupNo(),
   325→    memoryInterop: false,
   326→    createdAt: Date.now(),
   327→  };
   328→  writePool(app, [...readPool(app), group]);
   329→  // 建群系统消息（三.9）：居中灰字「群聊创建」，随群消息持久化
   330→  pushGroupEvent(group.id, '群聊创建', { type: 'create' });
   331→  return group;
   332→}

   334→/**
   335→ * 更新群（名字/头像/公告/互通开关/成员等）；返回更新后的群（群不存在返回 null）。
   336→ * 群资料变更（改名/换头像/改公告）自动落一条群事件系统消息（三.7/8：所有操作要生成对应系统消息），
   337→ * 机主无关调用方是谁（信息页/聊天页/宿主壳层统一在此处发事件，避免遗漏）。
   338→ */
   339→export function updateGroup(
   340→  groupId: string,
   341→  patch: Partial<Pick<ChatGroup, 'name' | 'remark' | 'avatar' | 'announcement' | 'memoryInterop' | 'memberIds'>>
   342→): ChatGroup | null {
   343→  const app = getGroup(groupId)?.app;
   344→  if (!app) return null;
   345→  const list = readPool(app);
   346→  const idx = list.findIndex((g) => g.id === groupId);
   347→  if (idx === -1) return null;
   348→  const prev = list[idx];
   349→  const next: ChatGroup = { ...prev, ...patch };
   350→  if (patch.name !== undefined) next.name = patch.name.trim() || next.name;
   351→  if (patch.remark !== undefined) next.remark = patch.remark.trim();
   352→  list[idx] = next;
   353→  writePool(app, list);
   354→  // 群资料事件（只在值真正变化时发，成员增删走专用函数不经过这里）
   355→  if (patch.name !== undefined && next.name !== prev.name) {
   356→    pushGroupEvent(groupId, `群名被修改为「${next.name}」`, { type: 'rename', extra: next.name });
   357→  }
   358→  if (patch.avatar !== undefined && (patch.avatar ?? null) !== (prev.avatar ?? null)) {
   359→    pushGroupEvent(groupId, '群头像已更新', { type: 'avatar' });
   360→  }
   361→  if (patch.announcement !== undefined && (patch.announcement ?? '') !== (prev.announcement ?? '')) {
   362→    next.annAt = Date.now();
   363→    pushGroupEvent(groupId, '群公告已更新', { type: 'announcement' });
   364→  }
   365→  return next;
   366→}

   412→/** 禁言剩余描述（「剩 X 分钟 / X 小时 / X 天」或「永久」；未禁言返回 null） */
   413→export function groupMuteLeftText(g: ChatGroup, contactId: string, now: number = Date.now()): string | null {
   414→  const until = g.mutes[contactId];
   415→  if (until === null) return '永久';
   416→  if (typeof until !== 'number' || until <= now) return null;
   417→  const left = Math.max(1, Math.ceil((until - now) / 60_000));
   418→  if (left < 60) return `剩 ${left} 分钟`;
   419→  const hours = Math.ceil(left / 60);
   420→  if (hours < 24) return `剩 ${hours} 小时`;
   421→  return `剩 ${Math.ceil(hours / 24)} 天`;
   422→}

   424→/** 时长毫秒 → 人类可读（事件文本用；固定档位优先用 GROUP_MUTE_PRESETS 的 label） */
   425→function muteDurationText(ms: number | null): string {
   426→  if (ms === null) return '永久';
   427→  const preset = GROUP_MUTE_PRESETS.find((p) => p.ms === ms);
   428→  if (preset) return preset.label;
   429→  if (ms % 60_000 === 0) {
   430→    const mins = ms / 60_000;
   431→    if (mins % 60 === 0) return `${mins / 60} 小时`;
   432→    return `${mins} 分钟`;
   433→  }
   434→  return `${Math.round(ms / 60_000)} 分钟`;
   435→}

   581→/** 事件相对时间标签（刚刚 / X 分钟前 / X 小时前 / 昨天 / M月D日 HH:MM） */
   582→export function groupEventAgo(ts: number, now: number = Date.now()): string {
   583→  const diff = now - ts;
   584→  if (diff < 60_000) return '刚刚';
   585→  if (diff < 60 * 60_000) return `${Math.floor(diff / 60_000)} 分钟前`;
   586→  if (diff < 24 * 60 * 60_000) return `${Math.floor(diff / (60 * 60_000))} 小时前`;
   587→  const d = new Date(ts);
   588→  const hh = String(d.getHours()).padStart(2, '0');
   589→  const mm = String(d.getMinutes()).padStart(2, '0');
   590→  const yest = new Date(now - 24 * 60 * 60_000);
   591→  if (d.getFullYear() === yest.getFullYear() && d.getMonth() === yest.getMonth() && d.getDate() === yest.getDate()) {
   592→    return `昨天 ${hh}:${mm}`;
   593→  }
   594→  return `${d.getMonth() + 1}月${d.getDate()}日 ${hh}:${mm}`;
   595→}

   605→/** 从 localStorage JSON map 里删掉一个键（unreads/flags/hidden/time-aware 同构清理用） */

   628→/**
   629→ * 解散群聊：删除群 + 群消息（红包/转账卡片状态随消息一并清除）+ 会话级附属数据
   630→ * （未读/标志/隐藏/时间感知/每成员的群记忆提取轮次计数/群聊天背景）+ 群来源记忆
   631→ * （通过 onGroupDissolved 钩子级联，见 memory 层 memPurgeGroupSource）。
   632→ * opts.purgeMemory = false 时跳过群来源记忆清理（用于「退出群聊」：群对其他成员仍然存在，
   633→ * AI 成员的群记忆应当保留，只有机主本机删除该群）。
   634→ * 返回被解散的群（供 UI 提示），群不存在返回 null。
   635→ */
   636→export function dissolveGroup(groupId: string, opts?: { purgeMemory?: boolean }): ChatGroup | null {
   637→  const g = getGroup(groupId);
   638→  if (!g) return null;
   639→  writePool(g.app, readPool(g.app).filter((x) => x.id !== groupId));
   640→  try {
   641→    kvDel(groupMsgsKey(g.app, groupId));
   642→    // 未读/标志/隐藏为 localStorage JSON map（键 = 会话 id；群会话 id = `group:<gid>`）
   643→    const maps = LS_MAPS[g.app];
   644→    removeLocalMapKey(maps.unreads, `group:${groupId}`);
   645→    removeLocalMapKey(maps.flags, `group:${groupId}`);
   646→    removeLocalMapKey(maps.hidden, `group:${groupId}`);
   647→    // 时间感知开关（localStorage map 里的键）
   648→    removeLocalMapKey('chat-time-aware', `${g.app}:group:${groupId}`);
   649→    // 每个成员的群记忆提取计数（旧 mem-round 轮次键 + 新 mem-msgcount 消息计数 / mem-anchor 锚点，均按群 scope）
   650→    for (const cid of g.memberIds) {
   651→      kvDel(`mem-round:${cid}:${g.app}:group:${groupId}`);
   652→      kvDel(`mem-msgcount:${cid}:${g.app}:group:${groupId}`);
   653→      kvDel(`mem-anchor:${cid}:${g.app}:group:${groupId}`);
   654→    }
   655→    // 群聊天背景图片本体（IndexedDB settings store，键同 contacts-store 的 chat-bg 前缀；群背景与单聊相互独立）
   656→    void localDB.delete('settings', `chat-bg:${g.app}:group:${groupId}`).catch(() => undefined);
   657→  } catch {
   658→    // 清理失败不阻塞解散
   659→  }
   660→  // 群来源记忆级联清理（钩子逐个调用；单个钩子异常不影响其余清理，更不阻塞解散）。
   661→  // opts.purgeMemory = false（退出群聊）时跳过：群对 AI 成员仍然存在，记忆保留。
   662→  if (opts?.purgeMemory !== false) {
   663→    for (const fn of dissolveHooks) {
   664→      try {
   665→        fn(g);
   666→      } catch {
   667→        // 忽略
   668→      }
   669→    }
   670→  }
   671→  return g;
   672→}

   674→/**
   675→ * 退出群聊（机主本人退群）：本机删除该群（群记录/消息/未读/标志/时间感知/背景等附属数据），
   676→ * 但不清 AI 成员的群来源记忆（群对其他成员仍然存在，他们记得群里发生过什么）。
   677→ * 与「解散群聊」的区别：解散 = 群对所有人消失且记忆级联清理；退出 = 仅机主本机移除。

   679→export function quitGroup(groupId: string): ChatGroup | null {

   705→function normalizeMsg(m: unknown): WxGroupMsg | null {
   706→  if (!m || typeof m !== 'object') return null;
   707→  const r = m as Partial<WxGroupMsg>;
   708→  if (typeof r.id !== 'string' || typeof r.content !== 'string' || typeof r.time !== 'number') return null;
   709→  return {
   710→    id: r.id,
   711→    role: r.role === 'me' ? 'me' : 'peer',
   712→    senderId: typeof r.senderId === 'string' ? r.senderId : 'unknown',
   713→    senderName: typeof r.senderName === 'string' ? r.senderName : '',
   714→    content: r.content,
   715→    time: r.time,
   716→    kind:
   717→      r.kind === 'notice' || r.kind === 'image' || r.kind === 'location' || r.kind === 'sticker' || r.kind === 'redpacket' || r.kind === 'transfer' || r.kind === 'forward'
   718→        ? r.kind
   719→        : 'text',
   720→    rp:
   721→      r.kind === 'redpacket' && r.rp && typeof r.rp.amount === 'number'
   722→        ? {
   723→            amount: r.rp.amount,
   724→            count: typeof r.rp.count === 'number' && r.rp.count >= 1 ? Math.floor(r.rp.count) : 1,
   725→            mode: r.rp.mode === 'lucky' || r.rp.mode === 'exclusive' ? r.rp.mode : 'normal',
   726→            blessing: typeof r.rp.blessing === 'string' ? r.rp.blessing : '恭喜发财，大吉大利',
   727→            targetId: typeof r.rp.targetId === 'string' ? r.rp.targetId : undefined,
   728→            targetName: typeof r.rp.targetName === 'string' ? r.rp.targetName : undefined,
   729→            claims: Array.isArray(r.rp.claims)
   730→              ? r.rp.claims
   731→                  .filter((c): c is GroupRpData['claims'][number] => Boolean(c) && typeof (c as { name?: unknown }).name === 'string')
   732→                  .map((c) => ({
   733→                    contactId: typeof c.contactId === 'string' ? c.contactId : '',
   734→                    name: c.name,
   735→                    avatar: typeof c.avatar === 'string' ? c.avatar : null,
   736→                    amount: typeof c.amount === 'number' ? c.amount : 0,
   737→                    ts: typeof c.ts === 'number' ? c.ts : 0,
   738→                  }))
   739→              : [],
   740→            sentAt: typeof r.rp.sentAt === 'number' ? r.rp.sentAt : r.time,
   741→            expired: r.rp.expired === true || undefined,
   742→            expiredAt: typeof r.rp.expiredAt === 'number' ? r.rp.expiredAt : undefined,
   743→            cid: typeof r.rp.cid === 'string' ? r.rp.cid : undefined,
   744→          }
   745→        : undefined,
   746→    tr:
   747→      r.kind === 'transfer' && r.tr && typeof r.tr.amount === 'number'
   748→        ? {
   749→            amount: r.tr.amount,
   750→            note: typeof r.tr.note === 'string' ? r.tr.note : '',
   751→            toId: typeof r.tr.toId === 'string' ? r.tr.toId : '',
   752→            toName: typeof r.tr.toName === 'string' ? r.tr.toName : '',
   753→            received: r.tr.received === true || undefined,
   754→            receivedAt: typeof r.tr.receivedAt === 'number' ? r.tr.receivedAt : undefined,
   755→            status: r.tr.status === 'returned' || r.tr.status === 'rejected' ? r.tr.status : undefined,
   756→            cid: typeof r.tr.cid === 'string' ? r.tr.cid : undefined,
   757→          }
   758→        : undefined,
   759→    noticeText: typeof r.noticeText === 'string' ? r.noticeText : undefined,
   760→    evt:
   761→      r.evt && typeof r.evt.type === 'string' && EVENT_TYPES.has(r.evt.type)
   762→        ? {
   763→            type: r.evt.type as GroupEventDetail['type'],
   764→            actorId: typeof r.evt.actorId === 'string' ? r.evt.actorId : undefined,
   765→            targetId: typeof r.evt.targetId === 'string' ? r.evt.targetId : undefined,
   766→            extra: typeof r.evt.extra === 'string' ? r.evt.extra : undefined,
   767→          }
   768→        : undefined,
   769→    notice:
   770→      r.notice && typeof r.notice.pre === 'string' && typeof r.notice.accent === 'string'
   771→        ? { icon: r.notice.icon === 'tr' || r.notice.icon === 'fam' ? r.notice.icon : 'rp', pre: r.notice.pre, accent: r.notice.accent }
   772→        : undefined,
   773→    img: r.img && typeof r.img.src === 'string' ? { src: r.img.src } : undefined,
   774→    loc:
   775→      r.loc && typeof r.loc.name === 'string' && typeof r.loc.address === 'string'
   776→        ? { name: r.loc.name, address: r.loc.address }
   777→        : undefined,
   778→    stk:
   779→      r.stk && typeof r.stk.url === 'string'
   780→        ? { url: r.stk.url, meaning: typeof r.stk.meaning === 'string' ? r.stk.meaning : '', sid: typeof r.stk.sid === 'string' ? r.stk.sid : undefined }
   781→        : undefined,
   782→    quote:
   783→      r.quote && typeof r.quote.name === 'string' && typeof r.quote.content === 'string'
   784→        ? {
   785→            name: r.quote.name,
   786→            content: r.quote.content,
   787→            // 引用源消息 ID（删除/撤回后把引用内容改写为「原消息已删除」用；旧数据无此字段照常渲染）
   788→            id: typeof r.quote.id === 'string' ? r.quote.id : undefined,
   789→          }
   790→        : undefined,
   791→    recalled: r.recalled === true,
   792→    fwd:
   793→      r.kind === 'forward' && r.fwd && typeof r.fwd.from === 'string'
   794→        ? {
   795→            from: r.fwd.from,
   796→            merged: r.fwd.merged === true || undefined,
   797→            title: typeof r.fwd.title === 'string' ? r.fwd.title : undefined,
   798→            records: Array.isArray(r.fwd.records)
   799→              ? r.fwd.records
   800→                  .filter((x): x is GroupFwdRecord => Boolean(x) && typeof (x as GroupFwdRecord).name === 'string' && typeof (x as GroupFwdRecord).text === 'string')
   801→                  .map((x) => ({
   802→                    name: x.name,
   803→                    role: x.role === 'me' ? ('me' as const) : ('peer' as const),
   804→                    text: x.text,
   805→                    quote: typeof x.quote === 'string' ? x.quote : undefined,
   806→                    time: typeof x.time === 'number' ? x.time : 0,
   807→                    avatar: typeof x.avatar === 'string' ? x.avatar : x.avatar === null ? null : undefined,
   808→                    kind: x.kind === 'sticker' || x.kind === 'image' ? x.kind : undefined,
   809→                    imgSrc: typeof x.imgSrc === 'string' ? x.imgSrc : undefined,
   810→                    stkMeaning: typeof x.stkMeaning === 'string' ? x.stkMeaning : undefined,
   811→                  }))
   812→              : undefined,
   813→          }
   814→        : undefined,
   815→  };
   816→}

   818→export function loadGroupMsgs(groupId: string): WxGroupMsg[] {
   819→  const app = getGroup(groupId)?.app ?? 'wx';
   820→  const raw = readJSON<WxGroupMsg[]>(groupMsgsKey(app, groupId));
   821→  if (!Array.isArray(raw)) return [];
   822→  return raw.map(normalizeMsg).filter((m): m is WxGroupMsg => m != null);
   823→}

   825→/** 落盘（保留最后 200 条，与私聊同量级） */
   826→export function saveGroupMsgs(groupId: string, msgs: WxGroupMsg[]): void {
   827→  const app = getGroup(groupId)?.app ?? 'wx';
   828→  writeJSON(groupMsgsKey(app, groupId), msgs.slice(-MSGS_CAP));
   829→}

   831→/** 该群最后一条非通知消息（会话列表预览用） */
   832→export function groupPreview(groupId: string): { text: string; time: number } {
   833→  const msgs = loadGroupMsgs(groupId);
   834→  const last = [...msgs].reverse().find((m) => m.kind !== 'notice');
   835→  if (!last) return { text: '', time: 0 };
   836→  if (last.recalled) return { text: `${last.role === 'me' ? '你' : last.senderName || '有人'}撤回了一条消息`, time: last.time };
   837→  if (last.kind === 'image') return { text: '[图片]', time: last.time };
   838→  if (last.kind === 'sticker') return { text: `[表情]${last.stk?.meaning ? ` ${last.stk.meaning}` : ''}`, time: last.time };
   839→  if (last.kind === 'location') return { text: `[位置] ${last.loc?.name ?? ''}`.trim(), time: last.time };
   840→  if (last.kind === 'redpacket') return { text: '[红包]', time: last.time };
   841→  if (last.kind === 'transfer') return { text: '[转账]', time: last.time };
   842→  if (last.kind === 'forward') return { text: last.fwd?.merged ? '[聊天记录]' : last.content, time: last.time };
   843→  const prefix = last.role === 'me' ? '我' : last.senderName;
   844→  return { text: `${prefix}：${last.content}`, time: last.time };
   845→}
