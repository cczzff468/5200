    44→/** HSL → #rrggbb（彩虹色相条选色用） */
    45→function hslToHex(h: number, s: number, l: number): string {
    46→  const f = (n: number) => {
    47→    const k = (n + h / 30) % 12;
    48→    const c = l - s * Math.min(l, 1 - l) * Math.max(-1, Math.min(k - 3, 9 - k, 1));
    49→    return Math.round(255 * c)
    50→      .toString(16)
    51→      .padStart(2, '0');
    52→  };
    53→  return `#${f(0)}${f(8)}${f(4)}`;
    54→}

    60→/** #rrggbb → 色相 0..360（无彩色返回 0） */
    61→function hexToHue(hex: string): number {
    62→  const n = parseInt(hex.slice(1), 16);
    63→  const r = ((n >> 16) & 255) / 255;
    64→  const g = ((n >> 8) & 255) / 255;
    65→  const b = (n & 255) / 255;
    66→  const mx = Math.max(r, g, b);
    67→  const mn = Math.min(r, g, b);
    68→  if (mx === mn) return 0;
    69→  const d = mx - mn;
    70→  let h: number;
    71→  if (mx === r) h = ((g - b) / d + (g < b ? 6 : 0)) * 60;
    72→  else if (mx === g) h = ((b - r) / d + 2) * 60;
    73→  else h = ((r - g) / d + 4) * 60;
    74→  return h;
    75→}

   437→      else if (tab === 'bottom') onChange2({ ...wear2, bottom: Number(key) });
   438→      else onChange2({ ...wear2, shoes: Number(key) });

   462→      if (col) t[slot] = col;

   489→          <motion.div

   491→            initial={{ y: '100%' }}
   492→            animate={{ y: 0 }}
