   195→interface SettingsState {
   196→  theme: ThemeMode;
   197→  wallpaperPreset: string;
   198→  /** 自定义壁纸的 ObjectURL（Blob 存 IndexedDB） */
   199→  customWallpaperUrl: string | null;
   200→  /** 锁屏壁纸预设（锁屏壁纸与主屏幕完全独立，各有各的预设/自定义） */
   201→  lockWallpaperPreset: string;
   202→  /** 锁屏自定义壁纸的 ObjectURL（Blob 存 IndexedDB） */
   203→  lockCustomWallpaperUrl: string | null;
   204→  apiConfig: ApiConfig;
   205→  /** 用户自己保存的 API 预设 */
   206→  apiPresets: ApiPreset[];
   207→  /** 识图模型配置（与 apiConfig 互相独立、互不覆盖；密文持久化） */
   208→  visionConfig: VisionConfig;
   209→  /** 用户自己保存的识图模型预设 */
   210→  visionPresets: VisionPreset[];
   211→  /** 锁屏密码配置（持久化） */
   212→  lockConfig: LockConfig;
   213→  /** 个人信息（头像/名字/标签，持久化） */
   214→  profile: Profile;
   215→  /** AI 对用户的称呼方式：用户已要求移除「AI 称呼方式」设置——昵称只是 App 显示昵称、
   216→   *  名字才是真名，AI 统一用真实名字称呼（load 时固定回 'name'，历史存储的 'nickname' 不再生效）；
   217→   *  【用户的称呼】人设注入保留，AI 仍知道昵称与真名同属一人 */
   218→  addressMode: AddressMode;
   219→  /** 语音 API（TTS）配置：与聊天/识图 API 相互独立（含 Key，密文持久化） */

   223→  /** 自定义 App 图标（AppId → ObjectURL，Blob 存 IndexedDB settings.customIcons） */
   224→  customIcons: Record<string, string>;
   225→  loaded: boolean;

   227→  load: () => Promise<void>;

   253→  /** 清空全部自定义图标（全部恢复默认） */
   254→  resetAllCustomIcons: () => void;

   257→export const useSettings = create<SettingsState>((set, get) => ({
   258→  // 主题默认浅色（用户要求）；load 时无记录/一次性迁移后也回退到浅色
   259→  theme: 'light',
   260→  wallpaperPreset: 'graphite',
   261→  customWallpaperUrl: null,
   262→  lockWallpaperPreset: 'graphite',
   263→  lockCustomWallpaperUrl: null,
   264→  apiConfig: { ...DEFAULT_API_CONFIG },
   265→  apiPresets: [],
   266→  visionConfig: { ...DEFAULT_VISION_CONFIG },
   267→  visionPresets: [],
   268→  lockConfig: { lockScreen: true, enabled: false, code: '', len: 4 },
   269→  profile: { ...DEFAULT_PROFILE },
   270→  addressMode: 'name',
   271→  ttsConfig: { ...DEFAULT_TTS_CONFIG },
   272→  ttsVoices: [],
   273→  customIcons: {},
   274→  loaded: false,
   275→
   276→  load: async () => {
   277→    if (get().loaded) return;
   278→    try {
   279→      const [themeRec, wallpaperRec, lockWallpaperRec, apiRec, presetsRec, visionRec, visionPresetsRec, lockRec, profileRec, iconsRec, ttsRec, ttsVoicesRec] = await Promise.all([
   280→        localDB.get('settings', 'theme'),
   281→        localDB.get('settings', 'wallpaper'),
   282→        localDB.get('settings', 'lockWallpaper'),
   283→        localDB.get('settings', 'apiConfig'),
   284→        localDB.get('settings', 'apiPresets'),
   285→        localDB.get('settings', 'visionConfig'),
   286→        localDB.get('settings', 'visionPresets'),
   287→        localDB.get('settings', 'lock'),
   288→        localDB.get('settings', 'profile'),
   289→        localDB.get('settings', 'customIcons'),
   290→        localDB.get('settings', 'ttsConfig'),
   291→        localDB.get('settings', 'ttsVoices'),
   292→      ]);
   293→
   294→      // 自定义 App 图标：{ AppId: Blob } → 为每个 Blob 建 ObjectURL
   295→      let customIcons: Record<string, string> = {};
   296→      if (iconsRec && typeof iconsRec.value === 'object' && iconsRec.value !== null) {
   297→        for (const [k, v] of Object.entries(iconsRec.value as Record<string, unknown>)) {
   298→          if (v instanceof Blob) customIcons[k] = URL.createObjectURL(v);
   299→        }
   300→      }
   301→
   302→      let profile: Profile = { ...DEFAULT_PROFILE };
   303→      if (profileRec && typeof profileRec.value === 'object' && profileRec.value !== null) {
   304→        const v = profileRec.value as Partial<Profile>;
   305→        profile = {
   306→          name: typeof v.name === 'string' ? v.name : '',
   307→          tag: typeof v.tag === 'string' ? v.tag : DEFAULT_PROFILE.tag,
   308→          avatar:
   309→            typeof v.avatar === 'string' && v.avatar.startsWith('data:image/') && v.avatar.length < 600_000
   310→              ? v.avatar
   311→              : null,
   312→        };
   313→      }
   314→
   315→      let lockConfig: LockConfig = { lockScreen: true, enabled: false, code: '', len: 4 };
   316→      if (lockRec && typeof lockRec.value === 'object' && lockRec.value !== null) {
   317→        const v = lockRec.value as Partial<LockConfig>;
   318→        lockConfig = {
   319→          lockScreen: v.lockScreen !== false, // 旧记录无此字段 → 默认开启
   320→          enabled: v.enabled === true,
   321→          code: typeof v.code === 'string' ? v.code : '',
   322→          len: v.len === 6 ? 6 : 4,
   323→        };
   324→      }
   325→
   326→      // 主题默认浅色（用户要求）：无记录回退浅色；历史存储里的「深色」一次性重置为浅色
   327→      //（否则用户之前测试时持久化的 dark 会让默认值改动不可见；重置后用户再手动切深色仍可保留）
   328→      let theme: ThemeMode = 'light';
   329→      if (themeRec && (themeRec.value === 'light' || themeRec.value === 'dark' || themeRec.value === 'auto')) {
   330→        theme = themeRec.value;
   331→      }
   332→      const themeResetRec = await localDB.get('settings', 'themeDefaultLight');
   333→      const themeReset = !themeResetRec;
   334→      if (themeReset && theme === 'dark') {
   335→        theme = 'light';
   336→        void localDB.put('settings', { key: 'theme', value: 'light' });
   337→      }
   338→      if (themeReset) void localDB.put('settings', { key: 'themeDefaultLight', value: true });
   339→
   340→      let wallpaperPreset = 'graphite';
   341→      let customWallpaperUrl: string | null = null;
   342→      if (wallpaperRec && typeof wallpaperRec.value === 'object' && wallpaperRec.value !== null) {
   343→        const w = wallpaperRec.value as { preset?: unknown; customBlob?: unknown };
   344→        if (w.customBlob instanceof Blob) {
   345→          customWallpaperUrl = URL.createObjectURL(w.customBlob);
   346→        } else if (typeof w.preset === 'string' && WALLPAPER_PRESETS.some((p) => p.id === w.preset)) {
   347→          wallpaperPreset = w.preset;
   348→        }
   349→      }
   350→
   351→      // 锁屏壁纸（完全独立，默认 graphite）：{ preset, customBlob } 同时记录，移除自定义后可回退到已选预设
   352→      let lockWallpaperPreset = 'graphite';
   353→      let lockCustomWallpaperUrl: string | null = null;
   354→      if (lockWallpaperRec && typeof lockWallpaperRec.value === 'object' && lockWallpaperRec.value !== null) {
   355→        const lw = lockWallpaperRec.value as LockWallpaperRecord;
   356→        if (typeof lw.preset === 'string' && WALLPAPER_PRESETS.some((p) => p.id === lw.preset)) {
   357→          lockWallpaperPreset = lw.preset;
   358→        }
   359→        if (lw.customBlob instanceof Blob) {
   360→          lockCustomWallpaperUrl = URL.createObjectURL(lw.customBlob);
   361→        }
   362→      }
   363→
   364→      let apiConfig: ApiConfig = { ...DEFAULT_API_CONFIG };
   365→      if (apiRec && typeof apiRec.value === 'object' && apiRec.value !== null) {
   366→        // 兼容：旧版明文记录 / 新版密文信封都从这里解出
   367→        const v = ((await decryptValue<Partial<ApiConfig>>(apiRec.value)) ?? apiRec.value) as Partial<ApiConfig>;
   368→        apiConfig = {
   369→          baseUrl: typeof v.baseUrl === 'string' && v.baseUrl ? v.baseUrl : DEFAULT_API_CONFIG.baseUrl,
   370→          apiKey: typeof v.apiKey === 'string' ? v.apiKey : '',
   371→          model: typeof v.model === 'string' && v.model ? v.model : DEFAULT_API_CONFIG.model,
   372→          temperature: typeof v.temperature === 'number' ? v.temperature : DEFAULT_API_CONFIG.temperature,
   373→          maxTokens: typeof v.maxTokens === 'number' ? v.maxTokens : DEFAULT_API_CONFIG.maxTokens,
   374→        };
   375→        // 旧版明文静默升级：立即以密文回写（有 apiKey 才有必要；无 key 的占位配置不动）
   376→        const wasPlain = !('__enc' in (apiRec.value as object));
   377→        if (wasPlain && typeof apiConfig.apiKey === 'string' && apiConfig.apiKey.length > 0) {
   378→          void encryptValue(apiConfig)
   379→            .then((enc) => localDB.put('settings', { key: 'apiConfig', value: enc }))
   380→            .catch(() => undefined);
   381→        }
   382→      }
   383→
   384→      let apiPresets: ApiPreset[] = [];
   385→      if (presetsRec && presetsRec.value) {
   386→        // 兼容：旧版明文数组 / 新版密文信封
   387→        const pv = (await decryptValue<ApiPreset[]>(presetsRec.value)) ?? presetsRec.value;
   388→        if (Array.isArray(pv)) {
   389→          apiPresets = pv.filter(
   390→            (p): p is ApiPreset =>
   391→              typeof p === 'object' &&
   392→              p !== null &&
   393→              typeof (p as ApiPreset).id === 'string' &&
   394→              typeof (p as ApiPreset).name === 'string' &&
   395→              typeof (p as ApiPreset).config === 'object'
   396→          );
   397→          // 旧版明文含 apiKey 时静默升级为密文
   398→          const hasKey = apiPresets.some((p) => typeof p.config?.apiKey === 'string' && p.config.apiKey.length > 0);
   399→          if (!('__enc' in (presetsRec.value as object)) && hasKey) {
   400→            void encryptValue(apiPresets)
   401→              .then((enc) => localDB.put('settings', { key: 'apiPresets', value: enc }))
   402→              .catch(() => undefined);
   403→          }
   404→        }
   405→      }
   406→
   407→      // 识图模型配置（含 apiKey，密文信封，与 apiConfig 同策略；未配置 = baseUrl 空）
   408→      let visionConfig: VisionConfig = { ...DEFAULT_VISION_CONFIG };
   409→      if (visionRec && typeof visionRec.value === 'object' && visionRec.value !== null) {
   410→        const v = ((await decryptValue<Partial<VisionConfig>>(visionRec.value)) ?? visionRec.value) as Partial<VisionConfig>;
   411→        visionConfig = {
   412→          baseUrl: typeof v.baseUrl === 'string' ? v.baseUrl : '',
   413→          apiKey: typeof v.apiKey === 'string' ? v.apiKey : '',
   414→          model: typeof v.model === 'string' ? v.model : '',
   415→        };
   416→        const visionWasPlain = !('__enc' in (visionRec.value as object));
   417→        if (visionWasPlain && visionConfig.apiKey.length > 0) {
   418→          void encryptValue(visionConfig)
   419→            .then((enc) => localDB.put('settings', { key: 'visionConfig', value: enc }))
   420→            .catch(() => undefined);
   421→        }
   422→      }
   423→
   424→      let visionPresets: VisionPreset[] = [];
   425→      if (visionPresetsRec && visionPresetsRec.value) {
   426→        const vv = (await decryptValue<VisionPreset[]>(visionPresetsRec.value)) ?? visionPresetsRec.value;
   427→        if (Array.isArray(vv)) {

   667→/** 由预设 id + 自定义 URL 解析壁纸背景样式（手机壳/锁屏/前景明暗共用，故导出）。

   670→ *  否则锁屏/主屏头几帧透明，会透出下层界面（刷新时主界面一闪而过的根因） */
   671→export function resolveWallpaperStyle(presetId: string, customUrl: string | null): CSSProperties {
   672→  if (customUrl) {
   673→    return {
   674→      backgroundColor: '#1c1c1e',
   675→      backgroundImage: `url(${customUrl})`,
   676→      backgroundSize: 'cover',

   678→      backgroundRepeat: 'no-repeat',

   684→  if (preset.css.startsWith('url(')) {

   691→      backgroundPosition: 'center',
   692→      backgroundRepeat: 'no-repeat',
   693→    };
   694→  }
   695→  return { backgroundColor: preset.base, backgroundImage: preset.css };
   696→}

   734→// ---------------- 壁纸亮度实测（锁屏文字跟随背景色） ----------------
   735→
   736→/** 区域平均亮度高于该值视为偏浅（该区域文字用黑色） */
   737→const LIGHT_LUMINANCE_THRESHOLD = 0.5;
   738→
   739→/** 图片壁纸亮度实测结果的模块级缓存（url → 三区域结果）：
   740→ *  同一张壁纸被多处独立 hook消费（图标底座 ×N / 锁屏 / 状态栏 / 页点），
   741→ *  共享缓存避免每个实例重复解码测量；后挂载的实例（如打开 Spotlight/多任务）
   742→ *  首帧即命中，不会闪「测量中回退静态标记」的错误颜色。上限 32 条防泄漏 */
   743→const wallpaperLightCache = new Map<string, { top: boolean; bottom: boolean; all: boolean }>();
   744→const WALLPAPER_LIGHT_CACHE_MAX = 32;
   745→
   746→/** 用已解码的图片实测三区域亮度并写入共享缓存（与 useMeasuredWallpaperLight 同一算法）。
   747→ *  开机预热调用：锁屏/主屏挂载首帧即拿到正确前景色，不闪「测量中回退静态标记」的跳变 */
   748→function measureWallpaperLightInto(url: string, img: HTMLImageElement): void {
   749→  if (wallpaperLightCache.has(url)) return;
   750→  try {
   751→    const W = 32;
   752→    const H = 64;
   753→    const canvas = document.createElement('canvas');
   754→    canvas.width = W;
   755→    canvas.height = H;
   756→    const ctx = canvas.getContext('2d', { willReadFrequently: true });
   757→    if (!ctx) return;
   758→    ctx.drawImage(img, 0, 0, W, H);
   759→    const { data } = ctx.getImageData(0, 0, W, H);
   760→    const rowLum: number[] = [];
   761→    for (let y = 0; y < H; y++) {
   762→      let sum = 0;
   763→      for (let x = 0; x < W; x++) {
   764→        const i = (y * W + x) * 4;
   765→        sum += luminanceOf([data[i], data[i + 1], data[i + 2]]);
   766→      }
   767→      rowLum.push(sum / W);
   768→    }
   769→    const avgRows = (a: number, b: number) => {
   770→      let sum = 0;
   771→      for (let y = a; y < b; y++) sum += rowLum[y];
   772→      return sum / (b - a);
   773→    };
   774→    if (wallpaperLightCache.size >= WALLPAPER_LIGHT_CACHE_MAX) {
   775→      wallpaperLightCache.delete(wallpaperLightCache.keys().next().value as string);
   776→    }
   777→    wallpaperLightCache.set(url, {
   778→      top: avgRows(0, Math.floor(H * 0.35)) > LIGHT_LUMINANCE_THRESHOLD,
   779→      bottom: avgRows(Math.floor(H * 0.86), H) > LIGHT_LUMINANCE_THRESHOLD,
   780→      all: avgRows(0, H) > LIGHT_LUMINANCE_THRESHOLD,
   781→    });
   782→  } catch {
   783→    // 画布被污染等异常：静默放弃，hook 内回退预设静态标记
   784→  }
   785→}

   787→/**
   788→ * 开机预载壁纸（修复刚打开网页时锁屏「闪一下」）：
   789→ * 黑屏开机门控期间把主屏/锁屏壁纸图提前拉取解码并预热亮度实测缓存，
   790→ * 锁屏/主屏挂载首帧即完整壁纸 + 正确前景色，不再「先铺底色 → 图片弹入」。
   791→ * data:/blob: 之外全部适用（data URL 无网络请求无需预载）；
   792→ * 整体带超时兜底，最坏多等 timeoutMs 即放行，绝不卡死开机。
   793→ */
   794→async function preloadWallpapersForBoot(urls: (string | null | undefined)[], timeoutMs = 1500): Promise<void> {
   795→  const list = [...new Set(urls.filter((u): u is string => !!u && !u.startsWith('data:')))];
   796→  if (list.length === 0) return;
   797→  const work = list.map(
   798→    (url) =>
   799→      new Promise<void>((res) => {
   800→        const img = new Image();
   801→        img.onload = () => {
   802→          measureWallpaperLightInto(url, img);
   803→          res();
   804→        };
   805→        img.onerror = () => res();
   806→        img.src = url;
   807→      })
   808→  );
   809→  await Promise.race([Promise.all(work), new Promise<void>((r) => setTimeout(r, timeoutMs))]);
   810→}

   827→/** 解析 CSS 颜色为 [r,g,b]（支持 #rgb/#rrggbb/#rrggbbaa 与 rgb()/rgba()） */
   828→function parseCssColor(input: string): [number, number, number] | null {
   829→  const s = input.trim();
   830→  let m = s.match(/^#([0-9a-f]{3,8})$/i);
   831→  if (m) {
   832→    let hex = m[1];
   833→    if (hex.length === 3 || hex.length === 4) hex = [...hex].map((c) => c + c).join('');
   834→    const r = parseInt(hex.slice(0, 2), 16);
   835→    const g = parseInt(hex.slice(2, 4), 16);
   836→    const b = parseInt(hex.slice(4, 6), 16);
   837→    if (Number.isFinite(r) && Number.isFinite(g) && Number.isFinite(b)) return [r, g, b];
   838→    return null;
   839→  }
   840→  m = s.match(/^rgba?\(([^)]+)\)$/i);
   841→  if (m) {
   842→    const parts = m[1].split(/[\s,/]+/).filter(Boolean).map(Number);
   843→    if (parts.length >= 3 && parts.slice(0, 3).every((n) => Number.isFinite(n))) {
   844→      return [parts[0], parts[1], parts[2]];
   845→    }
   846→  }
   847→  return null;
   848→}

   850→function luminanceOf([r, g, b]: [number, number, number]): number {
   851→  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255;
   852→}

   855→ * 分区域实测壁纸亮度（锁屏文字随背景变色）：
   856→ * - 图片壁纸：解码后缩到 32×64 逐行取 Rec.709 亮度，顶部 0–35%（状态栏/时钟/小组件）与
   857→ *   底部 86–100%（上滑提示/快捷按钮/Home 横杠）分别求均值 —— 解决「上亮下暗」壁纸
   858→ *   （如雾山上白山+下深林）一刀切判色导致某端文字看不清的问题；
   859→ * - 纯色/渐变壁纸：解析样式里的色值，按出现顺序（渐变自上而下）前半=顶/后半=底；
   860→ * 返回 true=偏浅（黑字）/ false=偏深（白字）/ null=测量中或失败（调用方回退预设静态标记）。
   861→ */
   862→export function useMeasuredWallpaperLight(style: CSSProperties): WallpaperLightness {
   863→  const bgImage = typeof style.backgroundImage === 'string' ? style.backgroundImage : '';
   864→  const bgColor = typeof style.backgroundColor === 'string' ? style.backgroundColor : '';
   865→  const url = useMemo(() => bgImageUrlOf(bgImage), [bgImage]);
   866→
   867→  // 纯色/渐变壁纸：同步纯函数推导
   868→  const colorLight = useMemo<WallpaperLightness>(() => {
   869→    if (url) return { top: null, bottom: null, all: null }; // 图片壁纸走异步实测
   870→    const tokens = `${bgColor} ${bgImage}`.match(/#[0-9a-fA-F]{3,8}\b|rgba?\([^)]+\)/g) ?? [];
   871→    const lums = tokens
   872→      .map(parseCssColor)
   873→      .filter((c): c is [number, number, number] => c !== null)
   874→      .map(luminanceOf);
   875→    if (lums.length === 0) return { top: null, bottom: null, all: null };
   876→    const half = Math.max(1, Math.ceil(lums.length / 2));
   877→    const avg = (arr: number[]) => arr.reduce((acc, v) => acc + v, 0) / arr.length;
   878→    return {
   879→      top: avg(lums.slice(0, half)) > LIGHT_LUMINANCE_THRESHOLD,
   880→      bottom: avg(lums.slice(Math.max(0, lums.length - half))) > LIGHT_LUMINANCE_THRESHOLD,
   881→      all: avg(lums) > LIGHT_LUMINANCE_THRESHOLD,
   882→    };
   883→  }, [bgColor, bgImage, url]);
   884→
   885→  // 图片壁纸：解码后异步实测。结果绑定当时的 url，防止换壁纸瞬间新旧串值
   886→  const [imgState, setImgState] = useState<{ url: string; top: boolean; bottom: boolean; all: boolean } | null>(null);
   887→  useEffect(() => {
   888→    if (!url) return;
   889→    // 共享缓存已命中：渲染时已同步消费，无需重复解码测量
   890→    if (wallpaperLightCache.has(url)) return;
   891→    let cancelled = false;
   892→    const img = new Image();
   893→    img.onload = () => {
   894→      if (cancelled) return;
   895→      try {
   896→        const W = 32;
   897→        const H = 64;
   898→        const canvas = document.createElement('canvas');
   899→        canvas.width = W;
   900→        canvas.height = H;
   901→        const ctx = canvas.getContext('2d', { willReadFrequently: true });
   902→        if (!ctx) return; // 保持 null → 回退静态标记
   903→        ctx.drawImage(img, 0, 0, W, H);
   904→        const { data } = ctx.getImageData(0, 0, W, H);
   905→        const rowLum: number[] = [];
   906→        for (let y = 0; y < H; y++) {
   907→          let sum = 0;
   908→          for (let x = 0; x < W; x++) {
   909→            const i = (y * W + x) * 4;
   910→            sum += luminanceOf([data[i], data[i + 1], data[i + 2]]);
   911→          }
   912→          rowLum.push(sum / W);
   913→        }
   914→        const avgRows = (a: number, b: number) => {
   915→          let sum = 0;
   916→          for (let y = a; y < b; y++) sum += rowLum[y];
   917→          return sum / (b - a);
   918→        };
   919→        const result = {
   920→          top: avgRows(0, Math.floor(H * 0.35)) > LIGHT_LUMINANCE_THRESHOLD,
   921→          bottom: avgRows(Math.floor(H * 0.86), H) > LIGHT_LUMINANCE_THRESHOLD,
   922→          all: avgRows(0, H) > LIGHT_LUMINANCE_THRESHOLD,
   923→        };
   924→        // 写入共享缓存（简单上限淘汰，防长会话换壁纸累积泄漏）
   925→        if (wallpaperLightCache.size >= WALLPAPER_LIGHT_CACHE_MAX) {
   926→          wallpaperLightCache.delete(wallpaperLightCache.keys().next().value as string);
   927→        }
   928→        wallpaperLightCache.set(url, result);
   929→        setImgState({ url, ...result });
   930→      } catch {
   931→        // 画布被污染等异常：保持 null 回退静态标记
   932→      }
   933→    };
   934→    img.onerror = () => {
   935→      if (!cancelled) setImgState(null);
   936→    };
   937→    img.src = url;
   938→    return () => {
   939→      cancelled = true;
   940→    };
   941→  }, [url]);
   942→
   943→  if (url) {
   944→    // 共享缓存命中：同步返回（后挂载的图标/多任务/Spotlight 首帧即拿到结果，
   945→    // 不闪「测量中回退静态标记」的错误颜色，也避免 effect 内 setState）
   946→    const cached = wallpaperLightCache.get(url);
   947→    if (cached) return cached;
   948→    return imgState && imgState.url === url
   949→      ? { top: imgState.top, bottom: imgState.bottom, all: imgState.all }
   950→      : { top: null, bottom: null, all: null };
   951→  }
   952→  return colorLight;
   953→}
