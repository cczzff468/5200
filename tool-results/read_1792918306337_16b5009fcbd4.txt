    88→    88→function hslToHex(h: number, s: number, l: number): string {
    89→    89→  const f = (n: number) => {
    90→    90→    const k = (n + h / 30) % 12;
    91→    91→    const c = l - s * Math.min(l, 1 - l) * Math.max(-1, Math.min(k - 3, 9 - k, 1));
    92→    92→    return Math.round(255 * c)
    93→    93→      .toString(16)
    94→    94→      .padStart(2, '0');
    95→    95→  };
    96→    96→  return `#${f(0)}${f(8)}${f(4)}`;
    97→    97→}

   104→   104→function hexToHue(hex: string): number {
   105→   105→  const n = parseInt(hex.slice(1), 16);
   106→   106→  const r = ((n >> 16) & 255) / 255;
   107→   107→  const g = ((n >> 8) & 255) / 255;
   108→   108→  const b = (n & 255) / 255;
   109→   109→  const mx = Math.max(r, g, b);
   110→   110→  const mn = Math.min(r, g, b);
   111→   111→  if (mx === mn) return 0;
   112→   112→  const d = mx - mn;
   113→   113→  let h: number;
   114→   114→  if (mx === r) h = ((g - b) / d + (g < b ? 6 : 0)) * 60;
   115→   115→  else if (mx === g) h = ((b - r) / d + 2) * 60;
   116→   116→  else h = ((r - g) / d + 4) * 60;
   117→   117→  return h;
   118→   118→}

   550→   550→
   551→   551→export default function Wardrobe({
   552→   552→  open,
   553→   553→  wear,
   554→   554→  wear2,
   555→   555→  onChange,
   556→   556→  onChange2,
   557→   557→  onClose,
   558→   558→  petWear,
   559→   559→  petCoat,
   560→   560→  onPetChange,
   561→   561→  initialChar,
   562→   562→}: WardrobeProps) {
   563→   563→  const [char, setChar] = useState<'girl' | 'boy' | 'pet'>(initialChar ?? 'girl');
   564→   564→  const [tab, setTab] = useState<WardTab>('hair');
   565→   565→  const inDressMode = wear.dress >= 0;
   566→   566→  const tabs = char === 'girl' ? TABS_GIRL : TABS_BOY;
   567→   567→
   568→   568→  /** 切换角色：保留当前分类（女生专属的连衣裙在男生侧回落到上衣；宠物栏不占用 tab） */
   569→   569→  const switchChar = (c: 'girl' | 'boy' | 'pet') => {
   570→   570→    if (c === char) return;
   571→   571→    sfx.tap();
   572→   572→    setChar(c);
   573→   573→    if (c === 'boy' && tab === 'dress') setTab('top');
   574→   574→  };
   575→   575→
   576→   576→  const items = useMemo(() => {
   577→   577→    if (char === 'boy') {
   578→   578→      if (tab === 'hair') return BOY_HAIR_ORDER.map((h) => ({ key: h, name: BOY_HAIR_NAMES[h] ?? h }));
   579→   579→      if (tab === 'top') return BOY_TOPS.map((t, i) => ({ key: String(i), name: t.name }));
   580→   580→      if (tab === 'bottom') return BOY_BOTTOMS.map((b, i) => ({ key: String(i), name: b.name }));
   581→   581→      return BOY_SHOES.map((s, i) => ({ key: String(i), name: s.name }));
   582→   582→    }
   583→   583→    if (tab === 'hair') return HAIR_ORDER.map((h) => ({ key: h, name: HAIR_NAMES[h] ?? h }));
   584→   584→    if (tab === 'dress') return DRESSES.map((d, i) => ({ key: String(i), name: d.name }));
   585→   585→    if (tab === 'top') return TOPS.map((t, i) => ({ key: String(i), name: t.name }));
   586→   586→    if (tab === 'bottom') return BOTTOMS.map((b, i) => ({ key: String(i), name: b.name }));
   587→   587→    return SHOES.map((s, i) => ({ key: String(i), name: s.name }));
   588→   588→  }, [char, tab]);
   589→   589→
   590→   590→  const isSel = (key: string) => {
   591→   591→    if (char === 'boy') {
   592→   592→      if (tab === 'hair') return wear2.hair === key;
   593→   593→      if (tab === 'top') return wear2.top === Number(key);
   594→   594→      if (tab === 'bottom') return wear2.bottom === Number(key);
   595→   595→      return wear2.shoes === Number(key);
   596→   596→    }
   597→   597→    if (tab === 'hair') return wear.hair === key;
   598→   598→    if (tab === 'dress') return inDressMode && wear.dress === Number(key);
   599→   599→    if (tab === 'top') return !inDressMode && wear.top === Number(key);
   600→   600→    if (tab === 'bottom') return !inDressMode && wear.bottom === Number(key);
   601→   601→    return wear.shoes === Number(key);
   602→   602→  };
   603→   603→
   604→   604→  const pick = (key: string) => {
   605→   605→    sfx.tap();
   606→   606→    if (char === 'boy') {
   607→   607→      if (tab === 'hair') onChange2({ ...wear2, hair: key as BoyHairStyle });
   608→   608→      else if (tab === 'top') onChange2({ ...wear2, top: Number(key) });
   609→   609→      else if (tab === 'bottom') onChange2({ ...wear2, bottom: Number(key) });
   610→   610→      else onChange2({ ...wear2, shoes: Number(key) });
   611→   611→      return;
   612→   612→    }
   613→   613→    if (tab === 'hair') onChange({ ...wear, hair: key as GirlHairStyle });
   614→   614→    else if (tab === 'dress') onChange({ ...wear, dress: Number(key) });
   615→   615→    else if (tab === 'top') onChange({ ...wear, dress: -1, top: Number(key) });
   616→   616→    else if (tab === 'bottom') onChange({ ...wear, dress: -1, bottom: Number(key) });
   617→   617→    else onChange({ ...wear, shoes: Number(key) });
   618→   618→  };
   619→   619→
   620→   620→  // —— 调色盘（当前角色 + 当前分类的槽位；null = 还原默认款色） ——
   621→   621→  const slot = TINT_SLOT[tab];
   622→   622→  const curTint = (char === 'girl' ? wear.tints?.[slot] : wear2.tints?.[slot]) ?? null;
   623→   623→  const paletteColors = tab === 'hair' ? HAIR_COLORS : CLOTH_COLORS;
   624→   624→
   625→   625→  const setTint = (col: string | null) => {
   626→   626→    sfx.tap();
   627→   627→    if (char === 'girl') {
   628→   628→      const t: WearTints = { ...(wear.tints ?? {}) };
   629→   629→      if (col) t[slot] = col;
   630→   630→      else delete t[slot];
   631→   631→      onChange({ ...wear, tints: Object.keys(t).length > 0 ? t : undefined });
   632→   632→    } else {
   633→   633→      const t: WearTints = { ...(wear2.tints ?? {}) };
   634→   634→      if (col) t[slot] = col;
   635→   635→      else delete t[slot];
   636→   636→      onChange2({ ...wear2, tints: Object.keys(t).length > 0 ? t : undefined });
   637→   637→    }
   638→   638→  };
   639→   639→
   640→   640→  // —— 宠物装扮（v8.3：缺省 = 奶油猫 + 樱红项圈；scarf/bow 为 null = 不戴） ——
   641→   641→  const pw: PetWear = petWear ?? {};
   642→   642→  const curCoat: PetCoat = petCoat ?? 'cream';
   643→   643→  const curCollar = pw.collar ?? '#e46a6a';
   644→   644→  const petParts = [PET_COAT_NAMES[curCoat], `${PET_COLOR_NAMES[curCollar] ?? '彩色'}项圈`];
   645→   645→  if (pw.scarf) petParts.push('戴围巾');
   646→   646→  if (pw.bow) petParts.push('戴蝴蝶结');
