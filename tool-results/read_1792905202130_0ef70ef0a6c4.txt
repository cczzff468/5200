     4→ * 全局聊天流式请求总线（微信 / QQ / 信息 App 共用）。
     5→ *
     6→ * 把「发送消息 → 请求 AI 回复 → 接收流式数据 → 落盘聊天记录」的完整生命周期
     7→ * 从聊天页组件中解耦出来，由本模块（页面之外的全局单例）管理：

    18→ *    并广播 chat-stream-finalized 事件供会话列表刷新预览；

   172→/**
   173→ * 从 sessionKey 解析参与拉黑的会话（wx:<contactId> / qq:<contactId> / sms:c:<contactId>）；
   174→ * 群聊（wx:group: / qq:group:）、小助手（sms:assistant）及其他非联系人会话返回 null（不参与拉黑）。
   175→ * 拉黑状态按 App × 联系人 ID 隔离，这里的解析与各 App 的会话键同源同义。
   176→ */
   177→function blockSessionOf(sessionKey: string): { app: BlockApp; contactId: string } | null {
   178→  // 群聊会话无拉黑关系天然放行（拉黑按 App × 单联系人 ID 隔离，群成员不参与拉黑）；
   179→  // AI 助手会话同理（sms:assistant / assistant）：助手是系统侧角色，不进入「角色 ↔ 用户」拉黑图谱
   180→  if (sessionKey.startsWith('wx:group:') || sessionKey.startsWith('qq:group:')) return null;
   181→  if (sessionKey === 'sms:assistant' || sessionKey === 'assistant') return null;
   182→  if (sessionKey.startsWith('sms:c:')) return { app: 'sms', contactId: sessionKey.slice('sms:c:'.length) };
   183→  if (sessionKey.startsWith('sms:')) return null; // 其余 sms 键（历史助手兼容等）不参与拉黑
   184→  const m = /^(wx|qq):(.+)$/.exec(sessionKey);
   185→  return m ? { app: m[1] as BlockApp, contactId: m[2] } : null;
   186→}

   190→/**
   191→ * 需求「二/三/四」（AI 认知隔离）的注入实现（v2：单库 + 键作用域 + per-app 账号）：
   192→ * - 某 App 当前账号为大号（main）时发起聊天，把用户各小号/匿名号与 AI 的聊天记录摘录算成 digest，

   203→/** 小号/匿名号作用域键内聊天记录的 kv 裸键前缀（与 idb-kv 迁移清单同源；sms- 为更早版本遗留键） */

   209→/** 单条摘录文本最大长度（超出截断，防长图描述/长句撑爆 prompt） */

   211→/** digest 总字符上限（多小号兜底，超出截断） */

   214→/** 摘录条目（mine=true=小号侧发出；peer=该会话对方角色名；time=消息时间戳；app=来源平台标注） */
   215→interface AltDigestEntry {
   216→  time: number;
   217→  mine: boolean;
   218→  peer: string;
   219→  text: string;
   220→  app: string;
   221→}
   222→
   223→/** 单条消息 → 摘录条目；不参与 AI 上下文的消息（撤回/错误/系统行/通知/拉黑卡等）与空消息返回 null。
   224→ *  消息形状三端兼容：role me|user|self=小号侧，其余（peer/assistant/…）=对方；文本取 text|content。
   225→ *  appLabel：会话所在平台（微信/QQ/信息，用户规则四：添加来源要说对，不能猜） */
   226→function altDigestEntryOf(raw: unknown, peer: string, appLabel: string): AltDigestEntry | null {
   227→  if (!raw || typeof raw !== 'object') return null;
   228→  const m = raw as Record<string, unknown>;
   229→  if (m.recalled === true || m.error === true) return null;
   230→  const kind = typeof m.kind === 'string' ? m.kind : '';
   231→  // 与各 App 历史上下文同口径：系统提示/通知行/拉黑卡/群邀请卡/通话卡不进上下文，摘录同样排除
   232→  if (kind === 'sys' || kind === 'notice' || kind === 'blockreq' || kind === 'groupcard' || kind === 'call') return null;
   233→  const mine = m.role === 'me' || m.role === 'user' || m.role === 'self';
   234→  const time = typeof m.time === 'number' && m.time > 0 ? m.time : 0;
   235→  let text = typeof m.text === 'string' ? m.text : typeof m.content === 'string' ? m.content : '';
   236→  text = text.trim();
   237→  // 好友验证消息/打招呼不是聊天内容（用户规则五）：加好友过程消息区分标注——加进摘录但绝不当作
   238→  // 「小号说过的话」（添加成功提示纯系统行直接丢弃）；AI 只能说「TA 加我时验证消息写的是…」
   239→  const fr = typeof m.fr === 'string' ? m.fr : '';
   240→  if (fr === 'added') return null;
   241→  if (fr === 'apply') text = `[好友验证消息] ${text}`;
   242→  else if (fr === 'greet') text = `[打招呼] ${text}`;
   243→  if (text.startsWith('data:')) text = ''; // QQ 图片消息 content=dataURL：不出原始图，按 kind 出占位
   244→  if (kind === 'image') {
   245→    // 图片消息：有识图描述时带出内容（wx 端 content 恒为「[图片]」占位，desc 才是有效信息）
   246→    const desc = (m.img as { desc?: unknown } | undefined)?.desc;
   247→    if (typeof desc === 'string' && desc.trim()) text = `[图片]（图片内容：${desc.trim()}）`;
   248→  }
   249→  if (!text) {
   250→    // 非文本消息按 kind 出占位/描述（与各 App 历史序列化占位口径一致；图片描述已在上方带出）
   251→    if (kind === 'image') {
   252→      text = '[图片]';
   253→    } else if (kind === 'voice') {
   254→      const v = m.voice as { transcript?: unknown; localText?: unknown } | undefined;
   255→      const t = typeof v?.transcript === 'string' ? v.transcript.trim() : '';
   256→      const l = typeof v?.localText === 'string' ? v.localText.trim() : '';
   257→      text = t || l || '[语音]';
   258→    } else if (kind === 'sticker') {
   259→      const meaning = (m.stk as { meaning?: unknown } | undefined)?.meaning;
   260→      text = typeof meaning === 'string' && meaning.trim() ? `[表情包：${meaning.trim()}]` : '[表情包]';
   261→    } else if (kind === 'textcard') {
   262→      const cardText = (m.card as { text?: unknown } | undefined)?.text;
   263→      text = typeof cardText === 'string' && cardText.trim() ? `[文字图片]（卡片上写着：${cardText.trim()}）` : '[文字图片]';
   264→    } else if (kind === 'redpacket') text = '[红包]';
   265→    else if (kind === 'transfer') text = '[转账]';
   266→    else if (kind === 'location') text = '[位置]';
   267→    else if (kind === 'forward') text = '[转发消息]';
   268→    else if (kind === 'family') text = '[亲属卡]';
   269→    else return null; // 无文本且非已知类型：跳过
   270→  }
   271→  // 压成单行 + 截断（摘录是给 AI 看的上下文，不保留换行排版）
   272→  text = text.replace(/\s+/g, ' ').trim();
   273→  if (!text) return null;
   274→  if (text.length > ALT_DIGEST_MSG_MAX_CHARS) text = `${text.slice(0, ALT_DIGEST_MSG_MAX_CHARS)}…`;
   275→  return { time, mine, peer, text, app: appLabel };
   276→}

   278→/** kv 裸键（base）→ 会话对方角色名；非聊天键返回 null（联系人会话按 contactId 查共享联系人名映射） */
   279→function altSessionPeerName(key: string, nameById: Map<string, string>): string | null {
   280→  if (key === ALT_ASSISTANT_KEY) return '小助手';
   281→  for (const p of ALT_CHAT_KEY_PREFIXES) {
   282→    if (!key.startsWith(p)) continue;
   283→    let id = key.slice(p.length);
   284→    if (id.startsWith('c:')) id = id.slice(2); // 信息端会话键形如 c:<contactId>
   285→    if (nameById.has(id)) return nameById.get(id) as string;
   286→    return p === 'wx-chat-msgs:' ? '微信联系人' : p === 'qq-chat-msgs:' ? 'QQ好友' : '联系人';
   287→  }
   288→  return null;
   289→}
   290→
   291→/** kv 裸键（base）→ 来源平台标注（用户规则四：添加来源/在哪聊的要准，AI 不许猜）：
   292→ *  微信键系→微信、QQ 键系→QQ、信息键系（含历史 sms-）→信息；其余返回 '' */
   293→function altSessionAppLabel(key: string): string {
   294→  if (key.startsWith('wx-chat-msgs:')) return '微信';
   295→  if (key.startsWith('qq-chat-msgs:')) return 'QQ';
   296→  if (key.startsWith('ios-chat-msgs:') || key.startsWith('sms-chat-msgs:')) return '信息';
   297→  if (key === ALT_ASSISTANT_KEY) return '信息';
   298→  return '';
   299→}

   301→/** 相对时间（小号摘录标题行用：最近一条消息距今多久） */
   302→function altRelTime(ts: number): string {
   303→  if (!ts) return '';
   304→  const diff = Date.now() - ts;
   305→  if (diff < 60_000) return '刚刚';
   306→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}分钟前`;
   307→  if (diff < 86_400_000) return `${Math.floor(diff / 3_600_000)}小时前`;
   308→  if (diff < 30 * 86_400_000) return `${Math.floor(diff / 86_400_000)}天前`;
   309→  return `${Math.floor(diff / (30 * 86_400_000))}个月前`;
   310→}

   312→/**
   313→ * 计算「用户的小号」聊天摘录 digest（仅在该 App 当前账号为大号时返回内容；否则/失败/无数据 → null）。
   314→ * v2 单库：一次读主库 kv 全部原始行（各账号键带 `--{id}` 后缀），按 parseScopedKey 解析出的
   315→ * 账号 id 归属；base 匹配聊天键前缀（或小助手精确键）的行即该账号的聊天消息；
   316→ * 联系人名字映射用共享库 contacts（三端共享联系人，id→名字一次建全）。
   317→ * 跨会话合并消息按时间倒序取最新 24 条；每账号一个块（标题行用注册表条目：小号名/匿名账号 + 号码），
   318→ * 块间空行；无聊天记录的账号跳过；主库读取失败返回 null（零注入）。
   319→ */
   320→async function computeAltAccountsDigest(app: AccountApp): Promise<string | null> {
   321→  if (typeof window === 'undefined' || !('indexedDB' in window)) return null;
   322→  // 门控：仅该 App 当前账号为大号时注入（小号/匿名号侧 AI 与用户互为陌生人）
   323→  const currentId = getActiveAccountIdFor(app);
   324→  if (currentId !== MAIN_ACCOUNT_ID) return null;
   325→  const alts = getAccounts().filter((a) => a && a.kind !== 'main' && a.id !== currentId);
   326→  if (alts.length === 0) return null;
   327→  // 主库一次 getAll 拿 kv 全部原始行（含各账号后缀键）+ contacts 全部记录
   328→  let kvRows: { key: string; value: unknown }[];
   329→  let contactRows: { id: string; name: string }[];
   330→  try {
   331→    const [kv, contacts] = await Promise.all([localDB.getAll('kv'), localDB.getAll('contacts')]);
   332→    kvRows = kv as { key: string; value: unknown }[];
   333→    contactRows = contacts as { id: string; name: string }[];
   334→  } catch {
   335→    return null; // 主库读取失败：本轮零注入，不影响聊天
   336→  }
   337→  const nameById = new Map<string, string>();
   338→  for (const c of contactRows) {
   339→    if (c && typeof c.id === 'string' && typeof c.name === 'string' && c.name.trim()) {
   340→      nameById.set(c.id, c.name.trim());
   341→    }
   342→  }
   343→  // 按账号收集摘录条目：键后缀归属小号账号 + base（裸键）匹配聊天键
   344→  const altIds = new Set(alts.map((a) => a.id));
   345→  const entriesByAcc = new Map<string, AltDigestEntry[]>();
   346→  for (const row of kvRows) {
   347→    if (!row || typeof row.key !== 'string' || !Array.isArray(row.value)) continue;
   348→    const scoped = parseScopedKey(row.key);
   349→    if (!scoped || !altIds.has(scoped.accId)) continue; // 大号原键/其他账号数据跳过
   350→    const peer = altSessionPeerName(scoped.base, nameById);
   351→    if (!peer) continue;
   352→    const appLabel = altSessionAppLabel(scoped.base);
   353→    const list = entriesByAcc.get(scoped.accId) ?? [];
   354→    for (const raw of row.value) {
   355→      const e = altDigestEntryOf(raw, peer, appLabel);
   356→      if (e) list.push(e);
   357→    }
   358→    entriesByAcc.set(scoped.accId, list);
   359→  }
   360→  const blocks: string[] = [];
   361→  for (const acc of alts) {
   362→    const entries = entriesByAcc.get(acc.id) ?? [];
   363→    if (entries.length === 0) continue; // 无聊天记录的账号跳过
   364→    // 跨会话按时间倒序取最新 N 条，再按时间正序输出（对话自然阅读顺序）
   365→    entries.sort((a, b) => b.time - a.time);
   366→    const picked = entries.slice(0, ALT_DIGEST_MAX_MSGS);
   367→    picked.sort((a, b) => a.time - b.time);
   368→    // 标题行用注册表条目（小号名/匿名账号 + 手机号），摘录格式与 v1 完全一致；
   369→    // C15：大号档账号（grade='main'，Task 33 引入）此前也被装进【用户的小号】标题下——
   370→    // 标题补「（大号）」让 AI 知道这是用户的大号马甲（语义上不再是「小号」）
   371→    const label =
   372→      acc.kind === 'anon'
   373→        ? '匿名账号'
   374→        : (acc.name.trim() || '小号') + (acc.grade === 'main' ? '（大号）' : '');
   375→    const phone = acc.phone ? `（${acc.phone}）` : '';
   376→    const recent = altRelTime(picked[picked.length - 1].time);
   377→    const lines = [`〔${label}〕${phone}与你的最近聊天${recent ? `（最近消息：${recent}）` : ''}：`];
   378→    for (const e of picked) {
   379→      // 每条带来源平台标注（微信/QQ/信息；用户规则四：AI 说「在哪加的/在哪聊的」必须准）：
   380→      // 「用户（微信）：想你」=小号在微信说的；peer 行=你（AI）在那个平台回复的
   381→      lines.push(`${e.mine ? '用户' : e.peer}${e.app ? `（${e.app}）` : ''}：${e.text}`);
   382→    }
   383→    blocks.push(lines.join('\n'));
   384→  }
   385→  if (blocks.length === 0) return null;
   386→  let digest = blocks.join('\n\n');
   387→  if (digest.length > ALT_DIGEST_MAX_CHARS) digest = `${digest.slice(0, ALT_DIGEST_MAX_CHARS)}…`;
   388→  return digest;
   389→}

   391→/** 把 digest 段合并进请求消息：追加到首条 system 消息（人设 prompt）末尾；
   392→ *  极端情况下请求里没有 system 消息（理论上不发生——人设/时间块恒在首位）时前置一条附加 system。 */
   393→function mergeAltDigestIntoMessages(messages: ChatPayloadMessage[], digest: string): ChatPayloadMessage[] {
   394→  const section = buildAltAccountsSection(digest);
   395→  if (!section) return messages;
   396→  const idx = messages.findIndex((m) => m.role === 'system');
   397→  if (idx === -1) return [{ role: 'system', content: section }, ...messages];
   398→  const out = messages.slice();
   399→  out[idx] = { ...out[idx], content: `${out[idx].content}\n\n${section}` };
   400→  return out;
   401→}

   413→/** 全部流（含已结束未清理的：留给页面重进时同步，超量后按时间淘汰） */

   416→const activePayloads = new Map<string, { messages: ChatPayloadMessage[]; config: ApiConfig }>();

   418→const finalizedSubs = new Set<(sessionKey: string) => void>();

   420→/**
   421→ * 新流开始前，若服务端还有关页期间生成中的接力回复，上报 cancel 让其作废：
   422→ * 否则用户回到网页马上发新消息时，会先收到迟到的旧接力回复、再收到新回复（双回复）。
   423→ * 尽力而为：发失败（如刚关页重开时序颠倒）也不影响本轮聊天，接力回复只会多到一条。
   424→ */
   425→function cancelBgRelay(sessionKey: string): void {
   426→  try {
   427→    void fetch('/api/chat/bg', {
   428→      method: 'POST',
   429→      headers: { 'Content-Type': 'application/json' },
   430→      body: JSON.stringify({ mode: 'cancel', sessionKey }),
   431→      keepalive: true,
   432→    }).catch(() => undefined);
   433→  } catch {
   434→    // 同上：尽力而为
   435→  }
   436→}

   438→function emit(): void {
   439→  subs.forEach((fn) => fn());
   440→}

   442→function emitFinalized(sessionKey: string): void {
   443→  finalizedSubs.forEach((fn) => {
   444→    try {
   445→      fn(sessionKey);
   446→    } catch {
   447→      // 监听器异常不影响其他监听器
   448→    }
   449→  });
   450→}

   452→/** 已结束的流最多保留 24 条（重进页面同步用），超出按开始时间淘汰最旧的 */
   453→function evictFinished(): void {
   454→  const finished = [...streams.entries()].filter(([, rt]) => rt.state.status !== 'streaming');
   455→  const overflow = finished.length - 24;
   456→  if (overflow <= 0) return;
   457→  finished
   458→    .sort((a, b) => a[1].state.startedAt - b[1].state.startedAt)
   459→    .slice(0, overflow)
   460→    .forEach(([key]) => streams.delete(key));
   461→}

   472→ * 此状态下 /api/chat 代理必然失败（服务端要求填写 API Key）、浏览器直连必然被 CORS 拦截，

   476→function isApiUnconfigured(cfg: ApiConfig): boolean {
   477→  if (cfg.apiKey.trim()) return false;
   478→  const raw = cfg.baseUrl.trim();
   479→  if (!raw) return true;
   480→  try {
   481→    return new URL(raw).hostname.toLowerCase() === 'api.openai.com';
   482→  } catch {
   483→    return false;
   484→  }

   487→async function runStream(rt: StreamRuntime, opts: BeginChatStreamOptions): Promise<void> {
   488→  const { apiConfig, messages } = opts;
   489→  // 流式分段器（回复条数 > 1 时启用）：边接收边切分，每凑齐一条完整句子立刻经 onSegment
   490→  // 投递上屏（不设条数上限：回复几条由生成端决定，切分端一句一条、不把溢出句堆进最后一条）；
   491→  // 原文只在本模块累计（raw），不写入展示状态 —— 流式期间页面只显示「正在输入」，
   492→  // 不再有「先全文后消失」的流式气泡；分段与展示状态完全分离（流式和分条不改同一块状态）
   493→  const multi = typeof opts.replyCount === 'number' && opts.replyCount > 1;
   494→  let scanner = multi && opts.onSegment ? createReplySegmentScanner(opts.onSegment) : null;
   495→  // 条数多时消息总长更长：抬高 max_tokens 下限，防止多条连发被截断（代理与浏览器直连共用该配置）
   496→  const effConfig =
   497→    multi && opts.replyCount
   498→      ? { ...apiConfig, maxTokens: Math.max(apiConfig.maxTokens, opts.replyCount * 120) }
   499→      : apiConfig;
   500→  /** 增量统一入口：原文累计 + 多条模式进分段器（立刻放出已凑齐的完整消息） */
   501→  let raw = '';
   502→  const onDelta = (delta: string): void => {
   503→    // 换号中止（abortStreamsByPrefix）：不再喂原文/分段器——流中分段就此截断，
   504→    // 旧账号会话不再产生新的投递（finalize 挡板只拦最后一条，不拦流中分段，这里补齐）
   505→    if (rt.aborted) return;
   506→    raw += delta;
   507→    scanner?.push(delta);
   508→  };
   509→  /**
   510→   * 发起一轮流式请求并读完整条流（增量统一交给 onDelta）。
   511→   * 服务器代理失败 / directOnly / 内网地址时回退浏览器直连（SSE 解析）；抛错交给调用方。
   512→   */
   513→  const streamOnce = async (roundMessages: ChatPayloadMessage[]): Promise<void> => {
   514→    const res = await fetch('/api/chat', {
   515→      method: 'POST',
   516→      headers: { 'Content-Type': 'application/json' },
   517→      body: JSON.stringify({
   518→        messages: roundMessages,
   519→        config: effConfig,
   520→      }),
   521→    });
   522→    if (!res.ok || !res.body) {
   523→      let detail = '';
   524→      let directOnly = false;
   525→      try {
   526→        const data: unknown = await res.json();
   527→        if (data && typeof data === 'object') {
   528→          const rec = data as { error?: unknown; directOnly?: unknown };
   529→          if (typeof rec.error === 'string') detail = rec.error;
   530→          directOnly = rec.directOnly === true;
   531→        }
   532→      } catch {
   533→        // 忽略解析失败
   534→      }
   535→      if (directOnly || isPrivateApiUrl(apiConfig.baseUrl)) {
   536→        // 内网地址 / 服务器建议直连：改用浏览器直连流式请求（SSE 解析）；
   537→        // 空回复不在此处兜底：finalize 由各 App 写入自己的兜底文案
   538→        try {
   539→          await directChatStream(effConfig, roundMessages, onDelta);
   540→        } catch (directErr) {
   541→          // 公网地址直连也失败：保留服务器侧原因，便于区分地区限制还是 CORS 问题
   542→          const directMsg = directErr instanceof Error && directErr.message ? directErr.message : '';
   543→          if (!isPrivateApiUrl(apiConfig.baseUrl) && detail) {
   544→            throw new Error(`${detail}；浏览器直连也不可用${directMsg ? `：${directMsg}` : ''}`);
   545→          }
   546→          throw directErr;
   547→        }
   548→      } else {
   549→        throw new Error(detail || `请求失败（${res.status}）`);
   550→      }
   551→    } else {
   552→      // 纯文本流（text/plain）：字节增量统一交给 onDelta（多条模式进分段器）
   553→      const reader = res.body.getReader();
   554→      const decoder = new TextDecoder();
   555→      for (;;) {
   556→        const { done, value } = await reader.read();
   557→        if (done) break;
   558→        onDelta(decoder.decode(value, { stream: true }));
   559→      }
   560→    }
   561→  };
   562→
   563→  /**
   564→   * 最终兜底：代理 + 浏览器直连都失败后，请求服务端内置模型（forceSdk）生成回复；
   565→   * 成功把全量文本交给 onDelta 返回 true，失败（含非 200 / 空文本）返回 false 保留原错误。
   566→   */
   567→  const sdkFallbackOnce = async (roundMessages: ChatPayloadMessage[]): Promise<boolean> => {
   568→    try {
   569→      const res = await fetch('/api/chat', {
   570→        method: 'POST',
   571→        headers: { 'Content-Type': 'application/json' },
   572→        body: JSON.stringify({ messages: roundMessages, forceSdk: true }),
   573→      });
   574→      if (!res.ok) return false;
   575→      const text = await res.text();
   576→      if (!text.trim()) return false;
   577→      onDelta(text);
   578→      return true;
   579→    } catch {
   580→      return false;
   581→    }
   582→  };
   583→
   584→  // ---- 40-B AI 认知隔离：主账号下注入「用户的小号」聊天摘录 ----
   585→  // 每轮最多计算一次（IndexedDB 读很快）；非主账号/无小号数据/读取失败 → null，零注入零破坏。
   586→  // 小号/匿名号侧不注入：AI 对小号是陌生人，摘录也绝不反向泄露大号内容。
   587→  let baseMessages = messages;
   588→  // v2 per-app 账号：由 sessionKey 推导所属 App（wx:/qq:/其余=sms），摘录/披露检测都按该 App 当前账号门控
   589→  const accApp = streamAppOf(opts.sessionKey);
   590→  const accBefore = getActiveAccountIdFor(accApp);
   591→  try {
   592→    const altDigest = await computeAltAccountsDigest(accApp);
   593→    // 摘录读取窗口内该 App 发生换号（v2 切号不刷新页面）：本轮作废，绝不把旧账号上下文
   594→    // 的回复落进新账号的会话。与 abortStreamsByPrefix 同口径清理（finalized 预置跳过落盘、
   595→    // 移除流状态与 payload，防流状态卡在 streaming 阻塞该会话后续新回合；切号事件监听方
   596→    // 通常已先行中止，此处为兜底幂等清理）：
   597→    if (getActiveAccountIdFor(accApp) !== accBefore) {
   598→      rt.finalized = true;
   599→      rt.aborted = true;
   600→      cancelBgRelay(rt.state.sessionKey);
   601→      streams.delete(rt.state.sessionKey);
   602→      activePayloads.delete(rt.state.sessionKey);
   603→      emit();
   604→      return;
   605→    }
   606→    if (altDigest) baseMessages = mergeAltDigestIntoMessages(messages, altDigest);
   607→  } catch {
   608→    // 摘录失败不影响本轮聊天
   609→  }
   610→  // 摘录读取窗口内发生换号中止：本轮作废（与识图等待后的中止检查同口径）
   611→  if (rt.aborted) return;
   612→
   613→  // ---- 大号侧「小号是我」披露检测（用户规则一/二：账号关联记忆） ----
   614→  // 主号私聊里用户发言命中「小号」披露表述 → ① 落账号关联（kv 设备级共享，幂等合并）+
   615→  // 给每个关联小号预写全部四 App 的披露粘性标记（小号/电话侧下次来聊自动「认出」，免再亮明）；
   616→  // ② 给当前联系人的【主号侧】记忆库补一条「用户亲口告知小号」事件碎片（持久化，AI 后续能自然提起）。
   617→  // 仅主号 + 与 AI 角色的私聊；检测/写入任何失败都不影响本轮聊天（增强能力）。
   618→  try {
   619→    const linkCid = streamContactIdOf(opts.sessionKey);
   620→    if (accBefore === MAIN_ACCOUNT_ID && linkCid) {
   621→      const lastUserText = [...messages].reverse().find((m) => m.role === 'user')?.content ?? '';
   622→      const hitAltIds = memDetectMainAltDisclosure(lastUserText);
   623→      if (hitAltIds.length > 0) {
   624→        memLinkAltAccounts(linkCid, hitAltIds, lastUserText);
   625→        // 机主称谓：联系人 App 机主真名优先（编辑自己后按最新），设置 profile 名仅兑底
   626→        const ownerLabel = cachedOwnerName() || useSettings.getState().profile.name?.trim() || '机主';
   627→        const altLabels = hitAltIds
   628→          .map((id) => getAccountById(id)?.name?.trim() || '小号')
   629→          .join('、');
   630→        memAddEventFragment(
   631→          linkCid,
   632→          accApp,
   633→          `${ownerLabel}亲口告知：「${altLabels}」是${ownerLabel}的小号——小号和大号都是${ownerLabel}本人（同一个人）`,
   634→          { sourceTag: 'acc-link' },
   635→        );
   636→      }
   637→    }
   638→  } catch {
   639→    // 检测/落盘失败不影响本轮聊天
   640→  }
   641→
   642→  /** 本轮实际发送的消息（识图成功后会被替换为「原图消息 + 图片描述」的组合） */
   643→  let workMessages: ChatPayloadMessage[] = baseMessages;
   644→
   645→  try {
   646→    // 识图前置步骤（设置 › 识图模型；未配置 / 本轮无图片时跳过，文字聊天零影响）：
   647→    // 识图模型只负责「看」——把图片转成描述，追加为一条 user 上下文消息；
   648→    // 最终回复仍由聊天模型生成；识图失败展示系统提示（不进上下文、不当角色台词），本轮按无图片继续
   649→    if (opts.vision && opts.vision.images.length > 0) {
   650→      const visionConfig = useSettings.getState().visionConfig;
   651→      // 第二十九轮「让 AI 知道上传的图片长什么样」：识图不再依赖用户配置——
   652→      // describeImagesAuto 配置了走用户模型（失败自动降级），未配置直接用内置识图兜底；
   653→      // 内置也失败才回落「图片还在加载中」临时上下文（本轮按无图继续，不阻断聊天）
   654→      try {
   655→        const desc = await describeImagesAuto(visionConfig, { images: opts.vision.images, text: opts.vision.text });
   656→        if (!desc) throw new Error('识图模型返回空内容');
   657→        const n = opts.vision.images.length;
   658→        // fix3-8 识图描述上下文行带发送者：群聊里是谁发的图片就说谁发了（speakerLabel），
   659→        // 未传时维持「我」（单聊=机主本人），单数/复数两种既有句式均不变
   660→        const speaker = opts.vision.speakerLabel || '我';
   661→        const prefix = n > 1 ? `（${speaker}发了 ${n} 张图片，图片内容分别是：` : `（${speaker}发了一张图片，图片内容是：`;
   662→        workMessages = [...baseMessages, { role: 'user' as const, content: `${prefix}${desc}）` }];
   663→        // 描述回写落盘（onVision 由各 App 提供）：之后的聊天历史 AI 都能读到图片内容；
   664→        // 换号中止后不再回写（旧账号会话的落盘动作全部停止）
   665→        try {
   666→          if (!rt.aborted) opts.onVision?.(desc);
   667→        } catch {
   668→          // 落盘失败不影响本轮回复（描述已进上下文）
   669→        }
   670→      } catch (err) {
   671→        const detail = err instanceof Error && err.message ? err.message : '未知原因';
   672→        // 兜底临时上下文：明确图片已收到但看不到内容，引导自然回应（不发编造细节）；
   673→        // 临时消息不落盘，与识图描述上下文同口径；【100-c】文案剧情化（不提幕后配置）
   674→        const n = opts.vision.images.length;
   675→        const speaker = opts.vision.speakerLabel || '我';
   676→        const note =
   677→          n > 1
   678→            ? `（${speaker}发了 ${n} 张图片，不过图片还在加载中，你暂时看不到内容——不要编造图片里有什么，可以泛泛回应，或自然让我重新发一次。）`
   679→            : `（${speaker}发了一张图片，不过图片还在加载中，你暂时看不到内容——不要编造图片里有什么，可以泛泛回应，或自然让我重新发一次。）`;
   680→        workMessages = [...baseMessages, { role: 'user' as const, content: note }];
   681→        patchState(rt, { visionNotice: `图片识别失败，本次回复未结合图片（${detail}）` });
   682→      }
   683→    }
   684→    // 换号中止（识图等待窗口内发生）：旧账号会话整体作废，不再登记 payload/发起请求
   685→    if (rt.aborted) return;
   686→    // 按各 App 组装的消息发起请求（条数指令已注入人设 system 消息，一轮发完、不做补发）；
   687→    // 记录实际 payload：用户在流结束前关闭网页时，pagehide 上报服务端接力生成（bg-turn）
   688→    activePayloads.set(rt.state.sessionKey, { messages: workMessages, config: effConfig });
   689→    if (isApiUnconfigured(apiConfig)) {
   690→      // 未配置用户 API（无 Key + 官方默认地址）：代理与直连必然失败，跳过等待直接走
   691→      // 服务端内置模型（与下游失败兜底同一函数）；兜底失败才落错误文案引导用户去配置
   692→      const viaSdk = await sdkFallbackOnce(workMessages);
   693→      if (viaSdk) {
   694→        patchState(rt, { status: 'done' });
   695→      } else {
   696→        patchState(rt, {
   697→          status: 'error',
   698→          error: '还没有配置 API Key，内置模型也不可用；请到「设置 › API 配置」填写后重试',
   699→        });
   700→      }
   701→    } else {
   702→      await streamOnce(workMessages);
   703→      patchState(rt, { status: 'done' });
   704→    }
   705→  } catch (err) {
   706→    // 换号中止后不再补救：旧账号会话的回复已整体作废（SDK 兜底请求与错误状态一并跳过）
   707→    if (rt.aborted) return;
   708→    // 代理与浏览器直连都失败：最后用服务端内置模型兜底一次（forceSdk），救回本轮回复；
   709→    // 兜底成功按正常完成收尾，失败才落错误文案（保留最先的代理侧错误，便于区分原因）；
   710→    // 流中已凑齐的分段已经过 onSegment 逐条投递上屏，出错不影响它们（不丢已到手的正文）
   711→    // 关键：进入 SDK 兜底前清空 raw 并重建分段器，避免「半截上游文本 + 完整 SDK 文本」拼接成乱文；
   712→    // 旧分段器已放出的分段保留上屏、不计入新分段器的条数计数
   713→    raw = '';
   714→    if (multi && opts.onSegment) {
   715→      scanner = createReplySegmentScanner(opts.onSegment);
   716→    } else {
   717→      scanner = null;
   718→    }
   719→    const viaSdk = await sdkFallbackOnce(workMessages);
   720→    if (viaSdk) {
   721→      patchState(rt, { status: 'done' });
   722→    } else {
   723→      patchState(rt, {
   724→        status: 'error',
   725→        error: err instanceof Error && err.message ? err.message : '消息没有送达，请稍后重试',
   726→      });
   727→    }
   728→  }
   729→  // ---- 收尾（status 补丁与 finalize 在同一微任务里，订阅方重渲染时落盘已完成）----
   730→  // 换号中止（abortStreamsByPrefix）：整段收尾跳过——finalize 不再落盘（finalized 已预置）；
   731→  // activePayloads 也不删：此刻同键可能已被换号后的新流重新登记，误删会丢新流的 pagehide 接力 payload
   732→  if (!rt.aborted) {
   733→    activePayloads.delete(rt.state.sessionKey);
   734→    if (!rt.finalized) {
   735→      rt.finalized = true;
   736→      try {
   737→        opts.finalize({
   738→          aiMsgId: rt.state.aiMsgId,
   739→          content: raw,
   740→          // 多条模式：tail = 分段器剩余的最后一条（第 N 条，可能含溢出合并的句子）；
   741→          // 单条模式：tail = 完整内容（finalize 按旧管线处理，行为与旧版一致）
   742→          tail: scanner ? scanner.finish() : raw,
   743→          error: rt.state.error,
   744→          startedAt: rt.state.startedAt,
   745→        });
   746→        patchState(rt, { content: raw });
   747→      } catch {
   748→        // 落盘失败不影响状态广播（与各 App 持久化失败静默的既有策略一致）
   749→      }
   750→    }
   751→    emitFinalized(rt.state.sessionKey);
   752→    emit();
   753→  }
   754→}

   758→/**
   759→ * 发起一条流式请求（全局发起，与页面生命周期解耦）。
   760→ * 同一会话已有进行中的流时拒绝并返回 false（调用方应回滚刚插入的用户消息）。
   761→ */
   762→export function beginChatStream(opts: BeginChatStreamOptions): boolean {
   763→  // 40-a 第二层兜底：用户拉黑角色（byUser）后，该 App 内 AI 不得发消息——除「申请解除拉黑」放行模式
   764→  //（charRequestOnlyOf，与各 App 回合入口守卫同一条件）外，这里不发起流。静默处理：返回 true 让
   765→  // 调用方按「已开始」收尾（不回滚已上屏的用户消息、不触发 useChatStream 错误显示路径），
   766→  // 实际不发请求、不产生任何流状态——挡住 bg-turn 接力等间接路径，群聊/助手会话天然放行。
   767→  const blkSession = blockSessionOf(opts.sessionKey);
   768→  if (blkSession) {
   769→    const blkEntry = loadBlock(blkSession.app, blkSession.contactId);
   770→    if (blkEntry.byUser && !charRequestOnlyOf(blkEntry)) return true;
   771→  }
   772→  const existing = streams.get(opts.sessionKey);
   773→  if (existing && existing.state.status === 'streaming') return false;
   774→  // 新回合接管：清除投递插入边界（#35）——之后投递的消息都是对本回合上下文里最新用户消息的回复，
   775→  // 照旧追加落库（边界只在「上一轮投递未完时用户插话」的窗口内生效）。
   776→  // M6（四轮审计）根因侧修复：群成员间切换/regen 不等投递尾巴就开始下一轮，旧实现无条件清边界
   777→  // 会让「还在按边界插入的旧尾巴」改成追加、倒挂在用户新消息之后落库——改为尾巴仍在时先排空批占位，
   778→  // 排空（旧回合分段全部落库）瞬间在其 onDrained 里同步清边界（先于下一批开始投递），
   779→  // 此后（含本回合经分段/finalize 新排队的批次）投递照旧追加，落库顺序=对话时序
   780→  if (isAiDelivering(opts.sessionKey)) {
   781→    void scheduleAiDelivery(opts.sessionKey, [], () => {}, {
   782→      onDrained: () => clearDeliverBoundary(opts.sessionKey),
   783→    });
   784→  } else {
   785→    clearDeliverBoundary(opts.sessionKey);
   786→  }
   787→  const rt: StreamRuntime = {
   788→    state: {
   789→      sessionKey: opts.sessionKey,
   790→      aiMsgId: opts.aiMsgId,
   791→      content: '',
   792→      status: 'streaming',
   793→      startedAt: Date.now(),
   794→      replyCount: opts.replyCount,
   795→    },
   796→    finalized: false,
   797→    aborted: false,
   798→  };
   799→  streams.set(opts.sessionKey, rt);
   800→  evictFinished();
   801→  emit();
   802→  // 服务端可能还留着上一次关页接力的生成中回复：本轮用户已回页接管，让它作废（防双回复）
   803→  cancelBgRelay(opts.sessionKey);
   804→  // 流一开始就登记初始 payload：pagehide 接力不再漏掉识图前置阶段（发图后立刻关页的窗口）。
   805→  // 识图成功后 runStream 会用 workMessages/effConfig 覆盖为实际 payload。
   806→  activePayloads.set(opts.sessionKey, { messages: opts.messages, config: opts.apiConfig });
   807→  void runStream(rt, opts);
   808→  return true;
   809→}

   847→/**
   848→ * 中止全部会话键以 prefix 开头的流并清掉对应流状态（换号 purge 用，审计 #3）：
   849→ * - 进行中的流：预先登记「已收尾」（finalized）并从状态表移除——HTTP 请求本身无法取消，
   850→ *   但收尾时的 finalize 回调不再执行（旧账号会话的回复不会落盘写进对应聊天记录键），
   851→ *   服务端关页接力生成（bg relay）一并上报 cancel 作废（防换号后迟到回复）；
   852→ * - 已结束未清理的流状态（重进页面同步用）：换号后不再需要，一并移除。
   853→ * 只影响前缀匹配的会话（如 'qq:'），其他端（wx: / sms:）的流不受影响。
   854→ * 返回被中止的进行中流数量（0 = 该前缀无在途流）。
   855→ */
   856→export function abortStreamsByPrefix(prefix: string): number {
   857→  let aborted = 0;
   858→  const keys: string[] = [];
   859→  streams.forEach((rt, key) => {
   860→    if (!key.startsWith(prefix)) return;
   861→    keys.push(key);
   862→    if (rt.state.status === 'streaming') {
   863→      rt.finalized = true; // runStream 收尾时跳过 finalize（防旧会话回复落盘）
   864→      // 同时预置中止标记：onDelta 停止喂分段器（流中分段不再产生）、收尾段整段跳过
   865→      //（含 activePayloads 清理——此刻起同键可能被换号后的新流重新登记，旧流不得误删）
   866→      rt.aborted = true;
   867→      cancelBgRelay(key); // 服务端接力生成一并作废
   868→      aborted += 1;
   869→    }
   870→  });
   871→  if (keys.length > 0) {
   872→    for (const key of keys) {
   873→      streams.delete(key);
   874→      activePayloads.delete(key);
   875→    }
   876→    emit();
   877→  }
   878→  return aborted;
   879→}

   910→export function useChatStreamFinalized(prefix: string, onFinalized: () => void): void {

   921→  }, [prefix]);
