    50→/** 默认木纹条地板（与旧版一致） */
    51→function makeFloorTexture(): THREE.CanvasTexture {
    52→  const S = 512;
    53→  const cv = document.createElement('canvas');
    54→  cv.width = S;
    55→  cv.height = S;
    56→  const ctx = cv.getContext('2d')!;
    57→  // 底色：云白（自定义颜色通过材质 color 与本纹理相乘生效）
    58→  ctx.fillStyle = '#f4f2ed';
    59→  ctx.fillRect(0, 0, S, S);
    60→  // 拼板：横向 4 条，微差灰白
    61→  const plank = S / 4;
    62→  for (let i = 0; i < 4; i++) {
    63→    const shade = i % 2 === 0 ? 'rgba(214,212,205,0.5)' : 'rgba(206,204,197,0.45)';
    64→    ctx.fillStyle = shade;
    65→    ctx.fillRect(0, i * plank, S, plank);
    66→    // 板间缝（浅灰）
    67→    ctx.fillStyle = 'rgba(178,175,167,0.4)';
    68→    ctx.fillRect(0, i * plank - 1.5, S, 3);
    69→    // 竖向接缝（错缝）
    70→    const seamX = ((i * 137) % S) + 40;
    71→    ctx.fillRect(seamX, i * plank + 2, 2.5, plank - 4);
    72→  }
    73→  // 极淡哑光纹理
    74→  ctx.strokeStyle = 'rgba(180,177,170,0.06)';
    75→  ctx.lineWidth = 1;
    76→  for (let i = 0; i < 26; i++) {
    77→    const y = Math.random() * S;
    78→    ctx.beginPath();
    79→    ctx.moveTo(0, y);
    80→    ctx.bezierCurveTo(S * 0.3, y + 5, S * 0.6, y - 5, S, y + 2);
    81→    ctx.stroke();
    82→  }
    83→  const tex = new THREE.CanvasTexture(cv);
    84→  tex.wrapS = THREE.RepeatWrapping;
    85→  tex.wrapT = THREE.RepeatWrapping;
    86→  tex.repeat.set(3, 2);
    87→  tex.colorSpace = THREE.SRGBColorSpace;
    88→  tex.anisotropy = 4;
    89→  return tex;
    90→}

   287→/** 相机对焦物（生活=小人 / 编辑=轨道中心）；墙体按「视线段是否穿过」判断隐藏 */
   288→interface FocusPos {

   292→
   293→interface WallDef {
   294→  key: string;

   306→
   307→/** 墙体世界 AABB（墙盒不旋转，中心±半尺寸） */
   308→function boxOf(def: WallDef): THREE.Box3 {
   309→  return new THREE.Box3(
   310→    new THREE.Vector3(def.pos[0] - def.size[0] / 2, def.pos[1] - def.size[1] / 2, def.pos[2] - def.size[2] / 2),
   311→    new THREE.Vector3(def.pos[0] + def.size[0] / 2, def.pos[1] + def.size[1] / 2, def.pos[2] + def.size[2] / 2),
   312→  );
   313→}
   314→
   315→const _segDir = new THREE.Vector3();
   316→
   317→/**
   318→ * 视线段 P0→P1 是否穿过 AABB（slab 法；t 限定 [0,1]）——
   319→ * 只统计「相机与对焦物之间」的墙：对焦物身后（t>1）与相机身后（t<0）的墙一律不隐藏，
   320→ * 杜绝「望穿小人身后的墙看到屋外」的过度剖面。
   321→ */
   322→function segHitsBox(p0: THREE.Vector3, p1: THREE.Vector3, b: THREE.Box3): boolean {
   323→  let tmin = 0;
   324→  let tmax = 1;
   325→  _segDir.subVectors(p1, p0);
   326→  for (const ax of ['x', 'y', 'z'] as const) {
   327→    const d = _segDir[ax];
   328→    if (Math.abs(d) < 1e-9) {
   329→      if (p0[ax] < b.min[ax] || p0[ax] > b.max[ax]) return false;
   330→    } else {
   331→      let t1 = (b.min[ax] - p0[ax]) / d;
   332→      let t2 = (b.max[ax] - p0[ax]) / d;
   333→      if (t1 > t2) {
   334→        const t = t1;
   335→        t1 = t2;
   336→        t2 = t;
   337→      }
   338→      if (t1 > tmin) tmin = t1;
   339→      if (t2 < tmax) tmax = t2;
   340→      if (tmin > tmax) return false;
   341→    }
   342→  }
   343→  return true;
   344→}
   345→
   346→const _focusP = new THREE.Vector3();
   347→/** 材质淡出前的基准值（透明/深度写状态也要还原，玻璃/光晕等本就透明的材质不受影响） */
   348→const _matBase = new WeakMap<THREE.Material, { o: number; t: boolean; d: boolean }>();
   349→/** mesh 原始 raycast（淡出期间临时禁用射线，恢复时还原） */
   350→const _rayBase = new WeakMap<THREE.Mesh, THREE.Mesh['raycast']>();
   351→const _noopRaycast = () => {};
   352→
   353→/**
   354→ * 墙体遮挡淡出（方案 A 核心）：
   355→ * - 遮挡判定 = 「相机→对焦物」视线段是否穿过墙体 AABB（boxes 任一命中即挡）；
   356→ * - 命中 → 组内所有材质 opacity 每帧阻尼淡出（透明期 depthWrite=false 防排序瑕疵，
   357→ *   接近全隐才 visible=false 跳过渲染）；移开 → 阻尼淡回，结束后按基准一次性还原交还声明式控制。
   358→ * - ⚑ 淡出期间同步禁用组内 mesh 的 raycast（可见即可穿，点击穿透到身后家具）；
   359→ *   否则半透明的门板/窗扇会挡住相机与家具之间的点击（真实 UX bug：看得见点不到）。
   360→ * 旋转时墙「丝滑变透明」而非啪地消失；只藏真正挡住视线的墙。
   361→ */
   362→function useWallOcclusionFade(
   363→  root: RefObject<THREE.Group | null>,
   364→  boxes: THREE.Box3[],
   365→  focusRef?: RefObject<FocusPos>,
   366→) {
   367→  const op = useRef(1);
   368→  const fading = useRef(false);
   369→  const _rayOn = useRef(true);
   370→  useFrame(({ camera }, rawDt) => {
   371→    const g = root.current;
   372→    if (!g) return;
   373→    const f = focusRef?.current;
   374→    _focusP.set(f?.x ?? 0, 1, f?.z ?? 2.4);
   375→    let blocked = false;
   376→    for (const b of boxes) {
   377→      if (segHitsBox(camera.position, _focusP, b)) {
   378→        blocked = true;
   379→        break;
   380→      }
   381→    }
   382→    const dt = Math.min(rawDt, 0.05);
   383→    op.current = THREE.MathUtils.damp(op.current, blocked ? 0 : 1, blocked ? 14 : 8, dt);
   384→    const v = op.current;
   385→    const vis = v > 0.015;
   386→    if (g.visible !== vis) g.visible = vis;
   387→    // ⚑ 可见即可穿：淡出中（含半透明过渡期）禁用射线；完全恢复才还原
   388→    const wantRay = v > 0.95;
   389→    if (wantRay !== _rayOn.current) {
   390→      _rayOn.current = wantRay;
   391→      g.traverse((o) => {
   392→        const mesh = o as THREE.Mesh;
   393→        if (!mesh.isMesh) return;
   394→        if (wantRay) {
   395→          const orig = _rayBase.get(mesh);
   396→          if (orig) mesh.raycast = orig;
   397→        } else {
   398→          if (!_rayBase.has(mesh)) _rayBase.set(mesh, mesh.raycast);
   399→          mesh.raycast = _noopRaycast;
   400→        }
   401→      });
   402→    }
   403→    if (!vis) return; // 全隐：材质保持淡出态，恢复时阻尼淡回
   404→    if (v < 0.998) {
   405→      fading.current = true;
   406→      g.traverse((o) => {
   407→        const mesh = o as THREE.Mesh;
   408→        if (!mesh.isMesh) return;
   409→        const mm = mesh.material as THREE.Material | THREE.Material[];
   410→        for (const m of Array.isArray(mm) ? mm : [mm]) {
   411→          let base = _matBase.get(m);
   412→          if (!base) {
   413→            base = { o: m.opacity, t: m.transparent, d: m.depthWrite };
   414→            _matBase.set(m, base);
   415→          }
   416→          m.opacity = base.o * v;
   417→          m.transparent = true;
   418→          m.depthWrite = false;
   419→        }
   420→      });
   421→    } else if (fading.current) {
   422→      // 淡出结束：按基准一次性还原（透明/深度写状态也还原），交还声明式控制
   423→      fading.current = false;
   424→      g.traverse((o) => {
   425→        const mesh = o as THREE.Mesh;
   426→        if (!mesh.isMesh) return;
   427→        const mm = mesh.material as THREE.Material | THREE.Material[];
   428→        for (const m of Array.isArray(mm) ? mm : [mm]) {
   429→          const base = _matBase.get(m);
   430→          if (base) {
   431→            m.opacity = base.o;
   432→            m.transparent = base.t;
   433→            m.depthWrite = base.d;
   434→          }
   435→        }
   436→      });
   437→    }
   438→  });
   439→}

   441→/**
   442→ * 隔墙（带门洞）随门位动态重建：门挪 → 墙段跟随。

   474→    { key: 'balWest', pos: [BAL_X0 - WALL_T / 2, H / 2, (BAL_Z0 + BAL_Z1) / 2], size: [WALL_T, H, BAL_Z1 - BAL_Z0], normal: new THREE.Vector3(-1, 0, 0), room: 'bal' },
   475→  ];
   476→}

   478→/** 门洞上方的门头梁（与所在整墙同隐同现；无踢脚线） */
   479→function DoorLintel({
   480→  pos,
   481→  alongAxis,
   482→  wallCol,
   483→  night,
   484→  focusRef,
   485→  hostBox,
   486→  wallTexMap,
   487→}: {
   488→  pos: [number, number, number];
   489→  alongAxis: 'x' | 'z';
   490→  wallCol?: string;
   491→  night: boolean;
   492→  focusRef?: RefObject<FocusPos>;
   493→  /** 所在整墙（含门洞跨度）的剖面盒：视线穿过墙上任意处 → 门梁随墙同隐 */
   494→  hostBox: THREE.Box3;
   495→  wallTexMap?: THREE.Texture;
   496→}) {
   497→  const ref = useRef<THREE.Group>(null);
   498→  useWallOcclusionFade(ref, useMemo(() => [hostBox], [hostBox]), focusRef);
   499→  const size: [number, number, number] = alongAxis === 'x' ? [DOOR_W + 0.04, WALL_H - 2.06, WALL_T] : [WALL_T, WALL_H - 2.06, DOOR_W + 0.04];
   500→  return (
   501→    <group ref={ref}>
   502→      <mesh position={pos} castShadow receiveShadow>
   503→        <boxGeometry args={size} />
   504→        <meshStandardMaterial color={wallCol ?? (night ? '#e9e4d8' : '#f7f3ec')} map={wallTexMap} roughness={0.95} />
   505→      </mesh>
   506→    </group>
   507→  );
   508→}

   511→function Wall({ def, night, wallCol, wallTexMap, focusRef }: { def: WallDef; night: boolean; wallCol?: string; wallTexMap?: THREE.Texture; focusRef?: RefObject<FocusPos> }) {
   512→  const ref = useRef<THREE.Group>(null);
   513→  useWallOcclusionFade(ref, useMemo(() => [boxOf(def)], [def]), focusRef);
   514→
   515→  return (
   516→    <group ref={ref}>
   517→      {/* 墙体：暖白（支持用户自定义色 + 墙纸纹理） */}
   518→      <mesh position={def.pos} castShadow receiveShadow>
   519→        <boxGeometry args={def.size} />
   520→        <meshStandardMaterial color={wallCol ?? (night ? '#e9e4d8' : '#f7f3ec')} map={wallTexMap} roughness={0.95} />
   521→      </mesh>
   522→      {/* 踢脚线：浅灰（内侧面贴墙） */}
   523→      {def.baseboard !== false && (
   524→        <mesh
   525→          position={[
   526→            def.pos[0] - def.normal.x * (WALL_T / 2 + 0.012),
   527→            0.055,
   528→            def.pos[2] - def.normal.z * (WALL_T / 2 + 0.012),
   529→          ]}
   530→          castShadow
   531→        >
   532→          <boxGeometry
   533→            args={[
   534→              Math.max(def.size[0] * 0.998, 0.02),
   535→              0.11,
   536→              Math.max(def.size[2] * 0.998, 0.02),
   537→            ]}
   538→          />
   539→          <meshStandardMaterial color="#e3e1da" roughness={0.9} />
   540→        </mesh>
   541→      )}
   542→    </group>
   543→  );
   544→}

   547→ * 墙段碰撞盒（生活模式相机防钻墙钳制用；不含门洞区域——相机可从开着的门洞穿过）。
   548→ * 随门位变化重建：门挪 → 墙段 AABB 跟随。
   549→ */
   550→export function buildWallBoxes(doorBed: number, doorKt: number, doorBath: number, doorBal: number): THREE.Box3[] {
   551→  return buildWalls(doorBed, doorKt, doorBath, doorBal).map(boxOf);

   554→// ---------------- 门牌（门洞上方：卧室 / 厨房 / 卫生间） ----------------
   555→
   556→/** 门牌纹理：白底圆角 + 樱粉描边 + 中文标签（canvas 绘制，每标签只建一次） */
   557→const signTexCache = new Map<string, THREE.CanvasTexture>();
   558→function makeSignTexture(label: string): THREE.CanvasTexture {
   559→  const hit = signTexCache.get(label);
   560→  if (hit) return hit;
   561→  const W = 256;
   562→  const H = 112;
   563→  const cv = document.createElement('canvas');
   564→  cv.width = W;
   565→  cv.height = H;
   566→  const ctx = cv.getContext('2d')!;
   567→  // 白底圆角卡片
   568→  ctx.fillStyle = '#fffdf8';
   569→  ctx.strokeStyle = '#f2b3c2';
   570→  ctx.lineWidth = 7;
   571→  const r = 26;
   572→  ctx.beginPath();
   573→  ctx.moveTo(r + 4, 4);
   574→  ctx.arcTo(W - 4, 4, W - 4, H - 4, r);
   575→  ctx.arcTo(W - 4, H - 4, 4, H - 4, r);
   576→  ctx.arcTo(4, H - 4, 4, 4, r);
   577→  ctx.arcTo(4, 4, W - 4, 4, r);
   578→  ctx.closePath();
   579→  ctx.fill();
   580→  ctx.stroke();
   581→  // 中文标签
   582→  ctx.fillStyle = '#4a4a4e';
   583→  ctx.font = '600 52px "PingFang SC", "Microsoft YaHei", sans-serif';
   584→  ctx.textAlign = 'center';
   585→  ctx.textBaseline = 'middle';
   586→  ctx.fillText(label, W / 2, H / 2 + 4);
   587→  const tex = new THREE.CanvasTexture(cv);
   588→  tex.colorSpace = THREE.SRGBColorSpace;
   589→  tex.anisotropy = 4;
   590→  signTexCache.set(label, tex);
   591→  return tex;
   592→}

   594→/** 门牌：挂在门洞正上方（面向门外走来的那一侧），随门位移动 */
   595→function DoorSign({ label }: { label: string }) {

   597→  return (

   599→      {/* 牌面（微微凸出墙面，双面均可读） */}

   601→        <boxGeometry args={[0.46, 0.21, 0.02]} />
   602→        <meshStandardMaterial color="#fffdf8" roughness={0.7} />
   603→      </mesh>
   604→      <mesh position={[0, 0, 0.02]}>
   605→        <planeGeometry args={[0.44, 0.19]} />
   606→        <meshBasicMaterial map={tex} toneMapped={false} />
   607→      </mesh>
   608→      <mesh position={[0, 0, -0.008]} rotation={[0, Math.PI, 0]}>
   609→        <planeGeometry args={[0.44, 0.19]} />
   610→        <meshBasicMaterial map={tex} toneMapped={false} />
   611→      </mesh>
   612→    </group>

   616→// ---------------- 门（三扇：可开合可换色；编辑模式点击=选中不开门） ----------------
   617→
   618→/**
   619→ * 门：门框（三个边条，略压进门洞形成企口）+ 门板（铰链 pivot，向门后侧屋内摆开）+ 双面把手。
   620→ * ry = 0 时门洞沿 x 走（z=3 隔墙）；ry = π/2 时门洞沿 z 走（x=7.5 共墙）。
   621→ */
   622→function HouseDoor({
   623→  pos,
   624→  ry,
   625→  open,
   626→  col,
   627→  night,
   628→  sign,
   629→  onToggle,
   630→  editMode,
   631→  selected,
   632→  onSelect,
   633→  onEditPointerDown,
   634→  focusRef,
   635→  hostBox,
   636→  introRef,
   637→}: {
   638→  pos: [number, number];
   639→  ry: number;
   640→  open: boolean;
   641→  /** 进屋过场：on 期间门自动迎宾打开（不写用户保存的门开态） */
   642→  introRef?: RefObject<{ on: boolean }>;
   643→  col?: string;
   644→  night: boolean;
   645→  /** 门牌标签（卧室/厨房/卫生间；空串不挂） */
   646→  sign?: string;
   647→  onToggle: () => void;
   648→  editMode: boolean;
   649→  selected: boolean;
   650→  onSelect: () => void;
   651→  onEditPointerDown: (e: ThreeEvent<PointerEvent>) => void;
   652→  focusRef?: RefObject<FocusPos>;
   653→  /** 所在整墙（含门洞跨度）的剖面盒：视线穿过墙上任意处 → 门随墙同隐（不浮空） */
   654→  hostBox: THREE.Box3;
   655→}) {
   656→  const panel = useRef<THREE.Group>(null);
   657→  const doorRoot = useRef<THREE.Group>(null);
   658→  const [hovered, setHovered] = useState(false);
   659→
   660→  // 门遮挡盒：门洞区域（含门框包边）；视线穿过门洞 → 门随墙同隐；从开着的门洞看过去也淡出门板
   661→  const doorBox = useMemo(() => {
   662→    const hw = DOOR_W / 2 + 0.09;
   663→    const ht = (WALL_T + 0.03) / 2 + 0.02;
   664→    const top = DOOR_H + 0.08;
   665→    return ry === 0
   666→      ? new THREE.Box3(
   667→          new THREE.Vector3(pos[0] - hw, 0, pos[1] - ht),
   668→          new THREE.Vector3(pos[0] + hw, top, pos[1] + ht),
   669→        )
   670→      : new THREE.Box3(
   671→          new THREE.Vector3(pos[0] - ht, 0, pos[1] - hw),
   672→          new THREE.Vector3(pos[0] + ht, top, pos[1] + hw),
   673→        );
   674→  }, [pos[0], pos[1], ry]);
   675→  useWallOcclusionFade(doorRoot, useMemo(() => [doorBox, hostBox], [doorBox, hostBox]), focusRef);
   676→
   677→  useFrame((_, dt) => {
   678→    if (!panel.current) return;
   679→    // 关 = 与隔墙同面；开 = 向屋内（ry=0 → +z；ry=π/2 → +x 卫生间）摆开约 100°
   680→    // 进屋过场期间门自动迎宾打开（introRef.on），过场结束自动回位
   681→    const effOpen = open || introRef?.current?.on === true;
   682→    panel.current.rotation.y = THREE.MathUtils.damp(panel.current.rotation.y, effOpen ? -1.78 : 0, 6, dt);
   683→  });
   684→
   685→  const doorCol = col ?? '#f3efe6';
   686→  const frameCol = useMemo(() => new THREE.Color(doorCol).multiplyScalar(0.86), [doorCol]);
   687→  const frameStr = `#${frameCol.getHexString()}`;
   688→  /** 门板宽/高：比门洞各边小 2cm（藏在门垛/门头梁企口后面，不露缝） */
   689→  const PANEL_W = DOOR_W - 0.04;
   690→  const PANEL_H = DOOR_H - 0.03;
   691→
   692→  const toggle = (e: { delta: number; stopPropagation: () => void }) => {
   693→    if (e.delta > 8) return; // 拖动视角不算点击
   694→    e.stopPropagation();
   695→    if (editMode) onSelect();
   696→    else onToggle();
   697→  };
   698→
   699→  const hoverProps = {
   700→    onPointerOver: (e: { stopPropagation: () => void }) => {
   701→      e.stopPropagation();
   702→      setHovered(true);
   703→      document.body.style.cursor = 'pointer';
   704→    },
   705→    onPointerOut: () => {
   706→      setHovered(false);
   707→      document.body.style.cursor = 'auto';
   708→    },
   709→  };
   710→
   711→  return (
   712→    <group position={[pos[0], 0, pos[1]]} rotation={[0, ry, 0]} ref={doorRoot}>
   713→      {/* 门牌（门洞上方，随门位移动） */}
   714→      {sign ? <DoorSign label={sign} /> : null}
   715→      {/* 门框：左右门垛（0.07 宽，压进门洞 2cm 盖住门板边）+ 门头梁包边（压下 4.5cm） */}
   716→      <mesh position={[-DOOR_W / 2 - 0.015, DOOR_H / 2 + 0.005, WALL_T / 2]} castShadow>
   717→        <boxGeometry args={[0.07, DOOR_H + 0.01, WALL_T + 0.03]} />
   718→        <meshStandardMaterial color={frameStr} roughness={0.8} />
   719→      </mesh>
   720→      <mesh position={[DOOR_W / 2 + 0.015, DOOR_H / 2 + 0.005, WALL_T / 2]} castShadow>
   721→        <boxGeometry args={[0.07, DOOR_H + 0.01, WALL_T + 0.03]} />
   722→        <meshStandardMaterial color={frameStr} roughness={0.8} />
   723→      </mesh>
   724→      <mesh position={[0, DOOR_H, WALL_T / 2]} castShadow>
   725→        <boxGeometry args={[DOOR_W + 0.16, 0.09, WALL_T + 0.03]} />
   726→        <meshStandardMaterial color={frameStr} roughness={0.8} />
   727→      </mesh>
   728→
   729→      {/* 门板（铰链在左侧门垛，往屋内方向开；点击开合 / 编辑模式选中） */}
   730→      <group
   731→        ref={panel}
   732→        position={[-DOOR_W / 2 + 0.02, 0, WALL_T / 2]}
   733→        onClick={toggle}
   734→        onPointerDown={(e) => {
   735→          if (!editMode) return;
   736→          e.stopPropagation();
   737→          onEditPointerDown(e);
   738→        }}
   739→        {...hoverProps}
   740→      >
   741→        <group position={[DOOR_W / 2 - 0.02, 0, 0]}>
   742→          {/* 门板主体（填满门洞，边缝全部藏进门框企口） */}
   743→          <mesh position={[0, PANEL_H / 2 + 0.005, 0]} castShadow receiveShadow>
   744→            <boxGeometry args={[PANEL_W, PANEL_H, 0.045]} />
   745→            <meshStandardMaterial color={doorCol} roughness={0.75} />
   746→          </mesh>
   747→          {/* 门板凹凸造型线（上下两块嵌板） */}
   748→          <mesh position={[0, PANEL_H * 0.7, 0.026]}>
   749→            <boxGeometry args={[PANEL_W - 0.18, 0.62, 0.012]} />
   750→            <meshStandardMaterial color={frameStr} roughness={0.8} />
   751→          </mesh>
   752→          <mesh position={[0, PANEL_H * 0.28, 0.026]}>
   753→            <boxGeometry args={[PANEL_W - 0.18, 0.5, 0.012]} />
   754→            <meshStandardMaterial color={frameStr} roughness={0.8} />
   755→          </mesh>
   756→          {/* 把手（双面） */}
   757→          <mesh position={[PANEL_W / 2 - 0.11, PANEL_H * 0.47, 0.045]}>
   758→            <sphereGeometry args={[0.032, 12, 10]} />
   759→            <meshStandardMaterial color="#d9cfb8" metalness={0.55} roughness={0.35} />
   760→          </mesh>
   761→          <mesh position={[PANEL_W / 2 - 0.11, PANEL_H * 0.47, -0.045]}>
   762→            <sphereGeometry args={[0.032, 12, 10]} />
   763→            <meshStandardMaterial color="#d9cfb8" metalness={0.55} roughness={0.35} />
   764→          </mesh>
   765→          {/* 夜里门缝透出的暖光提示 */}
   766→          {!open && night && (
   767→            <mesh position={[PANEL_W / 2 - 0.03, PANEL_H / 2, 0.026]}>
   768→              <boxGeometry args={[0.012, PANEL_H - 0.2, 0.004]} />
   769→              <meshBasicMaterial color="#ffd98f" transparent opacity={0.35} toneMapped={false} />
   770→            </mesh>
   771→          )}
   772→          {/* hover / 选中高亮 */}
   773→          {(hovered || selected) && (
   774→            <mesh position={[0, PANEL_H / 2, 0.03]}>
   775→              <planeGeometry args={[PANEL_W - 0.03, PANEL_H - 0.03]} />
   776→              <meshBasicMaterial color={selected ? '#f2a9bc' : '#ffffff'} transparent opacity={selected ? 0.3 : 0.14} depthWrite={false} />
   777→            </mesh>
   778→          )}
   779→        </group>
   780→      </group>
   781→    </group>

   979→function Ceiling({
   980→  x0,
   981→  x1,
   982→  z0,
   983→  z1,
   984→  col,
   985→  night,
   986→}: {
   987→  x0: number;
   988→  x1: number;
   989→  z0: number;
   990→  z1: number;
   991→  col?: string;
   992→  night: boolean;
   993→}) {
   994→  const ref = useRef<THREE.Mesh>(null);
   995→  const { camera } = useThree();
   996→  useFrame(() => {
   997→    if (!ref.current) return;
   998→    const inside =
   999→      camera.position.x > x0 - 0.15 &&
  1000→      camera.position.x < x1 + 0.15 &&
  1001→      camera.position.z > z0 - 0.15 &&
  1002→      camera.position.z < z1 + 0.15;
  1003→    ref.current.visible = inside && camera.position.y < WALL_H - 0.12;
  1004→  });
  1005→  return (
  1006→    <mesh ref={ref} position={[(x0 + x1) / 2, WALL_H, (z0 + z1) / 2]} rotation={[Math.PI / 2, 0, 0]}>
  1007→      <planeGeometry args={[x1 - x0, z1 - z0]} />
  1008→      <meshStandardMaterial color={col ?? (night ? '#e0ddd4' : '#fbf9f3')} roughness={0.96} />
  1009→    </mesh>
  1010→  );
  1011→}
