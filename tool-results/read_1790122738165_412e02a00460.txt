     3→/**
     4→ * 记忆库（跨 App 共享记忆）：记忆碎片 / 核心记忆 / 长期记忆 / 每联系人设置。
     5→ *
     6→ * 设计要点（「记忆库 APP」需求）：
     7→ * - 数据全部按【联系人 ID】隔离，localStorage 持久化（与聊天记录同层，重启保留）：
     8→ *   · mem-frag:<contactId>   记忆碎片（每条带来源 App/时间/消息 ID/权重/加强时间）
     9→ *   · mem-ltm:<contactId>    核心记忆（M 条碎片总结；键名沿用历史 ltm，语义=核心层）
    10→ *   · mem-long:<contactId>   长期记忆（K 条核心记忆总结出的最稳定画像，层级顶层）
    11→ *   · mem-settings:<contactId> 每联系人设置（提取频率/核心阈值/长期阈值/互通开关/失忆程度）
    12→ *   · mem-round:<contactId>:<app>  各 App 会话的对话轮次计数（提取节奏用）
    13→ * - 三层管线：碎片（每 N 轮提取）→ 核心记忆（积累 M 条碎片总结一条）→ 长期记忆（积累 K 条
    14→ *   未归档核心总结一条，默认 K=5）；核心被长期收编后标记 archivedAt（方案A：不再参与后续
    15→ *   总结与召回，由长期记忆代表，避免重复总结/重复注入）。
    16→ * - 互通开关（默认开）：开 = 四个 App（QQ/微信/信息/电话）共享该联系人全部记忆；
    17→ *   关 = 各 App 只召回自己来源的记忆（碎片/核心/长期均按来源 App 过滤）。
    18→ *   存储始终单一池 + App 标记：切换开关无需迁移数据，UI 管理视图始终展示该联系人全部记忆。
    19→ * - 不同联系人之间永远隔离：键即隔离边界，召回/提取都不跨联系人读取。
    20→ * - 【视角统一】提取/总结 prompt 一律用「用户真实名字 + 角色名字」指代双方，禁混用
    21→ *   「对方/用户/我」；memRepairPerspectiveNow 可把旧版「对方/用户」代称修复为真实名字。
    22→ * - 【过期淡化】失忆程度（快≈3天/中≈2周/慢≈2个月/从不）：超过有效期一半「淡化中」召回降权，
    23→ *   超过有效期「已归档」不再参与召回与总结；低权重记忆先淡化、高权重更持久；
    24→ *   重复提及/合并相似/手动「回忆一下」都会刷新淡化计时。
    25→ * - 【去重合并】同一事实的不同说法（相似度 ≥0.6）自动合并为一条：重复提取=加强而非重复占位；
    26→ *   设置页可手动「整理重复记忆」一次性清理；召回时跨层去重（长期→核心→碎片顺序，后层与已选
    27→ *   内容相似则跳过）。
    28→ * - 【权重优先级】姓名/关系/承诺/健康禁忌 → 高权重（自动分类 + 手动可调）；召回按权重×相关性排序。
    29→ * - 【时间感知】记忆与当前时间联动：碎片可带 eventTime（事件发生时间）/ expiresAt（过期时间，
    30→ *   空=永不过期）；提取时由模型按当前时间锚点判断（服务端校验 ±5 年防幻觉）；注入时头部带当前
    31→ *   时间、层内按有效时间从新到旧、每条带时间标签，并提示优先参考更近的记忆；过期碎片自动归档
    32→ *   （expiredAt）不再召回；新旧矛盾时模型标注 supersedes，旧记忆标记「已更新」不再召回。
    33→ *   核心/长期默认永不过期，可被用户手动设置过期；用户手动编辑过的时间（timeEditedAt）以用户
    34→ *   设置为准，自动流程（提取合并/去重/总结）一律不得覆盖。
    35→ * - 【来源追溯】每条碎片记录来源 App / 来源时间 / 来源消息 ID，回答「你怎么知道的」。
    36→ * - 提取/总结调用 /api/memory/extract 与 /api/memory/summarize（用户 API 配置优先，
    37→ *   服务端 SDK 兜底）；自动提取失败静默（下一窗口重试），手动「立即总结」失败给提示；
    38→ *   手动入口共四个且互斥：碎片页右上角（仅提取碎片）、核心页右上角（碎片→核心）、
    39→ *   长期页右上角（核心→长期）、设置页（按所选粒度执行）。
    40→ * - 删除联系人时由 contacts-store.deleteContact 调 memPurgeContact 级联清理全部记忆。
    41→ */

   122→function readJSON<T>(key: string): T | null {
   123→  try {
   124→    return kvGet<T>(key);

   126→    return null;

   130→function writeJSON(key: string, value: unknown): void {

   132→    // 内存同步 + 异步写穿 IndexedDB；存储失败不中断聊天（记忆是增强能力）

   135→    // 忽略
   136→  }

   172→export function getMemSettings(contactId: string): MemSettings {
   173→  const raw = readJSON<Partial<MemSettings>>(settingsKey(contactId));
   174→  if (!raw) return { ...DEFAULT_MEM_SETTINGS };
   175→  return {
   176→    interval: MEM_INTERVAL_OPTIONS.includes(raw.interval as MemSettings['interval'])
   177→      ? (raw.interval as MemSettings['interval'])
   178→      : DEFAULT_MEM_SETTINGS.interval,
   179→    threshold: MEM_THRESHOLD_OPTIONS.includes(raw.threshold as MemSettings['threshold'])
   180→      ? (raw.threshold as MemSettings['threshold'])
   181→      : DEFAULT_MEM_SETTINGS.threshold,
   182→    longThreshold: MEM_LONG_OPTIONS.includes(raw.longThreshold as MemSettings['longThreshold'])
   183→      ? (raw.longThreshold as MemSettings['longThreshold'])
   184→      : DEFAULT_MEM_SETTINGS.longThreshold,
   185→    share: raw.share !== false,
   186→    forget: raw.forget === 'fast' || raw.forget === 'medium' || raw.forget === 'slow' || raw.forget === 'never'
   187→      ? raw.forget
   188→      : DEFAULT_MEM_SETTINGS.forget,
   189→  };
   190→}

   268→/**
   269→ * 时间感知过期清扫：把 expiresAt 已到期的碎片标记 expiredAt（自动归档）。
   270→ * 惰性执行：召回（memRecallBlock）、一轮对话结束（memAfterAiTurn）、记忆库刷新时调用；
   271→ * 召回/总结的过滤都按 isMemExpired 实时判断，本清扫只负责落库标记与 UI 展示。
   272→ * 返回本次标记的条数。
   273→ */
   274→export function memSweepExpiry(contactId: string): number {
   275→  const now = Date.now();
   276→  const list = readFragments(contactId);
   277→  let changed = 0;
   278→  const next = list.map((f) => {
   279→    if (f.expiresAt != null && now > f.expiresAt && !f.expiredAt && !f.supersededAt) {
   280→      changed++;
   281→      return { ...f, expiredAt: now };
   282→    }
   283→    return f;
   284→  });
   285→  if (changed > 0) writeJSON(fragKey(contactId), next);
   286→  return changed;
   287→}

   289→/**
   290→ * 矛盾更新对比用：当前待总结（未消费/未更新/未过期/未淡化）碎片的 id+内容，最近 30 条。
   291→ * 随提取请求发给模型，让它判断新信息是否与已有记忆矛盾（supersedes 只接受这些 id）。
   292→ */
   293→function existingForConflict(contactId: string): { id: string; content: string }[] {
   294→  const { forget } = getMemSettings(contactId);
   295→  const now = Date.now();
   296→  return readFragments(contactId)
   297→    .filter(
   298→      (f) =>
   299→        !f.consumedAt &&
   300→        !f.supersededAt &&
   301→        !isMemExpired(f, now) &&
   302→        fadeState(f as FadeInput, forget, now) !== 'faded'
   303→    )
   304→    .slice(-30)
   305→    .map((f) => ({ id: f.id, content: f.content }));
   306→}

   316→/**
   317→ * 手动时间编辑入参（时间感知：用户在记忆库手动修改事件/过期时间）：
   318→ * - eventTime/expiresAt 传 null 表示清除（事件时间清空 / 恢复永不过期），传时间戳表示设置；
   319→ * - undefined 表示不改动该字段。
   320→ */
   321→export interface MemTimePatch {
   322→  eventTime?: number | null;
   323→  expiresAt?: number | null;
   324→}
   325→
   326→/**
   327→ * 三层通用：应用手动时间补丁。被手动编辑过的记忆会带上 timeEditedAt 标记，
   328→ * 此后自动流程（提取合并/去重/总结）一律以用户设置为准，不得改写其 eventTime/expiresAt。
   329→ */
   330→function applyTimePatch<
   331→  T extends { eventTime?: number; expiresAt?: number; expiredAt?: number; timeEditedAt?: number; editedAt?: number }
   332→>(rec: T, patch: MemTimePatch, now: number): T {
   333→  const next = { ...rec };
   334→  if (patch.eventTime !== undefined) {
   335→    if (patch.eventTime == null) delete next.eventTime;
   336→    else if (Number.isFinite(patch.eventTime)) next.eventTime = patch.eventTime;
   337→  }
   338→  if (patch.expiresAt !== undefined) {
   339→    if (patch.expiresAt == null) delete next.expiresAt;
   340→    else if (Number.isFinite(patch.expiresAt)) next.expiresAt = patch.expiresAt;
   341→  }
   342→  next.timeEditedAt = now;
   343→  next.editedAt = now;
   344→  // 手动把过期时间改到未来或清除 → 救回已过期归档的碎片（用户意图明确：重新有效）
   345→  if (next.expiresAt == null || next.expiresAt > now) delete next.expiredAt;
   346→  return next;
   347→}

   349→/** 手动修改碎片的事件/过期时间（设置 timeEditedAt，自动流程不再覆盖） */
   350→export function updateFragmentTime(contactId: string, id: string, patch: MemTimePatch): boolean {
   351→  const list = readFragments(contactId);
   352→  const idx = list.findIndex((f) => f.id === id);
   353→  if (idx < 0) return false;
   354→  list[idx] = applyTimePatch(list[idx], patch, Date.now());
   355→  writeJSON(fragKey(contactId), list);
   356→  return true;
   357→}

   379→export function updateFragment(contactId: string, id: string, content: string, weight?: MemWeight): boolean {
   380→  const list = readFragments(contactId);
   381→  const hit = list.find((f) => f.id === id);
   382→  if (!hit) return false;
   383→  hit.content = content.trim();
   384→  hit.editedAt = Date.now();
   385→  if (weight === 'high' || weight === 'normal' || weight === 'low') hit.weight = weight;
   386→  writeJSON(fragKey(contactId), list);
   387→  return true;
   388→}

   390→/** 「回忆一下」：手动救回淡化/归档的记忆（重置淡化计时，标记一次加强） */
   391→export function reinforceFragment(contactId: string, id: string): boolean {
   392→  const list = readFragments(contactId);
   393→  const hit = list.find((f) => f.id === id);

   464→/** 解散群聊 → 级联清理群来源记忆（钩子由 groups 层触发；依赖方向 memory → groups 单向，不反向 import） */

   467→/**
   468→ * 解散群聊时清理「群来源记忆」（onGroupDissolved 钩子调用；群删除时记忆一起清理的规则）：
   469→ * - 碎片：删除 source==='group' 且 sourceGroupId===groupId 的条目（无论是否已被总结消费/归档）；
   470→ * - 核心/长期：内容不删（可能含私聊来源的事实），只从 groupIds 里剔除该群；
   471→ *   剔除后不再有任何群来源时 privateSource 还原为 true（纯私聊语义，与旧数据兼容路径一致）。
   472→ * 成员维度只遍历群成员（只有成员参与过群聊，其他联系人不可能有该群来源的记忆）。
   473→ */
   474→export function memPurgeGroupSource(groupId: string, memberIds: string[]): void {
   475→  if (!groupId) return;
   476→  const ids = Array.from(new Set(memberIds.filter(Boolean)));
   477→  for (const contactId of ids) {
   478→    try {
   479→      // 碎片层：过滤掉本群来源的条目
   480→      const frags = readFragments(contactId);
   481→      const nextFrags = frags.filter((f) => !(f.source === 'group' && f.sourceGroupId === groupId));
   482→      if (nextFrags.length !== frags.length) writeJSON(fragKey(contactId), nextFrags);
   483→      // 核心层：剔除本群来源标记（内容保留）
   484→      const cores = readCores(contactId);
   485→      let coresChanged = false;
   486→      const nextCores = cores.map((m) => {
   487→        const gids = m.groupIds ?? [];
   488→        if (!gids.includes(groupId)) return m;
   489→        coresChanged = true;
   490→        const rest = gids.filter((x) => x !== groupId);
   491→        return { ...m, groupIds: rest.length > 0 ? rest : undefined, privateSource: rest.length > 0 ? m.privateSource : true };
   492→      });
   493→      if (coresChanged) writeJSON(coreKey(contactId), nextCores);
   494→      // 长期层：同核心层
   495→      const longs = readLongTerm(contactId);
   496→      let longsChanged = false;
   497→      const nextLongs = longs.map((m) => {
   498→        const gids = m.groupIds ?? [];
   499→        if (!gids.includes(groupId)) return m;
   500→        longsChanged = true;
   501→        const rest = gids.filter((x) => x !== groupId);
   502→        return { ...m, groupIds: rest.length > 0 ? rest : undefined, privateSource: rest.length > 0 ? m.privateSource : true };
   503→      });
   504→      if (longsChanged) writeJSON(longKey(contactId), nextLongs);
   505→    } catch {
   506→      // 单个成员清理失败不影响其他成员
   507→    }
   508→  }
   509→}

   513→/** 相关性打分：内容与当前上下文的重叠 + 新近加成（长期/核心/碎片各自组内排序用） */
   514→function relevanceScore(content: string, createdAt: number, context: string): number {
   515→  const ctx = contextGramsCache.get(context) ?? bigrams(context);
   516→  contextGramsCache.set(context, ctx);
   517→  if (contextGramsCache.size > 8) contextGramsCache.delete(contextGramsCache.keys().next().value as string);
   518→  let score = overlap(bigrams(content), ctx) * 2;
   519→  const age = Date.now() - createdAt;
   520→  if (age < 86400_000) score += 3;
   521→  else if (age < 7 * 86400_000) score += 2;
   522→  else if (age < 30 * 86400_000) score += 1;
   523→  return score;
   524→}
   525→
   526→function overlap(a: Set<string>, b: Set<string>): number {
   527→  let n = 0;
   528→  a.forEach((g) => {
   529→    if (b.has(g)) n++;
   530→  });
   531→  return n;
   532→}
   533→
   534→/** 上下文文本的 gram 缓存（同一次召回内复用） */
   535→const contextGramsCache = new Map<string, Set<string>>();
   536→
   537→/** 碎片召回得分：相关性 × 权重系数（高权重优先）× 淡化系数（淡化中 0.4；已归档直接不参与） */
   538→function fragRecallScore(f: MemFragment, forget: MemForget, context: string, now: number): { s: number; st: ReturnType<typeof fadeState> } {
   539→  const st = fadeState(f as FadeInput, forget, now);
   540→  const s =
   541→    relevanceScore(f.content, f.reinforcedAt ?? f.sourceTime, context) *
   542→    weightFactor(f.weight) *
   543→    (st === 'fading' ? 0.4 : 1);
   544→  return { s, st };
   545→}
   546→
   547→/**
   548→ * 召回该联系人（互通开关限定范围）的记忆，组装为注入 system 的文本块。
   549→ * 注入顺序（层级从高到低）：长期记忆 → 核心记忆 → 近期碎片；跨层去重：与已选内容相似（≥0.6）
   550→ * 的低层记忆不再重复注入（长期已收编的由长期代表、核心已收编碎片的由核心代表）。
   551→ * 容量上限：长期 8 条、核心 12 条、碎片 5 条（各按 权重×相关性×淡化 降序截断）；
   552→ * 已入长期（archivedAt）的核心、已入核心（consumedAt）与已归档的碎片不再召回。
   553→ * 该联系人没有记忆 / 互通关闭且当前 App 无来源记忆 → 返回空串（正常聊天，不报错）。
   554→ *
   555→ * 群聊互通（opts）：
   556→ * - mode 'private'（缺省）：跨 App 互通开关照旧；另读群聊来源记忆——碎片/总结按来源群
   557→ *   memoryInterop 判断（互斥于跨 App share 开关，两层开关都开才可见）；
   558→ * - mode 'group'（groupId 必传）：只读当前群来源记忆 + （该群互通开启时）自己的非群聊记忆；
   559→ *   其他群的记忆永不参与（群间隔离），角色隔离由存储键天然保证。
   560→ */
   561→export interface MemRecallOpts {
   562→  mode?: 'private' | 'group';
   563→  /** mode='group' 时必传：当前群聊 ID */
   564→  groupId?: string;
   565→  /** 群互通开关解析（缺省读群数据层的 memoryInterop） */
   566→  interopOn?: (groupId: string) => boolean;
   567→  /** 群名解析（来源标注「群聊·群名」用；缺省读群数据层） */
   568→  groupLabel?: (groupId: string) => string | null;
   569→}

   571→export function memRecallBlock(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts): string {
   572→  // 顶层容错（与 collectWbBlocks 同款约定）：记忆注入任何意外异常都不阻断消息发送，
   573→  // 返回空块 = 无记忆正常聊天（群聊/私聊同此规则）
   574→  try {
   575→    return memRecallBlockInner(contactId, app, contextText, opts);
   576→  } catch {
   577→    return '';
   578→  }
   579→}
   580→
   581→function memRecallBlockInner(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts): string {
   582→  if (!contactId) return '';
   583→  memSweepExpiry(contactId);
   584→  const { share, forget } = getMemSettings(contactId);
   585→  const now = Date.now();
   586→  const isGroupMode = opts?.mode === 'group' && !!opts.groupId;
   587→  const curGroup = opts?.groupId ?? '';
   588→  const interopOn = opts?.interopOn ?? defaultGroupInteropOn;
   589→  const groupLabelOf = opts?.groupLabel ?? defaultGroupLabel;
   590→  /** 核心/长期层级的群来源可见性：纯私聊总结恒可见（旧数据兼容）；带群来源的按各群互通判断 */
   591→  const summaryVisiblePrivate = (m: Pick<MemCore, 'groupIds' | 'privateSource'>): boolean => {
   592→    const gids = m.groupIds ?? [];
   593→    if (gids.length === 0) return true; // 纯私聊/朋友圈来源（含旧数据）
   594→    if (m.privateSource !== false) return true; // 混合来源（含私聊部分）：私聊恒可见
   595→    return gids.some((g) => interopOn(g)); // 纯群聊来源：任一来源群互通开启才可见
   596→  };
   597→  /** 群聊模式下的总结可见性：本群来源恒可见；纯私聊总结在互通开时可见；其他群的总结永不可见（群间隔离） */
   598→  const summaryVisibleInGroup = (m: Pick<MemCore, 'groupIds' | 'privateSource'>): boolean => {
   599→    const gids = m.groupIds ?? [];
   600→    if (gids.includes(curGroup)) return true;
   601→    if (gids.some((g) => g !== curGroup)) return false; // 沾了其他群来源：不进本群上下文
   602→    return interopOn(curGroup) && m.privateSource !== false;
   603→  };
   604→  /** 碎片级可见性（来源精确到条） */
   605→  const fragVisiblePrivate = (f: MemFragment): boolean => {
   606→    if (f.source !== 'group') return true;
   607→    return f.sourceGroupId ? interopOn(f.sourceGroupId) : false;
   608→  };
   609→  const fragVisibleInGroup = (f: MemFragment): boolean => {
   610→    if (f.source === 'group') return f.sourceGroupId === curGroup; // 只读当前群（群间隔离）
   611→    return interopOn(curGroup); // 自己的私聊/朋友圈记忆：互通开启才可见
   612→  };
   613→  // 长期记忆（顶层画像）：全量注入（安全上限 8 条防失控）；用户手动设置过期的长期不再注入
   614→  const longs = listLongTerm(contactId)
   615→    .filter((m) => !isMemExpired(m, now) && (share || m.apps.includes(app)))
   616→    .filter((m) => (isGroupMode ? summaryVisibleInGroup(m) : summaryVisiblePrivate(m)))
   617→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, contextText) }))
   618→    .sort((a, b) => b.s - a.s)
   619→    .slice(0, 8);
   620→  // 核心记忆：全量注入（安全上限 12 条）；已被长期收编的不再注入（由长期记忆代表）；
   621→  // 用户手动设置过期的核心不再注入（默认永不过期不受影响）
   622→  const cores = listCores(contactId)
   623→    .filter((m) => !m.archivedAt && !isMemExpired(m, now) && (share || m.apps.includes(app)))
   624→    .filter((m) => (isGroupMode ? summaryVisibleInGroup(m) : summaryVisiblePrivate(m)))
   625→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, contextText) }))
   626→    .sort((a, b) => b.s - a.s)
   627→    .slice(0, 12);
   628→  // 近期记忆碎片：按相关性取前 5 条；已消费的由核心记忆代表，已更新/已过期的不召回
   629→  const frags = listFragments(contactId)
   630→    .filter((f) => !f.consumedAt && !f.supersededAt && !isMemExpired(f, now) && (share || f.app === app))
   631→    .filter((f) => (isGroupMode ? fragVisibleInGroup(f) : fragVisiblePrivate(f)))
   632→    .map((f) => ({ f, ...fragRecallScore(f, forget, contextText, now) }))
   633→    .filter((x) => x.st !== 'faded')
   634→    .sort((a, b) => b.s - a.s)
   635→    .slice(0, 5);
   636→  if (longs.length === 0 && cores.length === 0 && frags.length === 0) return '';
   637→  // 跨层去重：按注入顺序遍历，与已选内容相似（≥0.6）的条目跳过（信息已由高层代表）
   638→  const accepted: string[] = [];
   639→  const isDup = (content: string) => accepted.some((a) => similarity(a, content) >= SIMILAR_MERGE_THRESHOLD);
   640→  const keepLongs = longs.filter(({ m }) => {
   641→    if (isDup(m.content)) return false;
   642→    accepted.push(m.content);
   643→    return true;
   644→  });
   645→  const keepCores = cores.filter(({ m }) => {
   646→    if (isDup(m.content)) return false;
   647→    accepted.push(m.content);
   648→    return true;
   649→  });
   650→  const keepFrags = frags.filter(({ f }) => {
   651→    if (isDup(f.content)) return false;
   652→    accepted.push(f.content);
   653→    return true;
   654→  });
   655→  if (keepLongs.length === 0 && keepCores.length === 0 && keepFrags.length === 0) return '';
   656→  // 时间感知：层内按有效时间（事件时间优先，其次加强/来源/创建时间）从新到旧排序
   657→  keepLongs.sort((a, b) => memEffectiveTime(b.m) - memEffectiveTime(a.m));
   658→  keepCores.sort((a, b) => memEffectiveTime(b.m) - memEffectiveTime(a.m));
   659→  keepFrags.sort((a, b) => memEffectiveTime(b.f) - memEffectiveTime(a.f));
   660→  const lines: string[] = [
   661→    `【关于对方的记忆（跨应用记忆库自动整理；当前时间：${memNowLabel(now)}；聊天时自然运用，不要逐条复述或主动承认看过记忆）】`,
   662→    '（时间越近的记忆越可信：优先参考时间更近的；同一事实新旧矛盾时，以时间更近的为准）',
   663→  ];
   664→  if (keepLongs.length > 0) {
   665→    lines.push('◇ 长期记忆（最稳定的画像；回复时应始终符合这些事实）：');
   666→    keepLongs.forEach(({ m }, i) => lines.push(`${i + 1}. （${memTimeLabel(memEffectiveTime(m), now)}）${m.content}`));
   667→  }
   668→  if (keepCores.length > 0) {
   669→    lines.push('◇ 核心记忆（长期事实；回复时应优先参考这些核心事实，保持前后一致）：');
   670→    keepCores.forEach(({ m }, i) => lines.push(`${i + 1}. （${memTimeLabel(memEffectiveTime(m), now)}）${m.content}`));
   671→  }
   672→  if (keepFrags.length > 0) {
   673→    lines.push('◇ 近期记忆碎片：');
   674→    keepFrags.forEach(({ f }, i) => {
   675→      // 事件时间优先（内容所指的时间），否则用来源对话时间；碎片额外带来源渠道标注
   676→      // （朋友圈/QQ动态/群聊来源的碎片与私聊区分标注，让 AI 知道这是哪里发生的事）
   677→      const srcLabel =
   678→        f.source === 'moments'
   679→          ? f.app === 'wx'
   680→            ? '朋友圈'
   681→            : 'QQ动态'
   682→          : f.source === 'group'
   683→            ? `群聊·${(f.sourceGroupId && groupLabelOf(f.sourceGroupId)) || '群聊'}`
   684→            : MEM_APP_LABEL[f.app];
   685→      const t = f.eventTime != null ? memTimeLabel(f.eventTime, now) : `${memTimeLabel(f.sourceTime, now)}·${srcLabel}`;
   686→      lines.push(`${i + 1}. （${t}）${f.content}`);
   687→    });
   688→  }
   689→  return lines.join('\n');
   690→}

   692→/** 召回预览（设置页）：与 memRecallBlock 同一套层级与排序，返回排序后的记忆对象 */
   693→export function memRecallPreview(contactId: string): { longs: MemLongTerm[]; cores: MemCore[]; frags: MemFragment[] } {
   694→  const { forget } = getMemSettings(contactId);
   695→  const now = Date.now();
   696→  const longs = listLongTerm(contactId)
   697→    .filter((m) => !isMemExpired(m, now))
   698→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, '') }))
   699→    .sort((a, b) => b.s - a.s)
   700→    .slice(0, 3)
   701→    .map((x) => x.m);
   702→  const cores = listCores(contactId)
   703→    .filter((m) => !m.archivedAt && !isMemExpired(m, now))
   704→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, '') }))
   705→    .sort((a, b) => b.s - a.s)
   706→    .slice(0, 3)
   707→    .map((x) => x.m);
   708→  const frags = listFragments(contactId)
   709→    .filter((f) => !f.consumedAt && !f.supersededAt && !isMemExpired(f, now))
   710→    .map((f) => ({ f, ...fragRecallScore(f, forget, '', now) }))
   711→    .filter((x) => x.st !== 'faded')
   712→    .sort((a, b) => b.s - a.s)
   713→    .slice(0, 5)
   714→    .map((x) => x.f);
   715→  return { longs, cores, frags };
   716→}

   727→/**
   728→ * 写入一条「朋友圈/QQ动态」来源的记忆碎片（动态与记忆双向打通：发动态/点赞/评论/回复都会入库）。
   729→ * - 内容由调用方用「机主真实名字 + 角色真实名字」视角拼好（禁「对方/用户/我」）；
   730→ * - 来源标注 source='moments' + 关联动态 id/互动类型，与私聊记忆区分（六.5 数据结构要求）；
   731→ * - 走 appendFragments 同款去重/合并/矛盾管线：同一事实重复写入只会加强，不会重复占位；
   732→ * - app 传动态所在平台（wx 朋友圈 / qq QQ动态）：互通开关关闭时，其它 App 不会召回这些记忆。
   733→ * 返回是否写入（新增或加强）成功。
   734→ */
   735→export function memAddMomentFragment(
   736→  contactId: string,
   737→  app: MemApp,
   738→  content: string,
   739→  src: MemMomentSource,
   740→  sourceTime?: number
   741→): boolean {
   742→  try {
   743→    const text = content.trim();
   744→    if (!contactId || !text) return false;
   745→    const res = appendFragments(
   746→      contactId,
   747→      app,
   748→      [{ text, weight: autoWeight(text) }],
   749→      sourceTime ?? Date.now(),
   750→      undefined,
   751→      { source: 'moments', sourcePostId: src.postId, sourceKind: src.kind, sourceCommentId: src.commentId }
   752→    );
   753→    return res.added.length > 0 || res.merged > 0 || res.superseded > 0;

   797→/** extract 接口返回的单条碎片：文本 + 可选权重 + 时间感知（事件/过期时间）+ 矛盾更新目标 */
   798→interface ExtractItem {
   799→  text: string;

   815→/**
   816→ * 模型时间字符串 → 时间戳（无时区字符串按北京时间解析，与提取锚点/注入标签一致）：
   817→ * - "YYYY-MM-DD" → 北京时间 0 点（仅日期事件，标签不带出无意义的 00:00）
   818→ * - "YYYY-MM-DD HH:mm" → 北京时间该时刻
   819→ * - 其余（含完整 ISO/带偏移）→ Date.parse 兜底
   820→ * 范围距当前 ±MEM_TIME_RANGE_YEARS 年，超出/解析失败返回 null（绝不让幻觉时间入库）。
   821→ */
   822→function parseMemTime(v: unknown, now: number): number | null {
   823→  if (typeof v !== 'string') return null;
   824→  const s = v.trim();
   825→  if (!s || s === 'null' || s === 'undefined') return null;
   826→  let d: Date | null = null;
   827→  const md = /^(\d{4})-(\d{1,2})-(\d{1,2})$/.exec(s);
   828→  const mdt = /^(\d{4})-(\d{1,2})-(\d{1,2})[ T](\d{1,2}):(\d{2})(?::\d{2})?$/.exec(s);
   829→  // 无时区字符串统一按北京时间（UTC+8）解析 —— 与提取锚点（Asia/Shanghai）、注入标签（bjParts）
   830→  // 一致，与设备时区无关；带时区/ISO 带偏移的字符串走 Date.parse 原语义
   831→  if (md) d = new Date(Date.UTC(Number(md[1]), Number(md[2]) - 1, Number(md[3])) - 8 * 3_600_000);
   832→  else if (mdt)
   833→    d = new Date(
   834→      Date.UTC(Number(mdt[1]), Number(mdt[2]) - 1, Number(mdt[3]), Number(mdt[4]), Number(mdt[5])) - 8 * 3_600_000
   835→    );
   836→  else {
   837→    const t = Date.parse(s);
   838→    if (!Number.isNaN(t)) d = new Date(t);
   839→  }
   840→  if (!d) return null;
   841→  const t = d.getTime();
   842→  const span = MEM_TIME_RANGE_YEARS * 366 * 86_400_000;
   843→  if (t < now - span || t > now + span) return null;
   844→  return t;
   845→}

   847→/** 把 extract 接口返回的碎片归一化（字符串/对象混合兼容 + 非法值兜底自动分类 + 时间解析/防幻觉） */
   848→function normalizeExtract(res: ExtractApiResult): ExtractItem[] {
   849→  const raw = Array.isArray(res.fragments) ? res.fragments : [];
   850→  const now = Date.now();
   851→  const out: ExtractItem[] = [];
   852→  for (const x of raw) {
   853→    if (typeof x === 'string') {
   854→      const t = x.trim();
   855→      if (t) out.push({ text: t, weight: autoWeight(t) });
   856→    } else if (x && typeof x === 'object' && typeof x.text === 'string' && x.text.trim()) {
   857→      const o = x as { text: string; weight?: unknown; eventTime?: unknown; expiresAt?: unknown; supersedes?: unknown };
   858→      const item: ExtractItem = { text: o.text.trim(), weight: normalizeWeight(o.weight, o.text) };
   859→      const et = parseMemTime(o.eventTime, now);
   860→      const ex = parseMemTime(o.expiresAt, now);
   861→      // 双保险：过期时间不得早于事件时间（服务端已校验一次，这里兜底防历史残留/异常数据）
   862→      if (et != null) item.eventTime = et;
   863→      if (ex != null && (et == null || ex >= et)) item.expiresAt = ex;
   864→      if (Array.isArray(o.supersedes)) {
   865→        const ids = o.supersedes
   866→          .filter((id): id is string => typeof id === 'string' && id.trim().length > 0)
   867→          .map((id) => id.trim())
   868→          .slice(0, 6);
   869→        if (ids.length > 0) item.supersedes = ids;
   870→      }
   871→      out.push(item);
   872→    }
   873→  }
   874→  return out.slice(0, 6);
   875→}

   877→async function callMemoryApi<T extends ExtractApiResult & SummarizeApiResult>(
   878→  op: 'extract' | 'summarize',
   879→  body: Record<string, unknown>,
   880→  apiConfig: ApiConfig
   881→): Promise<T> {
   882→  const res = await fetch(`/api/memory/${op}`, {
   883→    method: 'POST',
   884→    headers: { 'Content-Type': 'application/json' },
   885→    body: JSON.stringify({ ...body, config: apiConfig }),
   886→  });
   887→  const data = (await res.json().catch(() => ({}))) as T;
   888→  if (!res.ok) throw new Error(data?.error || `请求失败（${res.status}）`);
   889→  return data;
   890→}
   891→
   892→/** 归一化文本（去空白）——精确去重键 */
   893→const normText = (s: string) => s.replace(/\s+/g, '');
   894→
   895→/**
   896→ * 追加碎片（矛盾更新优先，其次三层去重）：
   897→ * 0) 矛盾更新（模型标注 supersedes 且目标存在）：旧记忆标记 supersededAt/supersededBy（「已更新」，
   898→ *    不再参与召回/总结），本条直接新增（事件/过期时间一并入库）；矛盾条目不走相似合并
   899→ *    （「不吃辣」绝不能被合并进「爱吃辣」）；
   900→ * 1) 精确重复 → 记一次「加强」（reinforcedAt/次数刷新），不新增；
   901→ * 2) 相似（2-gram ≥0.6，同一事实的不同说法）→ 合并进已有条目：内容取更完整的、权重取更高、
   902→ *    加强计时刷新、计数 +1，不新增占位（已更新/已过期/已消费的旧记忆不作为合并目标）；
   903→ * 3) 全新 → 新建（权重：LLM 判定优先，缺省按关键词自动分类）。
   904→ * 返回 { added: 新增条目; merged: 合并/加强次数; superseded: 矛盾更新条数 }
   905→ */
   906→function appendFragments(
   907→  contactId: string,
   908→  app: MemApp,
   909→  items: ExtractItem[],
   910→  sourceTime: number,
   911→  sourceMsgId?: string,
   912→  /** 额外字段（动态/群聊来源记忆用：source/sourcePostId/sourceKind/sourceCommentId/sourceGroupId/groupMembers 直通新碎片） */
   913→  extra?: Partial<Pick<MemFragment, 'source' | 'sourcePostId' | 'sourceKind' | 'sourceCommentId' | 'sourceGroupId' | 'groupMembers'>>
   914→): { added: MemFragment[]; merged: number; superseded: number } {
   915→  const list = readFragments(contactId);
   916→  const now = Date.now();
   917→  const seen = new Set(list.filter((f) => !f.supersededAt && !isMemExpired(f, now)).map((f) => normText(f.content)));
   918→  const added: MemFragment[] = [];
   919→  let merged = 0;
   920→  let superseded = 0;
   921→  const newItem = (content: string, item: ExtractItem, id?: string): MemFragment => ({
   922→    id: id ?? uid(),
   923→    contactId,
   924→    app,
   925→    content,
   926→    sourceTime,
   927→    createdAt: now,
   928→    eventTime: item.eventTime ?? undefined,
   929→    expiresAt: item.expiresAt ?? undefined,
   930→    weight: item.weight ?? autoWeight(content),
   931→    reinforcedAt: now,
   932→    reinforceCount: 0,
   933→    sourceMsgId,
   934→    ...(extra ?? {}),
   935→  });
   936→  for (const item of items) {
   937→    const content = item.text.trim();
   938→    if (!content) continue;
   939→    const key = normText(content);
   940→    // 0) 矛盾更新：标记被更正的旧记忆为「已更新」，本条直接新增
   941→    const targets = (item.supersedes ?? [])
   942→      .map((id) => list.find((f) => f.id === id))
   943→      .filter((f): f is MemFragment => f != null && f.supersededAt == null);
   944→    if (targets.length > 0) {
   945→      const newId = uid();
   946→      for (const t of targets) {
   947→        t.supersededAt = now;
   948→        t.supersededBy = newId;
   949→      }
   950→      superseded += targets.length;
   951→      seen.add(key);
   952→      added.push(newItem(content, item, newId));
   953→      continue;
   954→    }
   955→    // 1) 精确重复 → 加强已有记忆（已更新/已过期的旧条目不是目标）
   956→    if (seen.has(key)) {
   957→      const hit = list.find((f) => normText(f.content) === key && !f.supersededAt && !isMemExpired(f, now));
   958→      if (hit) {
   959→        hit.reinforcedAt = now;
   960→        hit.reinforceCount = (hit.reinforceCount ?? 0) + 1;
   961→        hit.weight = higherWeight(hit.weight, item.weight);
   962→        // 时间感知：未手动编辑过的记忆，重复提及且本次提取带新时间时刷新（事件时间保鲜）；
   963→        // 用户手动设置过（timeEditedAt）的一律以用户为准，绝不覆盖
   964→        if (!hit.timeEditedAt) {
   965→          if (item.eventTime != null) hit.eventTime = item.eventTime;
   966→          if (item.expiresAt != null) hit.expiresAt = item.expiresAt;
   967→        }
   968→        merged++;
   969→        continue;
   970→      }
   971→      // 命中的只是已更新/已过期的旧条目 → 落到下面按新增处理
   972→    }
   973→    // 2) 相似说法 → 合并进已有记忆（优先未消费的；内容保留更完整的一条）
   974→    const simHit = list
   975→      .filter((f) => !f.supersededAt && !isMemExpired(f, now) && f.content.length >= 4 && content.length >= 4)
   976→      .sort((a, b) => Number(Boolean(a.consumedAt)) - Number(Boolean(b.consumedAt)) || a.createdAt - b.createdAt)
   977→      .find((f) => similarity(f.content, content) >= SIMILAR_MERGE_THRESHOLD);
   978→    if (simHit) {
   979→      if (content.length > simHit.content.length) simHit.content = content;
   980→      simHit.reinforcedAt = now;
   981→      simHit.reinforceCount = (simHit.reinforceCount ?? 0) + 1;
   982→      simHit.weight = higherWeight(simHit.weight, item.weight);
   983→      // 时间感知（与精确重复同规则）：未手动编辑过 → 新时间保鲜；手动设置过 → 用户优先不覆盖
   984→      if (!simHit.timeEditedAt) {
   985→        if (item.eventTime != null) simHit.eventTime = item.eventTime;
   986→        if (item.expiresAt != null) simHit.expiresAt = item.expiresAt;
   987→      }
   988→      if (!simHit.sourceMsgId && sourceMsgId) simHit.sourceMsgId = sourceMsgId;
   989→      // 溯源字段补齐（动态来源记忆）：点赞/评论同动态连续写入时内容相近会被合并成一条，
   990→      // 合并目标（先写入的那条）若缺溯源字段，用本次传入的补齐——否则删评论时按
   991→      // sourceCommentId 清理会漏掉这条合并碎片（六.3 级联清理依赖这些字段）
   992→      if (extra) {
   993→        if (!simHit.source && extra.source) simHit.source = extra.source;
   994→        if (!simHit.sourcePostId && extra.sourcePostId) simHit.sourcePostId = extra.sourcePostId;
   995→        if (!simHit.sourceKind && extra.sourceKind) simHit.sourceKind = extra.sourceKind;
   996→        if (!simHit.sourceCommentId && extra.sourceCommentId) simHit.sourceCommentId = extra.sourceCommentId;
   997→        if (!simHit.sourceGroupId && extra.sourceGroupId) simHit.sourceGroupId = extra.sourceGroupId;
   998→        if (!simHit.groupMembers && extra.groupMembers) simHit.groupMembers = extra.groupMembers;
   999→      }
  1000→      seen.add(normText(simHit.content));
  1001→      merged++;
  1002→      continue;
  1003→    }
  1004→    // 3) 全新记忆
  1005→    seen.add(key);
  1006→    added.push(newItem(content, item));
  1007→  }
  1008→  if (added.length > 0 || merged > 0 || superseded > 0) writeJSON(fragKey(contactId), [...list, ...added]);
  1009→  return { added, merged, superseded };
  1010→}

  1012→/** 把当前待总结（未被更新/未过期/未消费/未归档）的碎片总结为核心记忆，并标记来源碎片已消费（阈值判断由调用方负责） */
  1013→async function summarizePendingIntoCore(contactId: string, apiConfig: ApiConfig, names?: MemNames | null): Promise<MemCore> {
  1014→  const { forget } = getMemSettings(contactId);
  1015→  const now = Date.now();
  1016→  const pending = readFragments(contactId).filter(
  1017→    (f) =>
  1018→      !f.consumedAt &&
  1019→      !f.supersededAt &&
  1020→      !isMemExpired(f, now) &&
  1021→      fadeState(f as FadeInput, forget, now) !== 'faded'
  1022→  );
  1023→  const { userName, peerName } = namesOf(names);
  1024→  const res = await callMemoryApi<SummarizeApiResult>(
  1025→    'summarize',
  1026→    {
  1027→      level: 'core',
  1028→      fragments: pending.map((f) => f.content),
  1029→      userName,
  1030→      peerName,
  1031→    },
  1032→    apiConfig
  1033→  );
  1034→  const summary = (res.summary || '').trim();
  1035→  if (!summary) throw new Error('总结结果为空');
  1036→  const core: MemCore = {
  1037→    id: uid(),
  1038→    contactId,
  1039→    content: summary,
  1040→    fragmentCount: pending.length,
  1041→    sourceIds: pending.map((f) => f.id),
  1042→    apps: Array.from(new Set(pending.map((f) => f.app))),
  1043→    createdAt: Date.now(),
  1044→    // 时间感知：来源碎片中带事件时间的，取最早一条作为核心的事件时间（可选，仅标注用）
  1045→    eventTime: (() => {
  1046→      const ets = pending.map((f) => f.eventTime).filter((t): t is number => t != null);
  1047→      return ets.length > 0 ? Math.min(...ets) : undefined;
  1048→    })(),
  1049→    // 群聊来源携带（互通与群间隔离过滤用）：混合来源 privateSource=true，纯群聊为 false
  1050→    groupIds: Array.from(new Set(pending.map((f) => f.sourceGroupId).filter((g): g is string => !!g))),
  1051→    privateSource: pending.some((f) => f.source !== 'group'),
  1052→  };
  1053→  const all = readCores(contactId);
  1054→  writeJSON(coreKey(contactId), [...all, core]);
  1055→  // 标记来源碎片已消费
  1056→  const consumedIds = new Set(core.sourceIds);
  1057→  writeJSON(
  1058→    fragKey(contactId),
  1059→    readFragments(contactId).map((f) => (consumedIds.has(f.id) ? { ...f, consumedAt: Date.now() } : f))
  1060→  );
  1061→  return core;
  1062→}

  1064→/** 把当前未归档且未过期的核心记忆总结为一条长期记忆，并标记来源核心已归档（方案A：不重复总结） */
  1065→async function summarizeCoresIntoLong(contactId: string, apiConfig: ApiConfig, names?: MemNames | null): Promise<MemLongTerm> {
  1066→  const now = Date.now();
  1067→  const pending = readCores(contactId).filter((m) => !m.archivedAt && !isMemExpired(m, now));
  1068→  const { userName, peerName } = namesOf(names);
  1069→  const res = await callMemoryApi<SummarizeApiResult>(
  1070→    'summarize',
  1071→    {
  1072→      level: 'long',
  1073→      fragments: pending.map((m) => m.content),
  1074→      userName,
  1075→      peerName,
  1076→    },
  1077→    apiConfig
  1078→  );
  1079→  const summary = (res.summary || '').trim();
  1080→  if (!summary) throw new Error('总结结果为空');
  1081→  const long: MemLongTerm = {
  1082→    id: uid(),
  1083→    contactId,
  1084→    content: summary,
  1085→    coreCount: pending.length,
  1086→    sourceIds: pending.map((m) => m.id),
  1087→    apps: Array.from(new Set(pending.flatMap((m) => m.apps))),
  1088→    createdAt: Date.now(),
  1089→    // 群聊来源携带（互通与群间隔离过滤用）：混合来源 privateSource=true，纯群聊为 false
  1090→    groupIds: Array.from(new Set(pending.flatMap((m) => m.groupIds ?? []))),
  1091→    privateSource: pending.some((m) => m.privateSource !== false),
  1092→  };
  1093→  const all = readLongTerm(contactId);
  1094→  writeJSON(longKey(contactId), [...all, long]);
  1095→  // 标记来源核心已归档（archivedAt）：召回由长期记忆代表，不再参与后续总结
  1096→  const archivedIds = new Set(long.sourceIds);
  1097→  writeJSON(
  1098→    coreKey(contactId),
  1099→    readCores(contactId).map((m) => (archivedIds.has(m.id) ? { ...m, archivedAt: Date.now() } : m))
  1100→  );
  1101→  return long;
  1102→}

  1125→/**
  1126→ * 一轮 AI 对话结束后的记忆管线（各聊天 App 的 finalize 成功分支调用）：
  1127→ * 1) 轮次 +1；2) 达到提取间隔 → 从最近对话提取记忆碎片（后台异步，不阻塞聊天）；
  1128→ * 3) 碎片积累达到核心阈值 → 自动总结核心记忆；4) 核心积累达到长期阈值 → 自动总结长期记忆。
  1129→ * 失败静默（console.warn），不打断聊天。
  1130→ * buildConvo 惰性调用：只有真的需要提取时才读取/整理对话文本。
  1131→ * getSourceMsgId 惰性调用：提取时取来源消息 ID（追溯用，可省略）。
  1132→ * names：用户真实名字 + 角色名字（视角统一注入提取/总结 prompt；缺省回退固定称呼）。
  1133→ * opts：群聊轮次传 roundScope + group（群记忆来源标记，轮次计数与私聊互不干扰）。
  1134→ */
  1135→export function memAfterAiTurn(
  1136→  contactId: string | null,
  1137→  app: MemApp,
  1138→  apiConfig: ApiConfig,
  1139→  buildConvo: () => MemConvoTurn[],
  1140→  getSourceMsgId?: () => string | undefined,
  1141→  names?: MemNames | null,
  1142→  opts?: MemTurnOpts
  1143→): void {
  1144→  if (!contactId) return;
  1145→  const scope = opts?.roundScope ? `:${opts.roundScope}` : '';
  1146→  try {
  1147→    // 时间感知：先把已到期的碎片标记归档（惰性清扫，召回/总结另有实时过滤兜底）
  1148→    memSweepExpiry(contactId);
  1149→    const key = `mem-round:${contactId}:${app}${scope}`;
  1150→    const count = (readJSON<number>(key) ?? 0) + 1;
  1151→    const { interval } = getMemSettings(contactId);
  1152→    if (count < interval) {
  1153→      writeJSON(key, count);
  1154→      return;
  1155→    }
  1156→    // 达到间隔：归零计数并异步提取（归零在先避免每轮重试轰炸；下个窗口自然重试）
  1157→    writeJSON(key, 0);
  1158→    const guard = `${contactId}:${app}${scope}`;
  1159→    if (inflight.has(guard)) return;
  1160→    inflight.add(guard);
  1161→    void (async () => {
  1162→      try {
  1163→        const convo = buildConvo();
  1164→        if (convo.length >= 4) {
  1165→          const res = await callMemoryApi<ExtractApiResult>(
  1166→            'extract',
  1167→            { conversation: convo, app, existing: existingForConflict(contactId), ...namesOf(names) },
  1168→            apiConfig
  1169→          );
  1170→          const items = normalizeExtract(res);
  1171→          if (items.length > 0) {
  1172→            // 群聊轮次：碎片带群来源标记（source/sourceGroupId/groupMembers），供互通召回过滤
  1173→            const extra = opts?.group
  1174→              ? { source: 'group' as const, sourceGroupId: opts.group.id, groupMembers: opts.group.members }
  1175→              : undefined;
  1176→            appendFragments(contactId, app, items, Date.now(), getSourceMsgId?.(), extra);
  1177→          }
  1178→        }
  1179→      } catch (err) {
  1180→        console.warn('[memory] 自动提取失败（下个窗口重试）', err);
  1181→      }
  1182→      // 核心总结与提取相互独立：即使本轮提取失败，只要已积累的未消费碎片达到阈值，
  1183→      // 仍应尝试凝结核心记忆（否则一次提取故障会连带把总结也卡到下个窗口）
  1184→      try {
  1185→        await maybeAutoSummarize(contactId, apiConfig, names);
  1186→      } catch (err) {
  1187→        console.warn('[memory] 自动总结核心记忆失败（下个窗口重试）', err);
  1188→      }
  1189→      // 长期总结与核心总结相互独立：核心达到长期阈值即触发（失败下个窗口重试）
  1190→      try {
  1191→        await maybeAutoLongSummarize(contactId, apiConfig, names);
  1192→      } catch (err) {
  1193→        console.warn('[memory] 自动总结长期记忆失败（下个窗口重试）', err);
  1194→      } finally {
  1195→        inflight.delete(guard);
  1196→      }
  1197→    })();
  1198→  } catch {
  1199→    // 记忆是增强能力：任何异常都不影响聊天
  1200→  }
  1201→}

  1310→// ---------------- 手动「整理重复记忆」 ----------------
  1311→
  1312→/**
  1313→ * 全库去重整理（设置页按钮）：两两扫描该联系人全部碎片，相似（≥0.6）的合并为一条。
  1314→ * 保留更早创建的为正本（若另一条已入核心则保留已入核心者），内容取更完整的、
  1315→ * 权重取更高、加强次数累加、加强时间取较新。返回合并掉的条数。
  1316→ */
  1317→export function memDedupeNow(contactId: string): number {
  1318→  let list = readFragments(contactId);
  1319→  const before = list.length;
  1320→  let changed = true;
  1321→  while (changed) {
  1322→    changed = false;
  1323→    outer: for (let i = 0; i < list.length; i++) {
  1324→      for (let j = i + 1; j < list.length; j++) {
  1325→        const a = list[i];
  1326→        const b = list[j];
  1327→        if (!a.content || !b.content || a.content.length < 4 || b.content.length < 4) continue;
  1328→        if (similarity(a.content, b.content) < SIMILAR_MERGE_THRESHOLD) continue;
  1329→        // 正本：已入核心者优先，其次更早创建者
  1330→        const keep = Number(Boolean(a.consumedAt)) - Number(Boolean(b.consumedAt)) !== 0
  1331→          ? (a.consumedAt ? a : b)
  1332→          : (a.createdAt <= b.createdAt ? a : b);
  1333→        const drop = keep === a ? b : a;
  1334→        if (drop.content.length > keep.content.length) keep.content = drop.content;
  1335→        keep.weight = higherWeight(keep.weight, drop.weight);
  1336→        // 时间感知（手动优先）：正本未被手动编辑时从副本补齐缺失的时间字段；
  1337→        // 副本被手动编辑过则把时间连同标记一并继承（合并体继续受用户设置保护）
  1338→        if (!keep.timeEditedAt) {
  1339→          if (keep.eventTime == null && drop.eventTime != null) keep.eventTime = drop.eventTime;
  1340→          if (keep.expiresAt == null && drop.expiresAt != null) keep.expiresAt = drop.expiresAt;
  1341→          if (drop.timeEditedAt) keep.timeEditedAt = drop.timeEditedAt;
  1342→        }
  1343→        keep.reinforceCount = (keep.reinforceCount ?? 0) + 1 + (drop.reinforceCount ?? 0);
  1344→        keep.reinforcedAt = Math.max(keep.reinforcedAt ?? 0, drop.reinforcedAt ?? 0);
  1345→        keep.sourceMsgId = keep.sourceMsgId ?? drop.sourceMsgId;
  1346→        keep.consumedAt = keep.consumedAt ?? drop.consumedAt;
  1347→        list = list.filter((f) => f.id !== drop.id);
  1348→        changed = true;
  1349→        break outer;
  1350→      }
  1351→    }
  1352→  }
  1353→  if (list.length !== before) writeJSON(fragKey(contactId), list);
  1354→  return before - list.length;
  1355→}

  1357→// ---------------- 视角修复（旧版「对方/用户」代称 → 真实名字） ----------------
  1358→
  1359→/** 同义反复短语：「小晨叫小晨 / 小晨名叫小晨 / 小晨的名字是小晨」等（名字身份事实无信息量） */
  1360→function tautologyClause(name: string): RegExp {
  1361→  const n = name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  1362→  return new RegExp(`${n}(?:叫|名叫|名字叫|的名字是|名字是)${n}[，,、]?`, 'g');
  1363→}

  1365→/**
  1366→ * 修复旧记忆的视角混乱（设置页按钮）：把碎片/核心/长期里「对方」替换为角色名字、「用户」
  1367→ * 替换为用户真实名字（旧版提取器固定用这两个代称，语义明确可安全替换）。
  1368→ * 只替换提供了对应名字的代称（名字缺失时代称保留，避免产生无主语碎片）；
  1369→ * 「我/你/TA」等代称语义不明不自动替换（可手动编辑）。
  1370→ * 同义反复清理：「用户叫小晨」→「小晨叫小晨」这类名字身份短语直接剥掉（已由全库用名隐含），
  1371→ * 剥完为空的条目删除。返回各层修复条数（含删除）。
  1372→ */
  1373→export function memRepairPerspectiveNow(
  1374→  contactId: string,
  1375→  names?: MemNames | null
  1376→): { frags: number; cores: number; longs: number } {
  1377→  const { userName, peerName } = namesOf(names);
  1378→  const hasUser = Boolean((names?.user ?? '').trim());
  1379→  const hasPeer = Boolean((names?.peer ?? '').trim());
  1380→  if (!hasUser && !hasPeer) return { frags: 0, cores: 0, longs: 0 };
  1381→  const fix = (content: string): string => {
  1382→    let out = content;
  1383→    if (hasPeer) out = out.replace(/对方/g, peerName);
  1384→    if (hasUser) out = out.replace(/用户/g, userName);
  1385→    // 同义反复清理：「用户叫小晨」替换后变成「小晨叫小晨」，名字身份事实已由全库用名隐含，
  1386→    // 直接剥掉该短语（拆不出有效内容则整条删除）
  1387→    out = out
  1388→      .replace(tautologyClause(userName), '')
  1389→      .replace(tautologyClause(peerName), '')
  1390→      .replace(/^[，,。、；;\s]+/, '')
  1391→      .replace(/[，,\s]+$/, '');
  1392→    return out;
  1393→  };
  1394→  /** 三层通用：替换 + 剥掉清空后为空的条目（返回修复条数） */
  1395→  const fixLayer = <T extends { content: string }>(list: T[]): { next: T[]; fixed: number } => {
  1396→    let fixed = 0;
  1397→    const out: T[] = [];
  1398→    for (const item of list) {
  1399→      const repaired = fix(item.content);
  1400→      if (repaired !== item.content) {
  1401→        fixed++;
  1402→        if (repaired) out.push({ ...item, content: repaired, editedAt: Date.now() });
  1403→      } else {
  1404→        out.push(item);
  1405→      }
  1406→    }
  1407→    return { next: out, fixed };
  1408→  };
  1409→  // 碎片
  1410→  const fragRes = fixLayer(readFragments(contactId));
  1411→  if (fragRes.fixed > 0) writeJSON(fragKey(contactId), fragRes.next);
  1412→  // 核心
  1413→  const coreRes = fixLayer(readCores(contactId));
  1414→  if (coreRes.fixed > 0) writeJSON(coreKey(contactId), coreRes.next);
  1415→  // 长期
  1416→  const longRes = fixLayer(readLongTerm(contactId));
  1417→  if (longRes.fixed > 0) writeJSON(longKey(contactId), longRes.next);
  1418→  return { frags: fragRes.fixed, cores: coreRes.fixed, longs: longRes.fixed };
  1419→}

  1437→/** 通用消息形状（各 App 消息的字段子集；卡片类映射为短标签，保证提取器可读） */
  1438→interface RawishMsg {

  1447→function cardLabel(kind: unknown): string | null {
  1448→  if (kind === 'redpacket') return '[红包]';
  1449→  if (kind === 'transfer') return '[转账]';
  1450→  if (kind === 'family') return '[亲属卡]';
  1451→  if (kind === 'image') return '[图片]';
  1452→  if (kind === 'sticker') return '[表情]';
  1453→  if (kind === 'location') return '[位置]';
  1454→  return null;
  1455→}

  1457→/** 把任意 App 的消息数组整理成 {role, text} 问答对（撤回/失败/空消息剔除） */
  1458→export function memConvoFromRaw(msgs: unknown[], peerName: string): MemConvoTurn[] {
  1459→  const out: MemConvoTurn[] = [];
  1460→  for (const raw of msgs) {
  1461→    if (!raw || typeof raw !== 'object') continue;
  1462→    const m = raw as RawishMsg;
  1463→    if (m.recalled === true || m.error === true) continue;
  1464→    const label = cardLabel(m.kind);
  1465→    const text =
  1466→      label ??
  1467→      ((typeof m.content === 'string' ? m.content.trim() : '') ||
  1468→        (typeof m.text === 'string' ? m.text.trim() : ''));
  1469→    if (!text) continue;
  1470→    const isMe = m.role === 'me' || m.role === 'user';
  1471→    out.push({ role: isMe ? 'me' : 'peer', text: text.slice(0, 400) });
  1472→  }
  1473→  void peerName;
  1474→  return out.slice(-60);
  1475→}

  1497→export function memMostRecentApp(contactId: string): { app: MemApp; convo: MemConvoTurn[] } | null {
  1498→  let best: { app: MemApp; ts: number; convo: MemConvoTurn[] } | null = null;
  1499→  for (const app of ['wx', 'qq', 'sms'] as MemApp[]) {
  1500→    const key =
  1501→      app === 'wx'
  1502→        ? `wx-chat-msgs:${contactId}`
  1503→        : app === 'qq'
  1504→          ? `qq-chat-msgs:${contactId}`
  1505→          : `ios-chat-msgs:c:${contactId}`;
  1506→    try {
  1507→      const parsed: unknown = kvGet(key);
  1508→      if (!Array.isArray(parsed) || parsed.length === 0) continue;
  1509→      const last = parsed[parsed.length - 1] as { time?: unknown } | null;
  1510→      const ts = typeof last?.time === 'number' ? last.time : 0;
  1511→      if (!best || ts > best.ts) best = { app, ts, convo: memConvoFromRaw(parsed, '') };
  1512→    } catch {
  1513→      continue;
  1514→    }
  1515→  }
  1516→  return best ? { app: best.app, convo: best.convo } : null;
  1517→}
