     4→ * 记忆库（跨 App 共享记忆）：记忆碎片 / 核心记忆 / 长期记忆 / 每联系人设置。
     5→ *
     6→ * 设计要点（「记忆库 APP」需求）：
     7→ * - 数据全部按【联系人 ID】隔离，localStorage 持久化（与聊天记录同层，重启保留）：
     8→ *   · mem-frag:<contactId>   记忆碎片（每条带来源 App/时间/消息 ID/权重/加强时间）
     9→ *   · mem-ltm:<contactId>    核心记忆（M 条碎片总结；键名沿用历史 ltm，语义=核心层）
    10→ *   · mem-long:<contactId>   长期记忆（K 条核心记忆总结出的最稳定画像，层级顶层）
    11→ *   · mem-settings:<contactId> 每联系人设置（提取频率/核心阈值/长期阈值/互通开关/失忆程度）
    12→ *   · mem-msgcount:<cid>[:<app>[:group:<gid>]]  自上次提取累计的消息条数（提取节奏用）：
    13→ *     群聊始终按群独立；私聊互通开 = 按角色一条计数（四 App 合并累计）、互通关 = 各 App 独立
    14→ *   · mem-anchor:<cid>:<app>[:group:<gid>]  各会话已计数的最后一条消息 ID（增量计数锚点：
    15→ *     每次 AI 回复后数锚点之后的新增消息（用户+AI 都算），汇入计数；切换互通只清计数不清锚点）
    16→ *   · 旧 mem-round:<cid>:<app> 轮次键已废弃不再读写（仅删除联系人时残留清扫）
    17→ * - 三层管线：碎片（每 N 条消息提取）→ 核心记忆（积累 M 条碎片总结一条）→ 长期记忆（积累 K 条
    18→ *   未归档核心总结一条，默认 K=5）；核心被长期收编后标记 archivedAt（方案A：不再参与后续
    19→ *   总结与召回，由长期记忆代表，避免重复总结/重复注入）。
    20→ * - 互通开关（默认开）：开 = 四个 App（QQ/微信/信息/电话）共享该联系人全部记忆；
    21→ *   关 = 各 App 只召回自己来源的记忆（碎片/核心/长期均按来源 App 过滤）。
    22→ *   存储始终单一池 + App 标记：切换开关无需迁移数据，UI 管理视图始终展示该联系人全部记忆。
    23→ * - 不同联系人之间永远隔离：键即隔离边界，召回/提取都不跨联系人读取。
    24→ * - 【视角统一】提取/总结 prompt 一律用「用户真实名字 + 角色名字」指代双方，禁混用
    25→ *   「对方/用户/我」；memRepairPerspectiveNow 可把旧版「对方/用户」代称修复为真实名字。
    26→ * - 【过期淡化】失忆程度（快≈3天/中≈2周/慢≈2个月/从不）：超过有效期一半「淡化中」召回降权，
    27→ *   超过有效期「已归档」不再参与召回与总结；低权重记忆先淡化、高权重更持久；
    28→ *   重复提及/合并相似/手动「回忆一下」都会刷新淡化计时。
    29→ * - 【去重合并】同一事实的不同说法（相似度 ≥0.6）自动合并为一条：重复提取=加强而非重复占位；
    30→ *   设置页可手动「整理重复记忆」一次性清理；召回时跨层去重（长期→核心→碎片顺序，后层与已选
    31→ *   内容相似则跳过）。
    32→ * - 【权重优先级】姓名/关系/承诺/健康禁忌 → 高权重（自动分类 + 手动可调）；召回按权重×相关性排序。
    33→ * - 【时间感知】记忆与当前时间联动：碎片可带 eventTime（事件发生时间）/ expiresAt（过期时间，
    34→ *   空=永不过期）；提取时由模型按当前时间锚点判断（服务端校验 ±5 年防幻觉）；注入时头部带当前
    35→ *   时间、层内按有效时间从新到旧、每条带时间标签，并提示优先参考更近的记忆；过期碎片自动归档
    36→ *   （expiredAt）不再召回；新旧矛盾时模型标注 supersedes，旧记忆标记「已更新」不再召回。
    37→ *   核心/长期默认永不过期，可被用户手动设置过期；用户手动编辑过的时间（timeEditedAt）以用户
    38→ *   设置为准，自动流程（提取合并/去重/总结）一律不得覆盖。
    39→ * - 【来源追溯】每条碎片记录来源 App / 来源时间 / 来源消息 ID，回答「你怎么知道的」。
    40→ * - 提取/总结调用 /api/memory/extract 与 /api/memory/summarize（用户 API 配置优先，
    41→ *   服务端 SDK 兜底）；自动提取失败静默（下一窗口重试），手动「立即总结」失败给提示；
    42→ *   手动入口共四个且互斥：碎片页右上角（仅提取碎片）、核心页右上角（碎片→核心）、
    43→ *   长期页右上角（核心→长期）、设置页（按所选粒度执行）。
    44→ * - 删除联系人时由 contacts-store.deleteContact 调 memPurgeContact 级联清理全部记忆。
    45→ */

   195→function readJSON<T>(key: string): T | null {
   196→  try {
   197→    return kvGet<T>(key);
   198→  } catch {
   199→    return null;
   200→  }
   201→}

   203→function writeJSON(key: string, value: unknown): void {
   204→  try {
   205→    // 内存同步 + 异步写穿 IndexedDB；存储失败不中断聊天（记忆是增强能力）
   206→    kvSet(key, value);
   207→  } catch {
   208→    // 忽略
   209→  }
   210→}

   258→export function getMemSettings(contactId: string): MemSettings {
   259→  const raw = readJSON<Partial<MemSettings>>(settingsKey(contactId));
   260→  if (!raw) return { ...DEFAULT_MEM_SETTINGS };
   261→  return {
   262→    interval: MEM_INTERVAL_OPTIONS.includes(raw.interval as MemSettings['interval'])
   263→      ? (raw.interval as MemSettings['interval'])
   264→      : DEFAULT_MEM_SETTINGS.interval,
   265→    threshold: MEM_THRESHOLD_OPTIONS.includes(raw.threshold as MemSettings['threshold'])
   266→      ? (raw.threshold as MemSettings['threshold'])
   267→      : DEFAULT_MEM_SETTINGS.threshold,
   268→    longThreshold: MEM_LONG_OPTIONS.includes(raw.longThreshold as MemSettings['longThreshold'])
   269→      ? (raw.longThreshold as MemSettings['longThreshold'])
   270→      : DEFAULT_MEM_SETTINGS.longThreshold,
   271→    share: raw.share !== false,
   272→    forget: raw.forget === 'fast' || raw.forget === 'medium' || raw.forget === 'slow' || raw.forget === 'never'
   273→      ? raw.forget
   274→      : DEFAULT_MEM_SETTINGS.forget,
   275→  };
   276→}

   278→export function saveMemSettings(contactId: string, patch: Partial<MemSettings>): MemSettings {
   279→  const prev = getMemSettings(contactId);
   280→  const next = { ...prev, ...patch };
   281→  writeJSON(settingsKey(contactId), next);
   282→  // 互通口径切换：计数清零重新累计（合并计数的数字对独立计数无意义，反之亦然；
   283→  // 锚点保留——已计过的消息不重复计，切换后只数新消息）
   284→  if (patch.share !== undefined && patch.share !== prev.share) {
   285→    try {
   286→      kvDelByPrefix(`mem-msgcount:${contactId}`);
   287→    } catch {
   288→      // 忽略
   289→    }
   290→  }
   291→  return next;
   292→}

   331→/** 未被核心总结消费、且未被更新/过期/归档的碎片数（设置页展示/总结触发判断用） */
   332→export function pendingFragmentCount(contactId: string): number {
   333→  const { forget } = getMemSettings(contactId);
   334→  const now = Date.now();
   335→  return readFragments(contactId).filter(
   336→    (f) =>
   337→      !f.consumedAt &&

   370→export function memSweepExpiry(contactId: string): number {
   371→  const now = Date.now();

   373→  let changed = 0;
   374→  const next = list.map((f) => {
   375→    if (f.expiresAt != null && now > f.expiresAt && !f.expiredAt && !f.supersededAt) {
   376→      changed++;
   377→      return { ...f, expiredAt: now };
   378→    }
   379→    return f;
   380→  });
   381→  if (changed > 0) writeJSON(fragKey(contactId), next);
   382→  return changed;
   383→}

   422→/**
   423→ * 手动时间编辑入参（时间感知：用户在记忆库手动修改事件/过期时间）：
   424→ * - eventTime/expiresAt 传 null 表示清除（事件时间清空 / 恢复永不过期），传时间戳表示设置；
   425→ * - undefined 表示不改动该字段。
   426→ */
   427→export interface MemTimePatch {
   428→  eventTime?: number | null;
   429→  expiresAt?: number | null;
   430→}
   431→
   432→/**
   433→ * 三层通用：应用手动时间补丁。被手动编辑过的记忆会带上 timeEditedAt 标记，
   434→ * 此后自动流程（提取合并/去重/总结）一律以用户设置为准，不得改写其 eventTime/expiresAt。
   435→ */
   436→function applyTimePatch<
   437→  T extends { eventTime?: number; expiresAt?: number; expiredAt?: number; timeEditedAt?: number; editedAt?: number }
   438→>(rec: T, patch: MemTimePatch, now: number): T {
   439→  const next = { ...rec };
   440→  if (patch.eventTime !== undefined) {
   441→    if (patch.eventTime == null) delete next.eventTime;
   442→    else if (Number.isFinite(patch.eventTime)) next.eventTime = patch.eventTime;
   443→  }
   444→  if (patch.expiresAt !== undefined) {
   445→    if (patch.expiresAt == null) delete next.expiresAt;
   446→    else if (Number.isFinite(patch.expiresAt)) next.expiresAt = patch.expiresAt;
   447→  }
   448→  next.timeEditedAt = now;
   449→  next.editedAt = now;
   450→  // 手动把过期时间改到未来或清除 → 救回已过期归档的碎片（用户意图明确：重新有效）
   451→  if (next.expiresAt == null || next.expiresAt > now) delete next.expiredAt;
   452→  return next;
   453→}

   485→export function updateFragment(contactId: string, id: string, content: string, weight?: MemWeight): boolean {
   486→  const list = readFragments(contactId);
   487→  const hit = list.find((f) => f.id === id);
   488→  if (!hit) return false;
   489→  hit.content = content.trim();
   490→  hit.editedAt = Date.now();
   491→  if (weight === 'high' || weight === 'normal' || weight === 'low') hit.weight = weight;
   492→  // 用户显式编辑保存 = 意图明确的「救回」。先记下更新者（supersededBy 指向的新记忆）再清标记：
   493→  // 1) 清被更新标记（supersededAt/supersededBy）让碎片重新参与召回/总结（否则编辑过内容也永不召回，#40）；
   494→  // 2) 清已入核心标记（consumedAt，#15）：已入核心的碎片此前编辑保存仍报「碎片已更新」，但召回
   495→  //    （memRecallBlock）与待总结统计/自动总结（pendingFragmentCount/summarizePendingIntoCore）均按
   496→  //    consumedAt 排除，编辑实际不生效——编辑救回后重新参与召回/总结；核心记忆里的旧总结滞后可接受。
   497→  const successorId = hit.supersededBy;
   498→  delete hit.supersededAt;
   499→  delete hit.supersededBy;
   500→  delete hit.consumedAt;
   501→  // 【#43】救回矛盾链只留一条活跃：A 原被 B 更新（supersededBy=B）→ A 救回后若 B 仍活跃，A/B 会
   502→  // 同时参与召回/总结形成矛盾对 → 反向压制 B（标记 B 被 A 更新），保证矛盾链同一时刻只有一条活跃。
   503→  // 妥协点（只处理一层）：若 B 自身已被 C 更新（B.supersededAt 已置、supersededBy=C），B 本就不参与
   504→  // 召回，保持其指向 C 的既有溯源不动；此时救回的 A 与 C 理论上可并存（链深 ≥2 罕见），由召回块
   505→  // 头部「同一事实新旧矛盾时以时间更近的为准」的注入提示兜底，不递归清理更深层。
   506→  const successor = successorId ? list.find((f) => f.id === successorId && f.id !== id) : undefined;
   507→  if (successor && successor.supersededAt == null) {
   508→    successor.supersededAt = Date.now();
   509→    successor.supersededBy = id;
   510→  }
   511→  writeJSON(fragKey(contactId), list);
   512→  return true;
   513→}

   684→/** 相关性打分：内容与当前上下文的重叠 + 新近加成（长期/核心/碎片各自组内排序用） */
   685→function relevanceScore(content: string, createdAt: number, context: string): number {
   686→  const ctx = contextGramsCache.get(context) ?? bigrams(context);
   687→  contextGramsCache.set(context, ctx);
   688→  if (contextGramsCache.size > 8) contextGramsCache.delete(contextGramsCache.keys().next().value as string);
   689→  let score = overlap(bigrams(content), ctx) * 2;
   690→  const age = Date.now() - createdAt;
   691→  if (age < 86400_000) score += 3;
   692→  else if (age < 7 * 86400_000) score += 2;
   693→  else if (age < 30 * 86400_000) score += 1;
   694→  return score;
   695→}
   696→
   697→function overlap(a: Set<string>, b: Set<string>): number {
   698→  let n = 0;
   699→  a.forEach((g) => {
   700→    if (b.has(g)) n++;
   701→  });
   702→  return n;
   703→}
   704→
   705→/** 上下文文本的 gram 缓存（同一次召回内复用） */
   706→const contextGramsCache = new Map<string, Set<string>>();

   708→/** 碎片召回得分：相关性 × 权重系数（高权重优先）× 淡化系数（淡化中 0.4；已归档直接不参与） */
   709→function fragRecallScore(f: MemFragment, forget: MemForget, context: string, now: number): { s: number; st: ReturnType<typeof fadeState> } {
   710→  const st = fadeState(f as FadeInput, forget, now);
   711→  const s =
   712→    relevanceScore(f.content, f.reinforcedAt ?? f.sourceTime, context) *
   713→    weightFactor(f.weight) *
   714→    (st === 'fading' ? 0.4 : 1);
   715→  return { s, st };
   716→}

   718→/**
   719→ * 召回该联系人（互通开关限定范围）的记忆，组装为注入 system 的文本块。
   720→ * 注入顺序（层级从高到低）：长期记忆 → 核心记忆 → 近期碎片；跨层去重：与已选内容相似（≥0.6）
   721→ * 的低层记忆不再重复注入（长期已收编的由长期代表、核心已收编碎片的由核心代表）。
   722→ * 容量上限：长期 8 条、核心 12 条、碎片 5 条（各按 权重×相关性×淡化 降序截断）；
   723→ * 已入长期（archivedAt）的核心、已入核心（consumedAt）与已归档的碎片不再召回。
   724→ * 该联系人没有记忆 / 互通关闭且当前 App 无来源记忆 → 返回空串（正常聊天，不报错）。
   725→ *
   726→ * 群聊互通（opts）：
   727→ * - mode 'private'（缺省）：跨 App 互通开关照旧；另读群聊来源记忆——碎片/总结按来源群
   728→ *   memoryInterop 判断（互斥于跨 App share 开关，两层开关都开才可见）；
   729→ * - mode 'group'（groupId 必传）：只读当前群来源记忆 + （该群互通开启时）自己的非群聊记忆；
   730→ *   其他群的记忆永不参与（群间隔离），角色隔离由存储键天然保证。
   731→ */
   732→export interface MemRecallOpts {
   733→  mode?: 'private' | 'group';
   734→  /** mode='group' 时必传：当前群聊 ID */
   735→  groupId?: string;
   736→  /** 群互通开关解析（缺省读群数据层的 memoryInterop） */
   737→  interopOn?: (groupId: string) => boolean;
   738→  /** 群名解析（来源标注「群聊·群名」用；缺省读群数据层） */
   739→  groupLabel?: (groupId: string) => string | null;
   740→  /**
   741→   * 小号「机主身份披露」检测文本（仅用户侧发言：当前消息 + 近期用户消息/验证留言）。
   742→   * 用户规则：小号 AI 默认纯陌生人（大号记忆零注入）；只有用户主动亮明身份
   743→   * （说「我是机主」「这是我大号」或报出机主名字等）才解锁大号记忆注入。
   744→   * 解锁通道一：文本命中披露词（含否定句守卫：「我不是机主」不算）；
   745→   * 解锁通道二（规则三→四闭环）：AI 近几轮刚用机主名做过猜测（「你是不是XX？」），
   746→   * 用户本轮整条消息是短确认（「对」「被你猜到了」）——也算亲口承认。
   747→   * 缺省/空 = 不检测 = 保持纯陌生人。
   748→   */
   749→  disclosureText?: string;
   750→  /** 机主（大号身份）显示名（宿主用 cachedOwnerName 现场传入）：披露措辞与名字关键词用 */
   751→  altMainName?: string;
   752→  /**
   753→   * 机主（大号）与该角色的关系标签（宿主传 peer.relation）：小号披露解锁后的「认出」
   754→   * 包装段用——认出后 AI 才知道与 TA 的真实关系（未披露时零读取，规则二.2）。
   755→   * 缺省/空 = 包装段不带关系行（零破坏）。
   756→   */
   757→  altMainRelation?: string;
   758→}

   760→export function memRecallBlock(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts): string {
   761→  setMemScopeForApp(app); // 多账号：按当前聊天 App 的账号取记忆（小号视角=空记忆=陌生人）
   762→  // 顶层容错（与 collectWbBlocks 同款约定）：记忆注入任何意外异常都不阻断消息发送，
   763→  // 返回空块 = 无记忆正常聊天（群聊/私聊同此规则）
   764→  try {
   765→    return memRecallBlockInner(contactId, app, contextText, opts);
   766→  } catch {
   767→    return '';
   768→  }
   769→}
   770→
   771→// ---------------- 小号「机主身份披露」门控（用户最新规则：默认纯陌生人） ----------------
   772→
   773→/**
   774→ * 用户规则（最终口径）：小号场景 AI 完全读不到大号记忆，除非用户主动告诉 AI 自己是谁。
   775→ * - 默认（未亮明身份）：小号/匿名号侧零注入大号记忆（memMainRecallBlockForAlt 返回空串），
   776→ *   AI 是纯陌生人——最多知道人设里「用户还有个别的身份」这层关系，不知道任何共同经历；
   777→ * - 用户在对话/验证消息里主动亮明（说「我是机主」「这是我大号」或报出机主名字等）
   778→ *   → AI「认出」TA：之后该（账号 × 联系人）粘性注入大号记忆块，可自然提起共同经历。
   779→ * 披露标记存 localStorage（召回链路是同步的，IndexedDB kv 异步不可用）；
   780→ * 键经 accLs 账号作用域（删除账号的 `--{id}` 兜底清扫覆盖），删联系人由 purgeChatTracesFor 清扫。
   781→ */
   782→
   783→function lsGetRaw(key: string): string {
   784→  try {
   785→    return window.localStorage.getItem(key) ?? '';
   786→  } catch {
   787→    return '';
   788→  }
   789→}
   790→
   791→function lsSetRaw(key: string, v: string): void {
   792→  try {
   793→    window.localStorage.setItem(key, v);
   794→  } catch {
   795→    // 存储异常静默：门控退化为「每次都按陌生人」
   796→  }
   797→}
   798→
   799→/** 披露标记键（账号作用域）：`mem-alt-disc:{app}:{cid}`（小号自动带 `--{accId}` 后缀） */
   800→function altDisclosedLsKey(app: MemApp, contactId: string): string {
   801→  try {
   802→    return accLs(`mem-alt-disc:${app}:${contactId}`, app);
   803→  } catch {
   804→    return `mem-alt-disc:${app}:${contactId}`;
   805→  }
   806→}
   807→
   808→/**
   809→ * 否定守卫：关键词/名字命中处的前 2 个字符含否定或疑问前缀 → 不算亮明身份。
   810→ *（「我不是机主」「这不是大号」「难道我是陈凡」都不算；「我是机主」「我大号」算）
   811→ */
   812→function altNegatedAt(t: string, idx: number): boolean {
   813→  const head = t.slice(Math.max(0, idx - 2), idx);
   814→  return /[不没别非无]|难道|是不是|何曾|岂/.test(head);
   815→}
   816→
   817→/** 名字句式命中位置扫描：任一出现的「我是X/我就是X/X是我」且非否定/疑问尾 → 命中 */
   818→function altNameClaimAt(t: string, nm: string): boolean {
   819→  for (const pat of [`我是${nm}`, `我就是${nm}`, `${nm}是我`, `${nm}本人`]) {
   820→    let from = 0;
   821→    for (;;) {
   822→      const i = t.indexOf(pat, from);
   823→      if (i === -1) break;
   824→      // 「我是陈凡吗？」= 疑问不是承认；否定前缀（不/没/难道…）同样排除
   825→      const tail = t.slice(i + pat.length, i + pat.length + 2);
   826→      if (!/^(吗|么|嘛)/.test(tail) && !altNegatedAt(t, i)) return true;
   827→      from = i + 1;
   828→    }
   829→  }
   830→  return false;
   831→}
   832→
   833→/**
   834→ * 披露信号句式表（收紧版）：裸出现「机主/大号/小号」不再解锁大号记忆——
   835→ * 必须落在第一人称声明/所有格句式里（「我是机主」「我的小号」「小号是我」等）。
   836→ * 「别人都用小号」「大号好累啊」这类旁观/感叹句不会误解锁（与主号侧第一人称正则口径对齐）。
   837→ */
   838→const ALT_DISCLOSE_PATTERNS: readonly string[] = [
   839→  // 机主身份声明
   840→  '我是机主',
   841→  '我就是机主',
   842→  '机主是我',
   843→  '机主就是我',
   844→  // 大号所有格 / 声明
   845→  '我的大号',
   846→  '我大号',
   847→  '是我大号',
   848→  '大号是我',
   849→  '大号就是我',
   850→  '这是我大号',
   851→  '这是我的大号',
   852→  '大号是我本人',
   853→  '我就是大号',
   854→  // 小号所有格 / 声明
   855→  '我的小号',
   856→  '我小号',
   857→  '是我小号',
   858→  '小号是我',
   859→  '小号就是我',
   860→  '这是我小号',
   861→  '这是我的小号',
   862→  '小号是我本人',
   863→  '我就是小号',
   864→];
   865→
   866→function altDiscloseHit(text: string, mainName: string): boolean {
   867→  const t = (text ?? '').trim();
   868→  if (!t) return false;
   869→  for (const pat of ALT_DISCLOSE_PATTERNS) {
   870→    let from = 0;
   871→    for (;;) {
   872→      const i = t.indexOf(pat, from);
   873→      if (i === -1) break;
   874→      if (!altNegatedAt(t, i)) return true;
   875→      from = i + 1;
   876→    }
   877→  }
   878→  const nm = mainName.trim();
   879→  if (nm && nm !== '机主' && altNameClaimAt(t, nm)) return true;
   880→  return false;
   881→}
   882→
   883→/**
   884→ * 用户短确认（规则三→四闭环）：AI 刚用机主名猜过（「你是不是陈凡？」），
   885→ * 用户回一句很短的肯定（「对」「是我」「没错」「哈哈被你猜到了」）= 亲口承认。
   886→ * 整条消息 ≤12 字且整句匹配肯定句式——普通闲聊里的「对啊怎么了」不会整句命中。
   887→ */
   888→const ALT_AFFIRM_RE =
   889→  /^(?:哈哈+|嘻嘻|嘿嘿|嘿|哎呀|额|em+|类)*(?:对的?|是呀?|是啊|是滴|是我[呀啊啦]?|就是我[呀啊啦]?|没错|没错就是我|被你猜到了?|被你发现了?|被你识破了?|猜对了?|猜对啦|猜中了?|答对了?|对头|对哒|嗯呢|嗯啊|bingo|yes|yeah)[啦哟哦呗哈呀啊嘛滴了恩~～!！。.？?]*$/i;
   890→
   891→function altShortAffirmHit(text: string): boolean {
   892→  const t = (text ?? '').trim();
   893→  if (!t || t.length > 12) return false;
   894→  return ALT_AFFIRM_RE.test(t);
   895→}
   896→
   897→/**
   898→ * AI 近几轮是否刚用机主名做过身份猜测（扫当前账号会话里 AI 侧发言，只读不写）：
   899→ * 命中条件 = 提到机主名 + 猜测语气（「猜/是不是/认出/难道」）。
   900→ * 给「AI 猜测 → 用户短确认」解锁链路用；读不到会话按没猜过（保守，不解锁）。
   901→ */
   902→function altAiNameGuessRecent(app: MemApp, contactId: string, mainName: string): boolean {
   903→  const nm = mainName.trim();
   904→  if (!nm || nm === '机主') return false;
   905→  try {
   906→    const turns = memRecentConvo(contactId, app)
   907→      .filter((t) => t.role === 'peer')
   908→      .slice(-4);
   909→    return turns.some((t) => t.text.includes(nm) && /猜|是不是|认出|难道/.test(t.text));
   910→  } catch {
   911→    return false;
   912→  }
   913→}
   914→
   915→/**
   916→ * 扫描披露文本：命中→落粘性标记并返回 true；已标记过直接 true；否则 false（保持纯陌生人）。
   917→ * 两条解锁通道：
   918→ * ① 直接亮明：披露词命中（含否定守卫）；
   919→ * ② 猜后确认（规则三→四闭环）：当前消息（首段；短线程下任一段）是短肯定句
   920→ *    + AI 近几轮刚用机主名猜过——用户对「你是不是陈凡？」回「对」也算亲口承认。
   921→ */
   922→function altDiscloseScan(app: MemApp, contactId: string, disclosureText: string | undefined, mainName: string): boolean {
   923→  const key = altDisclosedLsKey(app, contactId);
   924→  if (lsGetRaw(key) === '1') return true;
   925→  if (!disclosureText || !disclosureText.trim()) return false;
   926→  const t = disclosureText.trim();
   927→  let hit = altDiscloseHit(t, mainName);
   928→  if (!hit) {
   929→    // 肯定段判定：首段=当前消息（聊天调用方把当前消息放最前）；整段 ≤20 字时退化为任一段
   930→    //（好友验证线程调用方按时间正序拼接，短线程下最后一条回复也该参与）
   931→    const segs = t.split(/\s+/).filter(Boolean);
   932→    const affirm = (segs.length > 0 && altShortAffirmHit(segs[0])) || (t.length <= 20 && segs.some((seg) => altShortAffirmHit(seg)));
   933→    if (affirm && altAiNameGuessRecent(app, contactId, mainName)) hit = true;
   934→  }
   935→  if (!hit) return false;
   936→  lsSetRaw(key, '1');
   937→  return true;
   938→}
   939→
   940→/**
   941→ * 只读查询：该（账号 × 联系人）是否已亮明机主身份（粘性标记；不扫描文本）。
   942→ * 供好友决策（friend-state.decideCharFriendReq）在记忆召回后读取，切换「现实处境」行措辞。
   943→ * 内部先 setMemScopeForApp(app) 保证作用域就位（重复设置无害）。
   944→ */
   945→export function isAltMainDisclosed(app: MemApp, contactId: string): boolean {
   946→  try {
   947→    setMemScopeForApp(app);
   948→  } catch {
   949→    // 读不到账号按未披露
   950→  }
   951→  return lsGetRaw(altDisclosedLsKey(app, contactId)) === '1';
   952→}
   953→
   954→/**
   955→ * 多账号规则（用户最新口径）——小号侧「身份披露后」的大号记忆块：
   956→ *
   957→ * 小号/匿名号聊天时本账号记忆通常为空（AI 表现成纯陌生人），且【默认完全不注入】大号记忆；
   958→ * 只有用户亮明「机主」身份后才注入本块——两条通道：① 用户在对话里主动亮明（altDiscloseScan
   959→ * 命中并落粘性标记）；② 用户此前在大号侧亲口告知过「这个号是我的小号」（账号关联记忆，
   960→ * memLinkAltAccounts 预写标记/直接命中）。AI「认出」当前用户就是机主，可以自然提起共同经历。
   961→ *
   962→ * 边界：
   963→ * - 大号侧 → 返回空串（零注入，行为与旧版完全一致）；
   964→ * - 匿名号 → 返回空串（匿名聊天保持匿名：AI 不知道对方是谁，注入大号记忆会破坏匿名语义）；
   965→ * - 未披露且无账号关联 → 返回空串（纯陌生人：AI 读不到大号任何记忆）；
   966→ * - 已披露（本侧亮明或大号侧告知过账号关联）→ 认出块恒注入（含身份事实；大号有记忆时附记忆块，
   967→ *   无记忆时明确「身份已确认、共同经历暂无印象不编造」——用户规则一：小号侧问「我是谁」必须答得上来）；
   968→ * - 召回范围/层级/去重与 memRecallBlock 完全同款（互通开关、群来源可见性照旧生效）。
   969→ */
   970→export function memMainRecallBlockForAlt(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts): string {
   971→  let accId = '';
   972→  try {
   973→    accId = getActiveAccountIdFor(app);
   974→  } catch {
   975→    return '';
   976→  }
   977→  if (!accId || accId === MAIN_ACCOUNT_ID) return ''; // 大号侧零注入
   978→  try {
   979→    if (getAccountById(accId)?.kind === 'anon') return ''; // 匿名号保持匿名
   980→  } catch {
   981→    // 注册表读不到按小号处理（继续走披露门控）
   982→  }
   983→  const mainLabel = opts?.altMainName?.trim() || '机主';
   984→  // 【账号关联（用户规则一/二）】用户此前在大号侧亲口告知过 AI「这个号是我的小号」→ 小号侧自动「认出」，
   985→  // 无需在小号侧再亮明一次（memLinkAltAccounts 已预写粘性标记，altDiscloseScan 首行即命中，这里查关联记录双保险）
   986→  const linked = memAltLinked(contactId, accId);
   987→  // 【用户规则】默认纯陌生人：用户没亮明身份且没有账号关联 = 完全不注入大号记忆
   988→  if (!linked && !altDiscloseScan(app, contactId, opts?.disclosureText, mainLabel)) return '';
   989→  const accLabel = (() => {
   990→    try {
   991→      return getAccountById(accId)?.name?.trim() || '这个号';
   992→    } catch {
   993→      return '这个号';
   994→    }
   995→  })();
   996→  const savedSuffix = memScopeSuffix;
   997→  const savedName = memScopeName;
   998→  memScopeSuffix = ''; // 临时切到【大号】记忆作用域召回
   999→  memScopeName = '机主';
  1000→  let block = '';
  1001→  try {
  1002→    block = memRecallBlockInner(contactId, app, contextText, opts, {
  1003→      identitySegment: `现在和你聊天的这位用户就是「${mainLabel}」本人（TA 的常用身份）`,
  1004→    });
  1005→  } catch {
  1006→    block = '';
  1007→  } finally {
  1008→    memScopeSuffix = savedSuffix; // 恢复原作用域（召回是同步操作，立即还原防泄漏到后续写入）
  1009→    memScopeName = savedName;
  1010→  }
  1011→  // 认出块不再要求大号记忆非空（用户规则一：大号告知过关联后，小号侧问「我是谁」必须答得上来——
  1012→  // 哪怕大号侧还没有任何记忆条目，身份事实本身也要注入）
  1013→  const rel = opts?.altMainRelation?.trim() || '';
  1014→  return [
  1015→    `【你认出来了：当前和你聊天的就是「${mainLabel}」本人】`,
  1016→    // 认出来源二选一：用户在大号侧告知过「这个号是 TA 的」（账号关联）/ 用户本侧亲口亮明
  1017→    ...(linked
  1018→      ? [
  1019→          `- 用户之前亲口告诉过你：「${accLabel}」这个号也是 TA 的（TA 的小号）——TA 现在用这个小号来找你，所以你认出 TA 了；`,
  1020→        ]
  1021→      : [
  1022→          `- 当前用户已经亲口承认 TA 就是「${mainLabel}」（自己亮明身份或承认了你的猜测）——现在你认出 TA 了；`,
  1023→        ]),
  1024→    ...(rel ? [`- 你和「${mainLabel}」的关系：${rel}——认出后你可以按这份真实关系自然相处；`] : []),
  1025→    // 用户规则二：大小号是同一个人——绝不能说「你俩」
  1026→    `- 「${mainLabel}」和当前和你聊天的这位是同一个人（同一个人换了个号来找你），不是两个人：绝不能说「你俩」「你们两个」，也不要把 TA 和「${mainLabel}」当成两个人并列比较或一起提起；`,
  1027→    '- 可以像老朋友重逢一样自然反应（惊喜、嗔怪「怎么换个号来找我」「又是你」都可以，按你的人设来）；',
  1028→    '- 不要说「大号/小号」这类词，就像认出老朋友一样自然；',
  1029→    ...(block
  1030→      ? [`- 下面是你们（你与「${mainLabel}」）之间真实经历过的事，现在可以自然提起：`, block]
  1031→      : ['- 你们之间具体经历过的事你暂时没有印象（不要编造共同回忆），但身份本身你已经确认了；']),
  1032→  ].join('\n');
  1033→}
  1034→
  1035→/**
  1036→ * 小号侧未披露时的「陌生但隐约熟悉」规则块 = 陌生人边界 + 身份猜测（用户规则三/五）：
  1037→ *
  1038→ * 边界（E2E 发现的补丁）：AI 会顺着用户话题装作认识——用户提「团子」它就编
  1039→ * 「上次它把我的拖鞋藏起来」。给 AI 立规矩：不了解的往事不装记得、不编细节。
  1040→ * 猜测（本次新增）：TA 的说话语气、用词习惯、互动方式、称呼可能让你隐约觉得熟悉，

  1147→ * 大号侧披露检测（用户规则一）：主号聊天用户发言里出现第一人称归属表述——
  1148→ * 「我的小号」「我小号」「小号是我」「是我小号」等（纯提及「小号」二字不算，防误报；
  1149→ * 否定守卫：「不是我的小号」「难道是小号是我」不算）。
  1150→ * 返回应关联的小号账号 id：用户点名了小号名（注册表名）时只关联被点名的，泛指「我的小号」关联全部小号。
  1151→ * 仅限主号侧调用（调用方保证当前账号为 main）；未命中返回 []。
  1152→ */
  1153→export function memDetectMainAltDisclosure(text: string): string[] {
  1154→  const t = (text ?? '').trim();
  1155→  if (!t) return [];
  1156→  const re = /我(的)?小号|小号(就)?是我|是我(的)?小号|小号是我本人/g;
  1157→  let disclosed = false;
  1158→  for (let m = re.exec(t); m; m = re.exec(t)) {
  1159→    // 否定守卫：命中处前 2 字含否定/疑问前缀不算（「不是我的小号」「难道是我的小号」）
  1160→    const head = t.slice(Math.max(0, m.index - 2), m.index);
  1161→    if (!altNegatedAt(t, m.index) && !/[不没别非无]|难道|何曾|岂/.test(head)) {
  1162→      disclosed = true;
  1163→      break;
  1164→    }
  1165→  }
  1166→  re.lastIndex = 0;
  1167→  if (!disclosed) return [];
  1168→  let alts: { id: string; name: string }[] = [];
  1169→  try {
  1170→    alts = getAccounts()
  1171→      .filter((a) => a && a.kind === 'alt')
  1172→      .map((a) => ({ id: a.id, name: (a.name ?? '').trim() }));
  1173→  } catch {
  1174→    return [];
  1175→  }
  1176→  if (alts.length === 0) return [];
  1177→  const named = alts.filter((a) => a.name && t.includes(a.name));
  1178→  return (named.length > 0 ? named : alts).map((a) => a.id);
  1179→}

  1200→interface RecallHeaderOpts {

  1207→function memRecallBlockInner(contactId: string, app: MemApp, contextText: string, opts?: MemRecallOpts, headerOpts?: RecallHeaderOpts): string {
  1208→  if (!contactId) return '';
  1209→  memSweepExpiry(contactId);
  1210→  const { share, forget } = getMemSettings(contactId);
  1211→  const now = Date.now();
  1212→  const isGroupMode = opts?.mode === 'group' && !!opts.groupId;
  1213→  const curGroup = opts?.groupId ?? '';
  1214→  const interopOn = opts?.interopOn ?? defaultGroupInteropOn;
  1215→  const groupLabelOf = opts?.groupLabel ?? defaultGroupLabel;
  1216→  /** 群来源可见性（私聊侧）：互通开关 + 自己仍在该群成员表（被移出群聊后不能再读该群的记忆）。
  1217→   *  纯群来源总结：任一来源群同时满足才可见；碎片：来源群满足才可见。 */
  1218→  const groupMemVisible = (gid: string | null | undefined): boolean =>
  1219→    !!gid && interopOn(gid) && defaultGroupMemberVisible(gid, contactId);
  1220→  /** 核心/长期层级的群来源可见性：纯私聊总结恒可见（旧数据兼容）；带群来源的按各群互通判断 */
  1221→  const summaryVisiblePrivate = (m: Pick<MemCore, 'groupIds' | 'privateSource'>): boolean => {
  1222→    const gids = m.groupIds ?? [];
  1223→    if (gids.length === 0) return true; // 纯私聊/朋友圈来源（含旧数据）
  1224→    if (m.privateSource !== false) return true; // 混合来源（含私聊部分）：私聊恒可见
  1225→    return gids.some((g) => groupMemVisible(g)); // 纯群聊来源：任一来源群互通开启才可见
  1226→  };
  1227→  /** 群聊模式下的总结可见性：本群来源恒可见；纯私聊总结在互通开时可见；其他群的总结永不可见（群间隔离） */
  1228→  const summaryVisibleInGroup = (m: Pick<MemCore, 'groupIds' | 'privateSource'>): boolean => {
  1229→    const gids = m.groupIds ?? [];
  1230→    if (gids.includes(curGroup)) return true;
  1231→    if (gids.some((g) => g !== curGroup)) return false; // 沾了其他群来源：不进本群上下文
  1232→    return interopOn(curGroup) && m.privateSource !== false;
  1233→  };
  1234→  /** 碎片级可见性（来源精确到条） */
  1235→  const fragVisiblePrivate = (f: MemFragment): boolean => {
  1236→    if (f.source !== 'group') return true;
  1237→    return groupMemVisible(f.sourceGroupId);
  1238→  };
  1239→  const fragVisibleInGroup = (f: MemFragment): boolean => {
  1240→    if (f.source === 'group') return f.sourceGroupId === curGroup; // 只读当前群（群间隔离）
  1241→    return interopOn(curGroup); // 自己的私聊/朋友圈记忆：互通开启才可见
  1242→  };
  1243→  // 长期记忆（顶层画像）：全量注入（安全上限 8 条防失控）；用户手动设置过期的长期不再注入；
  1244→  // 被新记忆更正（supersededAt）的旧画像不再注入（B7：新旧矛盾以新为准）
  1245→  const longs = listLongTerm(contactId)
  1246→    .filter((m) => !m.supersededAt && !isMemExpired(m, now) && (share || m.apps.includes(app)))
  1247→    .filter((m) => (isGroupMode ? summaryVisibleInGroup(m) : summaryVisiblePrivate(m)))
  1248→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, contextText) }))
  1249→    .sort((a, b) => b.s - a.s)
  1250→    .slice(0, 8);
  1251→  // 核心记忆：全量注入（安全上限 12 条）；已被长期收编的不再注入（由长期记忆代表）；
  1252→  // 用户手动设置过期的核心不再注入（默认永不过期不受影响）；被更正的旧核心不再注入
  1253→  const cores = listCores(contactId)
  1254→    .filter((m) => !m.archivedAt && !m.supersededAt && !isMemExpired(m, now) && (share || m.apps.includes(app)))
  1255→    .filter((m) => (isGroupMode ? summaryVisibleInGroup(m) : summaryVisiblePrivate(m)))
  1256→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, contextText) }))
  1257→    .sort((a, b) => b.s - a.s)
  1258→    .slice(0, 12);
  1259→  // 近期记忆碎片：按相关性取前 5 条；已消费的由核心记忆代表，已更新/已过期的不召回；
  1260→  // 本轮已被人设【最近发生的事】(bondNotesFor) 选用的碎片不再重复注入（B8 跨块去重）
  1261→  const frags = listFragments(contactId)
  1262→    .filter((f) => !f.consumedAt && !f.supersededAt && !isMemExpired(f, now) && (share || f.app === app))
  1263→    .filter((f) => !bondExcludeIds.has(f.id)) // 本轮已被人设【最近发生的事】选用的碎片不再重复注入（B8 跨块去重）
  1264→    .filter((f) => !bondExcludeIds.has(f.id))
  1265→    .filter((f) => (isGroupMode ? fragVisibleInGroup(f) : fragVisiblePrivate(f)))
  1266→    .map((f) => ({ f, ...fragRecallScore(f, forget, contextText, now) }))
  1267→    .filter((x) => x.st !== 'faded')
  1268→    .sort((a, b) => b.s - a.s)
  1269→    .slice(0, 5);
  1270→  if (longs.length === 0 && cores.length === 0 && frags.length === 0) return '';
  1271→  // 跨层去重：按注入顺序遍历，与已选内容相似（≥0.6）的条目跳过（信息已由高层代表）
  1272→  const accepted: string[] = [];
  1273→  const isDup = (content: string) => accepted.some((a) => similarity(a, content) >= SIMILAR_MERGE_THRESHOLD);
  1274→  const keepLongs = longs.filter(({ m }) => {
  1275→    if (isDup(m.content)) return false;
  1276→    accepted.push(m.content);
  1277→    return true;
  1278→  });
  1279→  const keepCores = cores.filter(({ m }) => {
  1280→    if (isDup(m.content)) return false;
  1281→    accepted.push(m.content);
  1282→    return true;
  1283→  });
  1284→  const keepFrags = frags.filter(({ f }) => {
  1285→    if (isDup(f.content)) return false;
  1286→    accepted.push(f.content);
  1287→    return true;
  1288→  });
  1289→  if (keepLongs.length === 0 && keepCores.length === 0 && keepFrags.length === 0) return '';
  1290→  // 时间感知：层内按有效时间（事件时间优先，其次加强/来源/创建时间）从新到旧排序
  1291→  keepLongs.sort((a, b) => memEffectiveTime(b.m) - memEffectiveTime(a.m));
  1292→  keepCores.sort((a, b) => memEffectiveTime(b.m) - memEffectiveTime(a.m));
  1293→  keepFrags.sort((a, b) => memEffectiveTime(b.f) - memEffectiveTime(a.f));
  1294→  // 头部说明按互通状态分文案（跨 App 互通修复）：角色必须被告知「多端同一个人」以及
  1295→  // 本次能看到哪些来源的记忆——否则带「·微信」标注的碎片会被当成无关信息，甚至否认在别处的对话
  1296→  const headScope = isGroupMode
  1297→    ? '群聊模式：下面只给本群可见的共同经历'
  1298→    : share
  1299→      ? '下面是你在微信、QQ、信息、电话里和TA亲身经历过的事，带App标注的条目来自对应那一端'
  1300→      : '仅本App的经历（跨应用互通已关闭）：你在其他App和TA聊过的内容这里看不到';
  1301→  // 归属标注说明（朋友圈/QQ动态来源记忆自带「（AI角色本人）/（用户本人）」标注）：
  1302→  // 只要注入内容里出现这类标注，就附一行解释——AI 必须能区分「是我自己发的动态」还是「用户发的」，
  1303→  // 绝不能把自己（AI 本人）发的动态当成用户的行为（反之亦然）。
  1304→  // 【fix3-a 8】判定改为「显式 hasRoleTag 标记 || 字符串检测」：总结可能把标注磨掉，
  1305→  // 总结落库时聚合来源标记（hasRoleTag=true）后，这里仍能靠标记兜底展示说明行
  1306→  const showRoleTagNote =
  1307→    [...keepLongs, ...keepCores].some(({ m }) => m.hasRoleTag === true || memHasRoleTag(m.content)) ||
  1308→    keepFrags.some(({ f }) => f.hasRoleTag === true || memHasRoleTag(f.content));
  1309→  const identitySeg = headerOpts?.identitySegment ?? `现在和你聊天的人用的是：${memScopeAccountName()}`;
  1310→  const lines: string[] = [
  1311→    `【共同经历（${headScope}；${identitySeg}；当前时间：${memNowLabel(now)}；聊天时自然运用：这些都是你自己经历过、记得的事，不要逐条罗列复述，也不要说「记录」「记忆库」「看过资料」这类话）】`,
  1312→    '（长期记忆/核心记忆是稳定的画像，记忆碎片是最新进展：同一件事新旧不一致时，通常按更新的理解——人是在变化的；拿旧印象纠正对方刚说的话之前，先想想对方是不是已经变了）',
  1313→    ...(headerOpts?.extraNoteLines ?? []),
  1314→    ...(showRoleTagNote
  1315→      ? [

  1322→  const appsSrcLabel = (apps: MemApp[]): string => (apps.length > 0 ? apps.map((a) => MEM_APP_LABEL[a] ?? a).join('/') : '');

  1324→  // 记忆块可膨胀到 8k+ 字符；6000 字符是最后一道闸，裁剪顺序 碎片→核心→长期（越稳定越保留）

  1365→  }
  1366→  if (usedFrag.length > 0) {
  1367→    lines.push('◇ 近期记忆碎片：', ...usedFrag);
  1368→  }
  1369→    return lines.join('\n');
  1370→}

  1372→/** 召回预览（设置页）：与 memRecallBlock 同一套层级与排序，返回排序后的记忆对象 */
  1373→export function memRecallPreview(contactId: string): { longs: MemLongTerm[]; cores: MemCore[]; frags: MemFragment[] } {
  1374→  const { forget } = getMemSettings(contactId);
  1375→  const now = Date.now();
  1376→  const longs = listLongTerm(contactId)
  1377→    .filter((m) => !isMemExpired(m, now))
  1378→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, '') }))
  1379→    .sort((a, b) => b.s - a.s)
  1380→    .slice(0, 3)
  1381→    .map((x) => x.m);
  1382→  const cores = listCores(contactId)
  1383→    .filter((m) => !m.archivedAt && !isMemExpired(m, now))
  1384→    .map((m) => ({ m, s: relevanceScore(m.content, m.createdAt, '') }))
  1385→    .sort((a, b) => b.s - a.s)
  1386→    .slice(0, 3)
  1387→    .map((x) => x.m);
  1388→  const frags = listFragments(contactId)
  1389→    .filter((f) => !f.consumedAt && !f.supersededAt && !isMemExpired(f, now))
  1390→    .map((f) => ({ f, ...fragRecallScore(f, forget, '', now) }))
  1391→    .filter((x) => x.st !== 'faded')
  1392→    .sort((a, b) => b.s - a.s)
  1393→    .slice(0, 5)
  1394→    .map((x) => x.f);
  1395→  return { longs, cores, frags };
  1396→}

  1653→function parseMemTime(v: unknown, now: number): number | null {
  1654→  if (typeof v !== 'string') return null;
  1655→  const s = v.trim();
  1656→  if (!s || s === 'null' || s === 'undefined') return null;
  1657→  let d: Date | null = null;
  1658→  const md = /^(\d{4})-(\d{1,2})-(\d{1,2})$/.exec(s);
  1659→  const mdt = /^(\d{4})-(\d{1,2})-(\d{1,2})[ T](\d{1,2}):(\d{2})(?::\d{2})?$/.exec(s);
  1660→  // 无时区字符串统一按北京时间（UTC+8）解析 —— 与提取锚点（Asia/Shanghai）、注入标签（bjParts）
  1661→  // 一致，与设备时区无关；带时区/ISO 带偏移的字符串走 Date.parse 原语义
  1662→  if (md) d = new Date(Date.UTC(Number(md[1]), Number(md[2]) - 1, Number(md[3])) - 8 * 3_600_000);
  1663→  else if (mdt)
  1664→    d = new Date(
  1665→      Date.UTC(Number(mdt[1]), Number(mdt[2]) - 1, Number(mdt[3]), Number(mdt[4]), Number(mdt[5])) - 8 * 3_600_000
  1666→    );
  1667→  else {
  1668→    const t = Date.parse(s);
  1669→    if (!Number.isNaN(t)) d = new Date(t);
  1670→  }
  1671→  if (!d) return null;
  1672→  const t = d.getTime();
  1673→  const span = MEM_TIME_RANGE_YEARS * 366 * 86_400_000;
  1674→  if (t < now - span || t > now + span) return null;
  1675→  return t;
  1676→}

  1678→/** 把 extract 接口返回的碎片归一化（字符串/对象混合兼容 + 非法值兜底自动分类 + 时间解析/防幻觉） */
  1679→function normalizeExtract(res: ExtractApiResult): ExtractItem[] {
  1680→  const raw = Array.isArray(res.fragments) ? res.fragments : [];
  1681→  const now = Date.now();
  1682→  const out: ExtractItem[] = [];
  1683→  for (const x of raw) {
  1684→    if (typeof x === 'string') {
  1685→      const t = x.trim();
  1686→      if (t) out.push({ text: t, weight: autoWeight(t) });
  1687→    } else if (x && typeof x === 'object' && typeof x.text === 'string' && x.text.trim()) {
  1688→      const o = x as { text: string; weight?: unknown; eventTime?: unknown; expiresAt?: unknown; supersedes?: unknown };
  1689→      const item: ExtractItem = { text: o.text.trim(), weight: normalizeWeight(o.weight, o.text) };
  1690→      const et = parseMemTime(o.eventTime, now);
  1691→      const ex = parseMemTime(o.expiresAt, now);
  1692→      // 双保险：过期时间不得早于事件时间（服务端已校验一次，这里兜底防历史残留/异常数据）
  1693→      if (et != null) item.eventTime = et;
  1694→      if (ex != null && (et == null || ex >= et)) item.expiresAt = ex;
  1695→      if (Array.isArray(o.supersedes)) {
  1696→        const ids = o.supersedes
  1697→          .filter((id): id is string => typeof id === 'string' && id.trim().length > 0)
  1698→          .map((id) => id.trim())
  1699→          .slice(0, 6);
  1700→        if (ids.length > 0) item.supersedes = ids;
  1701→      }
  1702→      out.push(item);
  1703→    }
  1704→  }
  1705→  return out.slice(0, 6);
  1706→}

  1708→async function callMemoryApi<T extends ExtractApiResult & SummarizeApiResult>(
  1709→  op: 'extract' | 'summarize',
  1710→  body: Record<string, unknown>,
  1711→  apiConfig: ApiConfig
  1712→): Promise<T> {
  1713→  const res = await fetch(`/api/memory/${op}`, {
  1714→    method: 'POST',
  1715→    headers: { 'Content-Type': 'application/json' },
  1716→    body: JSON.stringify({ ...body, config: apiConfig }),
  1717→    // 上游挂起时 75s 超时强制 abort，否则 inflight Set 永不释放
  1718→    signal: AbortSignal.timeout(75_000),
  1719→  });
  1720→  const data = (await res.json().catch(() => ({}))) as T;
  1721→  if (!res.ok) throw new Error(data?.error || `请求失败（${res.status}）`);
  1722→  return data;
  1723→}
  1724→
  1725→/** 归一化文本（去空白）——精确去重键 */
  1726→const normText = (s: string) => s.replace(/\s+/g, '');
  1727→
  1728→/** 碎片容量软上限（按联系人）：appendFragments 写入后超出即择劣淘汰，防止只增不减无限膨胀 */
  1729→const MEM_FRAGMENT_CAP = 500;
  1730→
  1731→/** 权重淘汰序（低权重先淘汰）：high=2 / normal=1 / low=0，与 memory-core 的 higherWeight 同口径 */
  1732→function weightEvictRank(w?: MemWeight): number {
  1733→  return w === 'high' ? 2 : w === 'low' ? 0 : 1;
  1734→}
  1735→
  1736→/**
  1737→ * 碎片软上限淘汰：超出 MEM_FRAGMENT_CAP 时按「已消费（consumedAt 非空，召回已由核心/长期代表）
  1738→ * → 权重低 → createdAt 最旧」优先淘汰，直到回到上限内。
  1739→ * 幂等：排序键全为静态字段且比较器全序（同序保持原相对顺序），淘汰后条数回到上限内、
  1740→ * 下次调用不再触发；去重逻辑不受影响（淘汰条目随 seen 集合消失，同内容再提取会按全新记忆重建）。
  1741→ * batchIds = 本批新增（保护不淘汰：刚提取的记忆是最新信号，且需与返回值 added 保持一致）；
  1742→ * 兜底：非本批候选不足时（极端批量）从本批按同序补足，保证上限始终成立。
  1743→ */
  1744→function capFragments(next: MemFragment[], batchIds: Set<string>): MemFragment[] {
  1745→  if (next.length <= MEM_FRAGMENT_CAP) return next;
  1746→  const overflow = next.length - MEM_FRAGMENT_CAP;
  1747→  const byEvictOrder = (a: MemFragment, b: MemFragment): number =>
  1748→    Number(Boolean(a.consumedAt)) - Number(Boolean(b.consumedAt)) ||
  1749→    weightEvictRank(a.weight) - weightEvictRank(b.weight) ||
  1750→    a.createdAt - b.createdAt;
  1751→  const evict = new Set<string>();
  1752→  const pick = (list: MemFragment[]) => {
  1753→    for (const f of list) {
  1754→      if (evict.size >= overflow) return;
  1755→      evict.add(f.id);
  1756→    }
  1757→  };
  1758→  pick(next.filter((f) => !batchIds.has(f.id)).sort(byEvictOrder));
  1759→  if (evict.size < overflow) pick(next.filter((f) => batchIds.has(f.id)).sort(byEvictOrder));
  1760→  return evict.size > 0 ? next.filter((f) => !evict.has(f.id)) : next;
  1761→}
  1762→
  1763→/**
  1764→ * 追加碎片（矛盾更新优先，其次三层去重）：
  1765→ * 0) 矛盾更新（模型标注 supersedes 且目标存在）：旧记忆标记 supersededAt/supersededBy（「已更新」，
  1766→ *    不再参与召回/总结），本条直接新增（事件/过期时间一并入库）；矛盾条目不走相似合并
  1767→ *    （「不吃辣」绝不能被合并进「爱吃辣」）；
  1768→ * 1) 精确重复 → 记一次「加强」（reinforcedAt/次数刷新），不新增；
  1769→ * 2) 相似（2-gram ≥0.6，同一事实的不同说法）→ 合并进已有条目：内容取更完整的、权重取更高、
  1770→ *    加强计时刷新、计数 +1，不新增占位（已更新/已过期/已消费的旧记忆不作为合并目标）；
  1771→ * 3) 全新 → 新建（权重：LLM 判定优先，缺省按关键词自动分类）。
  1772→ * 返回 { added: 新增条目; merged: 合并/加强次数; superseded: 矛盾更新条数 }
  1773→ */
  1774→function appendFragments(
  1775→  contactId: string,
  1776→  app: MemApp,
  1777→  items: ExtractItem[],
  1778→  sourceTime: number,
  1779→  sourceMsgId?: string,
  1780→  /** 额外字段（动态/群聊来源记忆用：source/sourcePostId/sourceKind/sourceCommentId/sourceGroupId/groupMembers/sourceTag 直通新碎片） */
  1781→  extra?: Partial<Pick<MemFragment, 'source' | 'sourcePostId' | 'sourceKind' | 'sourceCommentId' | 'sourceGroupId' | 'groupMembers' | 'sourceTag'>>
  1782→): { added: MemFragment[]; merged: number; superseded: number } {
  1783→  const list = readFragments(contactId);
  1784→  const now = Date.now();
  1785→  const seen = new Set(list.filter((f) => !f.supersededAt && !isMemExpired(f, now)).map((f) => normText(f.content)));
  1786→  const added: MemFragment[] = [];
  1787→  let merged = 0;
  1788→  let superseded = 0;
  1789→  const newItem = (content: string, item: ExtractItem, id?: string): MemFragment => ({
  1790→    id: id ?? uid(),
  1791→    contactId,
  1792→    app,
  1793→    content,
  1794→    sourceTime,
  1795→    createdAt: now,
  1796→    eventTime: item.eventTime ?? undefined,
  1797→    expiresAt: item.expiresAt ?? undefined,
  1798→    weight: item.weight ?? autoWeight(content),
  1799→    reinforcedAt: now,
  1800→    reinforceCount: 0,
  1801→    sourceMsgId,
  1802→    ...(extra ?? {}),
  1803→    // 【fix3-a 8】内容自带角色归属标注（（AI角色本人）/（用户本人））时显式打标：
  1804→    // 总结聚合与召回兜底判定都依赖它（旧数据无字段时回退字符串检测）
  1805→    ...(memHasRoleTag(content) ? { hasRoleTag: true } : {}),
  1806→  });
  1807→  for (const item of items) {
  1808→    const content = item.text.trim();
  1809→    if (!content) continue;
  1810→    const key = normText(content);
  1811→    // 0) 矛盾更新：标记被更正的旧记忆为「已更新」，本条直接新增。
  1812→    // 目标三层都支持：碎片（裸 id）/ 核心（core:前缀）/ 长期（long:前缀）——
  1813→    // 核心/长期被更正后落 supersededAt，召回侧不再注入（B7：旧画像可被新事实更正）
  1814→    const supIds = item.supersedes ?? [];
  1815→    const fragTargets = supIds
  1816→      .map((id) => list.find((f) => f.id === id))
  1817→      .filter((f): f is MemFragment => f != null && f.supersededAt == null);
  1818→    const coreIds = supIds.filter((id) => id.startsWith('core:')).map((id) => id.slice(5));
  1819→    const longIds = supIds.filter((id) => id.startsWith('long:')).map((id) => id.slice(5));
  1820→    let coreChanged = false;
  1821→    let longChanged = false;
  1822→    if (coreIds.length > 0) {
  1823→      const cores = readCores(contactId);
  1824→      for (const m of cores) {
  1825→        if (coreIds.includes(m.id) && m.supersededAt == null) {
  1826→          m.supersededAt = now;
  1827→          m.supersededBy = 'pending';
  1828→          coreChanged = true;
  1829→        }
  1830→      }
  1831→      if (coreChanged) writeJSON(coreKey(contactId), cores);
  1832→    }
  1833→    if (longIds.length > 0) {
  1834→      const longs = readLongTerm(contactId);
  1835→      for (const m of longs) {
  1836→        if (longIds.includes(m.id) && m.supersededAt == null) {
  1837→          m.supersededAt = now;
  1838→          m.supersededBy = 'pending';
  1839→          longChanged = true;
  1840→        }
  1841→      }
  1842→      if (longChanged) writeJSON(longKey(contactId), longs);
  1843→    }
  1844→    const targets = fragTargets;
  1845→    if (targets.length > 0 || coreChanged || longChanged) {
  1846→      const newId = uid();
  1847→      for (const t of targets) {
  1848→        t.supersededAt = now;
  1849→        t.supersededBy = newId;
  1850→      }
  1851→      // 回填核心/长期的 supersededBy（上面先落 pending 占位，拿到新碎片 id 后指到真身）
  1852→      if (coreChanged) {
  1853→        const cores = readCores(contactId);
  1854→        for (const m of cores) if (m.supersededBy === 'pending') m.supersededBy = newId;
  1855→        writeJSON(coreKey(contactId), cores);
  1856→      }
  1857→      if (longChanged) {
  1858→        const longs = readLongTerm(contactId);
  1859→        for (const m of longs) if (m.supersededBy === 'pending') m.supersededBy = newId;
  1860→        writeJSON(longKey(contactId), longs);
  1861→      }
  1862→      superseded += targets.length + (coreChanged ? 1 : 0) + (longChanged ? 1 : 0);
  1863→      seen.add(key);
  1864→      added.push(newItem(content, item, newId));
  1865→      continue;
  1866→    }
  1867→    // 1) 精确重复 → 加强已有记忆（已更新/已过期的旧条目不是目标）
  1868→    if (seen.has(key)) {
  1869→      // seen 命中后既要查原始 list，也要查本批次已新增的 added
  1870→      // （否则同批次模型返回两条相同碎片时，第二条穿透到下方「全新」分支重复入库）
  1871→      const hit = [...list, ...added].find(
  1872→        (f) => normText(f.content) === key && !f.supersededAt && !isMemExpired(f, now)
  1873→      );
  1874→      if (hit) {
  1875→        hit.reinforcedAt = now;
  1876→        hit.reinforceCount = (hit.reinforceCount ?? 0) + 1;
  1877→        hit.weight = higherWeight(hit.weight, item.weight);
  1878→        // 【fix3-a 8】加强时角色归属标注 OR 保留（原条已标 或 本次内容带标注都保留）
  1879→        if (hit.hasRoleTag === true || memHasRoleTag(content)) hit.hasRoleTag = true;
  1880→        // 时间感知：未手动编辑过的记忆，重复提及且本次提取带新时间时刷新（事件时间保鲜）；
  1881→        // 用户手动设置过（timeEditedAt）的一律以用户为准，绝不覆盖
  1882→        if (!hit.timeEditedAt) {
  1883→          if (item.eventTime != null) hit.eventTime = item.eventTime;
  1884→          if (item.expiresAt != null) hit.expiresAt = item.expiresAt;
  1885→        }
  1886→        merged++;
  1887→        continue;
  1888→      }
  1889→      // 命中的只是已更新/已过期的旧条目 → 落到下面按新增处理
  1890→    }
  1891→    // 2) 相似说法 → 合并进已有记忆（优先未消费的；内容保留更完整的一条）
  1892→    // 相似判定同样需要把本批次 added 并入查找范围，否则本批次同义碎片穿透到「全新」分支
  1893→    // 同源事件标签（sourceTag）非空且两侧相同时跳过相似合并：同模板的不同事件（如每笔亲属卡消费）
  1894→    // 互不吞并，只靠步骤 1 的精确重复强化；与无标签/异标签碎片的合并行为不变
  1895→    const inTag = (extra?.sourceTag ?? '') as string;
  1896→    const simHit = [...list, ...added]
  1897→      .filter((f) => !f.supersededAt && !isMemExpired(f, now) && f.content.length >= 4 && content.length >= 4)
  1898→      .filter((f) => !(inTag && f.sourceTag && f.sourceTag === inTag))
  1899→      .sort((a, b) => Number(Boolean(a.consumedAt)) - Number(Boolean(b.consumedAt)) || a.createdAt - b.createdAt)
  1900→      .find((f) => similarity(f.content, content) >= SIMILAR_MERGE_THRESHOLD);
  1901→    if (simHit) {
  1902→      if (content.length > simHit.content.length) simHit.content = content;
  1903→      simHit.reinforcedAt = now;
  1904→      simHit.reinforceCount = (simHit.reinforceCount ?? 0) + 1;
  1905→      simHit.weight = higherWeight(simHit.weight, item.weight);
  1906→      // 【fix3-a 8】合并时角色归属标注 OR 保留（原条已标 或 合并后的内容带标注都保留）
  1907→      if (simHit.hasRoleTag === true || memHasRoleTag(simHit.content) || memHasRoleTag(content)) simHit.hasRoleTag = true;
  1908→      // 时间感知（与精确重复同规则）：未手动编辑过 → 新时间保鲜；手动设置过 → 用户优先不覆盖
  1909→      if (!simHit.timeEditedAt) {
  1910→        if (item.eventTime != null) simHit.eventTime = item.eventTime;
  1911→        if (item.expiresAt != null) simHit.expiresAt = item.expiresAt;
  1912→      }
  1913→      if (!simHit.sourceMsgId && sourceMsgId) simHit.sourceMsgId = sourceMsgId;
  1914→      // 溯源字段补齐（动态来源记忆）：点赞/评论同动态连续写入时内容相近会被合并成一条，
  1915→      // 合并目标（先写入的那条）若缺溯源字段，用本次传入的补齐——否则删评论时按
  1916→      // sourceCommentId 清理会漏掉这条合并碎片（六.3 级联清理依赖这些字段）
  1917→      if (extra) {
  1918→        if (!simHit.source && extra.source) simHit.source = extra.source;
  1919→        if (!simHit.sourcePostId && extra.sourcePostId) simHit.sourcePostId = extra.sourcePostId;
  1920→        if (!simHit.sourceKind && extra.sourceKind) simHit.sourceKind = extra.sourceKind;
  1921→        if (!simHit.sourceCommentId && extra.sourceCommentId) simHit.sourceCommentId = extra.sourceCommentId;
  1922→        if (!simHit.sourceGroupId && extra.sourceGroupId) simHit.sourceGroupId = extra.sourceGroupId;
  1923→        if (!simHit.groupMembers && extra.groupMembers) simHit.groupMembers = extra.groupMembers;
  1924→      }
  1925→      seen.add(normText(simHit.content));
  1926→      merged++;
  1927→      continue;
  1928→    }
  1929→    // 3) 全新记忆
  1930→    seen.add(key);
  1931→    added.push(newItem(content, item));
  1932→  }
  1933→  if (added.length > 0 || merged > 0 || superseded > 0) {
  1934→    // 容量软上限：写入前对合并后的全量做一次择劣淘汰（超出 500 条时优先丢
  1935→    // 已消费 → 低权重 → 更早创建的碎片），防止碎片库只增不减无限膨胀
  1936→    writeJSON(fragKey(contactId), capFragments([...list, ...added], new Set(added.map((f) => f.id))));
  1937→  }
  1938→  return { added, merged, superseded };
  1939→}

  1941→/** 把当前待总结（未被更新/未过期/未消费/未归档）的碎片总结为核心记忆，并标记来源碎片已消费（阈值判断由调用方负责） */
  1942→async function summarizePendingIntoCore(
  1943→  contactId: string,
  1944→  apiConfig: ApiConfig,
  1945→  names?: MemNames | null,
  1946→  /** 群聊来源成员显示名（透传总结 prompt：保留其他成员的名字，不强制二人视角） */
  1947→  participants?: string[]
  1948→): Promise<MemCore> {
  1949→  const { forget } = getMemSettings(contactId);
  1950→  const now = Date.now();
  1951→  const pending = readFragments(contactId).filter(
  1952→    (f) =>
  1953→      !f.consumedAt &&
  1954→      !f.supersededAt &&
  1955→      !isMemExpired(f, now) &&
  1956→      fadeState(f as FadeInput, forget, now) !== 'faded'
  1957→  );
  1958→  const { userName, peerName } = namesOf(names);
  1959→  const res = await callMemoryApi<SummarizeApiResult>(
  1960→    'summarize',
  1961→    {
  1962→      level: 'core',
  1963→      fragments: pending.map((f) => f.content),
  1964→      userName,
  1965→      peerName,
  1966→      ...(participants?.length ? { participants } : {}),
  1967→    },
  1968→    apiConfig
  1969→  );
  1970→  const summary = (res.summary || '').trim();
  1971→  if (!summary) throw new Error('总结结果为空');
  1972→  const core: MemCore = {
  1973→    id: uid(),
  1974→    contactId,
  1975→    content: summary,
  1976→    fragmentCount: pending.length,
  1977→    sourceIds: pending.map((f) => f.id),
  1978→    apps: Array.from(new Set(pending.map((f) => f.app))),
  1979→    createdAt: Date.now(),
  1980→    // 时间感知：来源碎片中带事件时间的，取最早一条作为核心的事件时间（可选，仅标注用）
  1981→    eventTime: (() => {
  1982→      const ets = pending.map((f) => f.eventTime).filter((t): t is number => t != null);
  1983→      return ets.length > 0 ? Math.min(...ets) : undefined;
  1984→    })(),
  1985→    // 群聊来源携带（互通与群间隔离过滤用）：混合来源 privateSource=true，纯群聊为 false
  1986→    groupIds: Array.from(new Set(pending.map((f) => f.sourceGroupId).filter((g): g is string => !!g))),
  1987→    privateSource: pending.some((f) => f.source !== 'group'),
  1988→    // 【fix3-a 8】总结 API 可能磨掉「（AI角色本人）/（用户本人）」标注：被消费的碎片任一带标注
